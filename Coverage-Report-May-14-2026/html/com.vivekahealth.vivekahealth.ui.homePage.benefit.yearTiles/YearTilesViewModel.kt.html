    fun findMissingStringPositions(): List<Int> {
        val firstSet = monthList
        val secondSet = workHourListPerYear.map { it.month }.toSet()

        val missingPositions = mutableListOf<Int>()
        for ((index, number) in firstSet.withIndex()) {
            if (!secondSet.contains(number)) {
                missingPositions.add(index)
            }
        }
        return missingPositions
    }

    fun addMissingMonths() {
        var missingList = findMissingStringPositions()
        missingList.forEach { item ->
            workHourListPerYear.add(
                item,
                WorkLog(
                    year = currentYear.toString(),
                    month = (item + 1).toString(),
                    addedItem = true
                )
            )
        }
    }

    fun loadReason(year: Int) {
        var result = workHourListV2.groupBy { it -> it.year.toInt() }
        Timber.d(result.toString())

        Timber.d(result[year].toString())

        contractDetail?.let { item ->
            noLookBack = !item.hasLookBackPeriod
            var months = HelperUtils.getMonthIntegers(
                item.lookBackPeriod,
                item.offsetMonth,
                currentMonth,
                currentYear
            )

            if (item.requiredAmount == null && item.requiredHours != null) {
                val hoursWorkedList = months.map { monthInt ->
                    val adjustedYear = if (monthInt > currentMonth) currentYear - 1 else currentYear

                    result[adjustedYear]?.find { it.month == monthInt.toString() }?.hoursWorked ?: 0.0
                }
                title2MyBenefit = resources.getString(
                    R.string.the_medical_benefit_requires_s_hours_of_work_in_s_you_have_s_hours_for_these_months,
                    item.benefitName,
                    HelperUtils.floor(item.requiredHours ?: 0.0),
                    HelperUtils.getFormattedMonthNames(
                        item.lookBackPeriod,
                        item.offsetMonth,
                        currentMonth,
                        currentYear,
                        resources
                    ),
                    HelperUtils.floor(hoursWorkedList.sum() ?: 0.0)

                )
            } else {
                val premiumWorkedList = months.map { monthInt ->
                    val adjustedYear = if (monthInt > currentMonth) currentYear - 1 else currentYear

                    result[adjustedYear]?.find { it.month == monthInt.toString() }?.memberContribution ?: 0.0
                }
                title2MyBenefit = resources.getString(
                    R.string.premium_text,
                    item.benefitName,
                    String.format(
                        Locale.getDefault(),
                        "%#.2f",
                        item.requiredAmount ?: 0.0
                    ),
                    HelperUtils.getFormattedMonthNames(
                        item.lookBackPeriod,
                        item.offsetMonth,
                        currentMonth,
                        currentYear,
                        resources
                    ),
                    HelperUtils.floor(premiumWorkedList.sum() ?: 0.0)
                )
            }
        }
    }
