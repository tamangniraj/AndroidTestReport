    fun loadInsuranceCard(selfId: String, cardType: String) {
        selectedMemberId = selfId
        compositeDisposable.add(
            benefitWorkRepository.getInsuranceCard(selfId, cardType, false).subscribeOn(
                Schedulers.io()
            )
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->

                    loading = false

                    _insuranceCardsInfo.value = Event(it.values)
//                memberCardList.clear()
                    val cardlist = it.values

                    memberCardList.clear()
                    benefitList.clear()

                    cardlist.forEach { cardtype ->
                        var card = cardtype.benefitType
                        card?.also {
                            benefitList.add(CardList(it, selected = false))
                        }
                    }
                    benefitList.add(
                        CardList(resources.getString(R.string.diagnoses), selected = false)
                    )
                    benefitList.add(
                        CardList(resources.getString(R.string.medications), selected = false)
                    )

                    _benefitListCreated.value = Event(Unit)
                    Timber.e(memberCardList.joinToString())

                    for (i in memberCardList.indices) {
                        when (i) {
                            0 -> {
                                hasCardOne = true
                                cardOneTitle = memberCardList[i]
                            }
                            1 -> {
                                hasCardTwo = true
                                cardTwoTitle = memberCardList[i]
                            }
                            2 -> {
                                hasCardThree = true
                                cardThreeTitle = memberCardList[i]
                            }
                            3 -> {
                                hasCardFour = true
                                cardFourTitle = memberCardList[i]
                            }
                            4 -> {
                                hasCardFive = true
                                cardFiveTitle = memberCardList[i]
                            }
                            5 -> {
                                hasCardSix = true
                                cardSixTitle = memberCardList[i]
                            }
                            6 -> {
                                hasCardSeven = true
                                cardSevenTitle = memberCardList[i]
                            }
                            7 -> {
                                hasCardEight = true
                                cardEightTitle = memberCardList[i]
                            }
                            8 -> {
                                hasCardNine = true
                                cardNineTitle = memberCardList[i]
                            }
                            9 -> {
                                hasCardTen = true
                                cardTenTitle = memberCardList[i]
                            }

                            else -> { // Note the block
                                // do nothing
                            }
                        }
                    }
                }, {
                    loading = false
                })
        )
    }
    fun sendInformation() {
        var cardTypes = benefitList.map { data ->
            if (data.selected && !(
                data.name.equals(resources.getString(R.string.medications)) || data.name.equals(
                        resources.getString(R.string.diagnoses)
                    )
                )
            ) {
                data.name
            } else {
                ""
            }
        }.filter { it.isNotEmpty() }

        benefitList.forEach { item ->
            if (item.name.equals(resources.getString(R.string.medications), true)) {
                medicationsSelected = item.selected
            } else if (item.name.equals(resources.getString(R.string.diagnoses), true)) {
                diagnosesSelected = item.selected
            }
        }

        var selectedCards = cardTypes.joinToString()

        if (isMail) {
            _navigateToMail.value = Event(selectedCards)
        } else if (isPrint) {
            if (selectedCards.isNullOrEmpty() && !diagnosesSelected && !medicationsSelected) {
                _navigateToNoCards.value = Event(Unit)
            } else {
                compositeDisposable.add(
                    homeRepository.downloadPDF(
                        selectedMemberId,
                        selectedCards,
                        diagnosesSelected,
                        medicationsSelected
                    ).subscribeOn(
                        Schedulers.io()
                    )
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                        }
                        .subscribe({ it ->

                            loading = false

                            _navigateToPDFDownloaded.value = Event(it)
                        }, {
                            loading = false
                            var error = it as RetrofitException
                            if (error.code?.contains("403") ?: false) {
                                _navigateToShareNeeded.value = Event(it.message.toString())
                            }
                        })
                )
            }
        } else {
            compositeDisposable.add(
                homeRepository.shareBenefitCards(
                    selectedMemberId,
                    sendEmail,
                    selectedCards,
                    diagnosesSelected,
                    medicationsSelected
                ).subscribeOn(
                    Schedulers.io()
                )
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        loading = true
                    }
                    .subscribe({ it ->

                        loading = false

                        _emailResponse.value = Event(it)
                    }, {
                        loading = false
                    })
            )
        }
    }

    fun setSelectedMember(memberInfo: MemberInfo) {
//
        val index = eligibleDependentMemberList.indexOf(memberInfo)

        for (item in eligibleDependentMemberList) {
            if (eligibleDependentMemberList.indexOf(item) == index) {
                eligibleDependentMemberList[index].apply {
                    isSelected = true
                }
            } else {
                eligibleDependentMemberList[eligibleDependentMemberList.indexOf(item)].apply {
                    isSelected = false
                }
            }
        }

        listForDependentAdapter = eligibleDependentMemberList.toList()
        _bubbleListGeneratedNew.value = Event(listForDependentAdapter)

        resetCardListSelection()
    }

    fun requestMail(selectedCards: String) {
        var referenceList: List<String> = listOf(selectedMemberId)
        var result: List<String> = selectedCards.split(",").map { it.trim() }

        compositeDisposable.add(
            homeRepository.mailBenefit(
                preferences.getString("memberID", "").toString(),
                RequestMail(
                    referenceList,
                    result
                )

            ).subscribeOn(
                Schedulers.io()
            )
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->

                    loading = false

                    _navigateToMailSuccess.value = Event(Unit)
                }, {
                    loading = false
                })
        )
    }
