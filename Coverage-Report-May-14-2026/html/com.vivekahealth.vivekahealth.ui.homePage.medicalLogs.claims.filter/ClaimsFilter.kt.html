package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.claims.filter

import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.lifecycle.ViewModelProvider
import com.vivekahealth.vivekahealth.data.local.home.ClaimsFilterData
import com.vivekahealth.vivekahealth.data.local.home.DiagnosisRange
import com.vivekahealth.vivekahealth.databinding.*
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseBottomSheetDialogFragment
import com.vivekahealth.vivekahealth.utils.EventObserver
import javax.inject.Inject

class ClaimsFilter : BaseBottomSheetDialogFragment(), Injectable {

    companion object {
        fun newInstance() = ClaimsFilter()
        const val TAG = "ClaimsFilterData"
    }

    var filterListener: ClaimsFilterListener? = null

    var filterOption: ClaimsFilterData? = null

    var initialRange: DiagnosisRange ? = null
    fun setCallback(filterListener: ClaimsFilterListener, filterOption: ClaimsFilterData) {
        this.filterListener = filterListener
        this.filterOption = filterOption
    }
    private lateinit var binding: ClaimsFilterFragmentBinding

    private lateinit var viewModel: ClaimsFilterViewModel
    var sentfilter = false

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = ClaimsFilterFragmentBinding.inflate(inflater, container, false)

        return binding.root
    }
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
    }

    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(ClaimsFilterViewModel::class.java)

        binding.viewModel = viewModel

        viewModel.claimsStatus = filterOption!!.claimsStatus
        viewModel.claimsTypes = filterOption!!.claimType

        viewModel.loadSetting()

        viewModel.navigateToClear.observe(
            viewLifecycleOwner,
            EventObserver {
            }
        )

        viewModel.navigateToPres.observe(
            viewLifecycleOwner,
            EventObserver {
                filterListener?.onClose()
            }
        )
        viewModel.navigateToDocFilter.observe(
            viewLifecycleOwner,
            EventObserver {
                filterListener!!.apply(
                    ClaimsFilterData(viewModel.claimsTypes, viewModel.claimsStatus)
                )
            }
        )
    }
}
