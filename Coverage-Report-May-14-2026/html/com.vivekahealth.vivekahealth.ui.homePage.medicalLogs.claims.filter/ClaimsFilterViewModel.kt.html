package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.claims.filter

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.local.VivekaDatabase
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.disposables.CompositeDisposable
import java.util.*
import javax.inject.Inject
import kotlin.collections.ArrayList

class ClaimsFilterViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository,
    private val vivekaDatabase: VivekaDatabase
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _navigateToPres = MutableLiveData<Event<Unit>>()
    val navigateToPres: LiveData<Event<Unit>>
        get() = _navigateToPres

    var claimsTypes: ArrayList<String> = arrayListOf()
    var claimsStatus: ArrayList<String> = arrayListOf()

    private val _navigateToDocFilter = MutableLiveData<Event<Unit>>()
    val navigateToDocFilter: LiveData<Event<Unit>>
        get() = _navigateToDocFilter

    private val _navigateToClear = MutableLiveData<Event<Unit>>()
    val navigateToClear: LiveData<Event<Unit>>
        get() = _navigateToClear

    var professional: Boolean = false
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.professional, null)
        }

    var facility: Boolean = false
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.facility, null)
        }

    var newOpt: Boolean = false
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.newOpt, null)
        }

    var pending: Boolean = false
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.pending, null)
        }

    var paid: Boolean = false
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.paid, null)
        }

    var rejected: Boolean = false
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.rejected, null)
        }
    var count: Int = 0
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.count, null)
        }

    fun loadSetting() {
        for (item in claimsTypes) {
            if (item.equals("2", true)) {
                professional = true
            }
            if (item.equals("1", true)) {
                facility = true
            }
        }

        for (item in claimsStatus) {
            if (item.equals("new", true)) {
                newOpt = true
            }
            if (item.equals("pending", true)) {
                pending = true
            }
            if (item.equals("paid", true)) {
                paid = true
            }
            if (item.equals("denied", true)) {
                rejected = true
            }
        }
        count = claimsStatus.size + claimsTypes.size
    }

    fun countSelected() {
        count = claimsStatus.size + claimsTypes.size
    }
    fun newOpt() {
        newOpt = !newOpt

        if (newOpt) {
            claimsStatus.add("new")
        } else {
            claimsStatus.removeAt(claimsStatus.indexOf("new"))
        }
        countSelected()
    }

    fun pen() {
        pending = !pending

        if (pending) {
            claimsStatus.add("pending")
        } else {
            claimsStatus.removeAt(claimsStatus.indexOf("pending"))
        }
        countSelected()
    }

    fun paid() {
        paid = !paid

        if (paid) {
            claimsStatus.add("paid")
        } else {
            claimsStatus.removeAt(claimsStatus.indexOf("paid"))
        }
        countSelected()
    }

    fun rejected() {
        rejected = !rejected
        if (rejected) {
            claimsStatus.add("denied")
        } else {
            claimsStatus.removeAt(claimsStatus.indexOf("denied"))
        }
        countSelected()
    }

    fun pro() {
        professional = !professional
        if (professional) {
            claimsTypes.add("2")
        } else {
            claimsTypes.removeAt(claimsTypes.indexOf("2"))
        }
        countSelected()
    }

    fun facility() {
        facility = !facility
        if (facility) {
            claimsTypes.add("1")
        } else {
            claimsTypes.removeAt(claimsTypes.indexOf("1"))
        }

        countSelected()
    }
    fun apply() {
        _navigateToDocFilter.value = Event(Unit)
    }

    fun close() {
        _navigateToPres.value = Event(Unit)
    }

    fun clear() {
        professional = false
        facility = false
        newOpt = false
        pending = false
        paid = false
        rejected = false
        claimsTypes.clear()
        claimsStatus.clear()
        count = 0
        _navigateToClear.value = Event(Unit)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
