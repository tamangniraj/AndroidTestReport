package com.vivekahealth.vivekahealth.ui.homePage.findCare.billPreview.deiveryOption

import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.lifecycle.ViewModelProvider
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.databinding.FragmentDelivaryOptionBinding
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseBottomSheetDialogFragment
import com.vivekahealth.vivekahealth.ui.homePage.findCare.billPreview.deiveryOption.bus.OptionCallback
import com.vivekahealth.vivekahealth.utils.EventObserver
import javax.inject.Inject

class DeliveryOption : BaseBottomSheetDialogFragment(), Injectable {

    companion object {
        fun newInstance() = DeliveryOption()
    }

    private lateinit var viewModel: DeliveryOptionViewModel

    private lateinit var binding: FragmentDelivaryOptionBinding

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory
    var successCallbackFrag: OptionCallback? = null
    fun setSuccessCallback(
        successCallback: OptionCallback
    ) {
        this.successCallbackFrag = successCallback
    }

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = FragmentDelivaryOptionBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setStyle(STYLE_NORMAL, R.style.CustomBottomSheetDialogTheme)
    }

    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            DeliveryOptionViewModel::class.java
        )
        binding.viewModel = viewModel

        viewModel.navigateToOK.observe(
            viewLifecycleOwner,
            EventObserver {
                successCallbackFrag?.onOkClick(viewModel.option)
            }
        )

        viewModel.navigateToBack.observe(
            viewLifecycleOwner,
            EventObserver {
                successCallbackFrag?.onBackClick()
            }
        )
    }
}
