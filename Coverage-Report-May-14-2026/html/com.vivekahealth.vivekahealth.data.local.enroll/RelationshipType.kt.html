package com.vivekahealth.vivekahealth.data.local.enroll

import android.os.Parcelable
import kotlinx.parcelize.Parcelize

enum class RelationshipType(val code: Int) {
    PRIMARY(18),
    SPOUSE(1),
    CHILD(19);

    companion object {
        fun fromCode(code: Int): RelationshipType? {
            return values().find { it.code == code }
        }
    }
}

@Parcelize
data class RootDocumentResponse(
    val documentList: List<DocumentRelation>
) : Parcelable

@Parcelize
data class DocumentRelation(
    val relationshipType: Int,
    val documents: List<DocumentItem>
) : Parcelable

@Parcelize
data class DocumentItem(
    val documentType: Int,
    val uploaded: Boolean,
    val required: Boolean,
    val documentName: String,
    val translatedDocumentName: String
) : Parcelable {
    fun toDocumentPlus(relationshipType: RelationshipType): DocumentItemPlus = DocumentItemPlus(
        documentType,
        uploaded,
        required,
        documentName,
        translatedDocumentName,
        relationshipType
    )
}

data class DocumentItemPlus(
    val documentType: Int,
    val uploaded: Boolean,
    val required: Boolean,
    val documentName: String,
    val translatedDocumentName: String,
    val relationshipType: RelationshipType
)
