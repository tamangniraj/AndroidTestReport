    fun getCategory(clientId: String) {
        docTypeList.clear()

        compositeDisposable.add(
            documentRepository.getType(clientId)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ typeList ->
                    Timber.d(typeList.toString())

                    var list = typeList.dtos
                    typeList.dtos.let { it1 ->

                        for (doc in it1) {
                            docTypeList.add(doc)
                        }
                    }

                    _typeListGenerated.value = Event(docTypeList.toList())

                    loading = false
                }, {
                    Timber.d(it.toString())

                    loading = false
                })
        )
    }

    fun setSelectedDoc(docType: Dto) {
        val index = docTypeList.indexOf(docType)

        for (item in docTypeList) {
            if (docTypeList.indexOf(item) == index) {
                docTypeList[index].apply {
                    isSelected = !(docTypeList[index].isSelected)
                }
            } else {
                docTypeList[docTypeList.indexOf(item)].apply {
                    isSelected = false
                }
            }
        }

        // set doc title

        docTypeTitle = "Select Type"
        docCategoryTitle = "Document Category"
        for (doc in docTypeList) {
            if (doc.isSelected) {
                docTypeTitle = doc.documentType
                docCategoryTitle = doc.documentCategory
                selectedDocType = doc
            }
        }
        Log.e("setIndexdoc", index.toString())
        Log.e("setdoc", docTypeList[index].toString())
        _typeListGenerated.value = Event(docTypeList.toList())
//        _newDocTypeListGenerated.value = Event(docTypeList.toList())
    }

    fun selectedReferenceMember(memberInfo: MemberInfo) {
        val index = bubbleMemberList.indexOf(memberInfo)

        for (item in bubbleMemberList) {

                    isSelected = !(bubbleMemberList[index].isSelected)
                }
            } else {
                bubbleMemberList[bubbleMemberList.indexOf(item)].apply {
                    isSelected = false
                }
            }
        }
        // set reference name

        memberName = "Select Person"
        for (mem in bubbleMemberList) {
            if (mem.isSelected) {
                memberName = mem.firstName.toString()
            }
        }

        _bubbleListGenerated.value = Event(bubbleMemberList.toList())
    }

    fun uploadDocument(docFile: File, docType: Int) {
        val file: File = docFile

        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    documentRepository.uploadDoc(
                        memberInfo.selfId, filePart, docType, categoryId,
                        HelperUtils.fileNamePending(
                            pendingType,
                            pendingDocument?.inReferenceTo ?: ""
                        ),
                        refrenceID, location?.longitude, location?.latitude, requestId
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                        }
                        .subscribe({
                            loading = false

                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().success(
                                it
                            )
                        }, {
                            loading = false

                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                                it.localizedMessage!!
                            )
                        })
                )
            }
        )
    }

    fun String.capitalize(): String {
        return this.replaceFirstChar {
            if (it.isLowerCase()) {
                it.titlecase(Locale.getDefault())
            } else {
                it.toString()
            }
        }
    }
