package com.vivekahealth.vivekahealth.ui.guestVerify

import android.content.SharedPreferences
import androidx.core.content.edit
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.HelperUtils
import javax.inject.Inject

class BiometricPermissionViewModel @Inject constructor(
    private val preferences: SharedPreferences
) : ViewModel() {

    private val _checkBiometricPermission = MutableLiveData<Event<Unit>>()
    val checkBiometricPermission: LiveData<Event<Unit>>
        get() = _checkBiometricPermission

    private val _locationRequest = MutableLiveData<Event<Unit>>()
    val locationRequest: LiveData<Event<Unit>>
        get() = _locationRequest

    fun checkForBiometricPermission(username: String) {
        preferences.edit {
            putString(Constants.GUEST_USERNAME, username)
            val key = "$username${Constants.EMAIL_VERIFIED}"
            putBoolean(key, false)
        }
        _checkBiometricPermission.value = Event(Unit)
    }

    fun proceedLocationRequest() {
        _locationRequest.value = Event(Unit)
    }

    fun saveBiometricPreference(email: String, password: String) {
        val encodedEmail = HelperUtils.encrypt(email)
        val encodedPassword = HelperUtils.encrypt(password)
        preferences.edit().putString(Constants.encodedEmail, encodedEmail).apply()
        preferences.edit().putString(Constants.encodedPassword, encodedPassword).apply()
    }
}
