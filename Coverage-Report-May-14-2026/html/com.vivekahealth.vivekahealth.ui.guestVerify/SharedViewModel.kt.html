package com.vivekahealth.vivekahealth.ui.guestVerify

import android.content.SharedPreferences
import android.graphics.Bitmap
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.gson.Gson
import com.google.mlkit.vision.barcode.common.Barcode
import com.google.mlkit.vision.barcode.common.Barcode.DriverLicense
import com.vivekahealth.vivekahealth.data.local.guest.DLImage
import com.vivekahealth.vivekahealth.data.remote.guest.ProcessLicenseResponse
import com.vivekahealth.vivekahealth.ui.guestHome.repository.GuestHomeRepository
import com.vivekahealth.vivekahealth.ui.guestVerify.documents.DocumentType
import com.vivekahealth.vivekahealth.ui.guestVerify.repository.GuestUploadRepository
import com.vivekahealth.vivekahealth.ui.enrollVTwo.UploadedDocument
import com.vivekahealth.vivekahealth.utils.DateUtils
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.HelperUtils
import io.reactivex.Single
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import okhttp3.MultipartBody
import org.json.JSONObject
import java.util.UUID
import javax.inject.Inject

class SharedViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    var uploadRepository: GuestUploadRepository,
    private val guestHomeRepository: GuestHomeRepository
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _isLoading = MutableLiveData<Event<Boolean>>()
    val isLoading: LiveData<Event<Boolean>> get() = _isLoading

    private val _frontImage = MutableLiveData<DLImage?>()
    val frontImage: LiveData<DLImage?> get() = _frontImage

    private val _backImage = MutableLiveData<DLImage?>()
    val backImage: LiveData<DLImage?> get() = _backImage

    private val _ssnImage = MutableLiveData<DLImage?>()
    val ssnImage: LiveData<DLImage?> get() = _ssnImage

    private val _selfieImage = MutableLiveData<DLImage?>()
    val selfieImage: LiveData<DLImage?> get() = _selfieImage

    private val _token = MutableLiveData<String?>()
    val token: LiveData<String?> get() = _token

    var email: String = ""
    var allowedDocumentTypes: List<String> = listOf("SSN")
    fun clearFront() {
        _frontImage.value = null
    }

    fun clearSsn() {
        _ssnImage.value = null
    }

    fun clearBack() {
        _backImage.value = null
    }

    fun clearSelfie() {
        _selfieImage.value = null
    }

    fun updateFrontImage(bitmap: Bitmap?) {
        _frontImage.value = DLImage(bitmap = bitmap)
    }

    fun updateSSNImage(bitmap: Bitmap?, ssnTxt: String?, isSsn: Boolean?) {
        _ssnImage.value = DLImage(bitmap = bitmap, ssn = ssnTxt, isSsn = isSsn)
    }

    fun updateSuccessTokenSSN(token: String) {
        val currentBitmap = _ssnImage.value // Get the current bitmap
        _ssnImage.value = currentBitmap?.copy(bitmapUploadedId = token) // Update with the new token
    }

    fun updateVerifiedBarCodeInServer(token: String) {
        val currentBitmap = _backImage.value // Get the current bitmap
        _backImage.value =
            currentBitmap?.copy(verifiedBarCodeInServer = token) // Update with the new token
    }

    fun updateSuccessTokenFront(token: String) {
        val currentBitmap = _frontImage.value // Get the current bitmap
        _frontImage.value =
            currentBitmap?.copy(bitmapUploadedId = token) // Update with the new token
    }

    fun updateSuccessTokenSelfie(token: String) {
        val currentBitmap = _selfieImage.value // Get the current bitmap
        _selfieImage.value =
            currentBitmap?.copy(bitmapUploadedId = token) // Update with the new token
    }

    fun updateSuccessTokenBack(token: String, dlToken: String, requiredSsn: Boolean) {
        val currentBackImage = _backImage.value // Get the current BackImage object
        _backImage.value = currentBackImage?.copy(
            bitmapUploadedId = token,
            driverLicenseUploadedId = dlToken,
            requiredSnn = requiredSsn
        ) // Update with the new token

        if (!requiredSsn) {
            clearSsn()
        }
    }

    fun updateBackImage(bitmap: Bitmap?, driverLicense: DriverLicense?, barcode: Barcode?) {
        var json = ""
        barcode?.let {
            val gson = Gson()
            val barCodeJson = gson.toJson(barcode)
            val jsonObject = JSONObject(barCodeJson)

            if (jsonObject.has("zza")) {
                val zza = jsonObject.getJSONObject("zza")
                if (zza.has("zza")) {
                    val zzaa = zza.getJSONObject("zza")
                    if (zzaa.has("zzo")) {
                        json = zzaa.getJSONObject("zzo").toString()
                    }
                }
            }
        }

        _backImage.value = DLImage(
            bitmap = bitmap,
            driverLicense = driverLicense,
            bitmapUploadedId = null,
            driverLicenseUploadedId = null,
            rawQRString = json
        )
    }

    fun updateSelfieImage(bitmap: Bitmap?, manualSelfie: Boolean? = false) {
        _selfieImage.value = DLImage(bitmap = bitmap, isManual = manualSelfie)
    }

    fun updateToken(newToken: String?) {
        _token.value = newToken
    }

    var loading = MutableLiveData<Event<Boolean>>()
    var isComingFromFailedBackCheck = false

    // Stores the parsed DL data returned by guest/v2/processLicense after DL_FRONT is captured
    var dlFrontData: ProcessLicenseResponse? = null
    val dlFrontDataLive = MutableLiveData<ProcessLicenseResponse?>()
    var gson = Gson()
    var hasSuccessfullyProcessedFront: Boolean = false

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    val primaryUploadList: MutableList<UploadedDocument> = mutableListOf()

    fun processFrontLicenseAndUpload(bitmap: Bitmap) {
        val docTypeId = 103

        compositeDisposable.add(
            Single.fromCallable {
                HelperUtils.createMultipartBodyPartFromBitmap(
                    bitmap,
                    "file",
                    "${UUID.randomUUID()}-${DocumentType.DL_FRONT.name}.jpg"
                )
            }
                .subscribeOn(Schedulers.io())
                .flatMap { part ->
                    guestHomeRepository.processLicense(part)
                        .doOnSuccess { response ->
                            if (response.isSuccessful) {
                                response.body()?.takeIf { !it.isEmpty }?.let { body ->
                                    dlFrontData = body
                                    dlFrontDataLive.postValue(body)
                                }
                            }
                        }
                        .flatMap { guestHomeRepository.uploadOpenFile(part, docTypeId.toString()) }
                }
                .observeOn(AndroidSchedulers.mainThread())
                .subscribe({ response ->
                    val uploadedFileId = response.id?.toIntOrNull() ?: return@subscribe
                    primaryUploadList.add(
                        UploadedDocument(
                            fileTypeId = docTypeId,
                            uploadedFileReferenceId = uploadedFileId
                        )
                    )
                }, { })
        )
    }

    fun uploadFrontAndProcess(
        documentType: DocumentType,
        file: Bitmap?,
        token: String
    ) {
        if (file == null) {
            return
        }

        val filePart: MultipartBody.Part = HelperUtils.createMultipartBodyPartFromBitmap(
            bitmap = file,
            "file",
            "${UUID.randomUUID()}-$documentType-${DateUtils.getCurrentLocalTime()}"
        )

        compositeDisposable.add(
            uploadRepository.uploadDoc(
                token,
                filePart,
                documentType
            )
                .subscribeOn(Schedulers.io())
                .flatMap { uploadFrontResult ->
                    uploadRepository.processDocument(token)
                }
                .observeOn(AndroidSchedulers.mainThread())
                .subscribe({ processDocumentResult ->
                    if (processDocumentResult.isSuccessful) {
                        hasSuccessfullyProcessedFront = true
                        updateSuccessTokenFront(
                            "${DateUtils.getCurrentLocalTime()}"
                        )
                    } else {
                        hasSuccessfullyProcessedFront = false
                    }
                }, {
                    hasSuccessfullyProcessedFront = false
                })
        )
    }
}
