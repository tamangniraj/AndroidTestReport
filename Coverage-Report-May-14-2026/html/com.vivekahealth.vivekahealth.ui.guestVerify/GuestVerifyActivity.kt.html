    override fun attachBaseContext(newBase: Context) {
        val langCode = Constants.LOCAL_LANG_MODIFIED ?: ""
        if (langCode.isNotEmpty()) {
            val locale = Locale(langCode)
            Locale.setDefault(locale)
            Timber.d("Locale1 ${locale.language}")

            val config = Configuration(newBase.resources.configuration)
            config.setLocale(locale) // <- API 17+ is okay; fallback below for API 16
            config.setLayoutDirection(locale)

            val updatedContext = newBase.createConfigurationContext(config)
            super.attachBaseContext(updatedContext)
        } else {
            super.attachBaseContext(newBase)
        }
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        val insetsController = WindowCompat.getInsetsController(window, window.decorView)
        insetsController.isAppearanceLightStatusBars = true
        insetsController.isAppearanceLightNavigationBars = true

        super.onCreate(savedInstanceState)

        executor = ContextCompat.getMainExecutor(this)

        binding = DataBindingUtil.setContentView(this, R.layout.activity_guest_verify)
        setSupportActionBar(binding.toolbar)
        fromHomePage = intent.getBooleanExtra(FROM_HOME_PAGE, false)
        email = intent.getStringExtra(EMAIL) ?: ""
        password = intent.getStringExtra(PASSWORD) ?: ""
        binding.toolbar.setNavigationOnClickListener {
            if (isFinishing || isDestroyed) return@setNavigationOnClickListener
            if (fromHomePage) {
                this.finish()
            } else {
                showCancellationDialog()
            }
        }
        navigationFragmentManager.fragmentManager = supportFragmentManager
        navigationFragmentManager.containerId = R.id.fragment

        viewModel = ViewModelProvider(this, viewModelFactory).get(
            BiometricPermissionViewModel::class.java

        sharedViewModel.updateToken(
            intent.getStringExtra
            (VERIFICATION_SESSION_TAG)
        )

        sharedViewModel.email = email

        durationTracker = DurationTracker(firebaseAnalytics, sharedViewModel)
        durationTracker.startFlow()
        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_STARTED)

        val navController = findNavController(R.id.fragment)

        // Get documentType from intent and set it as start destination argument
        val documentType = intent.getSerializableExtra(DOCUMENT_TYPE)
        Timber.d("language; received $documentType")

        // Always set the graph programmatically to ensure arguments are passed correctly
        val navGraph = navController.navInflater.inflate(R.navigation.guest_verify_nav)
        if (documentType != null) {
            val bundle = Bundle().apply {
                putSerializable("documentType", documentType)
            }
            navController.setGraph(navGraph, bundle)
        } else {
            navController.graph = navGraph
        }

        navController.addOnDestinationChangedListener { _: NavController?, destination: NavDestination, arguments: Bundle? ->
            val stepPairs = stepCounterViewModel.getStepsTextAndPercent(destination.id, arguments)

            }

            updateToolbarTitle(getTitleForActivity(destination.id, arguments))
        }

        viewModel.locationRequest.observe(
            this,
            EventObserver {
                locationRequest()
            }
        )

        viewModel.checkBiometricPermission.observe(
            this,
            EventObserver {
                if (biometricPopup != null) {
                    return@EventObserver
                }

                val biometricManager = BiometricManager.from(this)
                when (biometricManager.canAuthenticate(BIOMETRIC_STRONG)) {
                    BiometricManager.BIOMETRIC_SUCCESS -> {
                        // App can authenticate using biometrics.
                        durationTracker.startStep("biometric_dialog")
                        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_BIOMETRIC_DIALOG_SHOWN)
                        biometricPopup = GuestBiometricRequestDialog.newInstance()
                        biometricPopup?.isCancelable = false
                        biometricPopup?.setCallback(object : VerifyIdentityListener {
                            override fun onProceed(token: String?) {
                                logGuestVerifyEvent(
                                    FirebaseTags.GUEST_VERIFY_BIOMETRIC_PROCEED_CLICKED
                                )
                                biometricPrompt = BiometricPrompt(
                                    this@GuestVerifyActivity,
                                    executor,
                                    object : BiometricPrompt.AuthenticationCallback() {
                                        override fun onAuthenticationError(
                                            errorCode: Int,
                                            errString: CharSequence
                                        ) {
                                            super.onAuthenticationError(errorCode, errString)
                                            logGuestVerifyEvent(
                                                FirebaseTags.GUEST_VERIFY_BIOMETRIC_ERROR
                                            ) {
                                                putInt("error_code", errorCode)
                                            }
                                            viewModel.proceedLocationRequest()
                                        }

                                        override fun onAuthenticationSucceeded(
                                            result: BiometricPrompt.AuthenticationResult
                                        ) {
                                            super.onAuthenticationSucceeded(result)
                                            logGuestVerifyEvent(
                                                FirebaseTags.GUEST_VERIFY_BIOMETRIC_SUCCESS
                                            )
                                            if (email.isNotEmpty() && password.isNotEmpty()) {
                                                viewModel.saveBiometricPreference(email, password)
                                            }
                                            val hasRequestedBiometricKey = sharedPreferences.getString(
                                                Constants.USER_EMAIL,
                                                ""
                                            ) + Constants.BIO_PERMISSION_REQUESTED
                                            sharedPreferences.edit().putBoolean(
                                                hasRequestedBiometricKey,
                                                true
                                            ).apply()
                                            viewModel.proceedLocationRequest()
                                        }

                                        override fun onAuthenticationFailed() {
                                            super.onAuthenticationFailed()
                                            logGuestVerifyEvent(
                                                FirebaseTags.GUEST_VERIFY_BIOMETRIC_FAILED
                                            )
                                        }
                                    }
                                )

                                promptInfo = BiometricPrompt.PromptInfo.Builder()
                                    .setTitle(
                                        getString(
                                            R.string.biometric_login_for,
                                            getString(R.string.app_name)
                                        )
                                    )
                                    .setSubtitle(
                                        getString(
                                            R.string.log_in_using_your_biometric_credential
                                        )
                                    )
                                    .setNegativeButtonText(getString(R.string.cancel))
                                    .build()

                                biometricPrompt.authenticate(promptInfo)
                            }

                            override fun onSkip() {
                                logGuestVerifyEvent(
                                    FirebaseTags.GUEST_VERIFY_BIOMETRIC_SKIP_CLICKED
                                )
                                viewModel.proceedLocationRequest()
                            }

                            override fun onError(err: String) {
                            }
                        })

                        if (biometricPopup?.isVisible != true) {
                            biometricPopup?.show(supportFragmentManager, "GuestBiometric")
                        }
                    }

                    BiometricManager.BIOMETRIC_ERROR_NO_HARDWARE -> {
                        // No biometric features available on this device.
                        viewModel.proceedLocationRequest()
                    }

                    BiometricManager.BIOMETRIC_ERROR_HW_UNAVAILABLE -> {
                        // Biometric features are currently unavailable.
                        viewModel.proceedLocationRequest()
                    }

                    BiometricManager.BIOMETRIC_ERROR_NONE_ENROLLED -> {
                        // Should prompt the user to create credentials that your app accepts.
                        viewModel.proceedLocationRequest()
                    }

                    BiometricManager.BIOMETRIC_ERROR_SECURITY_UPDATE_REQUIRED -> {
                        // TODO: is this where we need to re-verify biometric ??
                    }

                    BiometricManager.BIOMETRIC_ERROR_UNSUPPORTED -> {
                        viewModel.proceedLocationRequest()
                    }

                    BiometricManager.BIOMETRIC_STATUS_UNKNOWN -> {
                        viewModel.proceedLocationRequest()
                    }
                }
            }
        )

        val hasRequestedBiometricKey = sharedPreferences.getString(Constants.USER_EMAIL, "") + Constants.BIO_PERMISSION_REQUESTED
        val hasAlreadyAskedBiometric = sharedPreferences.getBoolean(hasRequestedBiometricKey, false)

        if (!hasAlreadyAskedBiometric && documentType?.toString() != "SSN_ONLY") {
            viewModel.checkForBiometricPermission(email)
        } else {
            viewModel.proceedLocationRequest()
        }
    }

    // step 3: location Request
    fun locationRequest() {
        durationTracker.startStep("location_permission")
        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_LOCATION_REQUESTED)
        if (HelperUtils.isGpsEnabled(this)) {
            val permissionApproved =
                this.hasPermission(Manifest.permission.ACCESS_FINE_LOCATION)
            if (!permissionApproved) {
                if (ActivityCompat.shouldShowRequestPermissionRationale(
                        this,
                        Manifest.permission.ACCESS_FINE_LOCATION
                    )
                ) {
                    Timber.d("showLocationPermissionRationale")
                    // Show rationale to the user
//                    showLocationPermissionRationale()
                    cameraRequest()
                } else {
                    Timber.d("requestPermissio")

                    requestPermissions(
                        arrayOf(Manifest.permission.ACCESS_FINE_LOCATION),
                        REQUEST_FINE_LOCATION_PERMISSIONS_REQUEST_CODE
                    )
                }
            } else {
                cameraRequest()
            }
        } else {
            logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_GPS_DISABLED_SHOWN)
            DialogUtils(
                this,
                object : DialogListener {
                    override fun onNeutralClicked() {
                        cameraRequest()
                    }

                    override fun onPositiveClicked() {
                        if (HelperUtils.isGpsEnabled(this@GuestVerifyActivity)) {
                            locationRequest()
                        } else {
                            startActivity(
                                Intent(android.provider.Settings.ACTION_LOCATION_SOURCE_SETTINGS)
                            )
                        }
//
                    }

                    override fun onNegativeClicked() {
                        cameraRequest()
                    }
                }
            ).dialog(
                getString(R.string.gps_off),
                getString(R.string.ok_string),
                getString(R.string.cancel),
                ""
            )
        }
    }

    // step 4: Camera Request
    fun cameraRequest() {
        durationTracker.startStep("camera_permission")
        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_CAMERA_REQUESTED)
        val permissionApproved =
            this.hasPermission(Manifest.permission.CAMERA)
        if (!permissionApproved) {
            if (ActivityCompat.shouldShowRequestPermissionRationale(
                    this,
                    Manifest.permission.CAMERA
                )
            ) {
                // Show rationale to the user
                showCameraPermissionRationale()
            } else {
                requestPermissions(
                    arrayOf(Manifest.permission.CAMERA),
                    REQUEST_CAMERA_PERMISSION_REQUEST_CODE
                )
            }
        } else {
            captureDocument()
        }
    }

    fun captureDocument() {
        if (fromHomePage) {
            return

        val popup = VerifyIdentityDialog.newInstance()
        popup.requestApi = false
        popup.setCallback(object : VerifyIdentityListener {
            override fun onProceed(token: String?) {
                popup.dismiss()
            }

            override fun onError(err: String) {
                popup.dismiss()
            }

            override fun onSkip() {
            }
        })

        popup.show(this.supportFragmentManager, "popup")
    }

    private fun showCameraPermissionRationale() {
        AlertDialog.Builder(this)
            .setTitle(getString(R.string.camera_permission_needed))
            .setMessage(getString(R.string.camera_permission_rationale))
            .setPositiveButton(getString(R.string.grant_permission)) { _, _ ->
                // Request the permission again
                requestPermissions(
                    arrayOf(Manifest.permission.CAMERA),

    override fun onRequestPermissionsResult(
        requestCode: Int,
        permissions: Array<out String?>,
        grantResults: IntArray,
        deviceId: Int
    ) {
        if (requestCode == REQUEST_FINE_LOCATION_PERMISSIONS_REQUEST_CODE) {
            when {
                grantResults.isNotEmpty() -> {
                    if (grantResults[0] == PackageManager.PERMISSION_GRANTED) {
                        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_LOCATION_GRANTED)
                        // gotpo next step camera
                        cameraRequest()
                    } else {
                        if (!ActivityCompat.shouldShowRequestPermissionRationale(
                                this,
                                Manifest.permission.ACCESS_FINE_LOCATION
                            )
                        ) {
                            // User selected "Don't ask again"
                            logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_LOCATION_DENIED) {
                                putBoolean("is_permanent", true)
                            }
                            showPermissionDeniedDialog()
                        } else {
                            // User denied without "Don't ask again"
                            logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_LOCATION_DENIED) {
                                putBoolean("is_permanent", false)
                            }
                            cameraRequest()
//                            showLocationPermissionRationale()
                        }
                    }
                }
            }
        } else if (requestCode == REQUEST_CAMERA_PERMISSION_REQUEST_CODE) {
            when {
                grantResults.isNotEmpty() -> {
                    if (grantResults[0] == PackageManager.PERMISSION_GRANTED) {
                        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_CAMERA_GRANTED)
                        // gotpo next capture document
                        captureDocument()
                    } else {
                        if (!ActivityCompat.shouldShowRequestPermissionRationale(
                                this,
                                Manifest.permission.CAMERA
                            )
                        ) {
                            // User selected "Don't ask again"
                            logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_CAMERA_DENIED) {
                                putBoolean("is_permanent", true)
                            }
                            showPermissionDeniedDialogCamera()
                        } else {
                            // User denied without "Don't ask again"
                            logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_CAMERA_DENIED) {
                                putBoolean("is_permanent", false)
                            }
                            showCameraPermissionRationale()
                        }
                    }
                }
            }
        }
    }

    private fun showCancellationDialog() {
        // Inflate the dialog layout
        val dialogView = layoutInflater.inflate(R.layout.dialog_cancle_seemless, null)

        // Create the dialog
        val dialog = AlertDialog.Builder(this)
            .setView(dialogView)
            .create()

        dialog.window?.setBackgroundDrawableResource(android.R.color.transparent)

        dialog.setCancelable(false)
        // Find views in the dialog
        val proceedToLogin: MaterialButton = dialogView.findViewById(R.id.proceedToLoginButton)
        val subtextView: TextView = dialogView.findViewById(R.id.txtSubText)
        val messageText = getString(
            R.string.we_ve_created_your_account_and_send_you_and_email_at_s,
            email
        )
        val spannableString = android.text.SpannableString(messageText)
        val emailStart = messageText.indexOf(email)
        if (emailStart != -1) {
            spannableString.setSpan(
                android.text.style.StyleSpan(android.graphics.Typeface.BOLD),
                emailStart,
                emailStart + email.length,
                android.text.Spannable.SPAN_EXCLUSIVE_EXCLUSIVE
            )
        }
        subtextView.text = spannableString

        proceedToLogin.setOnClickListener {
            logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_CANCELLED)
            durationTracker.logTotalDuration(completed = false)
            if (fromHomePage) {
                this.finish()
            } else {
                val intent = Intent(this, LoginActivity::class.java)
                intent.flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK
                startActivity(intent)

                // Close all activities
                this.finishAffinity()
            }
        }

        val continueRegistration: TextView = dialogView.findViewById(R.id.continueRegistration)
        continueRegistration.setOnClickListener {
            dialog.dismiss()
        }

        // Show the dialog
        dialog.show()
    }

    fun getTitleForActivity(destinationId: Int, arguments: Bundle? = null): String {
        return when (destinationId) {
            R.id.documentScanFragment -> {
                val documentType = arguments?.getSerializable("documentType")?.toString()
                if (documentType == "SSN") {
                    getString(R.string.additional_verification)
                } else {
                    getString(R.string.verify_identity)
                }
            }

            R.id.zipUpdateFragment -> getString(R.string.additional_verification)
            else -> getString(R.string.verify_identity)
        }
    }

    private fun logGuestVerifyEvent(eventName: String, additionalParams: Bundle.() -> Unit = {}) {
        val bundle = Bundle().apply {
            putString(FirebaseTags.PARAM_TOKEN, sharedViewModel.token.value ?: "")
            additionalParams()
        }
        firebaseAnalytics.logEvent(eventName, bundle)
    }

class DurationTracker(
    private val analytics: FirebaseAnalytics,
    private val sharedViewModel: SharedViewModel
) {
    private var stepStartTime: Long = 0
    private var flowStartTime: Long = 0
    private var currentStep: String = ""

    private fun getCommonParams(): Bundle = Bundle().apply {
        putString(FirebaseTags.PARAM_TOKEN, sharedViewModel.token.value ?: "")
    }

    fun startFlow() {
        flowStartTime = System.currentTimeMillis()
    }

    fun startStep(stepName: String) {
        if (currentStep.isNotEmpty()) {
            logStepDuration()
        }
        currentStep = stepName
        stepStartTime = System.currentTimeMillis()
    }

    fun logStepDuration(): Long {
        if (currentStep.isEmpty()) return 0
        val duration = System.currentTimeMillis() - stepStartTime
        val bundle = getCommonParams().apply {
            putString("step_name", currentStep)
            putLong("duration_ms", duration)
        }
        analytics.logEvent(FirebaseTags.GUEST_VERIFY_STEP_DURATION, bundle)
        return duration
    }

    fun logTotalDuration(completed: Boolean): Long {
        if (currentStep.isNotEmpty()) {
            logStepDuration()
        }
        val duration = System.currentTimeMillis() - flowStartTime
        val bundle = getCommonParams().apply {
            putLong("duration_ms", duration)
            putBoolean("completed", completed)
        }
        analytics.logEvent(FirebaseTags.GUEST_VERIFY_TOTAL_DURATION, bundle)
        return duration
    }
