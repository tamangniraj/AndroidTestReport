package com.vivekahealth.vivekahealth.ui.login

import android.content.Context
import android.content.SharedPreferences
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.utils.AppUpdateHelper
import com.vivekahealth.vivekahealth.utils.Constants
import javax.inject.Inject

class LoginMainViewModel @Inject constructor(
    context: Context,
    preferences: SharedPreferences
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    private val _isAppUpdateAvailable = MutableLiveData<Boolean>()
    val getAppUpdateAvailable: LiveData<Boolean> get() = _isAppUpdateAvailable
    var update: Boolean = false

    init {
        // reset device prompt flag
        preferences.edit().putBoolean(Constants.REQUEST_DEVICE_PROMPT, false).apply()
        // reset magic login flags
        preferences.edit().putBoolean(Constants.IS_MAGIC_LOGIN, false).apply()
        preferences.edit().remove(Constants.MAGIC_TEMP_TOKEN).apply()

        preferences.edit().putBoolean(Constants.SHOWN_HOME_TUT_FOR_SESSION, false).apply()
        preferences.edit().putBoolean(Constants.SHOWN_MEDICAL_TUT_FOR_SESSION, false).apply()
        preferences.edit().putBoolean(Constants.SHOWN_MORE_TUT_FOR_SESSION, false).apply()
        preferences.edit().putBoolean(Constants.SHOWN_ACC_DET_TUT_FOR_SESSION, false).apply()
        preferences.edit().putBoolean(Constants.SHOWN_FIND_CARE_TUT_FOR_SESSION, false).apply()
        preferences.edit().putBoolean(Constants.SHOWN_CALENDAR_TUT_FOR_SESSION, false).apply()
        preferences.edit().putBoolean(Constants.SHOWN_BENEFITS_TUT_FOR_SESSION, false).apply()
        preferences.edit().putBoolean(Constants.SHOWN_CLAIMS_TUT_FOR_SESSION, false).apply()
        preferences.edit().putBoolean(Constants.SHOWN_DIAGNOSIS_TUT_FOR_SESSION, false).apply()
        preferences.edit().putBoolean(Constants.SHOWN_PRESCRIPTION_TUT_FOR_SESSION, false).apply()

        AppUpdateHelper.isAppUpdateAvailable(context = context) { appUpdate ->
            update = appUpdate
            _isAppUpdateAvailable.postValue(appUpdate)
        }
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }
}
