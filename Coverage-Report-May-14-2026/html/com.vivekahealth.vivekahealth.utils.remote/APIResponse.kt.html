package com.vivekahealth.vivekahealth.utils.remote

import androidx.annotation.Nullable
import com.bumptech.glide.load.HttpException
import com.vivekahealth.vivekahealth.data.remote.guest.ResponseOtpReceiveParams
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import java.io.IOException
import java.net.SocketTimeoutException

class APIResponse<T>() {

    lateinit var status: Status

    var data: T? = null

    var error: String? = null
        get() = if (field?.contains("Unable to resolve host") == true) {
            "Unable to proceed. Please check your internet connection and try again later."
        } else if (field?.contains("exceptions occurred.") == true) {
            "An unexpected error occurred."
        } else {
            field
        }
    constructor(status: Status, @Nullable data: T?, @Nullable error: String?) : this() {
        this.status = status
        this.data = data
        this.error = error
    }

    fun loading(): APIResponse<T> {
        return APIResponse<T>(Status.LOADING, null, null)
    }

    fun success(data: T): APIResponse<T> {
        return APIResponse<T>(Status.SUCCESS, data, null)
    }

    fun error(error: String): APIResponse<T> {
        return APIResponse<T>(Status.ERROR, null, error)
    }

    fun error(exception: Throwable): APIResponse<T> {
        return error(
            when (exception) {
                is SocketTimeoutException -> "Request timed out. Please try again."
                is IOException -> "No internet connection."
                is HttpException -> "HTTP ${exception.statusCode}: ${exception.message}"
                is RetrofitException -> {
                    when (exception.exception) {
                        is SocketTimeoutException -> "Request timed out. Please try again."
                        is IOException -> "No internet connection."
                        is HttpException -> "HTTP ${exception.code}: ${exception.message}"
                        else -> exception.localizedMessage ?: "An unexpected error occurred."
                    }
                }
                else -> exception.localizedMessage ?: "An unexpected error occurred."
            }
        )
    }
}

class APIResponseV2<T>() {

    lateinit var status: StatusV2

    var data: T? = null

    var error: String? = null

    constructor(status: StatusV2, @Nullable data: T?, @Nullable error: String?) : this() {
        this.status = status
        this.data = data
        this.error = error
    }

    fun clear(): APIResponseV2<T> {
        return APIResponseV2<T>(StatusV2.CLEAR, null, null)
    }

    fun loading(): APIResponseV2<T> {
        return APIResponseV2<T>(StatusV2.LOADING, null, null)
    }

    fun success(data: T): APIResponseV2<T> {
        return APIResponseV2<T>(StatusV2.SUCCESS, data, null)
    }

    fun error(error: String): APIResponseV2<T> {
        return APIResponseV2<T>(StatusV2.ERROR, null, error)
    }

    fun errorData(data: T): APIResponseV2<T> {
        return APIResponseV2<T>(StatusV2.ERROR_DATA, data, null)
    }
}

sealed class APIResponseState {
    data class SuccessOtp(val data: ResponseOtpReceiveParams) : APIResponseState()
    data class SuccessGeneric(val data: DataResponse) : APIResponseState()
    data class Error(val message: String) : APIResponseState()
    object Loading : APIResponseState()
}
