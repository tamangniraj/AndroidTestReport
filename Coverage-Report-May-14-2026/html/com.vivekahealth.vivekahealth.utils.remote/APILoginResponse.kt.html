package com.vivekahealth.vivekahealth.utils.remote

import androidx.annotation.Nullable

class APILoginResponse<T>() {

    lateinit var status: Status

    var data: T? = null

    var error: String? = null

    constructor(status: Status, @Nullable data: T?, @Nullable error: String?) : this() {
        this.status = status
        this.data = data
        this.error = error
    }

    fun loading(): APILoginResponse<T> {
        return APILoginResponse<T>(Status.LOADING, null, null)
    }

    fun success(data: T): APILoginResponse<T> {
        return APILoginResponse<T>(Status.SUCCESS, data, null)
    }

    fun error(error: String, data: T): APILoginResponse<T> {
        return APILoginResponse<T>(Status.ERROR, data, error)
    }
}
