package com.vivekahealth.vivekahealth.utils.remote

import com.google.gson.Gson
import com.google.gson.TypeAdapter
import com.vivekahealth.vivekahealth.data.remote.ServerError
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Constants.APOLOGY_CHECK_CONNECTION
import okhttp3.internal.readBomAsCharset
import java.io.IOException
import retrofit2.HttpException
import retrofit2.Response
import javax.net.ssl.SSLHandshakeException
import kotlin.text.Charsets.UTF_8

class RetrofitException
private constructor(
    message: String?,
    val code: String?,
    /**
     * The request URL which produced the error.
     */
    val url: String?,
    /**
     * Response object containing status code, headers, body, etc.
     */
    val response: Response<*>?,
    /**
     * The event kind which triggered this error.
     */
    val kind: Kind,
    val exception: Throwable
) : RuntimeException(message, exception) {

    override fun toString(): String {
        return super.toString() + " : " + kind + " : " + url + " : " + response?.peekErrorBodyString()
    }

    /**
     * Identifies the event kind which triggered a [RetrofitException].
     */
    enum class Kind {
        /**
         * An [IOException] occurred while communicating to the server.
         */
        NETWORK,

        /**
         * A non-200 HTTP status code was received from the server.
         */
        HTTP,

        /**
         * An internal error occurred while attempting to execute a request. It is best practice to
         * re-throw this exception so your application crashes.
         */
        UNEXPECTED
    }

    companion object {
        private fun httpError(
            url: String,
            response: Response<*>?,
            httpException: HttpException
        ): RetrofitException {
            var message = Constants.APOLOGY_MESSAGE
            val statusCode = response?.code()

            if (response?.errorBody() != null && response.errorBody()?.contentLength() != 0L) {
                try {
                    val contentType = response.errorBody()?.contentType()?.subtype
                    if (contentType == "json" || contentType == "plain") {
                        val gson = Gson()
                        val adapter: TypeAdapter<ServerError> = gson.getAdapter(
                            ServerError::class.java
                        )
                        val errorParser: ServerError = adapter.fromJson(
                            response.errorBody()!!.charStream()
                        )

                        message = errorParser.errorDescription
                            ?: errorParser.errorDetails
                            ?: message
                    } else if (statusCode == 403) {
                        message = "Access denied. You do not have permission to access this resource."
                    } else {
                        message = "Unexpected error format received from server."
                    }
                } catch (e: Exception) {
                    // Fallback in case JSON parsing fails
                    message = if (statusCode == 403) {
                        "Access denied. You do not have permission to access this resource."
                    } else {
                        "An error occurred but couldn't parse error details."
                    }
                }
            }

            return RetrofitException(
                message,
                statusCode.toString(),
                url,
                response,
                Kind.HTTP,
                httpException
            )
        }

        private fun networkError(exception: IOException): RetrofitException {
            val message = if (exception.message?.contains("Unable to resolve host") == true) {
                APOLOGY_CHECK_CONNECTION
            } else {
                exception.message
            }

            return RetrofitException(message, null, null, null, Kind.NETWORK, exception)
        }

        private fun unexpectedError(exception: Throwable): RetrofitException {
            return RetrofitException(
                exception.message,
                null,
                null,
                null,
                Kind.UNEXPECTED,
                exception
            )
        }

        fun asRetrofitException(throwable: Throwable): RetrofitException {
            if (throwable is RetrofitException) {
                return throwable
            }

            if (throwable is SSLHandshakeException) {
                return RetrofitException(
                    Constants.APOLOGY_CHECK_CONNECTION,
                    null,
                    null,
                    null,
                    Kind.NETWORK,
                    throwable
                )
            }

            // We had non-200 http error
            if (throwable is HttpException) {
                val response = throwable.response()
                return httpError(response?.raw()?.request?.url.toString(), response, throwable)
            }
            // A network error happened
            return if (throwable is IOException) {
                networkError(throwable)
            } else {
                unexpectedError(throwable)
            }
            // We don't know what happened. We need to simply convert to an unknown error
        }
    }
}

fun Response<*>.peekErrorBodyString(): String? {
    val peekSource = errorBody()?.source()?.peek()
    return peekSource?.let {
        val defaultCharset = errorBody()?.contentType()?.charset(UTF_8) ?: UTF_8
        val charset = it.readBomAsCharset(defaultCharset)
        // Use the line below in older versions of okhttp
        // val charset = Util.bomAwareCharset(it, defaultCharset)
        it.readString(charset)
    }
}
