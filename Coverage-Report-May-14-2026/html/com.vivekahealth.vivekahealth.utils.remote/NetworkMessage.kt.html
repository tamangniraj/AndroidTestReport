package com.vivekahealth.vivekahealth.utils.remote

import com.google.gson.reflect.TypeToken
import com.vivekahealth.vivekahealth.data.remote.ServerError
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.HelperUtils
import okhttp3.ResponseBody
import org.json.JSONObject

class NetworkMessage {
    companion object {
        fun getErrorMessage(errorBody: ResponseBody?): String {
            val value: MutableList<String> =
                mutableListOf()
            val builder = StringBuilder()

            val errorMessage = errorBody?.string() ?: ""
            if (errorMessage.isNotEmpty()) {
                val typeToken = object : TypeToken<ServerError>() {}
                HelperUtils.gson.fromJson<ServerError>(errorMessage, typeToken.type)?.let {
                    return it.errorDetails ?: it.errorDescription ?: Constants.APOLOGY_MESSAGE
                }

                val jsonObject = JSONObject(errorMessage)
                jsonObject.keys().forEach {
                    value.add(jsonObject.getString(it))
                }

                value.forEach {
                    builder.append(it + "\n")
                }
            }

            return builder.toString()
        }
    }
}
