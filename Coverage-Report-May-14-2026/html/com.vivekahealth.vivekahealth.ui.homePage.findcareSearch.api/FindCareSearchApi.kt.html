package com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.api

import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityModel.BillingCodePriceResponse
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityModel.BillingData
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityModel.Specialties
import io.reactivex.Single
import retrofit2.http.*

interface FindCareSearchApi {

    // finally found this one
    /**
     * displayPrice is added,
     * since this function is already used in various places, for now default is true, so that it will be compatible
     * for new places, the display price will be used as false
     */
    @GET("provider/v2/searchSpecialtyAndProcedures")
    fun getFindCareSearchList(
        @Query("searchText") searchText: String,
        @Query("memberId") memberId: String,
        @Query("latitude") latitude: String,
        @Query("longitude") longitude: String,
        @Query("providerType")providerType: String,
        @Query("radius") lastDistance: String,
        @Query("searchScope") searchScope: String?,
        @Query("displayPrice") displayPrice: Boolean ? = true
    ): Single<Specialties>

    // new pricing request
    @POST("provider/v1/billingCodePricingDetails")
    fun getPricingDetailsViaBillingCode(
        @Body billingData: BillingData,
        @Query("latitude") latitude: String,
        @Query("longitude") longitude: String,
        @Query("radius") lastDistance: String
    ): Single<BillingCodePriceResponse>
}
