package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.repository

import com.vivekahealth.vivekahealth.data.remote.home.*
import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.*
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.service.MedicalLogsService
import io.reactivex.Single
import java.util.*
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Repository that handles User objects.
 */
@Singleton
class MedicalLogsRepository @Inject constructor(
    private val medicalLogsService: MedicalLogsService
) {

    fun getClaimsMedicalLogs(
        memberID: String,
        type: String,
        sharedMemberIds: String,
        sort: String,
        sortBy: String,
        claimsType: String,
        claimsStatus: String,
        pageSize: Int = 30,
        pageNo: Int = 1
    ): Single<MedicalLogsAll> = medicalLogsService.getMedicalLogsApi().getClaimsLogsList(
        memberID,
        type,
        sharedMemberIds,
        sort,
        sortBy,
        claimsType,
        claimsStatus,
        pageSize,
        pageNo
    )

    fun getClaimsDetails(memberID: String, claimNumber: String): Single<ClaimsDetail> = medicalLogsService.getMedicalLogsApi().getClaimsDetails(
        memberID,
        claimNumber
    )
    fun getDiagnosisDetails(memberID: String, icdCode: String): Single<DiagnosisDetailList> = medicalLogsService.getMedicalLogsApi().getDiagnosisDetails(
        memberID,
        icdCode
    )

    fun getDiagnosisMedicalLogs(
        memberID: String,
        type: String,
        sharedMemberIds: String,
        sort: String,
        sortBy: String,
        range: String,
        pageSize: Int = 30,
        pageNo: Int = 1
    ): Single<MedicalLogsAll> = medicalLogsService.getMedicalLogsApi().getDiagnosisLogsList(
        memberID,
        type,
        sharedMemberIds,
        sort,
        sortBy,
        range,
        pageSize,
        pageNo
    )

    fun getImmunizationMedicalLogs(
        memberID: String,
        type: String,
        sharedMemberIds: String,
        sort: String,
        sortBy: String,
        typeFilter: String,
        pageSize: Int = 30,
        pageNo: Int = 1
    ): Single<MedicalLogsAll> = medicalLogsService.getMedicalLogsApi().getImmunizationLogsList(
        memberID,
        type,
        sharedMemberIds,
        sort,
        sortBy,
        typeFilter,
        pageSize,
        pageNo
    )

    fun getPrescriptionMedicalLogs(
        memberID: String,
        type: String,
        sharedMemberIds: String,
        sort: String,
        sortBy: String,
        drugName: String,
        typeFilter: String,
        pageSize: Int = 30,
        pageNo: Int = 1
    ): Single<MedicalLogsAll> = medicalLogsService.getMedicalLogsApi().getPresLogsList(
        memberID,
        type,
        sharedMemberIds,
        sort,
        sortBy,
        drugName,
        typeFilter,
        pageSize,
        pageNo
    )

    fun getMemberList(memberID: String, sharingTypes: String): Single<SharedMember> = medicalLogsService.getMedicalLogsApi().getDependentList(
        memberID,
        sharingTypes
    )

    fun getMedicalLogs(
        memberId: String,
        medicalLogType: String = "all",
        sharedMemberIds: String,
        sortAction: String = "desc",
        sortBy: String? = null,
        pageSize: Int = 30,
        pageNo: Int = 1
    ): Single<MedicalLogsAll> {
        return medicalLogsService.getMedicalLogsApi().getAllLogsList(
            memberId = memberId,
            medicalLogType = medicalLogType,
            sharedMemberIds = sharedMemberIds,
            sortAction = sortAction,
            sortBy = sortBy,
            pageSize = pageSize,
            pageNo = pageNo
        )
    }
}
