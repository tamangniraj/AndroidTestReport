    fun addItemToList(
        newItems: List<SharingDetails>,
        replace: Boolean,
        originalList: ArrayList<MemberInfoShare>
    ) {
        enableShare = true

        for (newItem in newItems) {
            val existingIndex = shareList.indexOfFirst {
                it.selfId == newItem.selfId && it.shareType == newItem.shareType
            }

            if (existingIndex != -1) {
                val existingItem = shareList[existingIndex]

                existingItem.shareRemove = newItem.shareRemove ?: existingItem.shareRemove
                existingItem.shareStartDate = newItem.shareStartDate ?: existingItem.shareStartDate
                existingItem.shareEndDate = newItem.shareEndDate ?: existingItem.shareEndDate

                if (replace) {
                    existingItem.excludedShareTypeDetail?.excludedCodes = newItem.excludedShareTypeDetail?.excludedCodes.orEmpty().distinct()
                } else {
                    // Merge excludedShareTypeDetail
                    existingItem.excludedShareTypeDetail?.excludedCodes = (
                        existingItem.excludedShareTypeDetail?.excludedCodes.orEmpty() +
                            newItem.excludedShareTypeDetail?.excludedCodes.orEmpty()
                        ).distinct()
                }
            } else {
                // Add new item if no matching item is found
                shareList.add(newItem)
            }
        }
    }

    fun addMissingSharingData(shareList: List<SharingDetails>, memberId: String): List<SharingDetails> {
        val updatedList = shareList.toMutableList()

        // Mutable set to track existing (selfId, shareType) pairs dynamically
        val existingPairs = shareList.mapNotNull { it.selfId?.let { id -> id to it.shareType } }.toMutableSet()

        // Group by selfId
        val groupedBySelfId = shareList.groupBy { it.selfId }

        groupedBySelfId.forEach { (selfId, shares) ->
            if (selfId == null || selfId == memberId) return@forEach

            val shareTypes = shares.mapNotNull { it.shareType }

            // Define which types to add based on conditions
            val typesToAdd = buildList {
                // Always add type 3 and 6 if missing
                addAll(listOf("3", "6"))

                // Only add type 4 if type 1 is NOT present
                if ("1" !in shareTypes) {
                    add("4")
                }
            }

            // 1. Add missing 3, 4, 6 if not present
            typesToAdd.forEach { type ->
                val key = selfId to type
                if (type !in shareTypes && key !in existingPairs) {
                    updatedList.add(
                        SharingDetails(
                            selfId = selfId,
                            shareType = type,
                            shareRemove = "True"
                        )
                    )
                    existingPairs.add(key) // track added entry
                }
            }

            // 2. Add type "1" if both 3 and 4 are missing
            if (!shareTypes.contains("3") && !shareTypes.contains("4")) {
                val key = selfId to "1"
                if (key !in existingPairs) {
                    updatedList.add(
                        SharingDetails(
                            selfId = selfId,
                            shareType = "1",
                            shareRemove = "True"
                        )
                    )
                    existingPairs.add(key) // track added entry
                }
            }
        }

        return updatedList
    }

    fun removeSelfSharing(shareList: List<SharingDetails>): List<SharingDetails> {
        return shareList.filter { it.selfId != memberID }
    }

    fun appendExcludedCode(
        shareList: List<SharingDetails>,
        sharedData: List<MemberInfoShare>?
    ): List<SharingDetails> {
        return shareList.map { shr ->
            if (!shr.shareRemove.equals("false", ignoreCase = true)) {
                return@map shr
            }

            val sourceMember = sharedData?.find { it.selfId == shr.selfId }

            if (sourceMember != null && shr.selfId != memberID) {
                val excludedCodes = when (shr.shareType) {
                    "1", "3", "4" ->
                        sourceMember.diagnosisList
                            ?.filter { it.checked != true }
                            ?.mapNotNull { it.id }
                            ?: emptyList()

                    "6" ->
                        sourceMember.prescriptionList
                            ?.filter { it.checked != true }
                            ?.mapNotNull { it.id }
                            ?: emptyList()

                    else -> emptyList()
                }

                val updatedDetail = shr.excludedShareTypeDetail ?: ExcludedShareTypeDetail()
                updatedDetail.excludedCodes = excludedCodes

                return@map shr.copy(
                    excludedShareTypeDetail = updatedDetail
                )
            }

            return@map shr
        }
    }

    fun clearExcludedCodesForNotShared(shareList: List<SharingDetails>): List<SharingDetails> {
        return shareList.map { shr ->
            if (shr.shareRemove.equals("true", ignoreCase = true)) {
                val clearedDetail = shr.excludedShareTypeDetail?.copy(excludedCodes = null)
                    ?: ExcludedShareTypeDetail(excludedCodes = null)

                return@map shr.copy(excludedShareTypeDetail = clearedDetail)
            }
            return@map shr
        }
    }

    fun addOrUpdateType1BasedOn3And4(shareList: List<SharingDetails>): List<SharingDetails> {
        val groupedBySelfId = shareList.groupBy { it.selfId }
        val updatedList = shareList.toMutableList()

        groupedBySelfId.forEach { (selfId, items) ->
            val threeOrFour = items.filter { it.shareType in listOf("3", "4") }

            val has3or4False = threeOrFour.any { it.shareRemove.equals("false", ignoreCase = true) }
            val hasBoth3and4True = listOf("3", "4").all { type ->
                threeOrFour.any {
                    it.shareType == type && it.shareRemove.equals(
                        "true",
                        ignoreCase = true
                    )
                }
            }

            val newShareRemoveValue = when {
                has3or4False -> "False"
                hasBoth3and4True -> "True"
                else -> null
            }

            if (newShareRemoveValue != null) {
                val existingType1Index = updatedList.indexOfFirst {
                    it.selfId == selfId && it.shareType == "1"
                }

                val _3or4 = threeOrFour.find { it.shareRemove.equals("false", ignoreCase = true) }

                val updatedSharingDetail = SharingDetails(
                    selfId = selfId,
                    shareType = "1",
                    shareRemove = newShareRemoveValue,
                    shareStartDate = _3or4?.shareStartDate,
                    shareEndDate = _3or4?.shareEndDate
                )

                if (existingType1Index >= 0) {
                    updatedList[existingType1Index] = updatedSharingDetail
                } else {
                    updatedList.add(updatedSharingDetail)
                }
            }
        }

        return updatedList
    }
