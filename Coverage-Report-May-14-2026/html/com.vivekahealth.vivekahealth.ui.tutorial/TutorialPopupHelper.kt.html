package com.vivekahealth.vivekahealth.ui.tutorial

import android.content.Context
import android.graphics.Typeface
import android.util.TypedValue
import android.view.Gravity
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.view.animation.AccelerateInterpolator
import android.widget.PopupWindow
import androidx.constraintlayout.widget.ConstraintSet
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.databinding.DescriptionCardviewBinding
import com.vivekahealth.vivekahealth.ui.tutorial.model.Tutorial

object TutorialPopupHelper {
    private var animationDuration = 250
    private var popupMenu: PopupWindow? = null
    fun showPopup(
        context: Context,
        tutorial: Tutorial?,
        onSkip: () -> Unit,
        onNext: (() -> Unit),
        onDismiss: (dontShowAgain: Boolean) -> Unit
    ) {
        if (popupMenu?.isShowing == true) {
            dismiss()
            popupMenu = null
        }
        val inflater =
            context.getSystemService(Context.LAYOUT_INFLATER_SERVICE) as LayoutInflater
        val view = DescriptionCardviewBinding.inflate(inflater)
        tutorial?.let { tutorial ->
            view.tvSkip.setOnClickListener {
                dismiss()
                onDismiss(false)
            }
            if (tutorial.isLast) {
                view.rlLast.visibility = View.VISIBLE
                view.rlNotLast.visibility = View.GONE
            } else {
                view.rlLast.visibility = View.GONE
                view.rlNotLast.visibility = View.VISIBLE
            }
            view.tvTitle.text = tutorial.title
            view.tvDescription.text = tutorial.message
            view.btnNext.setOnClickListener {
                onNext()
            }
            view.btnGotIt.setOnClickListener {
                onDismiss(view.cbDontShow.isChecked)
                dismiss()
            }
            view.tvSkip.setOnClickListener {
                onSkip()
            }
            view.root.measure(
                View.MeasureSpec.makeMeasureSpec(
                    context.resources.displayMetrics.widthPixels - 128,
                    View.MeasureSpec.UNSPECIFIED
                ),
                View.MeasureSpec.makeMeasureSpec(0, View.MeasureSpec.UNSPECIFIED)
            )
            view.scrollDesc.measure(
                View.MeasureSpec.makeMeasureSpec(0, View.MeasureSpec.UNSPECIFIED),
                View.MeasureSpec.makeMeasureSpec(0, View.MeasureSpec.UNSPECIFIED)
            )

            val location = IntArray(2)
            tutorial.actualAnchorView.getLocationInWindow(location)

            popupMenu = PopupWindow(
                view.root,
                context.resources.displayMetrics.widthPixels - 128,
                ViewGroup.LayoutParams.WRAP_CONTENT,
                false
            ).also { popupWindow ->
                popupWindow.isClippingEnabled = false
                popupWindow.update()
                popupWindow.contentView.measure(
                    View.MeasureSpec.makeMeasureSpec(
                        context.resources.displayMetrics.widthPixels - 128,
                        View.MeasureSpec.UNSPECIFIED
                    ),
                    View.MeasureSpec.makeMeasureSpec(0, View.MeasureSpec.UNSPECIFIED)
                )
            }

            val isActuallyDownward: Boolean =
                calculatePopUpDirection(context, tutorial, location, view)

            popupMenu?.showAtLocation(
                tutorial.actualAnchorView,
                if (isActuallyDownward) {
                    Gravity.NO_GRAVITY
                } else {
                    Gravity.BOTTOM or Gravity.START
                },
                64,
                if (isActuallyDownward) {
                    location[1] + tutorial.actualAnchorView.height + 20
                } else {
                    calculateHeightAndSetOffset(context, location)
                }
            )
            popupMenu!!.contentView.apply {
                alpha = 0f
                animate()
                    .alpha(1f)
                    .setDuration(animationDuration.toLong())
                    .setInterpolator(AccelerateInterpolator())
            }
        } ?: run {
            view.tvTitle.visibility = View.GONE
            view.tvDescription.text = context.resources.getString(R.string.don_t_show_tour)
            view.rlLast.visibility = View.VISIBLE
            view.rlNotLast.visibility = View.GONE
            view.btnGotIt.text = context.resources.getString(R.string.confirm)
            view.btnGotIt.setOnClickListener {
                onDismiss(view.cbDontShow.isChecked)
                dismiss()
            }
            view.tvDescription.setTypeface(null, Typeface.BOLD)
            view.tvDescription.setTextSize(TypedValue.COMPLEX_UNIT_DIP, 16f)
            popupMenu = PopupWindow(
                view.root,
                context.resources.displayMetrics.widthPixels - 128,
                ViewGroup.LayoutParams.WRAP_CONTENT,
                false
            )
            popupMenu?.showAtLocation(
                view.root,
                Gravity.NO_GRAVITY,
                64,
                context.resources.displayMetrics.heightPixels / 2
            )
        }
    }

    fun dismiss() {
        popupMenu?.dismiss()
        popupMenu = null
    }

    private fun calculateHeightAndSetOffset(
        context: Context,
        location: IntArray
    ): Int {
        val statusBarHeight =
            context.resources.getDimensionPixelSize(
                context.resources.getIdentifier(
                    "status_bar_height",
                    "dimen",
                    "android"
                )
            )
        var bottomNavBarHeight =
            context.resources.getDimensionPixelSize(
                context.resources.getIdentifier(
                    "navigation_bar_height",
                    "dimen",
                    "android"
                )
            )

        val screenHeight = context.resources.displayMetrics.heightPixels
        return screenHeight - (location[1] - statusBarHeight - bottomNavBarHeight - 20)
    }

    private fun calculatePopUpDirection(
        context: Context,
        tutorial: Tutorial,
        location: IntArray,
        view: DescriptionCardviewBinding
    ): Boolean {
        val screenHeight = context.resources.displayMetrics.heightPixels
        val statusBarHeight =
            context.resources.getDimensionPixelSize(
                context.resources.getIdentifier(
                    "status_bar_height",
                    "dimen",
                    "android"
                )
            )
        var bottomNavBarHeight =
            context.resources.getDimensionPixelSize(
                context.resources.getIdentifier(
                    "navigation_bar_height",
                    "dimen",
                    "android"
                )
            )

        var isActuallyDownward: Boolean = tutorial.isDownward

        if (location[1] + tutorial.actualAnchorView.height + 20 + view.root.measuredHeight <= screenHeight - statusBarHeight - bottomNavBarHeight) {
            // popup is ok to be displayed downward
            isActuallyDownward = true
        } else {
            // popup is not ok to be displayed downward
            // check if upwards has enough space
            if (location[1] - view.root.measuredHeight - 20 - statusBarHeight - bottomNavBarHeight >= 0) {
                // enough space to be displayed upward
                isActuallyDownward = false
            } else {
                // not enough space to be displayed upwards as well
                // calculate which direction has more space
                // set direction of tutorial to one that has more space
                if (location[1] - 20 > screenHeight - (location[1] + tutorial.actualAnchorView.height + 20)) {
                    // set calculated direction to upward
                    isActuallyDownward = false
                } else {
                    // set calculated direction to downward
                    isActuallyDownward = true
                }
                // make message text view scrollable with max height 100
                // FUTURE IMPROVEMENT: calculate the max height
                val cs = ConstraintSet()
                cs.clone(view.csLayout)
                cs.constrainMaxHeight(view.scrollDesc.id, 100)
                cs.constrainedHeight(view.scrollDesc.id, true)
                cs.applyTo(view.csLayout)
            }
        }
        return isActuallyDownward
    }
}
