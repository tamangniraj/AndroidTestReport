package com.vivekahealth.vivekahealth.ui.tutorial

import android.graphics.Bitmap
import android.graphics.Canvas
import android.graphics.Color
import android.graphics.Paint
import android.graphics.Path
import android.graphics.PorterDuff
import android.graphics.PorterDuffXfermode
import android.graphics.RectF

class BitmapRectHoleDrawer(
    backgroundColor: Int,
    holeWidth: Int,
    holeHeight: Int,
    var mWidth: Int = 0,
    var mHeight: Int = 0,
    var mX: Int = 0,
    var mY: Int = 0,
    var cornerRadius: Float = 40f
) : RectHoleDrawer(
    backgroundColor,
    holeWidth,
    holeHeight
) {

    private var mBitmap: Bitmap? = null

    private fun prepareBitmap(width: Int, height: Int) {
        if (mWidth != width || mHeight != height) {
            mWidth = width
            mHeight = height

            mBitmap = Bitmap.createBitmap(mWidth, mHeight, Bitmap.Config.ARGB_8888)
            mBitmap?.eraseColor(Color.TRANSPARENT)

            val c = Canvas(mBitmap!!)
            c.drawColor(mBackgroundColor)

            val p = Paint(Paint.ANTI_ALIAS_FLAG)
            p.style = Paint.Style.FILL
            p.color = -0x1
            p.xfermode = PorterDuffXfermode(PorterDuff.Mode.CLEAR)
            val corners = floatArrayOf(
                cornerRadius,
                cornerRadius, // Top left radius in px
                cornerRadius,
                cornerRadius, // Top right radius in px
                cornerRadius,
                cornerRadius, // Bottom right radius in px
                cornerRadius,
                cornerRadius // Bottom left radius in px
            )
            val path = Path()
            path.addRoundRect(
                RectF(
                    mX.toFloat(),
                    mY.toFloat(),
                    mX + rectWidth.toFloat(),
                    mY - rectHeight.toFloat()
                ),
                corners,
                Path.Direction.CW
            )
            c.drawPath(path, p)
        }
    }

    override fun draw(c: Canvas, width: Int, height: Int) {
        prepareBitmap(width, height)
        mBitmap?.let {
            c.drawBitmap(it, 0f, 0f, null)
        }
    }
}
