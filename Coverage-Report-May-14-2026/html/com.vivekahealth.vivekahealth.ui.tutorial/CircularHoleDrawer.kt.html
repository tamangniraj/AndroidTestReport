package com.vivekahealth.vivekahealth.ui.tutorial

import android.graphics.Canvas

abstract class CircularHoleDrawer(
    protected var mBackgroundColor: Int,
    protected var mHoleRadius: Int
) : HoleDrawer

abstract class RectHoleDrawer(
    protected var mBackgroundColor: Int,
    protected var rectWidth: Int,
    protected var rectHeight: Int
) : HoleDrawer

interface HoleDrawer {
    fun draw(c: Canvas, width: Int, height: Int)
}
