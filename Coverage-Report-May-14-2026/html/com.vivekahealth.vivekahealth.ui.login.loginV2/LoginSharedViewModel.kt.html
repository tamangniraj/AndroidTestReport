package com.vivekahealth.vivekahealth.ui.login.loginV2

import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.utils.NotificationDataParser

class LoginSharedViewModel : ViewModel() {

    private val _notificationData = MutableLiveData<NotificationDataParser?>()
    val notificationData: LiveData<NotificationDataParser?> get() = _notificationData

    fun updateNotificationIntentData(data: NotificationDataParser) {
        _notificationData.value = data
    }
}
