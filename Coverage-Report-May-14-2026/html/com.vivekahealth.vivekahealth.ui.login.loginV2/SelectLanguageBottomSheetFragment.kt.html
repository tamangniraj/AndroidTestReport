        loadLanguageList()
        val layoutManager = LinearLayoutManager(context)
        binding.selectLangRecyclerView.layoutManager = layoutManager
        val adapter = SelectLanguageAdapter(requireActivity(), languageList, onLangClick = {
            Timber.d("LanguageUpdate: $it")
            Constants.LOCAL_LANG_MODIFIED = it
            listener?.onLanguageChange(it)
            dismiss()
        })
        binding.selectLangRecyclerView.adapter = adapter
        binding.selectLangRecyclerView.addItemDecoration(
            DividerItemDecoration(requireContext(), layoutManager.orientation)
        )
        binding.cancelButton.setOnClickListener {
            dismiss()
        }
        binding.langSearchView.setOnQueryTextListener(object : SearchView.OnQueryTextListener {
            override fun onQueryTextChange(newText: String): Boolean {
                adapter.filter.filter(newText.trim())
                return true
            }

            override fun onQueryTextSubmit(query: String): Boolean {
                adapter.filter.filter(query.trim())
                return true
            }
        })
    }

    private fun getLanguageItemFor(
        @StringRes stringRes: Int,
        languageCode: String
    ): SelectLanguageItem {
        return SelectLanguageItem(
            requireContext().getStringByLocale(stringRes, Locale.forLanguageTag(languageCode)),
            requireContext().getStringByLocale(stringRes, Locale.forLanguageTag("en")),
            languageCode
        )
    }

fun Context.getStringByLocale(
    @StringRes stringRes: Int,
    locale: Locale,
    vararg formatArgs: Any
): String {
    val configuration = Configuration(resources.configuration)
    configuration.setLocale(locale)
    return createConfigurationContext(configuration).resources.getString(stringRes, *formatArgs)
}
