    fun checkUsername(username: String) {
        clearPreviousEnrollment()
        viewModelScope.launch {
            loginRepository.getGuestToken()
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _isLoading.value = Event(true)
                }
                .flatMap { response ->
                    response.body()?.let {
                        saveTokens(it)
                        val apisHeader: String = "${it.token_type} ${it.access_token}"
                        return@flatMap emailRepository.verifyEmailV2(username.trim(), apisHeader)
                            .subscribeOn(
                                Schedulers.io()
                            )
                            .observeOn(AndroidSchedulers.mainThread())
                    }

                    response.errorBody()?.let {
                        try {
                            val errorResponse: DataResponse = gson.fromJson(
                                it.string(),
                                DataResponse::class.java
                            )

                            return@flatMap Single.error(Throwable(errorResponse.errorDescription))
                        } catch (
                            ex: Exception
                        ) {
                            return@flatMap Single.error(Throwable(APOLOGY_MESSAGE))
                        }
                    }
                }.subscribe({
                    Timber.d("ErrorCode: ${it.code()} ${it.raw().code}")

                    if (it.isSuccessful) {
                        it.body()?.let {
                            if (it.code == "200") {
                                saveEmail()
                                saveRegistrationId(it.otherDetails?.sessionId)

                                _navigateToNextPage.value = Event(
                                    APIResponseV2<DataResponse>().success(it)
                                )

                                _shouldGoToLogin.value = Event(
                                    it.reason?.compareTo(
                                        "REGISTERED_AND_VERIFIED",
                                        true
                                    ) == 0
                                )
                            } else {
                                try {
                                    val errorResponse = it

                                    if (errorResponse.errorCode?.compareTo(
                                            "ALTERNATE_APP_AVAILABLE",
                                            true
                                        ) == 0
                                    ) {
                                        _alternativeApp.value = Event(
                                            ErrorResponse(
                                                android = errorResponse.otherDetails?.android.toString(),
                                                alternativeAppAvailable = errorResponse.otherDetails?.alternativeAppAvailable,
                                                error_description = errorResponse.errorDescription,
                                                appName = errorResponse.otherDetails?.appName ?: ""
                                            )
                                        )
                                    } else {
                                        _shouldGoToLogin.value = Event(
                                            errorResponse.errorCode?.compareTo(
                                                "REGISTERED_AND_VERIFIED",
                                                true
                                            ) == 0
                                        )

                                        _navigateToNextPage.value = Event(
                                            APIResponseV2<DataResponse>().errorData(
                                                errorResponse
                                            )
                                        )
                                    }
                                } catch (e: Exception) {
                                    _navigateToNextPage.value = Event(
                                        APIResponseV2<DataResponse>().error(APOLOGY_MESSAGE)
                                    )
                                }
                            }
                        }
                    } else {
                        try {
                            val errorResponse: DataResponse = gson.fromJson(
                                it.errorBody()?.string(),
                                DataResponse::class.java
                            )

                            if (errorResponse.errorType?.compareTo(
                                    "ALTERNATE_APP_AVAILABLE",
                                    true
                                ) == 0
                            ) {
                                _alternativeApp.value = Event(
                                    ErrorResponse(
                                        android = errorResponse.otherDetails?.android.toString(),
                                        alternativeAppAvailable = errorResponse.otherDetails?.alternativeAppAvailable,
                                        error_description = errorResponse.errorDescription,
                                        appName = errorResponse.otherDetails?.appName
                                    )
                                )
                            } else {
                                _shouldGoToLogin.value = Event(
                                    errorResponse.errorType?.compareTo(
                                        "REGISTERED_AND_VERIFIED",
                                        true
                                    ) == 0
                                )

                                _navigateToNextPage.value = Event(
                                    APIResponseV2<DataResponse>().errorData(
                                        errorResponse
                                    )
                                )
                            }
                        } catch (e: Exception) {
                            _navigateToNextPage.value = Event(
                                APIResponseV2<DataResponse>().error(APOLOGY_MESSAGE)
                            )
                        }
                    }
                    if (!isBiometricLogin || shouldGoToLogin.value?.peekContent() == false) {
                        _isLoading.value = Event(false)
                    }
                    email = username
                }, {
                    if ("Unable to resolve host" in it.message.toString() || "Unable to establish " in it.message.toString()) {
                        saveEmail()
                        val decryptedEmail = HelperUtils.decrypt(savedUserNameEncoded)
                        if (email == decryptedEmail) {
                            // Just go to login
                            _shouldGoToLogin.value = Event(true)
                        }
                    }

                    try {
                        _navigateToNextPage.value = Event(
                            APIResponseV2<DataResponse>().error(
                                it.message.toString()
                            )
                        )
                    } catch (exception: Exception) {
                        _navigateToNextPage.value = Event(
                            APIResponseV2<DataResponse>().error(APOLOGY_MESSAGE)
                        )
                    }

                    _isLoading.value = Event(false)
                })
        }
    }

    fun login(username: String, password: String, isBiometricLogin: Boolean = false) {
        fromMagicLogin = false
        compositeDisposable.add(
            loginRepository.getLogin(username, password, "", isBiometricLogin)
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _isLoading.value = Event(true)
                }
                .subscribe({
                    encourageVerification()
                    if (it.isSuccessful) {
                        it.body()?.let {
                            saveTokens(it)
                        }

                        preferences.edit().putBoolean(Constants.IS_MAGIC_LOGIN, false).apply()

                        saveForOffline(username, password)

                        getMemberId(it)
                        // Set email verified to true if logged in successfully
                        val key = "$username${Constants.EMAIL_VERIFIED}"
                        preferences.edit {
                            putBoolean(key, true)
                        }
                    } else {
                        _isLoading.value = Event(false)

                        val gson = Gson()
                        try {
                            val errorResponse: ErrorResponse? = gson.fromJson(
                                it.errorBody()?.charStream(),
                                ErrorResponse::class.java
                            )

                            if (it.raw().code == 403 && errorResponse?.alternativeAppAvailable == "true") {
                                _alternativeApp.value = Event(
                                    ErrorResponse(
                                        android = errorResponse.android.orEmpty(),
                                        alternativeAppAvailable = errorResponse.alternativeAppAvailable,
                                        error_description = errorResponse.error_description,
                                        appName = errorResponse.appName.orEmpty()
                                    )
                                )
                            } else {
                                _logInFail.value = errorResponse?.takeIf {
                                    !it.error_description.isNullOrBlank()
                                } ?: ErrorResponse(
                                    android = "",
                                    alternativeAppAvailable = "false",
                                    error_description = APOLOGY_MESSAGE,
                                    appName = ""
                                )
                            }
                        } catch (e: JsonSyntaxException) {
                            _logInFail.value = ErrorResponse(
                                android = "",
                                alternativeAppAvailable = "false",
                                error_description = APOLOGY_MESSAGE,
                                appName = ""
                            )
                        }
                    }
                }, {
                    _isLoading.value = Event(false)
                    it.message?.let { errMsg ->
                        val errRes = ErrorResponse()
                        errRes.error_description = errMsg
                        _logInFail.value = errRes
                    }
                })
        )
    }

    fun getMemberId(response: Response<AuthToken>) {

    fun getMemberFullId(response: Response<AuthToken>) {
        compositeDisposable.add(
            homeRepository.getMember()
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnDispose {
                    Timber.d("here")
                }.doOnSubscribe {
                    _isLoading.value = Event(true)
                }.doOnError {
                    _isLoading.value = Event(false)
                }.subscribe({
                    it.email?.let { email ->
                        preferences.edit().putString(Constants.USER_EMAIL, email).commit()
                        preferences.edit().putBoolean(
                            email + Constants.BIO_PERMISSION_REQUESTED,
                            biometricEnabled.value == true
                        ).commit()
                    }
                    it.selfId?.let { sid ->

                        fetchedMemberId = sid

                        preferences.edit().putString("memberID", it.selfId)
                            .apply()

                        var languageInServer = it.languagePreference

                        var updatedMemberInfo = it

                        if (languageToPush != null && languageToPush != languageInServer) {
                            updatedMemberInfo = it.copy(languagePreference = languageToPush)
                        }

                        Constants.CURRENT_LANGUAGE = updatedMemberInfo.languagePreference

                        homeRepository.insertMember(updatedMemberInfo)

                        if (languageToPush != null && languageToPush != languageInServer) {
                            postLanguage(it.selfId, languageToPush ?: "", response)
                        } else {
                            _isLoading.value = Event(false)
                            _logInSuccess.value = APILoginResponse<Response<AuthToken>>().success(
                                response
                            )
                        }
                    }
                }, {
                    _isLoading.value = Event(false)
                    _logInFail.value = ErrorResponse(
                        android = "",
                        alternativeAppAvailable = "false",
                        error_description = it.message ?: APOLOGY_MESSAGE,
                        appName = ""
                    )
                })
        )
    }

    fun saveTokens(data: AuthToken) {
        preferences.edit().putBoolean(Constants.IS_GUEST, data.isGuest).apply()
        preferences.edit().putString(Constants.ACCESS_TOKEN, data.access_token).apply()
        preferences.edit().putString(Constants.TOKEN_TYPE, data.token_type).apply()
        preferences.edit().putString(Constants.EXPIRES_IN, data.expires_in).apply()
        preferences.edit().putString(Constants.SCOPE, data.scope).apply()
        preferences.edit().putBoolean(Constants.HAS_SHOWN_SPLASH, false).apply()
        Constants.IS_OFFLINE = false
    }

    fun getFcmToken() {
        FirebaseMessaging.getInstance().token.addOnCompleteListener(
            OnCompleteListener { task ->
                if (!task.isSuccessful) {
                    return@OnCompleteListener
                }
                // Get new FCM registration token
                val token = task.result
                preferences.edit().putString("FCM_TOKEN", token).apply()
            }
        )
    }

    fun magicLogin(magicTempUUID: String) {
        preferences.edit().putString(Constants.MAGIC_TEMP_TOKEN, magicTempUUID).apply()
        compositeDisposable.add(
            loginRepository.getMagicLogin(magicTempUUID)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _isLoading.value = Event(true)
                }
                .subscribe({
                    _isLoading.value = Event(false)
                    if (it.isSuccessful) {
                        it.body()?.let {
                            saveTokens(it)
                        }
                        preferences.edit().putBoolean(Constants.IS_MAGIC_LOGIN, true).apply()
                        fromMagicLogin = true
                        getMemberId(it)
                    } else {
                        val gson = Gson()
                        val type = object : TypeToken<ErrorResponse>() {}.type
                        val errorResponse: ErrorResponse? = gson.fromJson(
                            it.errorBody()!!.charStream(),
                            type
                        )
                        errorResponse?.let { err ->
                            _logInFail.value = err
                        }
                    }
                }, {
                    _isLoading.value = Event(false)
                    it.message?.let { errMsg ->
                        val errRes = ErrorResponse()
                        errRes.error_description = errMsg
                        _logInFail.value = errRes
                    }
                })
        )
    }

    fun setBiometric() {
        if (biometricEnabled.value == true) {
            biometricEnabled.value = false
            preferences.edit()
                .putBoolean(
//                    preferences.getString("memberID", "") + Constants.BIO_PERMISSION_REQUESTED,
                    preferences.getString(Constants.USER_EMAIL, "") + Constants.BIO_PERMISSION_REQUESTED,
                    false
                )
                .apply()
        } else {
            biometricEnabled.value = true
            preferences.edit()
                .putBoolean(
//                    preferences.getString("memberID", "") + Constants.BIO_PERMISSION_REQUESTED,
                    preferences.getString(Constants.USER_EMAIL, "") + Constants.BIO_PERMISSION_REQUESTED,

    fun allowBiometric(): Boolean {
        // check new email or fallback to memberID
        val hasRequested = preferences.getBoolean(
            preferences.getString(Constants.USER_EMAIL, "") + Constants.BIO_PERMISSION_REQUESTED,
            false
        ) || preferences.getBoolean(
            preferences.getString(Constants.MEMBER_ID, "") + Constants.BIO_PERMISSION_REQUESTED,
            false
        )
        return preferences.contains(Constants.encodedPassword) && hasRequested
    }

    fun handleGuestLogin(guestUserInfo: GuestUserInfo) {
        preferences.edit().putString("memberID", guestUserInfo.id)
            .apply()
        preferences.edit().putString(Constants.USER_EMAIL, guestUserInfo.email).commit()

        guestEmail = guestUserInfo.email

        if (guestUserInfo.isOpenEnrollment == true) {
            _showPendingVerification.value = Event(guestUserInfo)
        } else {
            pendingSsnIdentification = guestUserInfo.pendingIdentityVerification
            if (pendingSsnIdentification) {
                sessionId = guestUserInfo.identificationSessionId
                _showPendingVerification.value = Event(guestUserInfo)
            } else {
                if (!guestUserInfo.requestInCustomerSupport) {
                    _showVerifyIdentity.value = Event(Unit)
                } else {
                    _showRequestInCustomerSupport.value = Event(Unit)
                }
            }
        }
    }
