    fun uploadAndIgnore(
        documentType: DocumentType,
        file: Bitmap?,
        token: String,
        isSsn: Boolean?
    ) {
        if (file == null) {
            _uploadState.value = Event(UploadState.Error("No file provided"))
            return
        }

        val filePart: MultipartBody.Part = HelperUtils.createMultipartBodyPartFromBitmap(
            bitmap = file,
            "file",
            "${UUID.randomUUID()}-$documentType-${DateUtils.getCurrentLocalTime()}"
        )

        viewModelScope.launch {
            if (documentType == DocumentType.DL_BACK) {
                compositeDisposable.add(
                    uploadRepository.uploadDoc(token, filePart, documentType)
                        .subscribeOn(Schedulers.io())
                        .subscribe({}, { error ->
                            timber.log.Timber.e(error, "Background upload failed")
                        })
                )
            } else if (documentType == DocumentType.SSN || documentType == DocumentType.SSN_ONLY) {
                compositeDisposable.add(
                    uploadRepository.uploadDoc(token, filePart, documentType)
                        .subscribeOn(Schedulers.io())
                        .subscribe({}, { error ->
                            timber.log.Timber.e(error, "Background SSN upload failed")
                        })
                )
            }
        }
    }

    fun uploadDocumentWithQR(
        documentType: DocumentType,
        file: Bitmap?,
        token: String,
        driverLicense: DriverLicense?,
        json: String?,
        id: String?,
        verifiedInServer: String?,
        isSsn: Boolean ? = null
    ) {
        if (file == null) {
            _uploadState.value = Event(UploadState.Error("No file provided"))
            return
        }
        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_UPLOAD_STARTED) {
            putString("doc_type", documentType.name)
        }

        val filePart: MultipartBody.Part = HelperUtils.createMultipartBodyPartFromBitmap(
            bitmap = file,
            "file",
            "${UUID.randomUUID()}-$documentType-${DateUtils.getCurrentLocalTime()}"
        )

        viewModelScope.launch {
            just(documentType)
                .flatMap({ docType ->
                    if (docType == DocumentType.DL_FRONT) {
                        val uploadFrontObservable = uploadRepository.uploadDoc(
                            token,
                            filePart,
                            documentType
                        ).toObservable()

                        uploadFrontObservable
                            .doOnNext { o -> status.onNext(LoadingState.PROCESS) }
                            ?.flatMap { uploadFrontResult ->

                                logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_PROCESS_FRONT_STARTED) {
                                    putString("doc_type", documentType.name)
                                    putInt("attempt_number", attemptNumber)
                                }

                                uploadRepository.processDocument(token)
                                    .toObservable()
                                    .map { processDocumentResult ->
                                        if (processDocumentResult.code() == 200) {
                                            logGuestVerifyEvent(
                                                FirebaseTags.GUEST_VERIFY_PROCESS_FRONT_SUCCESS
                                            ) {
                                                putString("doc_type", documentType.name)
                                                putInt("attempt_number", attemptNumber)
                                            }
                                            return@map UploadStep(
                                                documentType = documentType,
                                                uploadResponse = uploadFrontResult,
                                                processDocumentResponse = processDocumentResult.body()
                                            )
                                        } else if (processDocumentResult.code() == 400) {
                                            logGuestVerifyEvent(
                                                FirebaseTags.GUEST_VERIFY_PROCESS_FRONT_FAILED
                                            ) {
                                                putString("doc_type", documentType.name)
                                                putInt("attempt_number", attemptNumber)
                                            }
                                            val errorResponse: DataResponse = gson.fromJson(
                                                processDocumentResult.errorBody()?.string(),
                                                DataResponse::class.java
                                            )

                                            return@map UploadStep(
                                                error = errorResponse.errorDescription
                                                    ?: "Something went wrong!"
                                            )
                                        } else {
                                            return@map UploadStep(
                                                error = "Something went wrong"
                                            )
                                        }
                                    }
                            }
                    } else if (documentType == DocumentType.DL_BACK) {
                        if (verifiedInServer != null && verifiedInServer.isNotEmpty()) {
                            uploadRepository.verify(token)
                                .toObservable()
                                .map { verifyResponse ->
                                    if (verifyResponse.isSuccessful) {
                                        UploadStep(
                                            documentType = docType,
                                            verifyResponse = verifyResponse.body()
                                        )
                                    } else {
                                        val errorResponse: GenericErrorResponse = gson.fromJson(
                                            verifyResponse.errorBody()?.string(),
                                            GenericErrorResponse::class.java
                                        )
                                        UploadStep(error = errorResponse.getErrorMessage())
                                    }
                                }
                                .flatMap { uploadStep ->
                                    if (uploadStep.error != null) {
                                        return@flatMap just(uploadStep)
                                    }

                                    uploadRepository.additionalRequirements(token)
                                        .toObservable()
                                        .map { additionalRequirements ->
                                            uploadStep.copy(
                                                additionalRequirementResponse = additionalRequirements.body()
                                            )
                                        }
                                }
                        } else {
                            // upload back media in background we are not concerned with response

                            // upload the back barcode data; we are interested in this
                            // then upon success call verify and then additional details
                            uploadRepository.updateDetails(
                                token,
                                driverLicense.toUploadDetailsSendParams(
                                    json,
                                    attemptNumber
                                )
                            ).toObservable()
                                .doOnNext { o -> status.onNext(LoadingState.PROCESS) }
                                .map {
                                    if (it.isSuccessful) {
                                        return@map UploadStep(
                                            documentType = docType,
                                            qrUploadResult = it.body()
                                        )
                                    } else {
                                        val errorResponse: GenericErrorResponse = gson.fromJson(
                                            it.errorBody()?.string(),
                                            GenericErrorResponse::class.java
                                        )
                                        return@map UploadStep(
                                            error = errorResponse.getErrorMessage()
                                        )
                                    }
                                }.flatMap { uploadStep ->
                                    if (uploadStep.error != null) {
                                        return@flatMap just(uploadStep)
                                    }
                                    uploadRepository.verify(token)
                                        .toObservable().map {
                                            if (it.isSuccessful) {
                                                return@map uploadStep.copy(
                                                    verifyResponse = it.body()
                                                )
                                            } else {
                                                val errorResponse: GenericErrorResponse =
                                                    gson.fromJson(
                                                        it.errorBody()?.string(),
                                                        GenericErrorResponse::class.java
                                                    )
                                                return@map UploadStep(
                                                    error = errorResponse.getErrorMessage()
                                                )
                                            }
                                        }
                                }
                                .flatMap { uploadStep ->

                                    if (uploadStep.error != null) {
                                        return@flatMap just(uploadStep)
                                    }

                                    uploadRepository.additionalRequirements(token)
                                        .toObservable()
                                        .map { additionalRequirements ->
                                            if (additionalRequirements.isSuccessful) {
                                                return@map uploadStep.copy(
                                                    additionalRequirementResponse = additionalRequirements.body()
                                                )
                                            } else {
                                                val errorResponse: GenericErrorResponse =
                                                    gson.fromJson(
                                                        additionalRequirements.errorBody()
                                                            ?.string(),
                                                        GenericErrorResponse::class.java
                                                    )
                                                return@map UploadStep(
                                                    error = errorResponse.getErrorMessage()
                                                )
                                            }
                                        }
                                }
                        }
                    } else if (documentType == DocumentType.SSN || documentType == DocumentType.SSN_ONLY) {
                        uploadRepository.updateDetails(
                            token,
                            uploadSsnDetails(id, isSsn)
                        ).toObservable().map {
                            return@map UploadStep(
                                documentType = docType,
                                qrUploadResult = it.body()
                            )
                        }
                    } else {
                        return@flatMap just(UploadStep(error = ""))
                    }
                }, 5)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    status.onNext(LoadingState.UPLOAD)
                }
                .subscribe({
                    if (it.error != null) {
                        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_UPLOAD_ERROR) {
                            putString("doc_type", documentType.name)
                            putString("error_type", it.error)
                        }
                        _uploadState.value = Event(UploadState.Error(it.error))
                        return@subscribe
                    }

                    logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_UPLOAD_SUCCESS) {
                        putString("doc_type", documentType.name)
                    }

                    when (it.documentType) {
                        DocumentType.DL_FRONT -> {
                            _uploadState.value = Event(
                                UploadState.Success(null, it.uploadResponse)
                            )
                        }

                        DocumentType.DL_BACK -> {
                            attemptNumber = 1
                            _uploadState.value = Event(
                                UploadState.Success(
                                    it.qrUploadResult,
                                    it.uploadResponse,
                                    it.additionalRequirementResponse
                                )
                            )
                        }

                        DocumentType.SSN, DocumentType.SSN_ONLY -> {
                            _uploadState.value = Event(
                                UploadState.Success(it.qrUploadResult, it.uploadResponse)
                            )
                        }

                        else -> {
                        }
                    }

                    uploadAndIgnore(
                        documentType,
                        file,
                        token,
                        isSsn
                    )
                }, {
                    val errorMessage = when {
                        it.message?.contains(
                            "SocketException",
                            ignoreCase = true
                        ) == true -> Constants.APOLOGY_MESSAGE

                        else -> it.message ?: Constants.APOLOGY_MESSAGE
                    }
                    logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_UPLOAD_ERROR) {
                        putString("doc_type", documentType.name)
                        putString("error_type", errorMessage)
                    }
                    _uploadState.value = Event(UploadState.Error(errorMessage))
                })
        }
    }

                    status.onNext(LoadingState.UPLOAD)

                    if (it.isSuccessful) {
                        _uploadState.value = Event(UploadState.SuccessScan())

                        val errorResponse: GenericErrorResponse = gson.fromJson(

                            UploadState.FailedScan(errorResponse.getErrorMessage())
                        )
                    }
                }, {
                    _uploadState.value = Event(
                        UploadState.Error(it.message ?: Constants.APOLOGY_MESSAGE)
                    )
                })
        }

    fun verifyAndAdditionalDetails(token: String) {
        val verifyAndAddDet = uploadRepository.verify(token)
            .flatMap { verifyResponse ->
                if (verifyResponse.isSuccessful) {
                    uploadRepository.additionalRequirements(token)
                } else {
                    val errorResponse: GenericErrorResponse =
                        gson.fromJson(
                            verifyResponse.errorBody()
                                ?.string(),
                            GenericErrorResponse::class.java
                        )
                    Single.error(Exception(errorResponse.getErrorMessage()))
                }
            }
            .subscribeOn(Schedulers.io())
            .observeOn(AndroidSchedulers.mainThread())
            .doOnSubscribe { _uploadState.value = Event(UploadState.Loading(LoadingState.VERIFY)) }
            .subscribe({ additionalResponse ->
                if (additionalResponse.isSuccessful) {
                    _reUploadFrontVerifiedAndAdditionalDetails.value = Event(
                        additionalResponse.body()!!
                    )
                } else {
                    val errorResponse: GenericErrorResponse = gson.fromJson(
                        additionalResponse.errorBody()?.string(),
                        GenericErrorResponse::class.java
                    )
                    _reUploadFrontVerifyAndAdditionalDetailsError.value =
                        Event(errorResponse.message ?: "")
                }
            }, { error ->
                _reUploadFrontVerifyAndAdditionalDetailsError.value = Event(error.message ?: "")
            })
        compositeDisposable.add(verifyAndAddDet)
    }

    fun identify(token: String, reregister: Boolean? = false) {
        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_IDENTIFY_STARTED)
        viewModelScope.launch {
            compositeDisposable.clear()
            compositeDisposable.add(
                uploadRepository.identify(token, reregister)
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _guestSkipSsnUploadState.value = Event(
                            UploadState.Loading(LoadingState.IDENTIFY)
                        )
                    }.subscribe({
                        if (it.isSuccessful) {
                            it.body()?.let { responseBody ->
                                when (responseBody.code) {
                                    200 -> {
                                        logGuestVerifyEvent(
                                            FirebaseTags.GUEST_VERIFY_IDENTIFY_SUCCESS
                                        )
                                        _guestSkipSsnUploadState.value = Event(
                                            UploadState.Success()
                                        )
                                        _showSuccessDialog.value = Event(Unit)
                                    }

                                    else -> handleError(responseBody.error)
                                }
                            }
                        } else {
                            if (it.code() == 403) {
                                _guestSkipSsnUploadState.value = Event(UploadState.Error(""))
                                handleAlternateAppAvailable(it.errorBody())
                            } else {
                                handleErrorResponse(it.errorBody())
                            }
                        }
                    }, { error ->
                        _guestSkipSsnUploadState.value = Event(
                            UploadState.Error(error)
                        )
                    })
            )
        }
    }

    private fun handleError(error: ErrorDetail?) {
        when (error?.error) {
            "SENT_TO_CUSTOMER_SUPPORT" -> {
                logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_IDENTIFY_SUPPORT_NEEDED)
                _guestSkipSsnUploadState.value = Event(
                    UploadState.CustomerSupport
                )
            }

            "NO_MATCH" -> {
                logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_IDENTIFY_NO_MATCH)
                when {
                    error.requireCurrentZip -> _guestSkipSsnUploadState.value = Event(
                        UploadState.NoMatchAddZip
                    )

                    error.requireCurrentAddress -> _guestSkipSsnUploadState.value = Event(
                        UploadState.NoMatchAddAddress
                    )

                    else -> _guestSkipSsnUploadState.value = Event(
                        UploadState.Error(error.errorDescription)
                    )
                }
            }

            "MULTIPLE_MATCH" -> {
                logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_IDENTIFY_MULTIPLE_MATCH)
                when {
                    error.requireCurrentZip -> _guestSkipSsnUploadState.value = Event(
                        UploadState.AddZip
                    )

                    error.requireCurrentAddress -> _guestSkipSsnUploadState.value = Event(
                        UploadState.AddAddress
                    )

                    else -> _guestSkipSsnUploadState.value = Event(
                        UploadState.Error(error.errorDescription)
                    )
                }
            }

            "MEMBER_ALREADY_REGISTERED" -> {
                logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_IDENTIFY_ALREADY_REGISTERED)
                _guestSkipSsnUploadState.value = Event(
                    UploadState.MemberAlreadyRegistered(error.otherDetails?.email ?: "")
                )
            }

            else -> {
                logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_IDENTIFY_ERROR) {
                    putString("error_type", error?.error ?: "unknown")
                }
                _guestSkipSsnUploadState.value = Event(
                    UploadState.Error(error?.errorDescription ?: "")
                )
            }
        }
    }

    private fun handleErrorResponse(errorBody: ResponseBody?) {
        val errorBodyString = try {
            errorBody?.string()
        } catch (e: Exception) {
            null
        }

        if (errorBodyString.isNullOrBlank()) {
            _guestSkipSsnUploadState.value = Event(UploadState.Error("Unknown error occurred"))
            return
        }

        // Try to parse as JsonObject first to check structure
        val jsonObject = try {
            gson.fromJson(errorBodyString, JsonObject::class.java)
        } catch (e: Exception) {
            _guestSkipSsnUploadState.value = Event(UploadState.Error("Invalid server response"))
            return
        }

        val errorMessage = when {
            // Check if it matches IdentifyErrorResponse structure
            hasIdentifyErrorStructure(jsonObject) -> {
                parseIdentifyErrorResponse(errorBodyString)
            }
            // Check if it matches ErrorDetail structure
            hasErrorDetailStructure(jsonObject) -> {
                parseErrorDetail(errorBodyString)
            }
            // Try to extract any error message from generic structure
            else -> {
                extractGenericErrorMessage(jsonObject)
            }
        }

        _guestSkipSsnUploadState.value = Event(UploadState.Error(errorMessage))
    }

    private fun hasIdentifyErrorStructure(jsonObject: JsonObject): Boolean {
        return jsonObject.has("errorCode")
    }

    private fun hasErrorDetailStructure(jsonObject: JsonObject): Boolean {
        return jsonObject.has("error_description")
    }

    private fun parseIdentifyErrorResponse(jsonString: String): String {
        return try {
            val errorResponse = gson.fromJson(jsonString, IdentifyErrorResponse::class.java)
            errorResponse.error?.errorDescription
                ?: errorResponse.message
                ?: "Unknown error occurred"
        } catch (e: Exception) {
            "Error parsing server response"
        }
    }

    private fun parseErrorDetail(jsonString: String): String {
        return try {
            val errorResponse = gson.fromJson(jsonString, ErrorDetail::class.java)
            errorResponse.errorDescription ?: "Unknown error occurred"
        } catch (e: Exception) {
            "Error parsing server response"
        }
    }

    private fun extractGenericErrorMessage(jsonObject: JsonObject): String {
        return when {
            jsonObject.has("message") -> {
                jsonObject.get("message")?.asString ?: "Unknown error"
            }

            jsonObject.has("error") -> {
                val error = jsonObject.get("error")
                when {
                    error.isJsonPrimitive -> error.asString
                    error.isJsonObject -> error.asJsonObject.get("message")?.asString
                        ?: "Unknown error"

                    else -> "Unknown error"
                }
            }

            jsonObject.has("detail") -> {
                jsonObject.get("detail")?.asString ?: "Unknown error"
            }

            else -> "HTTP Status 404 – Not Found"
        }
    }

    private fun handleAlternateAppAvailable(errorBody: ResponseBody?) {
        val errorResponse: DataResponse? = gson.fromJson(
            errorBody?.string(),
            DataResponse::class.java
        )
        errorResponse?.let {
            _showAlternateAppAvailable.value = Event(it)
        }
    }

    /**
     *     SMS(1, "SMS"),
     *     VOICE(2, "Voice"),
     *     EMAIL(3, "Email"),
     *     DEMOGRAPHIC(4, "Demographic"),
     *     SELFIE_MATCH(5, "Selfie Match"), //verify selfie match with document photo
     *     AI_SELFIE(6, "AI Selfie"), //verify the selfie is valid
     *     LICENSE_FRONT(7, "License Front"),
     *     LICENSE_BACK(8, "License Back"),
     *     SSN_INSURANCE_CARD(9, "SSN Card or Insurance card");
     *
     * */
    fun logFailedDocumentScanAttempt(id: String, documentType: DocumentType) {
        val verificationType: Int = when (documentType) {
            DocumentType.DL_FRONT -> 7
            DocumentType.DL_BACK -> 8
            DocumentType.SSN -> 9
            DocumentType.SSN_ONLY -> 9
            DocumentType.INSURANCE_CARD -> 9
            DocumentType.SELFIE -> 5
            DocumentType.SELFIE_MANUAL -> 5
        }
        compositeDisposable.add(
            uploadRepository.logFailedDocumentAttempt(id, verificationType)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .subscribe({}, { error ->
                    timber.log.Timber.e(error, "Failed to log document scan attempt")
                })
        )
    }
}

    return UploadDetailsSendParams(

fun DriverLicense?.toUploadDetailsSendParams(
    json: String?,
    attemptCount: Int?
): UploadDetailsSendParams {
    if (json == null) {
        return UploadDetailsSendParams(
            drivingLicenseDto = Demographics(
                attemptNumber = attemptCount
            )
        )
    }

    return UploadDetailsSendParams(
        drivingLicenseDto = Demographics(
            documentId = "1",
            legalFirstName = this?.firstName.orEmpty(),
            legalLastName = this?.lastName.orEmpty(),
            legalMiddleName = this?.middleName.orEmpty(),
            dob = this?.birthDate.orEmpty().convertLicenseDate(),
            address = this?.addressStreet.orEmpty(),
            gender = this?.gender.orEmpty().toGenderApi(),
            expirationDate = this?.expiryDate.orEmpty().convertLicenseDate(),
            issuedDate = this?.issueDate.orEmpty().convertLicenseDate(),
            licenseNumber = this?.licenseNumber.orEmpty(),
            details = json,
            fullAddress = CurrentFullAddress(
                this?.addressZip,
                this?.addressStreet,
                "",
                this?.addressCity,
                this?.addressState
            ),
            attemptNumber = attemptCount
        )
    )
}
