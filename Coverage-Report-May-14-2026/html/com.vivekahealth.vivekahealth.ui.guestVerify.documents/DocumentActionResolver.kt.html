package com.vivekahealth.vivekahealth.ui.guestVerify.documents

sealed class DocumentAction {
    object ExtractSSN : DocumentAction()
    object ExtractInsurance : DocumentAction()
    data class Error(val message: String) : DocumentAction()
}

fun resolveDocumentAction(finalType: String, allowedTypes: List<String>): DocumentAction {
    val ssnOnly = allowedTypes.contains("SSN") && !allowedTypes.contains("InsuranceCard")
    val insuranceOnly = allowedTypes.contains("InsuranceCard") && !allowedTypes.contains("SSN")
    val errorMsg = when {
        ssnOnly -> "We couldn't detect a valid Social Security card. Please make sure you're uploading the correct document and that it's fully visible in the photo."
        insuranceOnly -> "We couldn't detect a valid insurance card. Please make sure you're uploading the correct document and that it's fully visible in the photo."
        else -> "We couldn't detect a valid Social Security card or insurance card. Please make sure you're uploading the correct document and that it's fully visible in the photo."
    }
    return when (finalType) {
        "Insurance" -> if (allowedTypes.contains("InsuranceCard")) {
            DocumentAction.ExtractInsurance
        } else {
            DocumentAction.Error(
                errorMsg
            )
        }
        "SSN" -> if (allowedTypes.contains("SSN")) {
            DocumentAction.ExtractSSN
        } else {
            DocumentAction.Error(
                errorMsg
            )
        }
        else -> DocumentAction.Error(errorMsg)
    }
}
