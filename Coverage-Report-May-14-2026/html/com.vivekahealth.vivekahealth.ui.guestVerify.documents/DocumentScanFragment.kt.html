open class DocumentScanFragment : BaseFragment(), Injectable {

    protected open fun getDocumentType(): DocumentType =
        navArgs<DocumentScanFragmentArgs>().value.documentType

    protected open fun requestCameraPermission() {
        (requireActivity() as GuestVerifyActivity).cameraRequest()
    }

    protected open fun isFromHomePage(): Boolean {
        return (requireActivity() as GuestVerifyActivity).fromHomePage
    }

    protected open fun onSsnFailed(bitmap: Bitmap) {
        showManualInputFragment()
    }

    protected open fun onFrontCaptured(originalBitmap: Bitmap) {}

    protected open fun onNavigateToNext() {
        val documentType = getDocumentType()
        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_DOC_NEXT_CLICKED) {
            putString("doc_type", documentType.name)
        }
        // if already uploaded the previous image, continue
        if (documentType == DocumentType.DL_FRONT) {
            sharedViewModel.frontImage.value?.let {
                if (it.bitmap != null && it.bitmapUploadedId != null) {
                    findNavController().navigate(
                        R.id.action_documentFrontFragment_to_documentBackFragment
                    )
                    return
                }
            }
        } else if (documentType == DocumentType.DL_BACK) {
            sharedViewModel.backImage.value?.let { bckIt ->
                if (bckIt.bitmap != null && bckIt.bitmapUploadedId != null) {
                    sharedViewModel.ssnImage.value?.let { ssnIt ->
                        if (ssnIt.bitmap != null && ssnIt.bitmapUploadedId != null) {
                            findNavController().navigate(
                                R.id.action_documentBackFragment_to_documentSnnFragment
                            )
                        } else {
                            if (bckIt.requiredSnn == true) {
//                                            stepCounterViewModel.enableSSN(true)
                                findNavController().navigate(
                                    R.id.action_documentBackFragment_to_documentSnnFragment
                                )
                            } else {
                                findNavController().navigate(
                                    R.id.action_documentBackFragment_to_selfieCapture
                                )
                            }
                        }
                    } ?: run {
                        if (bckIt.requiredSnn == true) {
                            findNavController().navigate(
                                R.id.action_documentBackFragment_to_documentSnnFragment
                            )
                        } else {
                            findNavController().navigate(
                                R.id.action_documentBackFragment_to_selfieCapture
                            )
                        }
                    }
                    return
                }
            }
        } else if (documentType == DocumentType.SSN) {
            sharedViewModel.ssnImage.value?.let {
                if (it.bitmap != null && it.bitmapUploadedId != null) {
                    findNavController().navigate(
                        R.id.action_documentBackFragment_to_selfieCapture
                    )
                    return
                }
            }
        }

        if (sharedViewModel.token.value == null) {
            Toast.makeText(
                requireContext(),
                "Session id Not found ",
                Toast.LENGTH_SHORT
            ).show()
            return
        }

        /**
         *  handle front doc upload and processing in the background {@see com.vivekahealth.vivekahealth.ui.guestVerify.SharedViewModel}
         *  and allow moving forward if this is the first time uploading the front and not coming from failed back and front
         */
        if (documentType == DocumentType.DL_FRONT &&
            !sharedViewModel.hasSuccessfullyProcessedFront &&
            !sharedViewModel.isComingFromFailedBackCheck
        ) {
            sharedViewModel.frontImage.value?.let {
                sharedViewModel.uploadFrontAndProcess(
                    documentType,
                    it.bitmap,
                    sharedViewModel.token.value!!
                )
            }
            findNavController().navigate(
                R.id.action_documentFrontFragment_to_documentBackFragment
            )
            return
        }

        // handle upload and processing in foreground
        if (documentType == DocumentType.DL_FRONT) {
            sharedViewModel.frontImage?.value?.let {
                viewModel.uploadDocumentWithQR(
                    documentType,
                    it.bitmap,
                    sharedViewModel.token.value!!,
                    it.driverLicense,
                    null,
                    null,
                    null
                )
            } ?: run {
                Toast.makeText(
                    requireContext(),
                    "File Not found ",
                    Toast.LENGTH_SHORT
                ).show()
                return
            }
        } else if (documentType == DocumentType.DL_BACK) {
            sharedViewModel.backImage?.value?.let {
                viewModel.uploadDocumentWithQR(
                    documentType,
                    it.bitmap,
                    sharedViewModel.token.value!!,
                    it.driverLicense,
                    it.rawQRString,
                    null,
                    it.verifiedBarCodeInServer
                )
            } ?: run {
                Toast.makeText(
                    requireContext(),
                    "File Not found ",
                    Toast.LENGTH_SHORT
                ).show()
                return
            }
        } else if (documentType == DocumentType.SSN || documentType == DocumentType.SSN_ONLY) {
            sharedViewModel.ssnImage?.value?.let {
                viewModel.uploadDocumentWithQR(
                    documentType,
                    it.bitmap,
                    sharedViewModel.token.value!!,
                    null,
                    null,
                    it.ssn,
                    null,
                    it.isSsn
                )
            } ?: run {
                Toast.makeText(
                    requireContext(),
                    "File Not found ",
                    Toast.LENGTH_SHORT
                ).show()
                return
            }
        }
    }

    protected open fun onNavigateToBack(documentType: DocumentType) {
        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_DOC_BACK_CLICKED) {
            putString("doc_type", documentType.name)
        }
        when (documentType) {
            DocumentType.DL_FRONT -> requireActivity().onBackPressed()
            DocumentType.DL_BACK -> findNavController().navigateUp()
            DocumentType.SSN -> {
                findNavController().navigateUp()
            }

            DocumentType.SSN_ONLY -> {
                requireActivity().onBackPressed()
            }

            else -> {
            }
        }
    }

    protected open fun onSsnExtracted(
        ssn: Bitmap?,
        type: String?,
        id: String?,
        name: String?,
        isManualInput: Boolean
    ) {
        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_SSN_EXTRACTED)

        if (isManualInput) {
            viewModel.onNextClicked()
        }
    }

    private fun logGuestVerifyEvent(eventName: String, additionalParams: Bundle.() -> Unit = {}) {
        val bundle = Bundle().apply {
            putString(FirebaseTags.PARAM_TOKEN, sharedViewModel.token.value ?: "")
            additionalParams()
        }
        firebaseAnalytics.logEvent(eventName, bundle)
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            DocumentScanViewModel::class.java
        )

        val documentType = getDocumentType()
        viewModel.documentType = documentType
        binding.documentType = documentType
        binding.executePendingBindings()

        if (documentType == DocumentType.SSN || documentType == DocumentType.SSN_ONLY) {
            updateScanTitleAndDescription(documentType)
        }

        // Log screen view
        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_DOC_SCREEN_VIEW) {
            putString("doc_type", documentType.name)
        }

        binding.viewModel = this@DocumentScanFragment.viewModel

        barcodeScanner = BarcodeScanning
            .getClient(
                BarcodeScannerOptions.Builder()
                    .setBarcodeFormats(FORMAT_PDF417)
                    .build()
            )

        val options = FaceDetectorOptions.Builder()
            .setClassificationMode(FaceDetectorOptions.CLASSIFICATION_MODE_NONE)
            .build()
        faceDetector = FaceDetection.getClient(options)

        textRecognizer = TextRecognition.getClient(TextRecognizerOptions.Builder().build())

        onScanPageCaptured = { uri, docType ->
            logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_DOC_CAPTURED) {
                putString("doc_type", docType.name)
            }
            val imageStream = requireActivity().contentResolver?.openInputStream(uri)
            val originalBitmap = BitmapFactory.decodeStream(imageStream)
            binding.previewMedia.setImageBitmap(originalBitmap)

            when {
                docType == DocumentType.DL_FRONT -> {
                    binding.nextEnable = true
                    sharedViewModel.updateFrontImage(originalBitmap)
                    onFrontCaptured(originalBitmap)
                }
                docType == DocumentType.DL_BACK -> extractBarCodeData(originalBitmap)
                docType == DocumentType.SSN || docType == DocumentType.SSN_ONLY -> {
                    extractId(
                        originalBitmap,
                        onResult = { bitmap, type, id, name, isManualInput ->
                            binding.nextEnable = true
                            binding.previewMedia.setImageBitmap(bitmap)
                            var normalized = id
                            if (type == "INSURANCE") {
                                id?.let {
                                    val normalizedResult = IdNormalizer().normalize(id)
                                    normalized = normalizedResult.normalized
                                }
                            }
                            sharedViewModel.updateSSNImage(bitmap, normalized, type == "SSN")
                            onSsnExtracted(bitmap, type, id, name, isManualInput)
                        },
                        onError = { _ ->
                            sharedViewModel.clearSsn()
                            onSsnFailed(originalBitmap)
                        }
                    )
                }
                else -> {}
            }
        }
        scannerClient.register(this, onScanPageCaptured!!)

        viewModel.navigateToSkip.observe(
            viewLifecycleOwner,
            Observer {
                it.getContentIfNotHandled()?.let {
                    logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_DOC_SKIP_CLICKED) {
                        putString("doc_type", documentType.name)
                    }
                    if (documentType == DocumentType.SSN) {
                        Constants.hasShownSSNFlowThisSession = true
                        findNavController().navigate(
                            R.id.action_documentSsnFragment_to_selfieCapture
                        )
                    } else {
                        requireActivity().onBackPressed()
                    }
                }
            }
        )

        viewModel.navigateToNext.observe(
            viewLifecycleOwner,
            Observer {
                it.getContentIfNotHandled()?.let {
                    onNavigateToNext()
                }
            }
        )

        var disposable = viewModel.status.observeOn(AndroidSchedulers.mainThread())
            .subscribe {
                // show progress only if in front driving while back has failed or any other document except front
                if (sharedViewModel.isComingFromFailedBackCheck || documentType != DocumentType.DL_FRONT) {
                    var message = ""

                    if (it == LoadingState.UPLOAD) {
                        message = getString(R.string.uploading_your_file_please_wait)
                    } else if (it == LoadingState.PROCESS) {
                        message = getString(R.string.processing_document)
                    }

                    showLoadingFrag(message)
                }
            }

        viewModel.addToDisposable(disposable)

        viewModel.uploadState.observe(viewLifecycleOwner) { state ->

            state.getContentIfNotHandled()?.let {
                when (it) {
                    is UploadState.Error -> {
                        dismissDialog()
                        viewModel.resetState()

                        val message = if (it.message.contains("Unable to resolve host")) {
                            getString(R.string.unable_to_proceed_please_try_again_later)
                        } else if (it.message.contains("timeout") || it.message.startsWith(
                                "Failed to connect to"
                            )
                        ) {
                            getString(R.string.no_internet_msg)
                        } else {
                            it.message
                        }
                        if (documentType == DocumentType.DL_FRONT && sharedViewModel.isComingFromFailedBackCheck) {
                            viewModel.attemptNumber = viewModel.attemptNumber.plus(1)
                            if (viewModel.attemptNumber > 2) {
                                showRetakeFront(true)
                            } else {
                                showOkMessageDialog(requireContext(), message)
                                binding.nextEnable = false
                            }
                            return@observe
                        }
                        showOkMessageDialog(requireContext(), message)
                    }

                    UploadState.Idle -> {
                    }

                    is UploadState.SuccessScan -> {
                        binding.nextEnable = true
                        dismissDialog()

                        sharedViewModel.updateVerifiedBarCodeInServer(
                            "${DateUtils.getCurrentLocalTime()}"
                        )
                    }

                    is UploadState.FailedScan -> {
                        if (viewModel.attemptNumber == 3) {
                            binding.nextEnable = true
                            dismissDialog()

                            sharedViewModel.updateVerifiedBarCodeInServer(
                                "${DateUtils.getCurrentLocalTime()}"
                            )
                        } else {
                            sharedViewModel.clearBack()

                            val message = if (it.msg.contains("Unable to resolve host")) {
                                getString(R.string.unable_to_proceed_please_try_again_later)
                            } else {
                                it.msg
                            }

                            showOkMessageDialog(requireContext(), message)

                            viewModel.attemptNumber = viewModel.attemptNumber.plus(1)
                        }
                    }

                    is UploadState.Success -> {
                        dismissDialog()
                        if (documentType == DocumentType.DL_FRONT) {
                            sharedViewModel.updateSuccessTokenFront(
                                "${DateUtils.getCurrentLocalTime()}"
                            )
                            /**
                             * Since this can happen only front previously failed and back failed and
                             * user is reuploading front, we need to call verify and additional details
                             * */
                            sharedViewModel.hasSuccessfullyProcessedFront = true
                            viewModel.verifyAndAdditionalDetails(
                                token = sharedViewModel.token.value!!
                            )
                        } else if (documentType == DocumentType.DL_BACK) {
                            sharedViewModel.updateSuccessTokenBack(
                                "${DateUtils.getCurrentLocalTime()}",
                                "qr successful",
                                it.additionalRequirement?.requiresAdditionalId ?: false
                            )
                        } else if (documentType == DocumentType.SSN) {
                            sharedViewModel.updateSuccessTokenSSN(
                                "${DateUtils.getCurrentLocalTime()}"
                            )
                        }

                        if (documentType == DocumentType.DL_FRONT) {
                            /**
                             * do nothing here as verify and additional details is being called
                             * handles the navigation in viewModel.reUploadFrontVerifiedAndAdditionalDetails
                             * */
//                            findNavController().navigate(
//                                R.id.action_documentFrontFragment_to_documentBackFragment
//                            )
                        } else if (documentType == DocumentType.SSN) {
                            findNavController().navigate(
                                R.id.action_documentSsnFragment_to_selfieCapture
                            )
                        } else if (documentType == DocumentType.SSN_ONLY) {
                            /**
                             * Call identify after successful SSN details upload
                             * */
                            viewModel.identify(sharedViewModel.token.value!!)
                        } else if (documentType == DocumentType.DL_BACK) {
                            // check for the additional Requiremnt here
                            it.additionalRequirement?.let { additionalRequirementResponse ->

                                stepCounterViewModel.enableSSN(
                                    additionalRequirementResponse.requiresAdditionalId
                                )

                                if (additionalRequirementResponse.requiresAdditionalId) {
                                    findNavController().navigate(
                                        R.id.action_documentBackFragment_to_documentSnnFragment
                                    )
                                } else {
                                    findNavController().navigate(
                                        R.id.action_documentBackFragment_to_selfieCapture
                                    )
                                }
                            } ?: run {
                                findNavController().navigate(
                                    R.id.action_documentBackFragment_to_selfieCapture
                                )
                            }
                        }

                        viewModel.resetState()
                    }

                    is UploadState.Loading -> {
                    }

                    else -> {
                    }
                }
            }
        }

        viewModel.openCamera.observe(
            viewLifecycleOwner,
            EventObserver {
                logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_DOC_CAPTURE_CLICKED) {
                    putString("doc_type", documentType.name)
                }
                // check camera persmission here
                if (!requireActivity().hasPermission(Manifest.permission.CAMERA)) {
                    requestCameraPermission()
                } else {
                    showPicker()
                    // Your delayed logic
                    Handler(Looper.getMainLooper()).postDelayed({
                        when (documentType) {
                            DocumentType.DL_FRONT -> sharedViewModel.clearFront()
                            DocumentType.DL_BACK -> sharedViewModel.clearBack()
                            DocumentType.SSN -> {
                                sharedViewModel.clearSsn()
                            }

                            else -> {
                            }
                        }
                    }, 200)
                }
            }
        )

        viewModel.navigateToBack.observe(
            viewLifecycleOwner,
            EventObserver {
                onNavigateToBack(getDocumentType())
            }
        )

        if (documentType == DocumentType.DL_FRONT) {
            sharedViewModel.frontImage.observe(viewLifecycleOwner) {
                it?.let {
                    binding.imageCaptured = true
                    binding.nextEnable = true
                    binding.previewMedia.post {
                        binding.previewMedia.setImageBitmap(it.bitmap)
                    }
                } ?: run {
                    sharedViewModel.hasSuccessfullyProcessedFront = false
                    binding.imageCaptured = false
                    binding.nextEnable = false
                    binding.previewMedia.setLicensePlaceholder(documentType)
                }
            }
        } else if (documentType == DocumentType.DL_BACK) {
            sharedViewModel.backImage.observe(viewLifecycleOwner) {
                it?.let {
                    binding.imageCaptured = true
                    binding.nextEnable = true
                    binding.previewMedia.post {
                        binding.previewMedia.setImageBitmap(it.bitmap)
                    }
                } ?: run {
                    binding.previewMedia.post {
                        binding.previewMedia.setImageBitmap(null)
                        binding.previewMedia.setLicensePlaceholder(documentType)
                        binding.imageCaptured = false
                        binding.nextEnable = false
                    }
                }
            }
        } else if (documentType == DocumentType.SSN || documentType == DocumentType.SSN_ONLY) {
            // this two line for on view created
            binding.nextEnable =
                enableForEmptySSN(documentType) || sharedViewModel.ssnImage.value != null
            binding.materialButton3.text = if (documentType == DocumentType.SSN_ONLY) {
                getString(R.string.proceed)
            } else {
                if (sharedViewModel.ssnImage.value?.ssn != null) {
                    getString(R.string.proceed)
                } else {
                    getString(
                        R.string.skip_for_now
                    )
                }
            }

            sharedViewModel.ssnImage.observe(viewLifecycleOwner) {
                // observe the change
                binding.nextEnable = enableForEmptySSN(documentType) || it?.ssn != null
                binding.materialButton3.text = if (documentType == DocumentType.SSN_ONLY) {
                    getString(R.string.proceed)
                } else {
                    if (it?.ssn != null) {
                        getString(R.string.proceed)
                    } else {
                        getString(
                            R.string.skip_for_now
                        )
                    }
                }
                it?.let {
                    binding.imageCaptured = true
                    binding.previewMedia.post {
                        binding.previewMedia.setImageBitmap(it.bitmap)
                    }
                } ?: run {
                    binding.previewMedia.post {
                        binding.previewMedia.setImageBitmap(null)
                        binding.previewMedia.setLicensePlaceholder(documentType)
                        binding.imageCaptured = false
                    }
                }
            }
        }

        documentClassifier = DocumentClassifier(requireContext())
        documentClassifier.loadModel()

        textClassifier = InsuranceTextClassifier(requireContext())

        viewModel.reUploadFrontVerifiedAndAdditionalDetails.observe(
            viewLifecycleOwner,
            EventObserver {
                dismissDialog()
                if (it.requiresAdditionalId) {
                    // mark the failed back scan success so that back is not uploaded
                    sharedViewModel.updateVerifiedBarCodeInServer(
                        "${DateUtils.getCurrentLocalTime()}"
                    )
                    sharedViewModel.updateSuccessTokenBack(
                        "${DateUtils.getCurrentLocalTime()}",
                        "qr successful",
                        true
                    )
                    // update step for SSN
                    stepCounterViewModel.enableSSN(true)
                    // first navigate to back (to maintain up navigation)
                    findNavController().navigate(
                        R.id.action_documentFrontFragment_to_documentBackFragment
                    )
                    // then navigate to SSN
                    findNavController().navigate(
                        R.id.action_documentBackFragment_to_documentSnnFragment
                    )
                } else {
                    // mark the failed back scan success so that back is not uploaded
                    sharedViewModel.updateVerifiedBarCodeInServer(
                        "${DateUtils.getCurrentLocalTime()}"
                    )
                    sharedViewModel.updateSuccessTokenBack(
                        "${DateUtils.getCurrentLocalTime()}",
                        "qr successful",
                        false
                    )
                    // update step for SSN
                    stepCounterViewModel.enableSSN(false)
                    // first navigate to back (to maintain up navigation)
                    findNavController().navigate(
                        R.id.action_documentFrontFragment_to_documentBackFragment
                    )
                    // then navigate to Selfie
                    findNavController().navigate(R.id.action_documentBackFragment_to_selfieCapture)
                }
            }
        )

        viewModel.reUploadFrontVerifyAndAdditionalDetailsError.observe(
            viewLifecycleOwner,
            EventObserver {
                dismissDialog()
                val message = if (it.contains("Unable to resolve host")) {
                    getString(R.string.unable_to_proceed_please_try_again_later)
                } else {
                    it
                }

                showOkMessageDialog(requireContext(), message)
                binding.nextEnable = false
            }
        )
        viewModel.showAlternateAppAvailable.observe(
            viewLifecycleOwner,
            EventObserver { errorResponse ->
                val alternateAppDialog = AlternateAppPromptFragment.newInstance(
                    errorResponse.errorDescription,

    // return image, type, id, isManualInput
    fun extractId(
        originalBitmap: Bitmap,
        onResult: (Bitmap?, String?, String?, String?, Boolean) -> Unit,
        onError: (String?) -> Unit
    ) {
        val timedResult = onResult
        val timedError = onError

        val inputImage = InputImage.fromBitmap(originalBitmap, 0)

        viewLifecycleOwner.lifecycleScope.launch {
            // Step 1: Run image classifier
            val imageResults = withContext(Dispatchers.IO) {
                documentClassifier.classify(originalBitmap)
            }

            if (imageResults.isEmpty()) {
                timedError("Failed to detect card type")
                return@launch
            }

            val imageTop = imageResults[0] // top label from image model
            val imageInsuranceConf = imageResults.find { it.index == 0 }?.confidence ?: 0f

            // Step 2: Run OCR → text classifier
            recognizer.process(inputImage)
                .addOnSuccessListener { visionText ->
                    val ocrText = visionText.text
                    val textResult = textClassifier.classify(ocrText)
                    val textLabel = textResult?.label ?: "Other"

                    // Step 3: Fusion decision logic
                    val finalType = when {
                        // image(Insurance) + text(Insurance) = Insurance
                        imageTop.label == "Insurance" && textLabel == "Insurance" -> "Insurance"

                        // image(SSN) + text(Other) = SSN
                        imageTop.label == "SSN" && textLabel == "Other" && imageTop.confidence > 0.60f -> "SSN"

                        // image(Other) + text(Other) = Other
                        imageTop.label == "Other" && textLabel == "Other" -> "Other"

                        // image(Other) + text(Insurance) + image:insurance > 10% = Insurance
                        imageTop.label == "Other" && textLabel == "Insurance" && imageInsuranceConf > 0.10f -> "Insurance"

                        // image(SSN) + text(Insurance) + image:insurance > 10% = Insurance
                        imageTop.label == "SSN" && textLabel == "Insurance" && imageInsuranceConf > 0.10f -> "Insurance"

                        // image(Insurance) + text(Other) + image:insurance > 50% = Insurance
                        imageTop.label == "Insurance" && textLabel == "Other" && imageInsuranceConf > 0.50f -> "Insurance"

                        // All other cases = Other
                        else -> "Other"
                    }
                    val allowedTypes = sharedViewModel.allowedDocumentTypes
                    val allowedFinalType = when {
                        finalType == "Insurance" && !allowedTypes.contains("InsuranceCard") -> "Other"
                        finalType == "SSN" && !allowedTypes.contains("SSN") -> "Other"
                        else -> finalType
                    }

                    when (allowedFinalType) {
                        "Insurance" -> extractInsuranceCard(inputImage, timedResult, timedError)
                        "SSN" -> extractSSN(inputImage, originalBitmap, timedResult, timedError)
                        else -> {
                            val errorMsg = when {
                                allowedTypes.contains("SSN") && !allowedTypes.contains(
                                    "InsuranceCard"
                                ) ->
                                    "We couldn't detect a valid Social Security card. Please make sure you're uploading the correct document and that it's fully visible in the photo."

                                allowedTypes.contains("InsuranceCard") && !allowedTypes.contains(
                                    "SSN"
                                ) ->
                                    "We couldn't detect a valid insurance card. Please make sure you're uploading the correct document and that it's fully visible in the photo."

                                else ->
                                    "We couldn't detect a valid Social Security card or insurance card. Please make sure you're uploading the correct document and that it's fully visible in the photo."
                            }
                            timedError(errorMsg)
                        }
                    }
                }
                .addOnFailureListener { e ->
                    // Fallback: use image classifier only
                    val allowedTypes = sharedViewModel.allowedDocumentTypes

                    when (imageTop.label) {
                        "Insurance" -> {
                            if (allowedTypes.contains("InsuranceCard")) {
                                timedError(
                                    "We can see your insurance card, but the text is too blurry to read. Please take a clearer photo, making sure the card is in focus and well-lit."
                                )
                            } else {
                                timedError(
                                    "We couldn't detect a valid Social Security card. Please make sure you're uploading the correct document and that it's fully visible in the photo."
                                )
                            }
                        }

                        "SSN" -> {
                            if (allowedTypes.contains("SSN")) {
                                timedError(
                                    "We can see your Social Security card, but the text is too blurry to read. Please take a clearer photo, making sure the card is in focus and well-lit."
                                )
                            } else {
                                timedError(
                                    "We couldn't detect a valid insurance card. Please make sure you're uploading the correct document and that it's fully visible in the photo."
                                )
                            }
                        }

                        else -> {
                            timedError(
                                "We couldn't read the text from your document. Please try taking another photo with good lighting and make sure the document is flat and in focus."
                            )
                        }
                    }
                }
        }
    }

    val recognizer = TextRecognition.getClient(TextRecognizerOptions.Builder().build())

    fun extractSSN(
        inputImage: InputImage,
        originalBitmap: Bitmap,
        onResult: (Bitmap?, String?, String?, String?, Boolean) -> Unit,
        onError: (String?) -> Unit
    ) {
        lifecycleScope.launch {
            var foundSSN: String? = null

            val result = withContext(Dispatchers.Default) {
                BwBackgroundRemover.processIterative(originalBitmap) { bitmap ->
                    withContext(Dispatchers.IO) {
                        runCatching {
                            recognizer.process(InputImage.fromBitmap(bitmap, 0)).await().text
                        }.getOrDefault("")
                    }
                }
            }
            result.finalSsn?.let {
                foundSSN = it
            }

            if (foundSSN != null) {
                onResult(inputImage.bitmapInternal, "SSN", foundSSN, "", false)
            } else {
                showManualInputFragment("SSN")
            }
        }
    }

    fun extractInsuranceCard(
        inputImage: InputImage,
        onResult: (Bitmap?, String?, String?, String?, Boolean) -> Unit,
        onError: (String?) -> Unit
    ) {
        recognizer.process(inputImage)
            .addOnSuccessListener { visionText ->
                val extractor = InsuranceCardExtractor()

                viewLifecycleOwner.lifecycleScope.launch {
                    val result = withContext(Dispatchers.IO) {
                        // Heavy processing off the main thread
                        val extractedData = extractor.extractFromDocument(visionText)

                        Pair(
                            if (extractedData.memberIds.isNotEmpty()) extractedData.memberIds[0] else null,
                            if (extractedData.names.isNotEmpty()) extractedData.names[0] else null
                        )
                    }

                    // Back on main thread, safe for UI updates
                    if (result.first != null && result.first != "N/A") {
                        onResult(
                            inputImage.bitmapInternal,
                            "INSURANCE",
                            result.first,
                            result.second,
                            false
                        )
                    } else {
                        showManualInputFragment("InsuranceCard")
                    }
                }
            }
            .addOnFailureListener {
                viewModel.logFailedDocumentScanAttempt(
                    sharedViewModel.token.value!!,
                    DocumentType.SSN_ONLY
                )

                onError(
                    "We couldn't find a valid ID number on your insurance card. Please make sure the entire card is visible and the photo is clear and in focus."
                )
            }
    }

    private fun extractBarCodeData(bitmap: Bitmap) {
        // if success only enable the next
        barcodeScanner?.process(InputImage.fromBitmap(convertToGrayscale(bitmap), 0))
            ?.addOnSuccessListener {
                val filteredList = it.filter { it.driverLicense != null }

                if (filteredList.size > 1) {
                    // Do something when more than one item has a non-null driverLicense
                    showOkMessageDialog(
                        requireContext(),
                        getString(R.string.multiple_cards_detected_please_try_again)
                    )

                    return@addOnSuccessListener
                }

                if (it.any { it.driverLicense != null }) {
                    var item = it.find { it.driverLicense != null }
                    logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_BARCODE_SUCCESS)

                    binding.nextEnable = true
                    sharedViewModel.updateBackImage(
                        bitmap,
                        item?.driverLicense,
                        item
                    )
                    onBarcodeDetected(item?.driverLicense)
                } else {
                    onBarcodeExtractionFailed(bitmap)
                }
            }
            ?.addOnFailureListener {
                viewModel.logFailedDocumentScanAttempt(
                    sharedViewModel.token.value!!,
                    DocumentType.DL_BACK
                )
                showOkMessageDialog(
                    requireContext(),
                    getString(R.string.fail_to_extract_the_barcode_data)
                )
            }
    }

                } else {
                    R.string.scan_insurance_only
                }

            }

            else -> return
        }

        binding.textViewTitle.text = getString(titleRes)
        binding.tvText.text = getString(descRes)
    }

    open fun onBarcodeExtractionFailed(bitmap: Bitmap) {
        logGuestVerifyEvent(FirebaseTags.GUEST_VERIFY_BARCODE_FAILED) {
            putInt("attempt_number", viewModel.attemptNumber)
        }

        sharedViewModel.token.value?.let {
            viewModel.logFailedDocumentScanAttempt(
                it,
                DocumentType.DL_BACK
            )
        }

        binding.nextEnable = false

        if (viewModel.attemptNumber == 3) {
            binding.nextEnable = true
            // mark the back scan as successful
            sharedViewModel.updateBackImage(
                bitmap,
                null,
                null
            )
            sharedViewModel.updateVerifiedBarCodeInServer(
                "${DateUtils.getCurrentLocalTime()}"
            )
            /**
             * Since back scan failed more than twice, we now check whether front has
             * been successfully processed. If the front has also failed then we show the
             * user a dialog informing they need to upload front again
             * */
            if (!sharedViewModel.hasSuccessfullyProcessedFront) {
                showRetakeFront()
            }
        } else {
            showOkMessageDialog(
                requireContext(),
                getString(
                    R.string.we_couldn_t_read_the_barcode_please_try_again_and_make_sure_the_barcode_is_clearly_visible_well_lit_and_not_blurred
                )
            )
            sharedViewModel.clearBack()
            viewModel.attemptNumber = viewModel.attemptNumber.plus(1)
        }
    }

fun convertToGrayscale(bitmap: Bitmap): Bitmap {
    // Create a new Bitmap with the same size as the original
    val grayscaleBitmap = Bitmap.createBitmap(bitmap.width, bitmap.height, Bitmap.Config.ARGB_8888)

    // Create a Canvas to draw the original bitmap onto the new one
    val canvas = Canvas(grayscaleBitmap)

    // Create a ColorMatrix for grayscale conversion
    val colorMatrix = ColorMatrix()
    colorMatrix.setSaturation(0f) // Set saturation to 0 to get grayscale

    // Create a Paint object and set the ColorMatrix filter
    val paint = Paint()
    paint.colorFilter = ColorMatrixColorFilter(colorMatrix)

    // Draw the original bitmap onto the canvas using the paint (which applies the grayscale filter)
    canvas.drawBitmap(bitmap, 0f, 0f, paint)

    return grayscaleBitmap
}
