package com.vivekahealth.vivekahealth.ui.guestVerify.documents

import android.content.SharedPreferences
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.local.guest.UploadState
import com.vivekahealth.vivekahealth.ui.guestVerify.repository.GuestUploadRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.launch
import javax.inject.Inject

class ManualIdViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    var uploadRepository: GuestUploadRepository
) : ViewModel() {

    private val gson = Gson()
    private val compositeDisposable = CompositeDisposable()

    private val _navigateToNext = MutableLiveData<Event<Unit>>()
    val navigateToNext: LiveData<Event<Unit>> get() = _navigateToNext

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>> get() = _navigateToBack

    private val _uploadState = MutableLiveData<Event<UploadState>>()
    val uploadState: LiveData<Event<UploadState>> get() = _uploadState

    private var debounceJob: Job? = null
    private var isDebouncing = false
    private val debounceInterval = 500L

    fun singleClick(action: () -> Unit) {
        if (isDebouncing) return
        action()
        isDebouncing = true
        debounceJob?.cancel()
        debounceJob = viewModelScope.launch {
            delay(debounceInterval)
            isDebouncing = false
        }
    }

    fun onNextClicked() {
        singleClick {
            _navigateToNext.value = Event(Unit)
        }
    }

    fun onPrevClicked() {
        singleClick {
            _navigateToBack.value = Event(Unit)
        }
    }

    fun uploadSSN(token: String, value: String?, isSsn: Boolean?) {
        val uploadDetails = uploadSsnDetails(value, isSsn)
        val disposable = uploadRepository.updateDetails(token, uploadDetails)
            .subscribeOn(Schedulers.io())
            .observeOn(AndroidSchedulers.mainThread())
            .subscribe({ response ->
                if (response.isSuccessful) {
                    _uploadState.value = Event(UploadState.Success())
                } else {
                    val errorMsg = try {
                        val errorBody = response.errorBody()?.string() ?: Constants.APOLOGY_MESSAGE
                        gson.fromJson(
                            errorBody,
                            com.vivekahealth.vivekahealth.data.remote.login.GenericErrorResponse::class.java
                        ).getErrorMessage()
                    } catch (e: Exception) {
                        Constants.APOLOGY_MESSAGE
                    }
                    _uploadState.value = Event(UploadState.Error(errorMsg))
                }
            }, { throwable ->
                val errorMessage = when {
                    throwable.message?.contains("SocketException", ignoreCase = true) == true ->
                        Constants.APOLOGY_MESSAGE
                    else -> throwable.message ?: Constants.APOLOGY_MESSAGE
                }
                _uploadState.value = Event(UploadState.Error(errorMessage))
            })
        compositeDisposable.add(disposable)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
