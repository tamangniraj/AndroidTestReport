    fun sortList(chatList: ArrayList<ChatDataModel>): ArrayList<ChatDataModel> {
        Collections.sort(
            chatList,
            Comparator<ChatDataModel> { o1, o2 ->
                getSectionSort(o1.createdOn).compareTo(
                    getSectionSort(o2.createdOn)
                )
            }
        )
        return chatList
    }

    fun getSectionSort(time: String): Date {

    fun locationParse(member: MemberAddress?): String {
        var street2 = ""
        if (!member?.street2.isNullOrEmpty()) {
            street2 = member?.street2.toString() + ", "
        }

        return member?.street1?.let { "$it, " } + street2 + member?.city?.let { "$it, " } + member?.state?.let { "$it, " } + member?.postalCode?.let { it }
    }

    fun saveSpeciality(speciality: String?) {

    fun getDependent(memberId: String): MemberInfo? {
        var dep = memberInfo.dependentsArrayList?.find { memberInfo ->

    fun chatData() {
        removeLoop()
        compositeDisposable.add(
            homeRepository.getChat("", null, null, page, pageSize, false)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({
//                    chatList.clear()

                    loading = false

                    chatList.addAll(it.values)
                    chatData = it

                    chatList.sortByDescending { value -> DateUtils.milliToDate(value.createdOn) }
//                    chatList.reverse()

                    if (page > 1) {
                        _messageReceived.value = Event(false)
                    } else {
                        _messageReceived.value = Event(true)
                    }

                    page++
                    isLastPage = it.paging?.totalPages == page
                    startLoop()
                }, {
                    Timber.d(it.toString())
                    startLoop()
                    loading = false
                })
        )
    }

    private fun getLastChatMessageId(chatList: ArrayList<ChatDataModel>): Long {
        val temp: ArrayList<ChatDataModel> = chatList
        for (chatDataModel in temp) {
            if (chatDataModel.selfId != null && chatDataModel.selfId != -1L) {
                return chatDataModel.selfId!!
            }
        }
        return 0
    }

    fun updateReadStatus() {
        val chatIds: ArrayList<Long> = arrayListOf()
        for (item in chatList) {
            if (item.selfId != null && !item.memberReadStatus) {
                chatIds.add(item.selfId!!)
            }
        }
        if (chatIds.isNotEmpty()) {
            removeLoop()
            compositeDisposable.add(
                homeRepository.chatRead(ChatRead(chatIds))
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                    }
                    .subscribe({
                        startLoop()
                    }, {
                        Timber.d(it.toString())
                        startLoop()
                    })
            )
        }
    }

    fun sessionClose(back: Boolean, newSession: Boolean) {
        compositeDisposable.add(
            homeRepository.getChatSessionClose(
                preferences.getString("memberID", "").toString(),
                sessionId
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                }
                .subscribe({
                    if (back) {
                        _sessionClose.value = Event(Unit)
                    }
                    if (newSession) {
                        sessionStart()
                    }
                }, {
                    if (back) {
                        _sessionClose.value = Event(Unit)
                    }
                })
        )
    }

    fun sendChatClicked() {
        var location = _currentLocation.value?.peekContent()

        var message = MessageData(message = text, sessionId = sessionId)
        location?.let {

        sendChat(message)
