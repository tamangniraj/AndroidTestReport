package com.vivekahealth.vivekahealth.ui.enrollVTwo.continueEnrollment

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.utils.Event
import javax.inject.Inject

class ContinuePreviousEnrollmentV2ViewModel @Inject constructor(val preferences: SharedPreferences) :
    ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    var fullName: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.name, null)
        }
    var lastEnrollmentDate: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.lastEnrollmentDate, null)
        }
    private val _onBack = MutableLiveData<Event<Unit>>()
    val onBack: LiveData<Event<Unit>>
        get() = _onBack
    private val _onContinue = MutableLiveData<Event<Unit>>()
    val onContinue: LiveData<Event<Unit>>
        get() = _onContinue
    private val _onStartNew = MutableLiveData<Event<Unit>>()
    val onStartNew: LiveData<Event<Unit>>
        get() = _onStartNew

    fun onBackClicked() {
        _onBack.value = Event(Unit)
    }
    fun onContinuePrev() {
        _onContinue.value = Event(Unit)
    }
    fun onStartNew() {
        _onStartNew.value = Event(Unit)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }
}
