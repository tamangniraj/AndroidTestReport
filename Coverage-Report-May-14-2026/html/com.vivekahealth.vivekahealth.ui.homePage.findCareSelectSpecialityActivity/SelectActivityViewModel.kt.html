package com.vivekahealth.vivekahealth.ui.homePage.findCareSelectSpecialityActivity

import android.content.SharedPreferences
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import com.vivekahealth.vivekahealth.ui.homePage.findcareSelectSpeciality.selectSpecialityAllModel.SpecialityAll
import com.vivekahealth.vivekahealth.ui.homePage.findcareSelectSpeciality.selectSpecialityRecentModel.Specialityrecent
import java.util.ArrayList
import javax.inject.Inject
import retrofit2.Call
import retrofit2.Callback
import retrofit2.Response

class SelectActivityViewModel @Inject constructor(val preferences: SharedPreferences) : BaseViewModel<SelectSpecialityVerifyNavigator>() {
    private var mutableSignup: MutableLiveData<SpecialityAll>? = null
    private var selectrecent: MutableLiveData<Specialityrecent>? = null
    private val TAG = "VerifyLoginViewModel"
    var ValueList1: ArrayList<SpecialityAll> = ArrayList()
    var tempListFlat: ArrayList<Any> = arrayListOf()
    var pref = preferences
    init {
        pref = preferences
    }
    var ValueList11: ArrayList<Specialityrecent> = ArrayList()
    var tempListFlat1: ArrayList<Any> = arrayListOf()

    var tempListFlat11: ArrayList<Any> = arrayListOf()
    var ValueList111: ArrayList<String> = ArrayList()

    fun selectAll(post: String): LiveData<SpecialityAll> {
        mutableSignup = MutableLiveData()
        selectSpecialityAll(post)
        return mutableSignup!!
    }

    fun selectRecent(post: String): LiveData<Specialityrecent> {
        selectrecent = MutableLiveData()
        selectRecentSpeciality(post)
        return selectrecent!!
    }

    private fun selectRecentSpeciality(post: String) {
        val call: Call<Specialityrecent> = RetrofitClient.getApi()!!.getAllRecentSpeciality(post)
        call.enqueue(object : Callback<Specialityrecent> {
            override fun onResponse(
                call: Call<Specialityrecent>,
                response: Response<Specialityrecent>
            ) {
                try {
                    selectrecent?.value = response.body()
                } catch (e: Exception) {
                }
            }

            override fun onFailure(call: Call<Specialityrecent>, t: Throwable) {
                getNavigator()?.onError("Email or Password Invalid")
                //  Log.d(TAG, "onFailure: ${t.localizedMessage}")
            }
        })
    }

    private fun selectSpecialityAll(post: String) {
        val call: Call<SpecialityAll> = RetrofitClient.getApi()!!.getAllAllSpeciality(post)
        call.enqueue(object : Callback<SpecialityAll> {
            override fun onResponse(
                call: Call<SpecialityAll>,
                response: Response<SpecialityAll>
            ) {
                try {
                    mutableSignup?.value = response.body()
                } catch (e: Exception) {
                }
            }

            override fun onFailure(call: Call<SpecialityAll>, t: Throwable) {
                getNavigator()?.onError("Email or Password Invalid")
                //  Log.d(TAG, "onFailure: ${t.localizedMessage}")
            }
        })
    }
}
