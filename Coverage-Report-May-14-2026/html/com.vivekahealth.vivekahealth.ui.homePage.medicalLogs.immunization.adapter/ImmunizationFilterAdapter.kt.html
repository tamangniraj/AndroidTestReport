package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.immunization.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.*
import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.AvailImmunizationFiltersItem
import com.vivekahealth.vivekahealth.databinding.*
import java.util.*
import timber.log.Timber

class ImmunizationFilterAdapter(private var onClick: (AvailImmunizationFiltersItem) -> Unit) : ListAdapter<AvailImmunizationFiltersItem, RecyclerView.ViewHolder>(
    FilterImmunizationDiffCallback()
) {

    var maxWidth: Int = 0
    var selectedPosition: Int = 0

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return ImmunizationFilterViewHolder(
            SingleItemImmunizationFilterBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            )
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as ImmunizationFilterViewHolder).binding.filter = item
        holder.binding.setClickListener {
            Timber.d(itemCount.toString())
            item.selected = !item.selected
            holder.binding.checkBox.isChecked = item.selected
            onClick(item)
            selectedPosition = position
        }
    }

    class ImmunizationFilterViewHolder(
        val binding: SingleItemImmunizationFilterBinding
    ) : RecyclerView.ViewHolder(binding.root) {

        init {
        }
    }
}

private class FilterImmunizationDiffCallback : DiffUtil.ItemCallback<AvailImmunizationFiltersItem>() {

    override fun areItemsTheSame(
        oldItem: AvailImmunizationFiltersItem,
        newItem: AvailImmunizationFiltersItem
    ): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(
        oldItem: AvailImmunizationFiltersItem,
        newItem: AvailImmunizationFiltersItem
    ): Boolean {
        return oldItem.value == newItem.value
    }
}
