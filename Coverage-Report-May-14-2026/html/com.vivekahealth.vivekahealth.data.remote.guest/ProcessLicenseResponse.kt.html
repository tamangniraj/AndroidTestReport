package com.vivekahealth.vivekahealth.data.remote.guest

import com.google.gson.annotations.SerializedName

data class ProcessLicenseResponse(
    @SerializedName("documentId") val documentId: String? = null,
    @SerializedName("legalFirstName") val legalFirstName: String? = null,
    @SerializedName("legalLastName") val legalLastName: String? = null,
    @SerializedName("legalMiddleName") val legalMiddleName: String? = null,
    @SerializedName("dob") val dob: String? = null,
    @SerializedName("gender") val gender: String? = null,
    @SerializedName("expirationDate") val expirationDate: String? = null,
    @SerializedName("height") val height: String? = null,
    @SerializedName("eyeColor") val eyeColor: String? = null,
    @SerializedName("issuedDate") val issuedDate: String? = null,
    @SerializedName("licenseNumber") val licenseNumber: String? = null,
    @SerializedName("fullAddress") val fullAddress: FullAddress? = null,
    @SerializedName("details") val details: String? = null,
    @SerializedName("attemptNumber") val attemptNumber: Int? = null
) {
    val isEmpty: Boolean
        get() = legalFirstName.isNullOrBlank() && legalLastName.isNullOrBlank()

    data class FullAddress(
        @SerializedName("state") val state: String? = null,
        @SerializedName("stateName") val stateName: String? = null,
        @SerializedName("city") val city: String? = null,
        @SerializedName("zip") val zip: String? = null,
        @SerializedName("zipExtension") val zipExtension: String? = null,
        @SerializedName("addressLine1") val addressLine1: String? = null,
        @SerializedName("addressLine2") val addressLine2: String? = null
    )
}
