class FindCareViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val findCareRepository: FindCareRepository,
    private val vivekaDatabase: VivekaDatabase,
    private val analyticsRepository: AnalyticsRepository,
    private val homeRepository: HomeRepository
) : ViewModel(), Observable {

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    var ValueList: ArrayList<DoctorResponseNew> = ArrayList()
    var eligibleValueList: ArrayList<DoctorResponseNew> = ArrayList()
    var docList: ArrayList<Value> = ArrayList()
    var listForDoctorAdapter: List<DoctorResponseNew> = listOf()
    var gson = Gson()
    var selected_location: String = "Nearby"

    var configDb =
        homeRepository.getClientDetail(preferences.getString("memberID", "").toString()).asLiveData()

    var nearByLocation: Location? = null
    var taxonomyGroupList: TaxonomyGroupList = TaxonomyGroupList()
    var valueString = ""

    // var memberInfoList  =  findCareRepository.getMemberFromDb().asLiveData()
    private var mutableSignup: MutableLiveData<SpecialityAll>? = null
    var memberInfoList = findCareRepository.getMemberFromDb().asLiveData()
    var memberInfoDb =
        findCareRepository.getMemberSid(preferences.getString("memberID", "").toString())
            .asLiveData()
    private val _logsListGenerated = MutableLiveData<Event<DoctorResponseNew>>()
    val logsListGenerated: LiveData<Event<DoctorResponseNew>>
        get() = _logsListGenerated

    var memberInfo = MemberInfo()
    var tempListFlat: ArrayList<Any> = arrayListOf()
    final var memberID = preferences.getString("memberID", "").toString()
    final var accessToken = preferences.getString(Constants.ACCESS_TOKEN, "").toString()

    private val _locationGeo = MutableLiveData<Event<Unit>>()
    val locationGeo: LiveData<Event<Unit>>
        get() = _locationGeo
    private val _findClick = MutableLiveData<Event<Unit>>()
    val findClick: LiveData<Event<Unit>>
        get() = _findClick
    private val _findAll = MutableLiveData<Event<Unit>>()
    val findAll: LiveData<Event<Unit>>
        get() = _findAll
    var filterOption: FindCareFilterModel = FindCareFilterModel(showSpeciality = true)
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.filterOption, null)
        }
    var wentToSetting: Boolean = false
    var loaderCount = 0
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    var showNoDoc: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showNoDoc, null)
        }

    var withIn: String = "10"
    var addressName: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.addressName, null)
        }
    var currentLocation: String = ""
    var enteredLocation: String = ""
    var workLocation: String = ""
    var homeLocation: String = ""
    private val _taxonomyListGenerated = MutableLiveData<Event<TaxonomyGroupList>>()
    val taxonomyListGenerated: LiveData<Event<TaxonomyGroupList>>
        get() = _taxonomyListGenerated

    init {
        getLogView()
    }

    fun selectAll(post: String): LiveData<SpecialityAll> {
        mutableSignup = MutableLiveData()
        selectSpecialityAll(post)
        return mutableSignup!!
    }

    fun locationParse(memberAddress: MemberAddress?): String = memberAddress?.fullAddress() ?: ""

    fun assignAddress(memberInfo: MemberInfo) {
        val locationHome = LatLng(
            memberInfo.homeAddress?.latitude ?: 0.0,
            memberInfo.homeAddress?.longitude ?: 0.0
        )
//        locationHome.latitude =   memberInfo.homeAddress?.latitude?:0.0
//        locationHome.longitude = memberInfo.homeAddress?.longitude?:0.0
        filterOption.home = locationHome
        val locationWork = LatLng(
            memberInfo.workAddress?.latitude ?: 0.0,
            memberInfo.workAddress?.longitude ?: 0.0
        )
//        locationWork.latitude =   memberInfo.workAddress?.latitude?:0.0
//        locationWork.longitude = memberInfo.workAddress?.longitude?:0.0
        filterOption.work = locationWork
        filterOption.showHome = memberInfo.homeAddress != null
        filterOption.showWork = memberInfo.workAddress != null

        _locationGeo.value = Event(Unit)
    }

    fun locationCurrent(currentLocation: Location) {
        filterOption.current = LatLng(currentLocation.latitude, currentLocation.longitude)
        filterOption.near = LatLng(currentLocation.latitude, currentLocation.longitude)
        filterOption = filterOption
        getDoctorList()
    }

    fun stringGson(value: Value): String {
        return gson.toJson(value)
    }

    fun viewAll() {
        _findAll.value = Event(Unit)
    }

    fun removeSpeciality() {
        filterOption.speciality = ""
        filterOption.selectedSubSpeciality.clear()
        filterOption.taxonomyCodes.clear()
    }

    fun getSpeciality() {
        filterOption.taxonomyGroup = ""
        val list = filterOption.selectedSubSpeciality.map { it.speciality }
        filterOption.speciality = list.joinToString(separator = ",") { it -> it }
        filterOption = filterOption
    }

    fun getTaxonomyGroup() {
        compositeDisposable.add(
            findCareRepository.getTaxonomyList()
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loaderCount++
                    loading = true
                }
                .subscribe({
                    loaderCount--
                    if (loaderCount == 0) {
                        loading = false
                    }
                    taxonomyGroupList = it

                    _taxonomyListGenerated.value = Event(it)
                }, {
                    loaderCount--
                    if (loaderCount == 0) {
                        loading = false
                    }
                })
        )
    }

    private fun selectSpecialityAll(post: String) {
        val call: Call<SpecialityAll> = RetrofitClient.getApi()!!.getAllAllSpeciality(post)
        call.enqueue(object : Callback<SpecialityAll> {
            override fun onResponse(
                call: Call<SpecialityAll>,
                response: Response<SpecialityAll>
            ) {
                try {
                    mutableSignup?.value = response.body()
                } catch (e: Exception) {
                }
            }

            override fun onFailure(call: Call<SpecialityAll>, t: Throwable) {
            }
        })
    }

    fun getLogView() {
        compositeDisposable.add(
            analyticsRepository.getLogPageView(
                preferences.getString(
                    "memberID",
                    ""
                ).toString(),
                PageType.FIND_DOCTOR_SCREEN.pageName
            )
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                }.subscribe({
                }, {
                })
        )
    }

    fun outNetworkClick() {
        filterOption.providerType = "1"
        filterOption.outNetwork = false
        filterOption.outNetwork = filterOption.outNetwork
    }

    fun findClick() {
        _findClick.value = Event(Unit)
    }

    var data: Boolean = false
    fun getDoctorList() {
        val memberId = preferences.getString("memberID", "").toString()

        Timber.d("MemebrId $memberId")

        if (filterOption.current != null) {
            compositeDisposable.add(
                findCareRepository.getDoctorList(
                    memberId,
                    filterOption.current?.latitude.toString(),
                    filterOption.current?.longitude.toString(),
                    "",
                    ""
                )
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        loaderCount++
                        loading = true
                    }
                    .subscribe({
                        showNoDoc = it.values.isEmpty()
                        docList.clear()
                        docList = it.values

                        listForDoctorAdapter = eligibleValueList.toList()
                        loaderCount--
                        if (loaderCount == 0) {
                            loading = false
                        }
                        _logsListGenerated.value = Event(it)
                    }, {
                        loaderCount--
                        if (loaderCount == 0) {
                            loading = false
                        }
                        showNoDoc = true
                    })
            )
        }
    }

    fun getMem() {
        viewModelScope.launch {
            memberInfoDb = findCareRepository.getMemberSid(memberID).asLiveData()
        }
        viewModelScope.launch {
            memberInfoList = findCareRepository.getMemberFromDb().asLiveData()
        }
        // Timber.d(memberInfoList?.value?.get(0)?.memberRelationship)
    }

    fun saveLocation(latitude: String, longitude: String) {
        preferences.edit().putString(Constants.LATITUDE, latitude).apply()
        preferences.edit().putString(Constants.LONGITUDE, longitude).apply()
    }

    fun saveLocationString(location: String) {
        preferences.edit().putString(Constants.LOCATIONSTRING, location).apply()
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }

    private val _tutorialsAdded = MutableLiveData(false)
    val tutorialsAdded: LiveData<Boolean>
        get() = _tutorialsAdded

    private val _showTutorial = MutableLiveData<Event<Unit>>()
    val showTutorial: LiveData<Event<Unit>>
        get() = _showTutorial

    var addedMyDoctorTutorial = false
    private var addedLookingForTutorial = false
    private var addedSearchTutorials = false

    var tutorialShown = false
    fun addMyDoctorTutorial() {
        addedMyDoctorTutorial = true
        showTutorialIfConditionMatched()
    }

    fun addLookingForTutorial() {
        addedLookingForTutorial = true
        showTutorialIfConditionMatched()
    }
    fun addSearchTutorials() {
        addedSearchTutorials = true
        updateTutorialAdditionStatus()
    }

    private fun updateTutorialAdditionStatus() {
        _tutorialsAdded.value = addedMyDoctorTutorial && addedSearchTutorials && addedLookingForTutorial
    }

    private fun showTutorialIfConditionMatched() {
        if (!preferences.getBoolean(
                Constants.SHOWN_FIND_CARE_TUT_FOR_SESSION,
                false
            ) && !tutorialShown && addedMyDoctorTutorial && addedLookingForTutorial
        ) {
            _showTutorial.value = Event(Unit)
            tutorialShown = true
            preferences.edit().putBoolean(Constants.SHOWN_FIND_CARE_TUT_FOR_SESSION, true).apply()
        }
    }

    fun dontShowTutorial() {
        preferences.edit().putBoolean(memberInfo.selfId + "findCareTutorialShown", true).apply()
    }

    fun resetTutorial() {
        _tutorialsAdded.value = false
    }

    fun shouldShowTutorial(): Boolean {
        return !preferences.getBoolean(memberInfo.selfId + "findCareTutorialShown", false)
    }
}
