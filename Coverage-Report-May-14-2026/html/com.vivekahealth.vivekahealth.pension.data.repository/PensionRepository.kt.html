package com.vivekahealth.vivekahealth.pension.data.repository

import com.google.gson.Gson
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.map
import com.vivekahealth.vivekahealth.pension.data.local.dao.PensionConfigDao
import com.vivekahealth.vivekahealth.pension.data.local.dao.SummaryImpactDao
import com.vivekahealth.vivekahealth.pension.data.local.dao.WorkHistoryDao
import com.vivekahealth.vivekahealth.pension.data.local.entity.PensionConfigEntity
import com.vivekahealth.vivekahealth.pension.data.local.entity.SummaryImpactEntity
import com.vivekahealth.vivekahealth.pension.data.local.entity.WorkHistoryEntity
import com.vivekahealth.vivekahealth.pension.data.model.PensionEstimateRequest
import com.vivekahealth.vivekahealth.pension.data.model.PensionEstimateResponse
import com.vivekahealth.vivekahealth.pension.data.model.PensionMemberResponse
import com.vivekahealth.vivekahealth.pension.data.model.WorkHistorySummary
import com.vivekahealth.vivekahealth.pension.data.model.SummaryImpactData
import com.vivekahealth.vivekahealth.pension.data.model.UserData
import com.vivekahealth.vivekahealth.pension.service.PensionService
import com.vivekahealth.vivekahealth.pension.data.model.AvailableOptions
import com.vivekahealth.vivekahealth.pension.data.model.RetirementOption
import com.vivekahealth.vivekahealth.pension.data.model.SurvivorOptionConfig
import io.reactivex.Single
import okhttp3.ResponseBody
import retrofit2.Response
import timber.log.Timber

import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class PensionRepository @Inject constructor(
    private val pensionService: PensionService,
    private val summaryImpactDao: SummaryImpactDao,
    private val workHistoryDao: WorkHistoryDao,
    private val pensionConfigDao: PensionConfigDao
) {
    // Create a separate Gson instance that doesn't exclude final fields
    // (Kotlin data class properties are final by default)
    private val gson: Gson = com.google.gson.GsonBuilder()
        .excludeFieldsWithModifiers(
            java.lang.reflect.Modifier.TRANSIENT,
            java.lang.reflect.Modifier.STATIC
        )
        .create()

    private val _userData = MutableStateFlow(UserData())
    val userData: StateFlow<UserData> = _userData.asStateFlow()

    private val _hasPensionData = MutableStateFlow(false)
    val hasPensionData: StateFlow<Boolean> = _hasPensionData.asStateFlow()

    fun getPensionMemberData(memberId: String): Single<Response<PensionMemberResponse>> {
        return pensionService.getPensionApi().getPensionMemberData(memberId)
    }

    fun updatePensionData(response: PensionMemberResponse) {
        val currentData = _userData.value
        val newData = currentData.copy(
            pensionMemberResponse = response
        )
        _userData.value = newData
        _hasPensionData.value = response.memberId != null
    }

    fun updateRetirementAge(retirementAge: RetirementOption) {
        val currentData = _userData.value
        val newData = currentData.copy(
            selectedRetirementAge = retirementAge
        )
        _userData.value = newData
    }

    fun updateExpectedHours(hours: String) {
        _userData.value = _userData.value.copy(expectedHours = hours)
    }

    fun updateSurvivorBenefit(benefit: SurvivorOptionConfig?) {
        _userData.value = _userData.value.copy(selectedSurvivorBenefit = benefit)
    }

    fun clearPensionData() {
        _userData.value = UserData()
        _hasPensionData.value = false
    }

    // Single Source of Truth - SummaryImpactData from Database
    fun getSummaryImpactData(memberId: String): Flow<SummaryImpactData?> {
        return summaryImpactDao.getSummaryImpactByMemberId(memberId)
            .map { entity -> entity?.toSummaryImpactData() }
    }

    suspend fun saveSummaryImpactData(memberId: String, data: SummaryImpactData) {
        val entity = SummaryImpactEntity.fromSummaryImpactData(memberId, data)
        summaryImpactDao.insertSummaryImpact(entity)
    }

    suspend fun deleteSummaryImpactData(memberId: String) {
        summaryImpactDao.deleteSummaryImpactByMemberId(memberId)
    }

    // Work History methods
    fun getWorkHistorySummary(memberId: String): Single<WorkHistorySummary> {
        return pensionService.getPensionApi().getWorkHistorySummary(memberId)
    }

    fun getWorkHistoryFromDb(memberId: String): Flow<WorkHistorySummary?> {
        return workHistoryDao.getWorkHistoryByMemberId(memberId)
            .map { entity -> entity?.toWorkHistorySummary() }
    }

    suspend fun getWorkHistoryFromDbOnce(memberId: String): WorkHistorySummary? {
        return workHistoryDao.getWorkHistoryByMemberIdOnce(memberId)?.toWorkHistorySummary()
    }

    suspend fun saveWorkHistory(memberId: String, data: WorkHistorySummary) {
        val entity = WorkHistoryEntity.fromWorkHistorySummary(memberId, data)
        workHistoryDao.insertWorkHistory(entity)
    }

    suspend fun deleteWorkHistory(memberId: String) {
        workHistoryDao.deleteWorkHistoryByMemberId(memberId)
    }

    suspend fun getWorkHistoryFetchedTime(memberId: String): Long? {
        return workHistoryDao.getFetchedTime(memberId)
    }

    // Pension Config methods
    fun getPensionConfig(memberId: String): Single<PensionEstimateResponse> {
        return pensionService.getPensionApi().getPensionConfig(memberId)
    }

    fun getPensionEstimate(
        memberId: String,
        request: PensionEstimateRequest
    ): Single<retrofit2.Response<PensionEstimateResponse>> {
        return pensionService.getPensionApi().getPensionEstimate(memberId, request)
    }

    fun getPensionConfigFromDb(memberId: String): Flow<AvailableOptions?> {
        return pensionConfigDao.getPensionConfigByMemberId(memberId)
            .map { entity ->
                entity?.let {
                    gson.fromJson<AvailableOptions>(
                        it.availableOptions,
                        AvailableOptions::class.java
                    )
                }
            }
    }

    suspend fun getPensionConfigFromDbOnce(memberId: String): AvailableOptions ? {
        val entity = pensionConfigDao.getPensionConfigByMemberIdSync(memberId)
        return entity?.let {
            gson.fromJson<AvailableOptions>(it.availableOptions, AvailableOptions::class.java)
        }
    }

    suspend fun savePensionConfig(memberId: String, availableOptions: AvailableOptions) {
        val json = gson.toJson(availableOptions)
        val entity = PensionConfigEntity(
            memberId = memberId,
            availableOptions = json,
            lastUpdated = System.currentTimeMillis()
        )
        pensionConfigDao.insertPensionConfig(entity)
        Timber.d("Pension config saved to database for memberId: $memberId")
    }

    suspend fun deletePensionConfig(memberId: String) {
        pensionConfigDao.deletePensionConfigByMemberId(memberId)
    }

    // Download Pension Application
    fun downloadPensionApplication(memberId: String): Single<ResponseBody> {
        return pensionService.getPensionApi().downloadPensionApplication(memberId)
    }
}
