package com.vivekahealth.vivekahealth.ui.homePage.messageCenter.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.*
import com.vivekahealth.vivekahealth.data.remote.home.messageCenter.MessageCenterData
import com.vivekahealth.vivekahealth.databinding.*

class NotificationListAdapter(private var onClick: (MessageCenterData) -> Unit) : ListAdapter<MessageCenterData, RecyclerView.ViewHolder>(
    NotificationListDiffCallback()
) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return MessageCenterListViewHolder(
            MessageCenterCellBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            )
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as MessageCenterListViewHolder).binding.data = item
        holder.binding.setClickListener {
            onClick(item)
        }
    }

    class MessageCenterListViewHolder(
        val binding: MessageCenterCellBinding
    ) : RecyclerView.ViewHolder(binding.root) {

        init {
        }
    }
}

private class NotificationListDiffCallback : DiffUtil.ItemCallback<MessageCenterData>() {
    override fun areItemsTheSame(oldItem: MessageCenterData, newItem: MessageCenterData): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(
        oldItem: MessageCenterData,
        newItem: MessageCenterData
    ): Boolean {
        return oldItem == newItem
    }
}
