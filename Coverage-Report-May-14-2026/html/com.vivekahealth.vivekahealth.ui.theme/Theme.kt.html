/**
 * Creates a Light ColorScheme using colors from XML resources.
 * This allows the theme to automatically adapt to different product flavors.
 */
@Composable
private fun getLightColorScheme(): ColorScheme {
    val colorPrimary = colorResource(R.color.colorPrimary)
    val primaryDarker = colorResource(R.color.primaryDarker)
    val primaryLight = colorResource(R.color.primaryLight)
    val primaryUltraLight = colorResource(R.color.primaryUltraLight)
    val gradientStart = colorResource(R.color.gradient_home_start)
    val gradientEnd = colorResource(R.color.gradient_home_end)

    return lightColorScheme(
        primary = colorPrimary,
        onPrimary = Color.White,
        primaryContainer = primaryUltraLight,
        onPrimaryContainer = Color(0xFF3F4348),

        secondary = colorPrimary,
        onSecondary = Color.White,
        secondaryContainer = primaryLight,
        onSecondaryContainer = Color(0xFF3F4348),

        tertiary = gradientStart,
        onTertiary = Color.White,
        tertiaryContainer = gradientEnd,
        onTertiaryContainer = Color(0xFF3F4348),

        error = Color(0xFFFF0000),
        onError = Color.White,
        errorContainer = Color(0xFFFFDAD6),
        onErrorContainer = Color(0xFF410002),

        background = Color.White,
        onBackground = Color(0xFF212121),

        surface = Color.White,
        onSurface = Color(0xFF212121),
        surfaceVariant = Color(0xFFF8F8F8),
        onSurfaceVariant = Color(0xFF707070),

        outline = Color(0xFFE0E0E0),
        outlineVariant = Color(0xFFF0F0F0),

        scrim = Color(0x80000000)
    )
}

/**
 * Viveka Health Material3 Theme
 *
 * This theme automatically adapts to different product flavors by reading colors from XML resources.
 * Features:
 * - Dynamic primary color from colors.xml (changes per flavor)
 * - Roboto font family
 * - Material3 design system
 *
 * Usage:
 * ```
 * VivekaTheme {
 *     // Your composable content
 *     // Access theme colors with MaterialTheme.colorScheme.primary
 * }
 * ```
 */
@Composable
fun VivekaTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    content: @Composable () -> Unit
) {
    // Create color scheme from XML resources (flavor-aware)
    val colorScheme = if (darkTheme) getDarkColorScheme() else getLightColorScheme()

    val view = LocalView.current
    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as Activity).window
            window.statusBarColor = colorScheme.primary.toArgb()
            WindowCompat.getInsetsController(window, view).isAppearanceLightStatusBars = !darkTheme
        }
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = VivekaTypography,
        content = content
    )
}
