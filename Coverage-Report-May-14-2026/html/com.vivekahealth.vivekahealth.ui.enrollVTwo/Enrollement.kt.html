package com.vivekahealth.vivekahealth.ui.enrollVTwo

import com.vivekahealth.vivekahealth.utils.DateUtils
import java.io.Serializable

data class Enrollment(
    val isAddDependent: Boolean,
    val childCount: Int = 1,
    var addSpouse: Boolean = false,
    var addChild: Boolean = false,
    val currentDependentList: ArrayList<DependentCheckItem> = arrayListOf(),
    val primaryFName: String,
    val primaryLName: String,
    val lastEnrollmentDate: String = DateUtils.getCurrentDate() ?: "",
    val subscriberGender: String? = null
) : Serializable {

    fun findChildByIndex(index: Int?): DependentCheckItem? {
        return currentDependentList.find {
            it.type == DependentType.CHILD &&
                it.index == index
        }
    }

    fun findChilds(): List<DependentCheckItem>? {
        return currentDependentList.filter {
            it.type == DependentType.CHILD
        }
    }

    fun findPrimary(): DependentCheckItem? {
        return currentDependentList.find {
            it.type == DependentType.PRIMARY
        }
    }

    fun findSpouse(): DependentCheckItem? {
        return currentDependentList.find {
            it.type == DependentType.SPOUSE
        }
    }

    fun prepareList(): ArrayList<DependentCheckItem> {
        val updatedList = ArrayList<DependentCheckItem>()

        // === self ===
        val previousSelf = currentDependentList.find { it.type == DependentType.PRIMARY }
        val selfItem = previousSelf ?: DependentCheckItem(
            title = "Primary Details",
            type = DependentType.PRIMARY,
            firstName = primaryFName,
            lastName = primaryLName
        )
        updatedList.add((selfItem))

        // === SPOUSE ===
        val previousSpouseItem = currentDependentList.find { it.type == DependentType.SPOUSE }

        if (addSpouse) {
            // Reuse existing spouse item or create new one
            val spouseItem = previousSpouseItem
                ?: DependentCheckItem(
                    title = "Spouse Details",
                    type = DependentType.SPOUSE
                )
            updatedList.add(spouseItem)
        }

        // === CHILDREN ===
        val previousChildren = currentDependentList.filter { it.type == DependentType.CHILD }
        if (addChild) {
            val childrenToKeep = previousChildren.take(childCount)

            // Add new children if needed
            val newChildren = (childrenToKeep.size + 1..childCount).map { index ->
                DependentCheckItem(
                    title = "Child $index Details",
                    type = DependentType.CHILD,
                    index = index
                )
            }

            updatedList.addAll(childrenToKeep + newChildren)
        }

        currentDependentList.clear()
        currentDependentList.addAll(updatedList)
        return currentDependentList
    }
}
