    fun findRequiredDocument(type: Int, primary: Boolean?, spouse: String?): List<DocumentItem> {
        val baseList =
            reqDocs?.documentList?.find { it.relationshipType == type }?.documents ?: emptyList()

        return if (type == RelationshipType.PRIMARY.code && primary == true
        ) {
            val extraItems = listOf(
                DocumentItem(
                    53,
                    false,
                    true,
                    "Proof of Previous Insurance",
                    resources.getString(R.string.proof_of_previous_insurance)
                )
            )
            baseList + extraItems
        } else if (type == RelationshipType.SPOUSE.code && spouse == "true") {
            val extraItems = listOf(
                DocumentItem(
                    501,
                    false,
                    true,
                    "Front of Insurance Card",
                    resources.getString(R.string.front_of_insurance_card)
                ),
                DocumentItem(
                    502,
                    false,
                    true,
                    "Back of Insurance Card",
                    resources.getString(R.string.back_of_insurance_card)
                )
            )
            baseList + extraItems
        } else {
            baseList
        }
    }

    fun updateChildDob(dobChild: String) {
        val currentEnrollment = _enrollmentData.value ?: return

        val dateFormat = SimpleDateFormat("yyyy-MM-dd", Locale.getDefault())

        val childAgeInYear = DateUtils.getDateDiff(
            dateFormat,
            dobChild,
            DateUtils.getCurrentDateServer()
        )

        val isNowMinor = childAgeInYear < 1

        val updatedList = currentEnrollment.currentDependentList.map { dependent ->
            if (dependent.index == index && dependent.type == DependentType.CHILD) {
                val updatedDependent = if (dependent.dob != null) {
                    val wasMinor = DateUtils.getDateDiff(
                        dateFormat,
                        dependent.dob,
                        DateUtils.getCurrentDateServer()
                    ) < 1

                    val changedFromMinorToAdult = wasMinor && !isNowMinor

                    if (changedFromMinorToAdult) {
                        dependent.copy(dob = dobChild, isCompleted = false)
                    } else {
                        dependent.copy(dob = dobChild)
                    }
                } else {
                    dependent.copy(dob = dobChild)
                }
                updatedDependent
            } else {
                dependent
            }
        }
        val updatedEnrollment = currentEnrollment.copy(
            currentDependentList = ArrayList(updatedList)
        )
        _enrollmentData.value = updatedEnrollment
    }

    fun updateUploadStatus(data: DependentCheckItem?, currentItem: ListDoc, uploadedFileId: String) {
        val currentEnrollment = _enrollmentData.value ?: return

        val updatedList = currentEnrollment.currentDependentList.map { dependent ->

            if (dependent.type == data?.type && dependent.index == data?.index) {
                val uploadedList = (dependent.uploadedList ?: mutableListOf()).apply {
                    val index = this.indexOfFirst { it.fileTypeId == currentItem.id }

                    // No upload for the file type; add
                    if (index == -1) {
                        add(UploadedDocument(currentItem.id, uploadedFileId.toInt()))
                    } else {
                        // Upload exists; update
                        this[index] = UploadedDocument(currentItem.id, uploadedFileId.toInt())
                    }
                }
                dependent.copy(uploadedList = uploadedList)
            } else {
                dependent
            }
        }

        val updatedEnrollment = currentEnrollment.copy(
            currentDependentList = ArrayList(updatedList)
        )
        _enrollmentData.value = updatedEnrollment
    }

    /**
     * MOB-3859 If the child has the same ssn as the spouse,
     * clear the ssn and make the child as incomplete
     */
    fun updateDependentDetails(
        type: DependentType,
        childIndex: Int?,
        firstName: String,
        lastName: String,
        dob: String,
        gender: String,
        ssn: String
    ) {
        val currentEnrollment = _enrollmentData.value ?: return
        val updatedList = currentEnrollment.currentDependentList.map { item ->
            when {
                type == DependentType.SPOUSE && item.type == DependentType.SPOUSE ->
                    item.copy(
                        firstName = firstName,
                        lastName = lastName,
                        dob = dob,
                        gender = gender,
                        ssn = ssn
                    )
                type == DependentType.CHILD && item.type == DependentType.CHILD && item.index == childIndex ->
                    item.copy(
                        firstName = firstName,
                        lastName = lastName,
                        dob = dob,
                        gender = gender,
                        ssn = ssn
                    )
                else -> item
            }
        }
        _enrollmentData.value = currentEnrollment.copy(
            currentDependentList = ArrayList(updatedList)
        )
    }

    fun checkChild(ssnDash: String) {
        val ssn = ssnDash.replace("-", "")
        val current = _enrollmentData.value ?: return

        val updatedChildren = current.currentDependentList.map { dependent ->
            if (dependent.type == DependentType.CHILD && dependent.ssn?.replace("-", "") == ssn) {
                // Mark child as incomplete and clear SSN
                dependent.copy(ssn = null, isCompleted = false)
            } else {
                dependent
            }
        }

        val updatedEnrollment = current.copy(
            currentDependentList = ArrayList(updatedChildren)
        )

        _enrollmentData.value = updatedEnrollment
    }

fun Enrollment.toEnrollMember(memId: String): EnrollMember {
    val spouse = this.findSpouse()
    val primary = this.findPrimary()
    val childs = this.findChilds()

    return EnrollMember(
        subscriber = Person(
            memberId = memId,
            healthInsuranceAvailable = if (primary?.previousInsurance != null) primary.previousInsurance.toString() else "", // step1
            signatureFileId = primary?.signatureFileId,
            uploadedDocuments = primary?.uploadedList,
            gender = this.subscriberGender // step 0
        ),
        spouse = spouse?.let {
            Person(
                firstName = it.firstName, // step1
                lastName = it.lastName, // step1
                dateOfMarriage = it.marriageDate, // step2
                gender = it.gender, // step3
                dateOfBirth = it.dob, // step4
                ssn = it.ssn, // step 5
                provideInsurance = it.insurance, // step 6
                selfEmployed = it.employed, // step 7
                employerName = it.employer, // step 7
                healthInsuranceAvailable = it.spouseInsured, // step 8
                signatureFileId = it.signatureFileId,
                uploadedDocuments = it.uploadedList
            )
        },
        children = childs?.map { child ->
            Person(
                firstName = child.firstName, // step 1
                lastName = child.lastName, // step 1
                gender = child.gender, // step 2
                dateOfBirth = child.dob, // step 3
                ssn = child.ssn, // step 4
                selfEmployed = child.employed, // step 5
                provideInsurance = child.spouseInsured,
                uploadedDocuments = child.uploadedList
            )
        }?.toCollection(ArrayList()) ?: arrayListOf()
    )
