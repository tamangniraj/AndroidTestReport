package com.vivekahealth.vivekahealth.ui.enrollVTwo

import com.vivekahealth.vivekahealth.capitalizeName
import java.io.Serializable

data class DependentCheckItem(
    val isCompleted: Boolean? = false,
    val title: String = "",
    val firstName: String? = "",
    val middleName: String? = "",
    val lastName: String? = "",
    val type: DependentType? = null,
    var marriageDate: String? = null,
    var gender: String? = null,
    var ssn: String? = null,
    var dob: String? = null,
    var insurance: String? = null,
    var employed: String? = null,
    var employer: String? = null,
    var spouseInsured: String? = null,
    var index: Int? = null,
    val uploadedList: MutableList<UploadedDocument>? = mutableListOf(),
    val previousInsurance: Boolean? = null,
    val isAffirmation: Boolean? = null,
    val signatureFileId: String? = null
) : Serializable {
    var nameWithTitle = ""
    var uploadedCount: Int ? = null
    fun updateNameWithTitle(details: String, child: String, spouse: String) {
        if (firstName?.isNotEmpty() == true) {
            nameWithTitle = "${firstName.capitalizeName()} $details"
        } else {
            if (type == DependentType.CHILD) {
                nameWithTitle = child
            } else if (type == DependentType.SPOUSE) {
                nameWithTitle = "$spouse $details"
            } else {
                nameWithTitle = title
            }
        }
    }
}
