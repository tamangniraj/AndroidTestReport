package com.vivekahealth.vivekahealth.ui.register.registrationSuccess

import android.content.SharedPreferences
import android.text.Editable
import android.text.TextWatcher
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.register.RegisterUser
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.register.step3.repository.EmailRepository
import com.vivekahealth.vivekahealth.ui.register.step8.repository.RegisterRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Coroutines
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.remote.APIResponse
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import javax.inject.Inject
import kotlinx.coroutines.Job
import org.apache.commons.lang3.StringUtils

class StepRegSuccessViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val emailRepository: EmailRepository,
    private val registerRepository: RegisterRepository
) : ViewModel(), Observable {

    var gson = Gson()
    var user = RegisterUser()
    private lateinit var job: Job
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _emailChanged = MutableLiveData<Event<String>>()

    val emailChanged: LiveData<Event<String>>
        get() = _emailChanged

    private val _changeEnable = MutableLiveData<Event<Unit>>()
    val changeEnable: LiveData<Event<Unit>>
        get() = _changeEnable
    private val _navigateToLogin = MutableLiveData<Event<Unit>>()
    val navigateToLogin: LiveData<Event<Unit>>
        get() = _navigateToLogin

    private val _navigateToReRegister = MutableLiveData<APIResponse<DataResponse>>()
    val navigateToReRegister: LiveData<APIResponse<DataResponse>>
        get() = _navigateToReRegister

    private val _navigateToResend = MutableLiveData<APIResponse<DataResponse>>()
    val navigateToResend: LiveData<APIResponse<DataResponse>>
        get() = _navigateToResend

    private val _navigateToSuccess = MutableLiveData<APIResponse<DataResponse>>()
    val navigateToSuccess: LiveData<APIResponse<DataResponse>>
        get() = _navigateToSuccess
    var email: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.email, null)
        }

    var originalEmail = ""
    var firstTextChange: Boolean = false

    var count = 0

    @get:Bindable
    val emailTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _emailChanged.value = Event(s.toString())
            firstTextChange = true
            count += 1
        }
    }

    var updateEnabled = false

    var Done = false
    var cancel = false
    fun checkEmail() {
        compositeDisposable.add(
            emailRepository.verifyEmail(email)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _navigateToReRegister.value = APIResponse<DataResponse>().loading()
                }
                .subscribe({
//                            val gson = Gson()
//                            val regEdit = preferences.edit();
//
//                            val regString = preferences.getString(Constants.Registration, "")
//
//                            val user = gson.fromJson(regString, RegisterUser::class.java)
//                            user.email = email
//
//                            val userString = gson.toJson(user)
//
//                            regEdit.putString(Constants.Registration, userString)
//                            regEdit.apply()

                    saveEmail()
                    _navigateToReRegister.value = APIResponse<DataResponse>().success(it)
                }, {
                    _navigateToReRegister.value = APIResponse<DataResponse>().error(
                        it.localizedMessage!!
                    )
                })
        )
    }

    fun reSend() {
        compositeDisposable.add(
            registerRepository.reSendEmail(email)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _navigateToResend.value = APIResponse<DataResponse>().loading()
                }
                .subscribe({
                    _navigateToResend.value = APIResponse<DataResponse>().success(it)
                }, {
                    _navigateToResend.value = APIResponse<DataResponse>().error(
                        it.localizedMessage!!
                    )
                })
        )
    }

    fun reRegister() {
        compositeDisposable.add(
            registerRepository.registerUser(user)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _navigateToSuccess.value = APIResponse<DataResponse>().loading()
                }
                .subscribe({
                    originalEmail = email

                    _navigateToSuccess.value = APIResponse<DataResponse>().success(it)
                }, {
                    _navigateToSuccess.value = APIResponse<DataResponse>().error(
                        it.localizedMessage!!
                    )
                })
        )
    }

    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.register, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.register, ""),
                        RegisterUser::class.java
                    )
                }

                firstTextChange = false
            },
            {
                email = user.email.toString()
                originalEmail = email
            }
        )
    }
    fun saveEmail() {
        user.email = email
        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.Registration, regString)
        regEdit.apply()
    }
    fun onBackClicked() {
    }

    fun removeReg() {
        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.Registration, "")
        regEdit.apply()
    }
    fun resendEmail() {
        reSend()
//        user.resendVerification = "true"

//        reRegister()
    }

    fun onOkClicked() {
        removeReg()

        _navigateToLogin.value = Event(Unit)
    }

    fun onUpdateClicked() {
        if (Done) {
            checkEmail()
        } else if (cancel) {
            _changeEnable.value = Event(Unit)
        } else {
            if (!updateEnabled) {
                updateEnabled = true

                _changeEnable.value = Event(Unit)
            } else {
                _changeEnable.value = Event(Unit)
            }
        }
    }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
