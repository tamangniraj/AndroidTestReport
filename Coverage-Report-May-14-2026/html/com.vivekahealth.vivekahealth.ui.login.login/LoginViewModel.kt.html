    fun onCreateClicked() {
        val regEdit = preferences.edit()

        var prevReg = preferences.getString(Constants.Registration, "")

        regEdit.putString(Constants.Registration, "")
        var prevEnroll = preferences.getString(Constants.enroll, "")

        regEdit.putString(Constants.enroll, "")

        var marriageStatus = preferences.getString(Constants.maritalStatus, "")
        preferences.edit().putString(Constants.maritalStatusTemp, marriageStatus).apply()
        preferences.edit().putString(Constants.maritalStatus, "").apply()

        var coverage = preferences.getString(Constants.ContinueCoverage, "")
        regEdit.putBoolean(Constants.continueRegistrationStarted, false)

        regEdit.putString(Constants.ContinueCoverageTemp, coverage)
        regEdit.remove(Constants.ContinueCoverage)

        regEdit.apply()

        _signUp.value = Event(Unit)

    fun removeConfData() {
        if (!preferences.getString(Constants.Registration, "").isNullOrBlank()) {
            user = gson.fromJson(
                preferences.getString(Constants.Registration, ""),
                RegisterUser::class.java
            )

            user.ssn = ""
            user.dob = ""

            val regString = gson.toJson(user)
            val regEdit = preferences.edit()
            regEdit.putString(Constants.Registration, regString)
            regEdit.apply()
        }
        if (!preferences.getString(Constants.enroll, "").isNullOrBlank()) {
            enroll = gson.fromJson(
                preferences.getString(Constants.enroll, ""),
                EnrollMember::class.java
            )

            enroll.spouse!!.ssn = ""
            enroll.spouse!!.dateOfBirth = ""

            for (child in enroll.children.indices) {
                enroll.children[child].ssn = ""
                enroll.children[child].dateOfBirth = ""
            }

            val enrollString = gson.toJson(enroll)

            val editEnroll = preferences.edit()

            editEnroll.putString(Constants.enroll, enrollString)
            editEnroll.apply()
        }
    }

    fun getFcmToken() {
        FirebaseMessaging.getInstance().token.addOnCompleteListener(
            OnCompleteListener { task ->
                if (!task.isSuccessful) {
                    return@OnCompleteListener
                }

                // Get new FCM registration token
                val token = task.result
                preferences.edit().putString("FCM_TOKEN", token).apply()

                // Log and toast
            }
        )
    }

    fun reloadPrevData() {
        if (preferences.getString(Constants.Registration, "")
            .isNullOrBlank() && !preferences.getString(Constants.RegistrationPrev, "")
                .isNullOrBlank()
        ) {
            val regEdit = preferences.edit()

            var prevReg = preferences.getString(Constants.RegistrationPrev, "")
            var coverage = preferences.getString(Constants.ContinueCoverageTemp, "")
            var prevEnroll = preferences.getString(Constants.enrollPrev, "")
            var marriageStatus = preferences.getString(Constants.maritalStatusTemp, "")

            regEdit.putString(Constants.maritalStatus, marriageStatus)
            regEdit.putString(Constants.Registration, prevReg)

            regEdit.putString(Constants.ContinueCoverage, coverage)

            regEdit.putBoolean(Constants.continueRegistrationStarted, true)

            regEdit.putString(Constants.enroll, prevEnroll)
            regEdit.apply()
            removeEnrollPrev()
            removeRegPrev()
        }
        removeConfData()
    }

    fun login() {
        compositeDisposable.add(

            loginRepository.getLogin(userName = email, password = password, "", false)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _logInSuccess.value = APILoginResponse<Response<AuthToken>>().loading()
                }
                .subscribe({
                    if (it.isSuccessful || it.raw().code == 200 || it.raw().code == 400 || it.raw().code == 401 || it.raw().code == 404) {
                        _logInSuccess.value = APILoginResponse<Response<AuthToken>>().success(it)
                    } else if (it.raw().code == 403) {
                        try {
                            val errorResponse: ErrorResponse? = gson.fromJson(
                                it.errorBody()?.string(),
                                ErrorResponse::class.java
                            )
                            errorResponse?.let {
                                _alternativeApp.value = Event(errorResponse)
                            }
                        } catch (e: Exception) {
                            _logInFail.value = Event("Something went wrong, Please try again.")
                        }
                    } else {
                        _logInFail.value = Event(it.raw().code.toString())
                    }
                }, {
                    _logInFail.value = Event(it.message.toString())
                })
        )
    }

    fun magicLogin(magicTempUUID: String) {
        preferences.edit().putString(Constants.MAGIC_TEMP_TOKEN, magicTempUUID).apply()
        compositeDisposable.add(
            loginRepository.getMagicLogin(magicTempUUID)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    preferences.edit().putBoolean(Constants.IS_MAGIC_LOGIN, true).apply()
                    _logInSuccess.value = APILoginResponse<Response<AuthToken>>().loading()
                }
                .subscribe({
                    _logInSuccess.value = APILoginResponse<Response<AuthToken>>().success(it)
                }, {
                    _logInFail.value = Event(it.message.toString())
                })
        )
    }

    fun savePassword() {
        preferences.edit().remove(Constants.INACTIVITY_TIME_STAMP).apply()
        val encoded = HelperFingerAuth.encryptString(password)
        preferences.edit().putString(
            Constants.KEY_PASSWORD,
            encoded
        ).apply()
//        val encodedEmail = HelperFingerAuth.encryptString(email)
        if (preferences.getString(Constants.emailLogin, "").equals(email, true)) {
            preferences.edit().putString(
                Constants.emailLogin,
                email
            ).apply()
        } else {
            preferences.edit().putBoolean(
                Constants.BIO_PERMISSION_REQUESTED,
                false
            ).apply()
            preferences.edit().putString(
                Constants.emailLogin,
                email
            ).apply()
        }
    }
