package com.vivekahealth.vivekahealth.ui.enroll.fragments.legal

import android.content.Intent
import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.fragment.app.DialogFragment
import androidx.lifecycle.ViewModelProvider
import androidx.navigation.fragment.findNavController
import com.github.gcacace.signaturepad.views.SignaturePad
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.databinding.FragmentEnrollmentLegalAffirmationBinding
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseFragment
import com.vivekahealth.vivekahealth.ui.enroll.EnrollActivity
import com.vivekahealth.vivekahealth.ui.enroll.NextClickListener
import com.vivekahealth.vivekahealth.ui.enroll.fragments.enrollmentEnd.SuccessCallback
import com.vivekahealth.vivekahealth.ui.enroll.fragments.enrollmentEnd.SuccessEnrollmentFragment
import com.vivekahealth.vivekahealth.ui.login.LoginActivity
import com.vivekahealth.vivekahealth.utils.*
import com.vivekahealth.vivekahealth.utils.remote.Status
import javax.inject.Inject

class LegalAffirmation : BaseFragment(), Injectable, NextClickListener {

    companion object {
        fun newInstance() = LegalAffirmation()
    }

    private lateinit var viewModel: LegalAffirmationViewModel

    private lateinit var binding: FragmentEnrollmentLegalAffirmationBinding

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory
    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = FragmentEnrollmentLegalAffirmationBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            LegalAffirmationViewModel::class.java
        ).apply {
            nextClickListener = this@LegalAffirmation
        }
        binding.viewModel = viewModel

        binding.firstShow = true
        binding.secondShow = false
        binding.thirdShow = false

        binding.nextEnable = false
        binding.disclaimerText = resources.getString(R.string.disclaimer_legal)
        intSig()
        viewModel.loadData()
        (activity as EnrollActivity).changeState(6)
        (activity as EnrollActivity).focusRight()

        viewModel.filesDir = requireActivity().filesDir

        binding.totalStep = "1"
        viewModel.navigateToBack.observe(
            viewLifecycleOwner,
            EventObserver {
                when {
                    binding.firstShow == true -> {
                        (activity as EnrollActivity).changeState(5)
                        (activity as EnrollActivity).focusLeft()

                        viewModel.spouseDone = false
                        viewModel.memberDone = false
                        findNavController().navigateUp()
                    }
//                binding.secondShow == true -> {
//                    viewModel.currentStep = "1"
//                    binding.firstShow = true
//                    binding.secondShow = false
//                    binding.thirdShow = false
//                    binding.nextEnable = true
//                }
                    binding.thirdShow == true -> {
                        viewModel.currentStep = "1"
//                    binding.firstShow = false
//                    binding.secondShow = true
//                    binding.thirdShow = false
                        binding.checkbox.isChecked = false
                        viewModel.checkBox = false
                        binding.signaturePad.clear()

                        binding.firstShow = true
                        binding.secondShow = false
                        binding.thirdShow = false

                        binding.nextEnable = viewModel.memberDone
                        if (binding.showSpouse == true) {
                            binding.nextEnable = viewModel.spouseDone
                        }
                    }
                }
            }
        )

        viewModel.navigateToEnroll.observe(viewLifecycleOwner) {
            when (it.status) {
                Status.LOADING -> {
                    (activity as EnrollActivity).showLoading(getString(R.string.loading))
                }
                Status.SUCCESS -> {
                    (activity as EnrollActivity).dismissLoading()

                    if (it.data?.code.equals(Constants.SUCCESS_CODE)) {
                        success()
                    } else {
                        DialogUtils(
                            requireActivity(),
                            object : DialogListener {
                                override fun onNeutralClicked() {
                                }

                                override fun onPositiveClicked() {
                                    val intent = Intent(requireContext(), LoginActivity::class.java)
                                    startActivity(intent)
                                    requireActivity().finish()
                                }

                                override fun onNegativeClicked() {
                                }
                            }
                        ).dialog(

                            it.data!!.message,
                            getString(R.string.ok_string),
                            "",
                            ""
                        )
                    }
                }
                Status.ERROR -> {
                    DialogUtils(
                        requireActivity(),
                        object : DialogListener {
                            override fun onNeutralClicked() {
                            }

                            override fun onPositiveClicked() {
                                val intent = Intent(requireContext(), LoginActivity::class.java)
                                startActivity(intent)
                                requireActivity().finish()
                            }

                            override fun onNegativeClicked() {
                            }
                        }
                    ).dialog(

                        it.error.toString(),
                        getString(R.string.ok_string),
                        "",
                        ""
                    )

//                    //Timber.d(it.error)
                }
            }
        }
        viewModel.navigateToUpload.observe(viewLifecycleOwner) {
            when (it.status) {
                Status.LOADING -> {
                    (activity as EnrollActivity).showLoading(getString(R.string.loading))
                }
                Status.SUCCESS -> {
                    viewModel.setDocumentName(it.data?.id.toString(), it.data?.fileName.toString())

                    binding.signaturePad.clear()
                    binding.checkbox.isChecked = false
                    viewModel.checkBox = false

                    binding.firstShow = true
                    binding.secondShow = false
                    binding.thirdShow = false

                    if (viewModel.currentPerson == viewModel.user.subscriber) {
                        binding.memberUploaded = true
                        viewModel.memberDone = true
                    } else {
                        viewModel.spouseDone = true

                        binding.spouseUploaded = true
                    }

                    if (binding.showSpouse == false) {
//                        binding.nextEnable = binding.memberUploaded == true
                        binding.nextEnable = viewModel.memberDone
                    } else {
//                        binding.nextEnable =
//                            binding.memberUploaded == true && binding.spouseUploaded == true

                        binding.nextEnable =
                            viewModel.memberDone && viewModel.spouseDone
                    }

                    if (viewModel.married && viewModel.memberDone && viewModel.spouseDone) {
                        binding.nextEnable = true
                    } else if (!viewModel.married && viewModel.memberDone) {
                        binding.nextEnable = true
                    }
                    (activity as EnrollActivity).dismissLoading()
                }
                Status.ERROR -> {
                    // Timber.d(it.error)
                }
            }
        }

        viewModel.navigateToNext.observe(
            viewLifecycleOwner,
            EventObserver {
                if (binding.memberUploaded == true && binding.spouseUploaded == true) {
                    viewModel.uploadData((activity as EnrollActivity).enrollId())
                } else {
                    if (binding.showSpouse == true) {
                        if (binding.firstShow == true) {
                            viewModel.currentStep = "1"
                            binding.firstShow = false
                            binding.secondShow = true
                            binding.thirdShow = true
                            binding.checkbox.isChecked = false
                            viewModel.checkBox = false
                            binding.nextEnable = false
                        }
//                    else if (binding.secondShow == true) {
//                        viewModel.currentStep = "1"
//                        binding.firstShow = false
//                        binding.secondShow = false
//                        binding.thirdShow = true
//                        binding.nextEnable = false
//
//                    }
                        else if (binding.thirdShow == true) {
                            if (viewModel.signed) {
                                val file = HelperUtils.persistImage(
                                    viewModel.signatureBitmap,
                                    "sign",
                                    requireContext()
                                )
                                viewModel.uploadSign(file)
                            }
                        }
                    } else {
                        if (viewModel.married && viewModel.memberDone && viewModel.spouseDone) {
                            viewModel.uploadData((activity as EnrollActivity).enrollId())
                        } else if (!viewModel.married && viewModel.memberDone) {
                            viewModel.uploadData((activity as EnrollActivity).enrollId())
                        } else {
                            if (binding.thirdShow == true) {
                                if (viewModel.signed) {
                                    val file = HelperUtils.persistImage(
                                        viewModel.signatureBitmap,
                                        "sign",
                                        requireContext()
                                    )
                                    viewModel.uploadSign(file)
                                }
                            }
                        }
                    }
                }
            }
        )

        viewModel.memberClicked.observe(
            viewLifecycleOwner,
            EventObserver {
                viewModel.memberDone = false
                viewModel.signed = false
                binding.memberUploaded = false
                viewModel.currentStep = "1"

                binding.firstShow = false
                binding.secondShow = true
                binding.thirdShow = true
                binding.nextEnable = false
            }
        )

        viewModel.spouseClicked.observe(
            viewLifecycleOwner,
            EventObserver {
                viewModel.spouseDone = false
                viewModel.signed = false
                binding.spouseUploaded = false

                viewModel.currentStep = "1"

                binding.firstShow = false
                binding.secondShow = true
                binding.thirdShow = true
                binding.nextEnable = false
            }
        )

        viewModel.check.observe(
            viewLifecycleOwner,
            EventObserver {
                binding.checkbox.isChecked = it

                binding.nextEnable = it && viewModel.signed
            }
        )
        viewModel.clearClicked.observe(
            viewLifecycleOwner,
            EventObserver {
                binding.signaturePad.clear()
                binding.nextEnable = false
            }
        )

        viewModel.childRetrieved.observe(
            viewLifecycleOwner,
            EventObserver { list ->

                if (viewModel.married) {
                    binding.showSpouse = !list.spouse?.firstName.isNullOrEmpty()
                }
            }
        )
    }

    fun success() {
        val dialog: DialogFragment = SuccessEnrollmentFragment.newInstance()

        dialog.isCancelable = false

        (dialog as SuccessEnrollmentFragment).setSuccessCallback(
            object : SuccessCallback {
                override fun onBackClick() {
                    dialog.dismiss()
                    val intent = Intent(requireActivity(), LoginActivity::class.java)
                    requireActivity().startActivity(intent)
                    requireActivity().finish()
                }

                override fun onOkClick() {
                    dialog.dismiss()

                    val intent = Intent(requireActivity(), LoginActivity::class.java)
                    requireActivity().startActivity(intent)
                    requireActivity().finish()
                }
            },
            false
        )

        dialog.show(parentFragmentManager, "tag")
    }

    private fun intSig() {
        binding.signaturePad.setOnSignedListener(object : SignaturePad.OnSignedListener {
            override fun onStartSigning() {
                // Event triggered when the pad is touched
                viewModel.signed = true
            }

            override fun onSigned() {
                viewModel.signatureBitmap = binding.signaturePad.transparentSignatureBitmap

                binding.nextEnable = viewModel.checkBox && viewModel.signed
                viewModel.checkBoxErrorVisibility = if (viewModel.signed) {
                    if (viewModel.checkBox) View.GONE else View.VISIBLE
                } else {
                    View.GONE
                }
            }

            override fun onClear() {
                viewModel.signed = false

                // Event triggered when the pad is cleared
            }
        })
    }
}
