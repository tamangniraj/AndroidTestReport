    private val _navigateToNext = MutableLiveData<Event<Unit>>()
    val navigateToNext: LiveData<Event<Unit>>
        get() = _navigateToNext

    private val _clearClicked = MutableLiveData<Event<Unit>>()
    val clearClicked: LiveData<Event<Unit>>
        get() = _clearClicked

    private val _navigateToUpload = MutableLiveData<APIResponse<DocumentUploadSuccessEnroll>>()
    val navigateToUpload: LiveData<APIResponse<DocumentUploadSuccessEnroll>>
        get() = _navigateToUpload
    private val _memberClicked = MutableLiveData<Event<Unit>>()
    val memberClicked: LiveData<Event<Unit>>
        get() = _memberClicked
    private val _spouseClicked = MutableLiveData<Event<Unit>>()
    val spouseClicked: LiveData<Event<Unit>>
        get() = _spouseClicked
    private val _childRetrieved = MutableLiveData<Event<EnrollMember>>()
    val childRetrieved: LiveData<Event<EnrollMember>>
        get() = _childRetrieved

    private val _navigateToEnroll = MutableLiveData<APIResponse<DataResponse>>()
    val navigateToEnroll: LiveData<APIResponse<DataResponse>>
        get() = _navigateToEnroll
    lateinit var signatureBitmap: Bitmap

    var signed: Boolean = false
    var currentStep: String = ""

    val check: LiveData<Event<Boolean>>

    var currentNameFirst: String = ""

    private lateinit var job: Job

    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.enroll, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.enroll, ""),
                        EnrollMember::class.java
                    )
                }
            },
            {
                setupList()

                _childRetrieved.value = Event(user)
            }
        )
    }
    fun memberLegalClick() {
        currentNameFirst = user.subscriber?.firstName.toString()
        currentNameLast = user.subscriber?.lastName.toString()
        currentPerson = user.subscriber!!

        _memberClicked.value = Event(Unit)
    }

    fun setupList() {
        memberName = user.subscriber?.lastName + ", " + user.subscriber?.firstName

        spouseName = user.spouse?.lastName + ", " + user.spouse?.firstName
    }
    fun spouseLegalClick() {
        currentNameFirst = user.spouse?.firstName.toString()
        currentNameLast = user.spouse?.lastName.toString()
        currentPerson = user.spouse!!
        _spouseClicked.value = Event(Unit)
    }
    fun onPrevClicked() {
        _navigateToBack.value = Event(Unit)
    }

    fun clearClicked() {
        _clearClicked.value = Event(Unit)
    }
    fun onNextClicked() {
        _navigateToNext.value = Event(Unit)
    }

    fun uploadData(memberID: String) {
        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    enrollRepository.enrollMember(
                        user,
                        memberID
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            _navigateToEnroll.value = APIResponse<DataResponse>().loading()
                            loading = true
                        }
                        .subscribe({
                            _navigateToEnroll.value = APIResponse<DataResponse>().success(it)
                            loading = false
                        }, {
                            _navigateToEnroll.value = APIResponse<DataResponse>().error(
                                it.localizedMessage!!
                            )
                            loading = false
                        })
                )
            }
        )
    }

    fun setDocumentName(id: String, fileName: String) {
        job = Coroutines.ioToMainThread(
            {
                when (currentPerson) {
                    user.subscriber -> {
                        user.subscriber!!.signatureFileId = id
                        user.subscriber!!.signatureFileName = fileName

                        memberDone = true
                    }
                    user.spouse -> {
                        user.spouse!!.signatureFileId = id
                        user.spouse!!.signatureFileName = fileName
                        spouseDone = true
                    }
                }
            },
            {
                val regString = gson.toJson(user)
                val regEdit = preferences.edit()
                regEdit.putString(Constants.enroll, regString)
                regEdit.apply()

//                uploadData()
            }
        )
    }

    fun uploadSign(pic: File) {
        val file: File = File(pic?.path)

        val filePart: MultipartBody.Part = MultipartBody.Part.createFormData(
            "file",

            file.name,
            file.asRequestBody("image/*".toMediaTypeOrNull())
        )

        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    enrollRepository.uploadDoc(
                        filePart,
                        HelperUtils.fileName(Constants.Signature, currentNameFirst, currentNameLast)
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().loading()
                        }
                        .subscribe({
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().success(
                                it
                            )
                        }, {
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                                it.localizedMessage!!
                            )
                        })
                )
            }
        )
    }
    fun persistImage(bitmap: Bitmap, name: String) {
        val filesDir: File = filesDir
        val imageFile = File(filesDir, "$name.png")
        val os: OutputStream
        try {
            os = FileOutputStream(imageFile)
            bitmap.compress(Bitmap.CompressFormat.PNG, 100, os)
            os.flush()
            os.close()
        } catch (e: Exception) {
            e.stackTrace
        }
        uploadSign(imageFile)
    }

    fun ssnRadio() {
        checkBox = !checkBox

        _check.value = Event(checkBox)
        checkBoxErrorVisibility = if (signed) {
            if (checkBox) View.GONE else View.VISIBLE
        } else {
            View.GONE
        }
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
