package com.vivekahealth.vivekahealth.ui.enroll.fragments.legal

import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.lifecycle.ViewModelProvider
import com.vivekahealth.vivekahealth.databinding.FragmentEnrollmentLegalSignatureBinding
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseFragment
import com.vivekahealth.vivekahealth.ui.enroll.NextClickListener
import javax.inject.Inject

class LegalSignature : BaseFragment(), Injectable, NextClickListener {

    companion object {
        fun newInstance() = LegalSignature()
    }

    private lateinit var viewModel: LegalSignatureViewModel

    private lateinit var binding: FragmentEnrollmentLegalSignatureBinding

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = FragmentEnrollmentLegalSignatureBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            LegalSignatureViewModel::class.java
        ).apply {
            nextClickListener = this@LegalSignature
        }
        binding.viewModel = viewModel // TODO: Use the ViewModel
    }
}
