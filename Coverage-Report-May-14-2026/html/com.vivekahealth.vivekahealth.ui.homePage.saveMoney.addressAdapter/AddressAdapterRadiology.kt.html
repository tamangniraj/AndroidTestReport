package com.vivekahealth.vivekahealth.ui.homePage.saveMoney.addressAdapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.*
import com.vivekahealth.vivekahealth.databinding.SingleCellSaveMoneyBinding

class AddressAdapterRadiology() : ListAdapter<Radiologist, RecyclerView.ViewHolder>(
    RadiologistDiffCallback()
) {

    var showAddress: Boolean = false
    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return AddressViewHolder(
            SingleCellSaveMoneyBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            )
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as AddressViewHolder).binding.data = item
        holder.binding.showAddress = false

        holder.binding.setClickListener {
            holder.binding.showAddress = !holder.binding.showAddress
        }
    }

    class AddressViewHolder(
        val binding: SingleCellSaveMoneyBinding
    ) : RecyclerView.ViewHolder(binding.root) {

        init {
        }
    }
}

private class RadiologistDiffCallback : DiffUtil.ItemCallback<Radiologist>() {
    override fun areItemsTheSame(oldItem: Radiologist, newItem: Radiologist): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: Radiologist, newItem: Radiologist): Boolean {
        return oldItem.displayName == newItem.displayName
    }
}
