package com.vivekahealth.vivekahealth.ui.headshots

import com.vivekahealth.vivekahealth.data.local.home.ClientDetailsDao
import com.vivekahealth.vivekahealth.data.local.home.MemberInfoDao
import com.vivekahealth.vivekahealth.data.remote.home.AppInstallUpdate
import com.vivekahealth.vivekahealth.data.remote.home.ClientDetailResponse
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.data.remote.home.SwitcherResponse
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import io.reactivex.Single
import jakarta.inject.Inject
import kotlinx.coroutines.flow.Flow

class HeadShotRepository @Inject constructor(
    private val memberInfoDao: MemberInfoDao,
    private val clientDetailsDao: ClientDetailsDao,
    private val homeRepository: HomeRepository
) {

    fun getMemberInfoFlow(memberId: String): Flow<MemberInfo?> {
        return memberInfoDao.getMemberInfoFlow(memberId)
    }

    fun getClientDetailFlow(memberId: String): Flow<ClientDetailResponse?> {
        return clientDetailsDao.getClientDetailLiveData(memberId)
    }

    fun sendAppInstallUpdateData(
        installInfo: AppInstallUpdate,
        memberId: String
    ): Single<DataResponse> {
        return homeRepository.sendAppInstallUpdateData(installInfo, memberId)
    }

    fun getSwitcher(memberId: String): Single<SwitcherResponse> {
        return homeRepository.getSwitcher(memberId)
    }
}
