package com.vivekahealth.vivekahealth.ui.headshots

import android.content.SharedPreferences
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.vivekahealth.vivekahealth.data.remote.home.AppInstallUpdate
import com.vivekahealth.vivekahealth.data.remote.home.ClientDetailResponse
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.data.remote.home.SwitcherResponse
import com.vivekahealth.vivekahealth.utils.Constants
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import kotlinx.coroutines.ExperimentalCoroutinesApi
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.distinctUntilChanged
import kotlinx.coroutines.flow.filterNotNull
import kotlinx.coroutines.flow.flatMapLatest
import kotlinx.coroutines.flow.flow
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.suspendCancellableCoroutine
import timber.log.Timber
import javax.inject.Inject
import kotlin.coroutines.resume
import kotlin.coroutines.resumeWithException

class HeadShotViewModel @Inject constructor(
    val preferences: SharedPreferences,
    val headShotRepository: HeadShotRepository
) : ViewModel() {

    // Reactive member ID that can be changed
    private val _memberId = MutableStateFlow<String?>(
        preferences.getString(Constants.MEMBER_ID, null)
    )

    val memberId: StateFlow<String?> = _memberId

    var currentTimeInMills = System.currentTimeMillis().toString()

    // Expose repository flows
    @OptIn(ExperimentalCoroutinesApi::class)
    val memberInfo: StateFlow<MemberInfo?> = _memberId
        .filterNotNull()
        .distinctUntilChanged()
        .flatMapLatest { memberId ->
            Timber.d("Loading member info for: $memberId")
            headShotRepository.getMemberInfoFlow(memberId)
        }
        .stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(),
            initialValue = null
        )

    @OptIn(ExperimentalCoroutinesApi::class)
    val switchedUser: StateFlow<SwitcherResponse?> = _memberId
        .filterNotNull()
        .distinctUntilChanged()
        .flatMapLatest { memberId ->
            flow {
                try {
                    Timber.d("Loading switcher data for member: $memberId")
                    val result = suspendCancellableCoroutine<SwitcherResponse> { continuation ->
                        val disposable = headShotRepository.getSwitcher(memberId)
                            .subscribeOn(Schedulers.io())
                            .observeOn(AndroidSchedulers.mainThread())
                            .subscribe(
                                { response ->
                                    continuation.resume(response)
                                },
                                { error ->
                                    continuation.resumeWithException(error)
                                }
                            )

                        continuation.invokeOnCancellation {
                            disposable.dispose()
                        }
                    }
                    emit(result)
                } catch (e: Exception) {
                    Timber.e(e, "Error loading switcher for $memberId")
                    emit(null)
                }
            }
        }
        .stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(),
            initialValue = null
        )

    @OptIn(ExperimentalCoroutinesApi::class)
    val clientDetailsFlow: StateFlow<ClientDetailResponse?> =
        _memberId
            .filterNotNull()
            .distinctUntilChanged()
            .flatMapLatest { memberId ->
                Timber.d("Loading client details for member: $memberId")
                headShotRepository.getClientDetailFlow(memberId)
            }
            .stateIn(
                scope = viewModelScope,
                started = SharingStarted.WhileSubscribed(5000),
                initialValue = null
            )

    private val _dependentsFlow = MutableStateFlow<List<MemberInfo>>(emptyList())
    val dependents: StateFlow<List<MemberInfo>> = _dependentsFlow

    private val _selectedMember = MutableStateFlow<MemberInfo?>(null)
    val selectedMember: StateFlow<MemberInfo?> = _selectedMember

    var actualSelectedId: String? = null

    val headShots: StateFlow<List<MemberInfo>> =
        combine(memberInfo, _selectedMember, switchedUser, clientDetailsFlow) { memberInfo, selected, switchUser, clientDetails ->

            var dependents = memberInfo?.dependentsArrayList ?: emptyList<MemberInfo>()

            val unsortedHeadshots = mutableListOf<MemberInfo>()
            // Auto-select memberInfo if no selection

            if ((actualSelectedId == null || selectedMember.value?.selfId == null) && memberInfo?.selfId != null) {
                actualSelectedId = memberInfo.selfId
                _selectedMember.value = memberInfo
            }

            // Step1: Add insured
            memberInfo?.insured?.let { insured ->
                if (unsortedHeadshots.none { it.selfId == insured.selfId || it.memberRelationship == "insured" }) {
                    unsortedHeadshots.add(
                        insured.copy(isSelected = insured.selfId == actualSelectedId)
                    )
                }
            }

            // Add Self
            if (memberInfo != null) {
                memberInfo.isInsured = false
                unsortedHeadshots.add(
                    memberInfo.copy(isSelected = memberInfo.selfId == actualSelectedId)
                )
            }

            // Add Dependent
            dependents.filter { !it.hideProfileFromHome }.forEach { dependent ->
                if (unsortedHeadshots.none { it.selfId == dependent.selfId }) {
                    unsortedHeadshots.add(
                        dependent.copy(isSelected = dependent.selfId == actualSelectedId)
                    )
                }
            }

            if (clientDetails?.addDependentDisabled != true &&
                (
                    memberInfo?.insured == null || memberInfo.memberRelationship?.equals(
                            "spouse",
                            ignoreCase = true
                        ) == true
                    )
            ) {
                // Your logic here
                unsortedHeadshots.add(
                    MemberInfo(
                        isInsured = memberInfo?.insured == null || memberInfo.relationDesc.equals(
                            "spouse",
                            true
                        ),
                        isDummy = true,
                        isSelected = false
                    )
                )
            }

            if (switchUser?.switcherAvailable == true) {
                unsortedHeadshots.find { it.selfId == switchUser.memberId }?.let {
                    it.relatedId = switchUser.relatedMemberId
                    it.loggedInId = switchUser.memberId
                }
            }

            var sorted = SortingUtils.sortHeadShots((unsortedHeadshots))

            if (selected?.selfId != actualSelectedId && !sorted.map { it.selfId }.contains(
                    actualSelectedId
                )
            ) {
                _selectedMember.value = memberInfo
            }

            return@combine sorted
        }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(), emptyList())

    fun selectHeadShot(memberInfo: MemberInfo) {
        if (!memberInfo.isDummy) {
            actualSelectedId = memberInfo.selfId
            _selectedMember.value = memberInfo
        }
    }

    fun clearHeadShotSelected(meberId: String) {
        actualSelectedId = meberId
        _selectedMember.value = null
    }

    val currentUserAndSelectedUser: StateFlow<Pair<MemberInfo?, MemberInfo?>?> = combine(
        memberInfo,
        selectedMember
    ) { memberInfo, selectedMember ->
        Pair(memberInfo, selectedMember)
    }.stateIn(
        viewModelScope,
        SharingStarted.WhileSubscribed(),
        null
    )

    val clientAndMember: StateFlow<Pair<ClientDetailResponse?, MemberInfo?>?> =
        combine(clientDetailsFlow, memberInfo) { clientDetails, member ->
            Pair(clientDetails, member)
        }.stateIn(
            viewModelScope,
            SharingStarted.WhileSubscribed(),
            null
        )

    fun initiateHome() {
        _memberId.value = null
        val newMemberId = preferences.getString(Constants.MEMBER_ID, null)
        _memberId.value = newMemberId
    }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    var installInfo: AppInstallUpdate = AppInstallUpdate()

    fun callSendAppInstallUpdateData() {
        val isGuestUser = preferences.getBoolean(Constants.IS_GUEST, false)
        if (isGuestUser) {
            return
        }

        compositeDisposable.add(
            headShotRepository.sendAppInstallUpdateData(
                installInfo,
                preferences.getString(Constants.MEMBER_ID, null) ?: ""
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .subscribe({
                }, {
                })
        )
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
