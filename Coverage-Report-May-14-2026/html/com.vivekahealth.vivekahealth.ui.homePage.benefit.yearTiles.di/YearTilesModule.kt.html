package com.vivekahealth.vivekahealth.ui.homePage.benefit.yearTiles.di

import android.content.Context
import android.content.SharedPreferences
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.ui.homePage.benefit.yearTiles.YearTilesFragment
import com.vivekahealth.vivekahealth.ui.homePage.service.HomeService
import dagger.Module
import dagger.Provides
import dagger.android.ContributesAndroidInjector
import okhttp3.OkHttpClient

@Module
abstract class YearTilesModule {

    @ContributesAndroidInjector
    abstract fun contributesYearTilesFragment(): YearTilesFragment

    companion object {
        @JvmStatic
        @Provides
        fun provideYearTilesFragmentService(
            context: Context,
            gson: Gson,
            builder: OkHttpClient.Builder,
            sharedPreferences: SharedPreferences
        ): HomeService = HomeService(
            context = context,
            gson = gson,
            builder = builder,
            preferences = sharedPreferences
        )
    }
}
