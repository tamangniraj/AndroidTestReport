package com.vivekahealth.vivekahealth.ui.homePage.documents.sortSheet

import android.content.SharedPreferences
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.local.home.SortData
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.disposables.CompositeDisposable
import javax.inject.Inject

class SortSheetViewModel @Inject constructor(private val preferences: SharedPreferences) :
    ViewModel(),
    Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    var type = "doc"

    var sortAction: String = ""
    var sortBy: String = ""

    var recentUpdate: String = ""

    private val _navigateToSort = MutableLiveData<Event<SortData>>()
    val navigateToSort: LiveData<Event<SortData>>
        get() = _navigateToSort
    private val _navigateToAsc = MutableLiveData<Event<Unit>>()
    val navigateToAsc: LiveData<Event<Unit>>
        get() = _navigateToAsc

    private val _navigateToCost = MutableLiveData<Event<Unit>>()
    val navigateToCost: LiveData<Event<Unit>>
        get() = _navigateToCost
    private val _navigateToRelevance = MutableLiveData<Event<Unit>>()
    val navigateToRelevance: LiveData<Event<Unit>>
        get() = _navigateToRelevance
    private val _navigateToName = MutableLiveData<Event<Unit>>()
    val navigateToName: LiveData<Event<Unit>>
        get() = _navigateToName

    private val _navigateToDsc = MutableLiveData<Event<Unit>>()
    val navigateToDsc: LiveData<Event<Unit>>
        get() = _navigateToDsc

    private val _navigateToDrug = MutableLiveData<Event<Unit>>()
    val navigateToDrug: LiveData<Event<Unit>>
        get() = _navigateToDrug
    private val _navigateToSubmit = MutableLiveData<Event<Unit>>()
    val navigateToSubmit: LiveData<Event<Unit>>
        get() = _navigateToSubmit
    private val _navigateToRecentUpdate = MutableLiveData<Event<Unit>>()
    val navigateToRecentUpdate: LiveData<Event<Unit>>
        get() = _navigateToRecentUpdate

    private val _navigateToRecentlyAdded = MutableLiveData<Event<Unit>>()
    val navigateToRecentlyAdded: LiveData<Event<Unit>>
        get() = _navigateToRecentlyAdded

    private val _navigateToICD = MutableLiveData<Event<Unit>>()
    val navigateToICD: LiveData<Event<Unit>>
        get() = _navigateToICD

    private val _navigateToRecent = MutableLiveData<Event<Unit>>()
    val navigateToRecent: LiveData<Event<Unit>>
        get() = _navigateToRecent
    private val _navigateToClose = MutableLiveData<Event<Unit>>()
    val navigateToClose: LiveData<Event<Unit>>
        get() = _navigateToClose

    fun loadSetting() {
        if (type.equals("billPreview", true)) {
            if (sortBy.equals("cost", true)) {
                _navigateToCost.value = Event(Unit)
            } else if (sortBy.equals("code", true)) {
                _navigateToName.value = Event(Unit)
            } else if (sortBy.equals("relevance", true)) {
                _navigateToRelevance.value = Event(Unit)
            }
        } else {
            if (sortAction.equals("asc", true)) {
                _navigateToAsc.value = Event(Unit)
            }
            if (sortAction.equals("desc", true)) {
                _navigateToDsc.value = Event(Unit)
            }
            if (sortBy.equals("dateOfService", true)) {
                _navigateToRecent.value = Event(Unit)
            }
            if (sortBy.equals("recentlyAdded", true)) {
                _navigateToRecentlyAdded.value = Event(Unit)
            }

            if (sortBy.equals("icdCodes", true)) {
                _navigateToICD.value = Event(Unit)
            }

            if (sortBy.equals("drugName", true)) {
                _navigateToDrug.value = Event(Unit)
            }
            if (sortBy.equals("submitted", true)) {
                _navigateToSubmit.value = Event(Unit)
            }
            if (sortBy.equals("recentlyUpdated", true)) {
                _navigateToRecentUpdate.value = Event(Unit)
            }
        }
    }

    fun recentDate() {
        sortBy = "recentlyAdded"
        sortAction = "desc"
    }
    fun icd() {
        sortBy = "icdCodes"
        sortAction = "asc"
    }

    fun asc() {
        sortAction = "asc"
    }

    fun showRecentUpdate() {
        sortBy = "recentlyUpdated"
        sortAction = "desc"
    }

    fun submitRecent() {
        sortBy = "submitted"

        sortAction = "desc"
    }

    fun submitOld() {
        sortBy = "submitted"
        sortAction = "asc"
    }

    fun alpha() {
        sortBy = "drugName"
        sortAction = "asc"
    }
    fun dsc() {
        sortAction = "desc"
    }

    fun service() {
        sortBy = "dateOfService"
        sortAction = "desc"
    }

    fun costLow() {
        sortBy = "cost"
        sortAction = "asc"
    }
    fun costHigh() {
        sortBy = "cost"
        sortAction = "desc"
    }
    fun ascAlpha() {
        sortBy = "name"
        sortAction = "asc"
    }
    fun dscAlpha() {
        sortBy = "name"
        sortAction = "desc"
    }
    fun ascCode() {
        sortBy = "code"
        sortAction = "asc"
    }
    fun dscCode() {
        sortBy = "code"
        sortAction = "desc"
    }

    fun dscRelevance() {
        sortBy = "relevance"
        sortAction = ""
    }
    fun apply() {
        _navigateToSort.value = Event(SortData(sortBy, sortAction))
    }
    fun close() {
        _navigateToClose.value = Event(Unit)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
