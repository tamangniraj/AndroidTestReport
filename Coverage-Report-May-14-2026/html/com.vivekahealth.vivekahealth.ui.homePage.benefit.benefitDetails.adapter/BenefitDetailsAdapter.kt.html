package com.vivekahealth.vivekahealth.ui.homePage.benefit.benefitDetails.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.MemberBenefitDto
import com.vivekahealth.vivekahealth.databinding.SingleRowBubbleBenefitDetailsBinding
import timber.log.Timber

class BenefitDetailsAdapter(private var onClick: (MemberBenefitDto) -> Unit) : ListAdapter<MemberBenefitDto, RecyclerView.ViewHolder>(
    BenefitBubbleDiffCallback()
) {

    var selectedPosition: Int = 0

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return MemberViewHolder(
            SingleRowBubbleBenefitDetailsBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick,
            selectedPosition
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val member = getItem(position)

//        (holder as MemberViewHolder).bind(member)

        (holder as MemberViewHolder).binding.setClickListener {
            notifyItemChanged(selectedPosition)
//
//            selectedPosition = position
//
//            holder.binding.clicked = selectedPosition == position

            Timber.d(itemCount.toString())
            onClick(member)
            selectedPosition = position
//            if(position != itemCount-1) {
//                onClick(member)
//            }
//
//            if (position != itemCount - 1) {
//                selectedPosition = position
//            }
            notifyItemChanged(position)
        }

        holder.binding.member = member
//        holder.binding.hasImage = member.hasAvatar.toBooleanStrict()

        holder.binding.clicked = member.isSelected
//                selectedPosition == position

        holder.binding.hasImage = member.selfieAvailable

        holder.binding.isAdd = false
    }

    class MemberViewHolder(
        val binding: SingleRowBubbleBenefitDetailsBinding,
        private val onClick: (MemberBenefitDto) -> Unit,
        private var selectedPosition: Int
    ) : RecyclerView.ViewHolder(binding.root)
}

private class BenefitBubbleDiffCallback : DiffUtil.ItemCallback<MemberBenefitDto>() {

    override fun areItemsTheSame(oldItem: MemberBenefitDto, newItem: MemberBenefitDto): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: MemberBenefitDto, newItem: MemberBenefitDto): Boolean {
        return oldItem.isSelected == newItem.isSelected
    }
}
