package com.vivekahealth.vivekahealth.ui.guestAccNdetails

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import androidx.lifecycle.asLiveData
import com.vivekahealth.vivekahealth.data.remote.guest.GuestUserInfo
import com.vivekahealth.vivekahealth.ui.guestHome.repository.GuestHomeRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Event
import javax.inject.Inject

class GuestAccountDetailsViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val guestHomeRepository: GuestHomeRepository
) : ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private val _navigateToChangeEmail = MutableLiveData<Event<Unit>>()
    val navigateToChangeEmail: LiveData<Event<Unit>>
        get() = _navigateToChangeEmail
    private val _navigateToChangePhone = MutableLiveData<Event<Unit>>()
    val navigateToChangePhone: LiveData<Event<Unit>>
        get() = _navigateToChangePhone

    val guestInfoFromDb: LiveData<GuestUserInfo> = guestHomeRepository.getGuestInfoFromDb(
        preferences.getString(Constants.MEMBER_ID, "") ?: ""
    ).asLiveData()

    var email: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.email, null)
        }

    var phoneNumber: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.phoneNumber, null)
        }

    fun phoneChange() {
        _navigateToChangePhone.value = Event(Unit)
    }

    fun emailChange() {
        _navigateToChangeEmail.value = Event(Unit)
    }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }
}
