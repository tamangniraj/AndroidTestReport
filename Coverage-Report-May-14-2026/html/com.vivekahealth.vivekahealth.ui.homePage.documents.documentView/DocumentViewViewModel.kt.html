package com.vivekahealth.vivekahealth.ui.homePage.documents.documentView

import androidx.lifecycle.MutableLiveData

import com.bumptech.glide.load.engine.GlideException
import com.bumptech.glide.request.RequestListener
import com.bumptech.glide.request.target.Target
import com.bumptech.glide.request.transition.DrawableCrossFadeTransition
import com.vivekahealth.vivekahealth.data.remote.home.Value
import com.vivekahealth.vivekahealth.ui.homePage.documents.repository.DocumentRepository
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers

import java.io.FileOutputStream
import java.io.IOException
import javax.inject.Inject

class DocumentViewViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val documentRepository: DocumentRepository
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack
    private val _navigateToShare = MutableLiveData<Event<File>>()
    val navigateToShare: LiveData<Event<File>>
        get() = _navigateToShare
    var isEob: Boolean = false
    var eobID: String = ""
    private val _navigateToPDFDownloaded = MutableLiveData<Event<ResponseBody>>()
    val navigateToPDFDownloaded: LiveData<Event<ResponseBody>>
        get() = _navigateToPDFDownloaded

    var myFile: File? = null
    var imageBitmap: Bitmap? = null

    var fileName: String = ""
    var myImageFile: File? = null

    var loadedFile: Boolean = false
    var value: Value? = null

    var memberID: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.memberID, null)
        }

    var type: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.type, null)
        }

    var showImage: Boolean = true
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showImage, null)
        }

    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }

    var contentTypePDF = "application/pdf"

    private val _navigateToImageDownloaded = MutableLiveData<Event<Unit>>()
    val navigateToImageDownloaded: LiveData<Event<Unit>>
        get() = _navigateToImageDownloaded
    var typeID: Int = 1
    init {

//    getAttachment()
    }

    fun saveBitmapToFile(
        dir: File?,
        fileName: String?,
        quality: Int
    ): Boolean {
        var type: Bitmap.CompressFormat? = null
        if (value?.contentType.equals("image/jpeg")) {
            type = Bitmap.CompressFormat.JPEG
        } else if (value?.contentType.equals("image/png")) {
            type = Bitmap.CompressFormat.PNG
        }
        var correctedFileName = fileName
        val docExtRegEx = Regex("(\\.(pdf|jpg|png))(?!.+)")
        val hasExtension = docExtRegEx.containsMatchIn(fileName!!)
        if (!hasExtension) {
            val ext = type!!.name.lowercase()
            correctedFileName = "$fileName.$ext"
        }
        val format: Bitmap.CompressFormat? = type
        myImageFile = File(dir, correctedFileName!!)
        var fos: FileOutputStream? = null
        try {
            fos = FileOutputStream(myImageFile)
            imageBitmap!!.compress(format!!, quality, fos)
            fos.close()
            return true
        } catch (e: IOException) {
            if (fos != null) {
                try {
                    fos.close()
                } catch (e1: IOException) {
                    e1.printStackTrace()
                }
            }
        }
        return false
    }

    fun getAttachment() {
        if (value?.contentType.equals("image/jpeg", true) || value?.contentType.equals(
                "image/png",
                true
            )
        ) {
            showImage = true

            value?.selfId?.let {
                memberID = it
            }

            loading = true

            value?.let {
                if (it.type.equals("COMMUNICATION", true)) {
                    type = "downloadCommunication"
                    typeID = 2
                } else if (it.type.equals("ATTACHMENT", true)) {
                    type = "downloadAttachment"
                    typeID = 1
                }
            }
        } else {
            showImage = false

            loading = true

            value?.selfId?.let {
                memberID = it
            }

            value?.let {
                if (it.type.equals("COMMUNICATION", true)) {
                    type = "downloadCommunication"
                    typeID = 2
                } else if (it.type.equals("ATTACHMENT", true)) {
                    type = "downloadAttachment"
                    typeID = 1
                }
            }

            var url: String = "communication/$memberID/$type"

            getPDF(url)
        }
    }

    @get:Bindable
    val listener = object : RequestListener<Drawable> {
        override fun onLoadFailed(
            e: GlideException?,
            model: Any?,
            target: Target<Drawable>?,
            isFirstResource: Boolean
        ): Boolean {
            loading = false

            return false
        }

        override fun onResourceReady(
            resource: Drawable?,
            model: Any?,
            target: Target<Drawable>?,
            dataSource: DataSource?,
            isFirstResource: Boolean
        ): Boolean {
            loading = false

            imageBitmap = (resource as BitmapDrawable).bitmap
            target?.onResourceReady(
                resource,
                DrawableCrossFadeTransition(100, isFirstResource)
            )
            logView()
            _navigateToImageDownloaded.value = Event(Unit)

            return true
        }
    }

    fun logView() {
        compositeDisposable.add(
            documentRepository.logDocumentView(
                preferences.getString("memberID", "").toString(),
                value!!.selfId.toString(),
                typeID,
                value!!.referencedMemberId.toString()
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                }.subscribe({
                }, {
                })
        )
    }

    fun getPDF(url: String) {
        compositeDisposable.add(
            documentRepository.getPDF(url)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({
                    loading = false

                    _navigateToPDFDownloaded.value = Event(it)
                }, {
                    loading = false
                })
        )
    }

    fun onCloseClicked() {
        _navigateToBack.value = Event(Unit)
    }

    fun onShareClicked() {
        var file: File? = null
        if (!isEob) {
            if (value?.contentType?.contains("image") == true) {
                file = myImageFile
            } else {
                file = myFile
            }
        } else {
            file = myFile
        }

        file?.let {
            _navigateToShare.value = Event(file)
        }
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
