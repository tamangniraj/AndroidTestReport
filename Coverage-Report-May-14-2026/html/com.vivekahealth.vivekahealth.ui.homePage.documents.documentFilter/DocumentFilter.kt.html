package com.vivekahealth.vivekahealth.ui.homePage.documents.documentFilter

import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.lifecycle.ViewModelProvider
import androidx.recyclerview.widget.RecyclerView
import androidx.recyclerview.widget.SimpleItemAnimator
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.databinding.DocumentFilterFragmentBinding
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseDialogFragment
import com.vivekahealth.vivekahealth.ui.homePage.documents.documentFilter.adapter.DocumentFilterAdapter
import com.vivekahealth.vivekahealth.utils.EventObserver
import java.util.ArrayList
import javax.inject.Inject

class DocumentFilter : BaseDialogFragment(), Injectable {

    companion object {
        fun newInstance() = DocumentFilter()
        const val TAG = "DocumentFilter"
    }
//    var successCallbackFrag: SuccessCallback? = null
//    fun setSuccessCallback(successCallback: SuccessCallback) {
//        this.successCallbackFrag = successCallback
//
//    }

    var filterListener: FilterListener? = null

    fun setCallback(filterListener: FilterListener) {
        this.filterListener = filterListener
    }
    private lateinit var binding: DocumentFilterFragmentBinding

    private lateinit var viewModel: DocumentFilterViewModel
    var sentfilter = false

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory

    private val adapter = DocumentFilterAdapter(
        onClick = { memberInfo ->
            viewModel.setSelectedMember(memberInfo)
        }
    )

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = DocumentFilterFragmentBinding.inflate(inflater, container, false)

        return binding.root
    }
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setStyle(STYLE_NORMAL, R.style.FullscreenDialogTheme)
    }

    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            DocumentFilterViewModel::class.java
        )

        binding.viewModel = viewModel

        binding.dependentView.adapter = adapter
        val sent = arguments?.getBoolean("sentSelected")
        val receive = arguments?.getBoolean("receiveSelected")
        val open = arguments?.getBoolean("openSelected")
        val unOpen = arguments?.getBoolean("unOpenSelected")
        val memberList: ArrayList<MemberInfo> = Gson().fromJson(
            arguments?.getString("memberList"),
            object : TypeToken<ArrayList<MemberInfo>>() {}.type
        )
        // load list
        viewModel.loadList(memberList)

        if (sent != null) {
            viewModel.sentSelected = sent
        }
        if (receive != null) {
            viewModel.receiveSelected = receive
            viewModel.showHideOpenCloseMail()
        }
        if (open != null) {
            viewModel.openSelected = open
        }
        if (unOpen != null) {
            viewModel.unOpenSelected = unOpen
        }

        val animator: RecyclerView.ItemAnimator? = binding.dependentView.itemAnimator
        if (animator is SimpleItemAnimator) {
            (animator as SimpleItemAnimator).supportsChangeAnimations = false
        }
//        viewModel.memberInfoDb.observe(viewLifecycleOwner, Observer{
// //            viewModel.memberInfo = it
//            viewModel.loadData(it)
//
//        })
        viewModel.dependentListGenerated.observe(
            viewLifecycleOwner,
            EventObserver {
                adapter.submitList(it)
            }
        )

        // document filter dismiss
        viewModel.navigateToDoc.observe(
            viewLifecycleOwner,
            EventObserver {
                filterListener?.onClose()
            }
        )
        viewModel.navigateToDocFilter.observe(
            viewLifecycleOwner,
            EventObserver {
                filterListener?.onFilter(
                    viewModel.sentSelected,
                    viewModel.receiveSelected,
                    viewModel.openSelected,
                    viewModel.unOpenSelected,
                    viewModel.dependentMemberList
                )
            }
        )
    }
}
