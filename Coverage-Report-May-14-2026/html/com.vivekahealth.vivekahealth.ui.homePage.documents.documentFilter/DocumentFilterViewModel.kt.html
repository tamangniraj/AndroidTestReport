    fun loadList(memList: ArrayList<MemberInfo>) {
        dependentMemberList.clear()

        dependentMemberList = memList
        listForDependentAdapter = dependentMemberList.toList()

        _dependentListGenerated.value = Event(listForDependentAdapter)
    }
    fun String.capitalize(): String {
        return this.replaceFirstChar {
            if (it.isLowerCase()) {
                it.titlecase(Locale.getDefault())
            } else {
                it.toString()
            }
        }
    }

    fun applyFilter() {
        _navigateToDocFilter.value = Event(Unit)
    }
    fun applyButtonEnabled() {
        applyEnabled = hasChangedUO || hasChangeSR || hasChangedMem
    }
    fun onCrossClicked() {
        _navigateToDoc.value = Event(Unit)
    }
    fun showHideOpenCloseMail() {
        if (receiveSelected) {
            hideOpenCloseMail = false
        } else {
            openSelected = false
            unOpenSelected = false
            hideOpenCloseMail = sentSelected
        }
    }

    fun onSentClicked() {
        hasChangeSR = true
        sentSelected = !sentSelected

        showHideOpenCloseMail()

        applyButtonEnabled()
    }
    fun onReceiveClicked() {
        hasChangeSR = true
        receiveSelected = !receiveSelected
        showHideOpenCloseMail()

        applyButtonEnabled()
    }
    fun onOpenClicked() {
        hasChangedUO = true

        openSelected = !openSelected
        if (openSelected) {
            receiveSelected = true
        }
        applyButtonEnabled()
    }
    fun onUnOpenClicked() {
        hasChangedUO = true
        unOpenSelected = !unOpenSelected
        if (unOpenSelected) {
            receiveSelected = true
        }
        applyButtonEnabled()
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
