package com.vivekahealth.vivekahealth.ui.enrollVTwo.child.ssn

import android.content.SharedPreferences
import android.content.res.Resources
import android.text.Editable
import android.text.TextWatcher
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.remote.APIResponse
import io.reactivex.disposables.CompositeDisposable
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.launch
import javax.inject.Inject

class ChildSSNViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val resources: Resources
) : ViewModel(), Observable {

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack

    private val _navigateToNext = MutableLiveData<Event<Unit>>()
    val navigateToNext: LiveData<Event<Unit>>
        get() = _navigateToNext

    var gson = Gson()
    private lateinit var job: Job
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    val showError: LiveData<APIResponse<DataResponse>>
        get() = _showError
    private val _showError = MutableLiveData<APIResponse<DataResponse>>()

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun onPrevClicked() {
        singleClick {
            _navigateToBack.value = Event(Unit)
        }
    }

    fun onNextClicked() {
        singleClick {
            _navigateToNext.value = Event(Unit) // Process the first click immediately
        }
    }

    private var debounceJob: Job? = null
    private var isDebouncing = false
    private val debounceInterval = 500L // Set your debounce interval in milliseconds
    var ssnOptional = false

    fun singleClick(action: () -> Unit) {
        if (isDebouncing) return
        action()
        isDebouncing = true
        debounceJob?.cancel()
        debounceJob = viewModelScope.launch {
            delay(debounceInterval)
            isDebouncing = false
        }
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }

    fun ssnPrevious(ssn: String?, fname: String?) {
        fname?.let {
            firstName = it
        }

        ssn?.let {
            _ssnChangedPrev.value = Event(ssn.toString())
        }

        _ssnChanged.value = Event(ssn ?: "")
    }

    var firstName: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.firstName, null)
        }

    private val _ssnChanged = MutableLiveData<Event<String>>()
    val ssnChanged: LiveData<Event<String>>
        get() = _ssnChanged

    var ssn: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.ssn, null)
        }

    var noSsnCard: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.noSsnCard, null)
        }

    private val _ssnChangedPrev = MutableLiveData<Event<String>>()
    val ssnChangedPrev: LiveData<Event<String>>
        get() = _ssnChangedPrev

    @get:Bindable
    val ssnTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
            _ssnChanged.value = Event(s.toString())
        }
    }

    var errorMessageSsn: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.errorMessageSsn, null)
        }

    var errorMessageAlert: String = resources.getString(
        R.string.once_the_dependent_reaches_12_months_of_age_and_you_will_be_required_to_submit_a_valid_ssn
    )

        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.errorMessageAlert, null)
        }
}
