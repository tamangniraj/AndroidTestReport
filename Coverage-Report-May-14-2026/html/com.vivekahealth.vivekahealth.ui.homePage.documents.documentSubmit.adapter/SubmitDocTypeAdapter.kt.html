package com.vivekahealth.vivekahealth.ui.homePage.documents.documentSubmit.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.Dto
import com.vivekahealth.vivekahealth.databinding.DoctypeCellBinding
import java.util.*

class SubmitDocTypeAdapter(private var onClick: (Dto) -> Unit) : ListAdapter<Dto, RecyclerView.ViewHolder>(
    TypeListDiffCallback()
) {

    var selectedPosition: Int = -1
    var lastSelected: Int = -1
    var count: Int = 0
    var currentYearPos: Int = 0
    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return TypeViewHolder(
            DoctypeCellBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick,
            selectedPosition
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as TypeViewHolder).binding.setClickListener {
            onClick(item)
            notifyItemChanged(selectedPosition)

            selectedPosition = position

            notifyItemChanged(position)
        }
        holder.binding.doc = item
    }

    class TypeViewHolder(
        val binding: DoctypeCellBinding,
        private val onClick: (Dto) -> Unit,
        private var selectedPosition: Int
    ) : RecyclerView.ViewHolder(binding.root)
}

private class TypeListDiffCallback : DiffUtil.ItemCallback<Dto>() {

    override fun areItemsTheSame(oldItem: Dto, newItem: Dto): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: Dto, newItem: Dto): Boolean {
        return oldItem.isSelected == newItem.isSelected
    }
}
