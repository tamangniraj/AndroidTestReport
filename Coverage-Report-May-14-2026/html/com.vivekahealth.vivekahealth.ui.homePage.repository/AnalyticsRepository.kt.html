package com.vivekahealth.vivekahealth.ui.homePage.repository

import com.vivekahealth.vivekahealth.data.remote.home.LocationData
import com.vivekahealth.vivekahealth.data.remote.home.insuranceCard.LocationDataInsurance
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.homePage.service.HomeService
import com.vivekahealth.vivekahealth.utils.thread.AppExecutors
import io.reactivex.Single
import okhttp3.ResponseBody
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class AnalyticsRepository @Inject constructor(
    private val appExecutors: AppExecutors,
    private val homeService: HomeService
) {
    fun getLogPageView(memberId: String, pageName: String, sourcePage: String? = null): Single<ResponseBody> = homeService.getHomeApi().logScreenViews(
        memberId,
        pageName,
        sourcePage
    )

    fun logInsuranceCardOpen(memberId: String, locationDataInsurance: LocationDataInsurance): Single<DataResponse> = homeService.getHomeApi().logInsuranceCardOpen(
        memberId,
        locationDataInsurance
    )
    fun logAppOpen(memberId: String, locationData: LocationData?): Single<DataResponse> = homeService.getHomeApi().logAppOpen(
        memberId,
        locationData
    )
}
