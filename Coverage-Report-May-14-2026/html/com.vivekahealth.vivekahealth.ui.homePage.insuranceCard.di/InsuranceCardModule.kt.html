package com.vivekahealth.vivekahealth.ui.homePage.insuranceCard.di

import android.content.SharedPreferences
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.ui.homePage.benefit.service.BenefitWorkService
import com.vivekahealth.vivekahealth.ui.homePage.insuranceCard.InsuranceCardsFragment
import dagger.Module
import dagger.Provides
import dagger.android.ContributesAndroidInjector
import okhttp3.OkHttpClient

@Module
abstract class InsuranceCardModule {

    @ContributesAndroidInjector
    abstract fun contributesInsuranceCardsFragment(): InsuranceCardsFragment

    companion object {
        @JvmStatic
        @Provides
        fun provideInsuranceCardsService(
            gson: Gson,
            builder: OkHttpClient.Builder,
            sharedPreferences: SharedPreferences
        ): BenefitWorkService = BenefitWorkService(
            gson = gson,
            builder = builder,
            preferences = sharedPreferences
        )
    }
}
