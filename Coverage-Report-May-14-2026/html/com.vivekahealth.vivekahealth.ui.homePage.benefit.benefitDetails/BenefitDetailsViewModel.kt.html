    fun chatCounter() {
        compositeDisposable.add(
            homeRepository.getChat("new", null, null, null, null, false)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                }
                .subscribe({
                    Timber.d(it.toString())
                    if (!it.values.isNullOrEmpty()) {
                        chat = it.values!!.count()
                        _chatCount.value = Event(it.values!!.count())
                    } else {
                        chat = 0

                        _chatCount.value = Event(0)
                    }
                }, {
                    Timber.d(it.toString())
                })
        )
    }

    fun setSelectedMember(memberInfo: MemberBenefitDto) {
        val index = bubbleMemberList.indexOf(memberInfo)

        Timber.tag("Benefit").d("Selected Member ---> $index")

        for (item in bubbleMemberList) {
            if (bubbleMemberList.indexOf(item) == index) {
                bubbleMemberList[index].apply {
                    isSelected = true
                    if (item.inEligibilityReasons.isNullOrBlank()) {
                        noData = false
                        reason = ""
                    } else {
                        noData = true
                        reason = item.inEligibilityReasons
                    }
                }
            } else {
                bubbleMemberList[bubbleMemberList.indexOf(item)].apply {
                    isSelected = false
                }
            }
        }
        val updated = bubbleMemberList[index]
        bubbleMemberList[index] = updated
        listForHomeBubbleAdapter = bubbleMemberList.toList()
        _bubbleListGeneratedNew.value = Event(listForHomeBubbleAdapter)
    }

    fun barInfoLoad(selfId: String) {
        compositeDisposable.add(
            benefitWorkRepository.getBarInfo(
                selfId,
                false,
                year.toInt(),
                month.toInt()
            )
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                    loading = true
                }.subscribe({ value ->
                    barInfo = value.values?.find {
                        it.name?.contains(benefit) == true
                    }
                    Timber.d(barInfo.toString())
                    if (barInfo != null) {
                        _barInfoLoaded.value = Event(barInfo)
                    } else {
                        outGone = true
                        inGone = true
                    }
                    loading = false
                }, {
                    Timber.d(it.toString())
                    loading = false
                })
        )
    }

    fun loadContractDetails(selfId: String) {
        compositeDisposable.add(
            benefitWorkRepository.getContractDetails(
                selfId,
                year.toInt(),
                month.toInt(),
                benefitId,
                contractId
            ).subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                loading = true
            }.subscribe({
                if (!it.contracts.isNullOrEmpty()) {
                    val item = it.contracts[0]

                    if (item.benefits[0].planExpenses != null) {
                        showDataView = true

                        if (!item.benefits[0].planExpenses!!.expenses.isNullOrEmpty()) {
                            val expenses = item.benefits[0].planExpenses!!.expenses[0]

                            expenses.copayAmount?.let { value ->
                                copayAmount = value
                            }
                            expenses.coinsuranceAmount?.let { value ->
                                coInsuranceAmount = value
                            }
                            expenses.deductibleAmount?.let { value ->
                                deductiblePayedAmount = value
                            }
                            expenses.deductibleMaxAmount?.let { value ->
                                maxDeductibleAmount = value
                            }
                            expenses.outOfPocketAmount?.let { value ->
                                outOfPocketAmount = value
                            }
                            expenses.outOfPocketMaxAmount?.let { value ->
                                maxOutOfPocketAmount = value
                            }
                            expenses.oonBalanceAmount?.let { value ->
                                outOfNetworkAmount = value
                            }
                            expenses.memberResponsibleAmount?.let { value ->
                                totalMemberAmount = value
                            }
                            expenses.planResponsibleAmount?.let { value ->
                                totalPlanAmount = value
                            }

                            val toggle = item.benefits[0]

                            copayAmountShow = toggle.showCopay
                            coInsuranceAmountShow = toggle.showCoinsurance
                            outOfNetworkAmountShow = toggle.showOONBalance
                            totalMemberAmountShow = toggle.showMemberResponsible
                            totalPlanAmountShow = toggle.showPlanResponsible
                            barDeductibleShow = toggle.showDeductibleBar
                            barOppShow = true
                        }
                    } else {
                        showDataView = false
                    }
                }
                loading = false

                if (!it.contracts.isNullOrEmpty()) {
                    if (!it.contracts[0].benefits[0].insuranceCardType.isNullOrEmpty()) {
                        loadInsuranceCard(selfId, it.contracts[0].benefits[0].insuranceCardType)
                    }
                }
                barInfoLoad(selfId)
            }, {
                loading = false
            })
        )
    }

    }

    fun loadMembers(memberBenefitDto: List<MemberBenefitDto>) {
        memberBenefitDto.let { listItem ->

            var grouped = listItem.groupBy { value -> value.relation }
            grouped.forEach {
                it.value.sortedByDescending { item ->
                    DateUtils.stringServerDateToDate(
                        item.dateOfBirth.toString()
                    )
                }
                it.value.reversed()
            }

            bubbleMemberList.clear()

            grouped.forEach {
                if (it.key.equals("Insured", true)) {
                    memberInfo = it.value.first()
                    it.value.first().isSelected = true
                    bubbleMemberList.add(0, it.value[0])
                } else if (it.key.equals("Spouse", true)) {
                    if (grouped.containsKey("Insured")) {
                        bubbleMemberList.add(1, it.value[0])
                    } else {
                        bubbleMemberList.add(0, it.value[0])
                    }
                } else if (it.key.equals("Child", true)) {
                    val dateGroup = it.value.groupBy { date ->
                        DateUtils.stringServerDateToDate(
                            date.dateOfBirth.toString()
                        )
                    }

                    dateGroup.forEach { item ->
                        val sorted = item.value.sortedBy { item1 -> item1.firstName }
                        bubbleMemberList.addAll(sorted)
                    }
                }
            }
            val tempOthers: ArrayList<MemberBenefitDto> = arrayListOf()
            grouped.forEach {
                if (!it.key.equals("Insured", true) && !it.key.equals(
                        "Spouse",
                        true
                    ) && !it.key.equals("Child", true)
                ) {
                    tempOthers.addAll(it.value)
                }
            }

            val dateGroup = tempOthers.groupBy { date ->
                DateUtils.stringServerDateToDate(
                    date.dateOfBirth.toString()
                )
            }
            dateGroup.forEach { item ->
                val sorted = item.value.sortedBy { item1 -> item1.firstName }
                bubbleMemberList.addAll(sorted)
            }
        }

        listForHomeBubbleAdapter = bubbleMemberList.toList()
        _bubbleListGenerated.value = Event(listForHomeBubbleAdapter)
    }
