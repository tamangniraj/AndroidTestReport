package com.vivekahealth.vivekahealth.ui.homePage.findCare.billPreview.enterValue

import android.content.SharedPreferences
import android.content.res.Resources
import android.text.Editable
import android.text.TextWatcher
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.data.remote.home.findCare.FindCareFilterModel
import com.vivekahealth.vivekahealth.ui.enroll.repository.EnrollRepository
import com.vivekahealth.vivekahealth.data.remote.home.findCare.doctorModelsNew.FeeSchedule
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.repository.FindCareSearchRepository
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import javax.inject.Inject

class PriceEnterViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val enrollRepository: EnrollRepository,
    private val findCareSearchRepository: FindCareSearchRepository,
    private val resources: Resources
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack
    private val _navigateToAck = MutableLiveData<Event<Unit>>()
    val navigateToAck: LiveData<Event<Unit>>
        get() = _navigateToAck
    private val _navigateToOk = MutableLiveData<Event<Unit>>()
    val navigateToOK: LiveData<Event<Unit>>
        get() = _navigateToOk
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    var minPrice: Double? = null
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.minPrice, null)
        }
    var maxPrice: Double? = null
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.maxPrice, null)
        }
    var filterOption: FindCareFilterModel = FindCareFilterModel()

    fun onNextClicked() {
        _navigateToOk.value = Event(Unit)
    }
    var price: String = ""
    private val _nameChanged = MutableLiveData<Event<String>>()
    val nameChanged: LiveData<Event<String>>
        get() = _nameChanged
    var item: FeeSchedule? = null
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.item, null)
        }
    var tele: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.tele, null)
        }
    var errorMessage: String =
        resources.getString(R.string.you_must_enter_a_price_for_this_procedure_to_be_considered)
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.errorMessage, null)
        }

    private val _showPrice = MutableLiveData<Event<Unit>>()
    val showPrice: LiveData<Event<Unit>>
        get() = _showPrice
    var showError: Boolean = true
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showError, null)
        }

    @get:Bindable
    val nameTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
            price = s.toString()
            _nameChanged.value = Event(s.toString())
        }
    }
    fun onBackClicked() {
        _navigateToBack.value = Event(Unit)
    }

    fun getSpecialityList() {
        compositeDisposable.add(
            findCareSearchRepository.getProcedureList(
                item?.codeValue!!,
                preferences.getString("memberID", "").toString(),
                filterOption.current?.latitude.toString(),
                filterOption.current?.longitude.toString(),
                filterOption.providerType.toString(),
                filterOption.radius.toString(),
                searchScope = "procedures"
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->

                    loading = false

                    val item = it.procedures.find { it.code == item?.codeValue }
                    minPrice = item?.minPrice
                    maxPrice = item?.maxPrice
                    _showPrice.value = Event(Unit)
                }, {
                    loading = false
                })
        )
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
