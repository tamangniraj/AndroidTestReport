@BindingAdapter("clientId", "imageFromUrl", "loadListener", requireAll = false)
fun bindImageFromUrl(
    view: ImageView,
    clientId: String?,
    imageUrl: String?,
    loadListener: GlideDrawableLoadListener?
) {
    val pref: SharedPreferences = PreferenceManager.getDefaultSharedPreferences(view.context)
    if (!Constants.IS_OFFLINE) {
        if (!imageUrl.isNullOrEmpty() && !clientId.isNullOrBlank()) {
            if (!pref.getString("uri$clientId", "").isNullOrBlank()) {
                Glide.with(view.context)
                    .load(pref.getString("uri$clientId", ""))
                    .diskCacheStrategy(DiskCacheStrategy.NONE)
                    .listener(loadListener)
                    .into(view)
            }
            Glide.with(view.context)
                .load(imageUrl)
                .transition(DrawableTransitionOptions.withCrossFade())
                .diskCacheStrategy(DiskCacheStrategy.AUTOMATIC)
                .listener(loadListener)
                .into(object : CustomTarget<Drawable?>() {
                    override fun onResourceReady(
                        @NonNull resource: Drawable,
                        @Nullable transition: Transition<in Drawable?>?
                    ) {
                        val bitmap = (resource as BitmapDrawable).bitmap
                        view.setImageBitmap(bitmap)
                        HelperUtils.saveImage(view.context, bitmap, clientId, pref)
                    }

                    override fun onLoadCleared(@Nullable placeholder: Drawable?) {}
                    override fun onLoadFailed(@Nullable errorDrawable: Drawable?) {
                    }
                })
        }
    } else {
        if (!pref.getString("uri$clientId", "").isNullOrBlank()) {
            Glide.with(view.context)
                .load(pref.getString("uri$clientId", ""))
                .diskCacheStrategy(DiskCacheStrategy.NONE)
                .listener(loadListener)
                .into(view)
        }
    }
}

@BindingAdapter("memberIdDoc", "typeDoc", "loadListenerGlide", requireAll = true)
fun bindImageDoc(
    view: ImageView,
    memberIdDoc: String?,
    typeDoc: String,
    loadListenerGlide: RequestListener<Drawable>
) {
    val pref: SharedPreferences = PreferenceManager.getDefaultSharedPreferences(view.context)
    if (!memberIdDoc.isNullOrEmpty()) {
        var glideUrl: GlideUrl = GlideUrl(

            BuildConfig.SERVER_URL + "communication/$memberIdDoc/$typeDoc",
            LazyHeaders.Builder()
                .addHeader("Authorization", HeaderUtils.createApisHeader(pref))
                .addHeader("Device-Id", Constants.deviceUUID)
                .addHeader("App-Version", Constants.versionNumber)
                .addHeader("Device-Info", Constants.deviceID)
                .addHeader("Client-ID", Constants.clientID)
                .build()
        )
        Glide.get(view.context).clearMemory()
        Glide.with(view.context)
            .load(glideUrl)
            .diskCacheStrategy(DiskCacheStrategy.AUTOMATIC)
            .signature(ObjectKey(System.currentTimeMillis()))
            .listener(loadListenerGlide)
            .into(view)
    }
}

@BindingAdapter("memberId", "loadListener", requireAll = false)
fun bindImageFromAuthUrl(
    view: CircleImageView,
    memberId: String?,
    loadListener: GlideDrawableLoadListener?
) {
    if (memberId.isNullOrEmpty()) {
        view.setImageResource(R.drawable.ic_head_placeholder)
        return
    }

    val context = view.context
    val pref = PreferenceManager.getDefaultSharedPreferences(context)
    val glideUrl = GlideUrl(
        "${BuildConfig.SERVER_URL}member/$memberId/getAvatar?v=${System.currentTimeMillis()}",
        LazyHeaders.Builder()
            .addHeader("Authorization", HeaderUtils.createApisHeader(pref))
            .addHeader("App-Version", Constants.versionNumber)
            .addHeader("Device-Id", Constants.deviceUUID)
            .addHeader("Device-Info", Constants.deviceID)
            .addHeader("Client-ID", Constants.clientID)
            .build()
    )

    Glide.with(context)
        .load(glideUrl)
        .placeholder(R.drawable.ic_head_progress)
        .error(R.drawable.ic_head_placeholder)
        .diskCacheStrategy(DiskCacheStrategy.AUTOMATIC)
        .into(view)
}

@BindingAdapter("memberId", "currentTime", "loadListener", requireAll = false)
fun bindImageFromAuthUrl(
    view: CircleImageView,
    memberId: String?,
    currentTime: String?,
    loadListener: GlideDrawableLoadListener?
) {
    if (memberId.isNullOrEmpty()) {
        view.setImageResource(R.drawable.ic_head_placeholder)
        return
    }

    val context = view.context
    val pref = PreferenceManager.getDefaultSharedPreferences(context)

    if (Constants.IS_OFFLINE) {
        Glide.with(view.context)
            .load(R.drawable.ic_head_progress)
            .into(view)

        val uri = pref.getString("uri$memberId", "")

        Glide.with(view.context)
            .load(uri)
            .placeholder(R.drawable.ic_head_progress)
            .error(R.drawable.ic_head_placeholder)
            .diskCacheStrategy(DiskCacheStrategy.NONE)
            .listener(loadListener)
            .into(view)
        return
    }

    // this is for place holder
    Glide.with(view.context)
        .load(R.drawable.ic_head_progress)
        .into(view)

    val glideUrl = GlideUrl(
        "${BuildConfig.SERVER_URL}member/$memberId/getAvatar?v=$currentTime",
        LazyHeaders.Builder()
            .addHeader("Authorization", HeaderUtils.createApisHeader(pref))
            .addHeader("App-Version", Constants.versionNumber)
            .addHeader("Device-Id", Constants.deviceUUID)
            .addHeader("Device-Info", Constants.deviceID)
            .addHeader("Client-ID", Constants.clientID)
            .build()
    )

    Glide.with(context)
        .load(glideUrl)
        .placeholder(R.drawable.ic_head_progress)
        .error(R.drawable.ic_head_placeholder)
        .diskCacheStrategy(DiskCacheStrategy.AUTOMATIC)
        .listener(loadListener)
        .into(object : CustomTarget<Drawable?>() {
            override fun onResourceReady(
                @NonNull resource: Drawable,
                @Nullable transition: Transition<in Drawable?>?
            ) {
                val bitmap = (resource as BitmapDrawable).bitmap
                view.setImageBitmap(bitmap)
                HelperUtils.saveImage(view.context, bitmap, memberId, pref)
            }

            override fun onLoadCleared(@Nullable placeholder: Drawable?) {}
            override fun onLoadFailed(@Nullable errorDrawable: Drawable?) {
                view.setImageDrawable(errorDrawable)
            }
        })
}
