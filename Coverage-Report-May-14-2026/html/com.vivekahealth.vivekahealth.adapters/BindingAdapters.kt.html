@BindingAdapter("lastUpdated", "premium", "expanded", requireAll = false)

        if (!lastUpdated.isNullOrBlank()) {

                var date = DateUtils.getMonthYearFromServerDateTimePlane(lastUpdated)
                if (premium) {
                    textView.text = textView.context.getString(R.string.premium_last_reported, date)
                } else {
                    textView.text = textView.context.getString(R.string.hours_last_reported, date)
                }
            }
        } else {
            textView.visibility = View.GONE
        }
    } else {
        textView.visibility = View.GONE
    }
}

@BindingAdapter("premium", "hoursTile", requireAll = true)
fun bindHoursView(textView: TextView, premium: Double?, hoursTile: Double?) {
    var premium: Double? = premium
    var hours: Double? = hoursTile
    var t = (
        (premium == null || premium == 0.0) &&
            (hours == 0.0 && premium == 0.0)
        )
    var p = ((hours == null || hours == 0.0) && (premium != null && premium != 0.0))
    if (t) {
        textView.visibility = View.VISIBLE
    } else {
        if (p) {
            textView.visibility = View.GONE
        } else {
            textView.visibility = View.VISIBLE
        }
    }
}

@BindingAdapter("locationType", "miles", "locationName", requireAll = true)
fun bindLocationFilterText(
    toolbar: TextView,
    locationType: String?,
    miles: String?,
    locationName: String?
) {
    if (locationType != null && miles != null) {
        if (locationType.equals("NearBy", true) || locationType.equals("1", true)) {
            if (miles.isBlank()) {
                toolbar.text = toolbar.context.getString(
                    R.string.location_with_in_nolimit,
                    toolbar.context.getString(R.string.near_by),
                    toolbar.context.getString(R.string.no_limits)
                )
            } else {
                toolbar.text =
                    toolbar.context.getString(
                        R.string.location_with_in_miles,
                        toolbar.context.getString(R.string.near_by),
                        miles
                    )
            }
        } else if (locationType.equals("Home", true) || locationType.equals("2", true)) {
            if (miles.isBlank()) {
                toolbar.text = toolbar.context.getString(
                    R.string.location_with_in_nolimit,
                    toolbar.context.getString(R.string.home),

                    toolbar.context.getString(R.string.no_limits)
                )
            } else {
                toolbar.text =
                    toolbar.context.getString(
                        R.string.location_with_in_miles,
                        toolbar.context.getString(R.string.home),
                        miles
                    )
            }
        } else if (locationType.equals("Work", true) || locationType.equals("3", true)) {
            if (miles.isBlank()) {
                toolbar.text = toolbar.context.getString(
                    R.string.location_with_in_nolimit,
                    toolbar.context.getString(R.string.work),

                    toolbar.context.getString(R.string.no_limits)
                )
            } else {
                toolbar.text =
                    toolbar.context.getString(
                        R.string.location_with_in_miles,
                        toolbar.context.getString(R.string.work),
                        miles
                    )
            }
        } else {
            if (!locationName.isNullOrEmpty()) {
                if (miles.isBlank()) {
                    toolbar.text =
                        toolbar.context.getString(R.string.custom_within_nolimits, locationName)
                } else {
                    toolbar.text =
                        toolbar.context.getString(
                            R.string.custom_miles_location,
                            miles,
                            locationName
                        )
                }
            }
        }
    }
}

@BindingAdapter("spanFirstName", "spanDateEnroll", requireAll = true)
fun bindSpanEnrollTextDate(
    textView: TextView,
    spanFirstName: String?,
    spanDateEnroll: String?
) {
    if (spanFirstName.isNullOrBlank() || spanDateEnroll.isNullOrBlank()) {
        textView.text = ""
        return
    }

    val context = textView.context
    val fullText = context.getString(R.string.enrollment_greeting, spanFirstName, spanDateEnroll)
    val stringBuilder = SpannableStringBuilder(fullText)

    // Find and style first name
    val firstNameStart = fullText.indexOf(spanFirstName)
    if (firstNameStart >= 0) {
        val firstNameEnd = firstNameStart + spanFirstName.length
        stringBuilder.setSpan(
            StyleSpan(Typeface.BOLD),
            firstNameStart,
            firstNameEnd,
            Spannable.SPAN_INCLUSIVE_INCLUSIVE
        )
        stringBuilder.setSpan(
            ForegroundColorSpan(ContextCompat.getColor(context, R.color.TextGrey)),
            firstNameStart,
            firstNameEnd,
            Spannable.SPAN_INCLUSIVE_INCLUSIVE
        )
    }

    // Find and style date
    val dateStart = fullText.indexOf(spanDateEnroll)
    if (dateStart >= 0) {
        val dateEnd = dateStart + spanDateEnroll.length
        stringBuilder.setSpan(
            StyleSpan(Typeface.BOLD),
            dateStart,
            dateEnd,
            Spannable.SPAN_INCLUSIVE_INCLUSIVE
        )
        stringBuilder.setSpan(
            ForegroundColorSpan(ContextCompat.getColor(context, R.color.TextGrey)),
            dateStart,
            dateEnd,
            Spannable.SPAN_INCLUSIVE_INCLUSIVE
        )
    }

    textView.text = stringBuilder
}

@BindingAdapter("completeData", "uploadCount")
fun bindCompleteData(
    materialTextView: MaterialTextView,
    docsComplete: Boolean,
    fieldsComplete: Boolean
) {
    val useUploadCount = materialTextView.context.resources.getBoolean(
        R.bool.complete_data_use_upload_count
    )
    if (useUploadCount) {
        when {
            docsComplete -> {
                materialTextView.setCompoundDrawablesWithIntrinsicBounds(
                    R.drawable.ic_checked,
                    0,
                    0,
                    0
                )
                materialTextView.text = materialTextView.context.getString(R.string.completed)
                materialTextView.setTextColor(
                    ContextCompat.getColor(materialTextView.context, R.color.check_green)
                )
            }
            fieldsComplete -> {
                val amber = ContextCompat.getColor(
                    materialTextView.context,
                    R.color.partial_complete_amber
                )
                val icon = ContextCompat.getDrawable(materialTextView.context, R.drawable.ic_info)
                    ?.mutate()
                    ?.also { DrawableCompat.setTint(it, amber) }
                materialTextView.setCompoundDrawablesWithIntrinsicBounds(icon, null, null, null)
                materialTextView.text = materialTextView.context.getString(
                    R.string.partially_completed
                )
                materialTextView.setTextColor(amber)
            }
            else -> {
                materialTextView.setCompoundDrawablesWithIntrinsicBounds(R.drawable.error, 0, 0, 0)
                materialTextView.text = materialTextView.context.getString(
                    R.string.enter_details
                )
                materialTextView.setTextColor(
                    ContextCompat.getColor(materialTextView.context, R.color.red)
                )
            }
        }
    } else {
        if (docsComplete) {
            materialTextView.setCompoundDrawablesWithIntrinsicBounds(R.drawable.ic_checked, 0, 0, 0)
            materialTextView.text = materialTextView.context.getString(R.string.completed)
            materialTextView.setTextColor(
                ContextCompat.getColor(materialTextView.context, R.color.check_green)
            )
        } else {
            materialTextView.setCompoundDrawablesWithIntrinsicBounds(R.drawable.error, 0, 0, 0)
            materialTextView.text = materialTextView.context.getString(R.string.complete_details)
            materialTextView.setTextColor(
                ContextCompat.getColor(materialTextView.context, R.color.red)
            )
        }
    }
}

@BindingAdapter("boldText")
fun setBoldText(textView: TextView, text: String?) {
    if (text != null) {
        val fullText = textView.context.getString(R.string.did_you_mean_s, text)
        val spannableString = SpannableString(fullText)
        val start = fullText.indexOf(text)
        val end = start + text.length
        spannableString.setSpan(
            StyleSpan(Typeface.BOLD),
            start,
            end,
            Spannable.SPAN_EXCLUSIVE_EXCLUSIVE
        )
        textView.text = spannableString
    }
}
