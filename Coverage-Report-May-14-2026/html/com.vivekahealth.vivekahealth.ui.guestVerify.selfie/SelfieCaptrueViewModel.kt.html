    val openCamera: LiveData<Event<Unit>>

    var gson = Gson()
    private lateinit var job: Job
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    var isFront: Boolean = true

    val showError: LiveData<APIResponse<DataResponse>>
        get() = _showError

    private val _phoneEmailPrev = MutableLiveData<Event<String>>()

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)

    private var debounceJob: Job? = null
    private var isDebouncing = false
    private val debounceInterval = 500L // Set your debounce interval in milliseconds

    fun singleClick(action: () -> Unit) {
        if (isDebouncing) return
        action()
        isDebouncing = true
        debounceJob?.cancel()
        debounceJob = viewModelScope.launch {
            delay(debounceInterval)
            isDebouncing = false
        }
    }

    fun onNextClicked() {
        singleClick {
            _navigateToNext.value = Event(Unit)
        }
    }

    fun onPrevClicked() {
        singleClick {
            _navigateToBack.value = Event(Unit)
        }
    }

    fun onOpenCamera() {
        singleClick {
            _openCamera.value = Event(Unit)
        }
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }

    private val _uploadState = MutableLiveData<Event<UploadState>>()
    val uploadState: LiveData<Event<UploadState>> get() = _uploadState

    private val _showAlternateAppAvailable = MutableLiveData<Event<DataResponse>>()
    val showAlternateAppAvailable: LiveData<Event<DataResponse>> get() = _showAlternateAppAvailable

    fun uploadSelfie(
        documentType: DocumentType,
        file: Bitmap?,
        token: String
    ) {
        if (file == null) {
            _uploadState.value = Event(UploadState.Error("No file provided"))
            return
        }

        val compressed = compressBitmap(file)

        val filePart: MultipartBody.Part = HelperUtils.createMultipartBodyPartFromBitmap(
            bitmap = compressed,
            "file",
            "${UUID.randomUUID()}-$documentType-${DateUtils.getCurrentLocalTime()}"
        )

        viewModelScope.launch {
            compositeDisposable.add(
                uploadRepository.uploadDoc(
                    token,
                    filePart,
                    documentType
                ).subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _uploadState.value = Event(UploadState.Loading(LoadingState.UPLOAD))
                    }
                    .subscribe({
                        if (it.status) {
                            _uploadState.value = Event(UploadState.Uploaded)
                            identify(token)
                        } else {
                            _uploadState.value = Event(
                                UploadState.Error(
                                    it.message ?: "Failed to upload image"
                                )
                            )
                        }
                    }, {
                        val errorMessage = when {
                            it.message?.contains("SocketException", ignoreCase = true) == true -> Constants.APOLOGY_MESSAGE
                            else -> it.message ?: Constants.APOLOGY_MESSAGE
                        }
                        _uploadState.value = Event(UploadState.Error(errorMessage))
                    })
            )
        }
    }

    fun resetState() {
        _uploadState.value = Event(UploadState.Idle)
    }

    fun verify(token: String) {
        viewModelScope.launch {
            compositeDisposable.clear()
            compositeDisposable.add(
                uploadRepository.verify(token)
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _uploadState.value = Event(UploadState.Loading(LoadingState.VERIFY))
                    }.subscribe({
                        if (it.isSuccessful) {
                            identify(token)
                        } else {
                            val errorResponse: GenericErrorResponse = gson.fromJson(
                                it.errorBody()?.string(),
                                GenericErrorResponse::class.java
                            )
                            _uploadState.value = Event(
                                UploadState.Error(errorResponse.getErrorMessage())
                            )
                        }
                    }, { error ->

                        _uploadState.value = Event(
                            UploadState.Error(error)
                        )
                    })
            )
        }
    }

    fun identify(token: String, reregister: Boolean? = false) {
        viewModelScope.launch {
            compositeDisposable.clear()
            compositeDisposable.add(
                uploadRepository.identify(token, reregister)
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _uploadState.value = Event(UploadState.Loading(LoadingState.IDENTIFY))
                    }.subscribe({
                        if (it.isSuccessful) {
                            it.body()?.let { responseBody ->
                                when (responseBody.code) {
                                    200 -> _uploadState.value = Event(UploadState.Success())
                                    else -> handleError(responseBody.error)
                                }
                            }
                        } else {
                            if (it.code() == 403) {
                                _uploadState.value = Event(UploadState.Error(""))
                                handleAlternateAppAvailable(it.errorBody())
                            } else {
                                handleErrorResponse(it.errorBody())
                            }
                        }
                    }, { error ->
                        _uploadState.value = Event(
                            UploadState.Error(error)
                        )
                    })
            )
        }
    }

    private fun handleError(error: ErrorDetail?) {
        when (error?.error) {
            "SENT_TO_CUSTOMER_SUPPORT" -> _uploadState.value = Event(UploadState.CustomerSupport)
            "NO_MATCH" -> {
                when {
                    error.requireCurrentZip -> _uploadState.value = Event(UploadState.NoMatchAddZip)
                    error.requireCurrentAddress -> _uploadState.value = Event(
                        UploadState.NoMatchAddAddress
                    )

                    else -> _uploadState.value = Event(UploadState.Error(error.errorDescription))
                }
            }

            "MULTIPLE_MATCH" -> {
                when {
                    error.requireCurrentZip -> _uploadState.value = Event(UploadState.AddZip)
                    error.requireCurrentAddress -> _uploadState.value = Event(
                        UploadState.AddAddress
                    )

                    else -> _uploadState.value = Event(UploadState.Error(error.errorDescription))
                }
            }

            "MEMBER_ALREADY_REGISTERED" -> {
                _uploadState.value = Event(
                    UploadState.MemberAlreadyRegistered(error.otherDetails?.email ?: "")
                )
            }

            else -> _uploadState.value = Event(UploadState.Error(error?.errorDescription ?: ""))
        }
    }

    private fun handleErrorResponse(errorBody: ResponseBody?) {
        val errorBodyString = try {
            errorBody?.string()
        } catch (e: Exception) {
            null
        }

        if (errorBodyString.isNullOrBlank()) {
            _uploadState.value = Event(UploadState.Error("Unknown error occurred"))
            return
        }

        // Try to parse as JsonObject first to check structure
        val jsonObject = try {
            gson.fromJson(errorBodyString, JsonObject::class.java)
        } catch (e: Exception) {
            _uploadState.value = Event(UploadState.Error("Invalid server response"))
            return
        }

        val errorMessage = when {
            // Check if it matches IdentifyErrorResponse structure
            hasIdentifyErrorStructure(jsonObject) -> {
                parseIdentifyErrorResponse(errorBodyString)
            }
            // Check if it matches ErrorDetail structure
            hasErrorDetailStructure(jsonObject) -> {
                parseErrorDetail(errorBodyString)
            }
            // Try to extract any error message from generic structure
            else -> {
                extractGenericErrorMessage(jsonObject)
            }
        }

        _uploadState.value = Event(UploadState.Error(errorMessage))
    }

    private fun hasIdentifyErrorStructure(jsonObject: JsonObject): Boolean {
        return jsonObject.has("errorCode")
    }

    private fun hasErrorDetailStructure(jsonObject: JsonObject): Boolean {
        return jsonObject.has("error_description")
    }

    private fun parseIdentifyErrorResponse(jsonString: String): String {
        return try {
            val errorResponse = gson.fromJson(jsonString, IdentifyErrorResponse::class.java)
            errorResponse.error?.errorDescription
                ?: errorResponse.message
                ?: "Unknown error occurred"
        } catch (e: Exception) {
            "Error parsing server response"
        }
    }

    private fun parseErrorDetail(jsonString: String): String {
        return try {
            val errorResponse = gson.fromJson(jsonString, ErrorDetail::class.java)
            errorResponse.errorDescription ?: "Unknown error occurred"
        } catch (e: Exception) {
            "Error parsing server response"
        }
    }

    private fun extractGenericErrorMessage(jsonObject: JsonObject): String {
        return when {
            jsonObject.has("message") -> {
                jsonObject.get("message")?.asString ?: "Unknown error"
            }

            jsonObject.has("error") -> {
                val error = jsonObject.get("error")
                when {
                    error.isJsonPrimitive -> error.asString
                    error.isJsonObject -> error.asJsonObject.get("message")?.asString
                        ?: "Unknown error"

                    else -> "Unknown error"
                }
            }

            jsonObject.has("detail") -> {
                jsonObject.get("detail")?.asString ?: "Unknown error"
            }

            else -> "HTTP Status 404 – Not Found"
        }
    }

    private fun handleAlternateAppAvailable(errorBody: ResponseBody?) {
        val errorResponse: DataResponse? = gson.fromJson(
            errorBody?.string(),
            DataResponse::class.java
        )
        errorResponse?.let {
            _showAlternateAppAvailable.value = Event(it)
        }
    }

    fun compressBitmap(bitmap: Bitmap, quality: Int = 50): Bitmap {
        val resizedBitmap = resizeBitmap(bitmap, 1080, 1920)
        val outputStream = ByteArrayOutputStream()
        resizedBitmap.compress(Bitmap.CompressFormat.JPEG, quality, outputStream)
        val byteArray = outputStream.toByteArray()
        return BitmapFactory.decodeByteArray(byteArray, 0, byteArray.size)
    }

    fun resizeBitmap(bitmap: Bitmap, maxWidth: Int, maxHeight: Int): Bitmap {
        val aspectRatio = bitmap.width.toFloat() / bitmap.height.toFloat()
        val (newWidth, newHeight) = if (bitmap.width > bitmap.height) {
            Pair(maxWidth, (maxWidth / aspectRatio).toInt())
        } else {
            Pair((maxHeight * aspectRatio).toInt(), maxHeight)
        }
        return Bitmap.createScaledBitmap(bitmap, newWidth, newHeight, true)
    }
