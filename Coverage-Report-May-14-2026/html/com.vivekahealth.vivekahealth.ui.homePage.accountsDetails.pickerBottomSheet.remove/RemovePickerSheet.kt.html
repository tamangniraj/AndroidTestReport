package com.vivekahealth.vivekahealth.ui.homePage.accountsDetails.pickerBottomSheet.remove

import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.lifecycle.ViewModelProvider
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.databinding.FragmentRemovePickerSheetBinding
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseBottomSheetDialogFragment
import com.vivekahealth.vivekahealth.utils.EventObserver
import javax.inject.Inject

class RemovePickerSheet : BaseBottomSheetDialogFragment(), Injectable {

    companion object {
        fun newInstance() = RemovePickerSheet()
    }
    var pickerButtonListener: PickerRemoveButtonListener? = null

    fun setCallback(pickerButtonListener: PickerRemoveButtonListener) {
        this.pickerButtonListener = pickerButtonListener
    }
    private lateinit var viewModel: RemovePickerSheetViewModel
    private lateinit var binding: FragmentRemovePickerSheetBinding

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory
    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = FragmentRemovePickerSheetBinding.inflate(inflater, container, false)
        return binding.root
    }
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setStyle(STYLE_NORMAL, R.style.CustomBottomSheetDialogTheme)
    }
    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            RemovePickerSheetViewModel::class.java
        )
        binding.viewModel = viewModel

        val hasImage = arguments?.getBoolean("hasImage")
        if (hasImage != null) {
            binding.hasImage = hasImage
        }

        viewModel.navigateToCancel.observe(
            viewLifecycleOwner,
            EventObserver {
                pickerButtonListener?.onClose()
                dialog?.dismiss()
            }
        )
        viewModel.navigateToRemove.observe(
            viewLifecycleOwner,
            EventObserver {
                pickerButtonListener?.onRemove()
                dialog?.dismiss()
            }
        )

        viewModel.navigateToRemoveDependent.observe(
            viewLifecycleOwner,
            EventObserver {
                pickerButtonListener?.onRemoveDependent()
                dialog?.dismiss()
            }
        )
    }
}
