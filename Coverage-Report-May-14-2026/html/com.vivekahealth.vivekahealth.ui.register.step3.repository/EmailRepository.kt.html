package com.vivekahealth.vivekahealth.ui.register.step3.repository

import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.data.remote.register.ValidateRequest
import com.vivekahealth.vivekahealth.data.remote.register.ValidateResponse
import com.vivekahealth.vivekahealth.ui.register.step3.service.EmailService
import io.reactivex.Single
import retrofit2.Response
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Repository that handles User objects.
 */
@Singleton
class EmailRepository @Inject constructor(
    private val emailService: EmailService
) {

    fun verifyEmail(email: String): Single<DataResponse> =
        emailService.getEmailApi().checkEmailService(
            email
        )

    fun verifyEmailV2(email: String): Single<Response<DataResponse>> =
        emailService.getEmailApi().checkEmailServiceV2(
            email
        )

    fun verifyEmailV2(email: String, token: String): Single<Response<DataResponse>> =
        emailService.getEmailApi().checkEmailServiceV2(
            email,
            token
        )

    fun validateEmail(
        email: String,
        checkExistingAccount: Boolean,
        registrationId: String
    ): Single<Response<ValidateResponse>> = emailService.getEmailApi().validateEmail(
        ValidateRequest(
            email,
            checkExistingAccount,
            registrationId
        )
    )
}
