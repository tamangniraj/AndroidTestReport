package com.vivekahealth.vivekahealth.ui.guestHome.repository

import com.vivekahealth.vivekahealth.data.local.home.GuestClientDetailsDao
import com.vivekahealth.vivekahealth.data.local.home.GuestInfoDao
import com.vivekahealth.vivekahealth.data.remote.enroll.DocumentUploadSuccessEnroll
import com.vivekahealth.vivekahealth.data.local.enroll.RootDocumentResponse
import com.vivekahealth.vivekahealth.data.remote.enroll.OpenEnrollmentData
import com.vivekahealth.vivekahealth.data.remote.enroll.RootMissingDocumentResponse
import com.vivekahealth.vivekahealth.data.remote.guest.ProcessLicenseResponse
import okhttp3.MultipartBody
import retrofit2.Response
import com.vivekahealth.vivekahealth.data.remote.guest.GuestClientDetails
import com.vivekahealth.vivekahealth.data.remote.guest.GuestUserInfo
import com.vivekahealth.vivekahealth.data.remote.guest.LanguagePreference
import com.vivekahealth.vivekahealth.data.remote.guest.StartIdentificationsReceiveParams
import com.vivekahealth.vivekahealth.data.remote.home.findCare.HtmlFindCareResponse
import com.vivekahealth.vivekahealth.data.remote.home.messageCenter.MessageCenterList
import com.vivekahealth.vivekahealth.data.remote.home.messageCenter.NotificationCount
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.data.remote.resetPassword.ChangePasswordV3DTO
import com.vivekahealth.vivekahealth.ui.guestHome.service.GuestHomeService
import io.reactivex.Single
import kotlinx.coroutines.flow.Flow
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class GuestHomeRepository @Inject constructor(
    private val homeService: GuestHomeService,
    private val guestInfoDao: GuestInfoDao,
    private val guestClientDetailsDao: GuestClientDetailsDao
) {

    fun getGuestInfo(): Single<GuestUserInfo> {
        return homeService.getGuestHomeApi().getGuestInfo()
    }

    fun getGuestClientDetails(): Single<GuestClientDetails> {
        return homeService.getGuestHomeApi().getGuestClientDetails()
    }

    fun updateGuestInfo(guestInfo: GuestUserInfo) {
        guestInfoDao.update(guestInfo)
    }

    fun getGuestInfoFromDb(id: String): Flow<GuestUserInfo> {
        return guestInfoDao.getGuestInfoLiveData(id)
    }

    fun insertGuestInfo(guestInfo: GuestUserInfo) {
        guestInfoDao.insert(guestInfo)
    }

    fun updateGuestClientDetails(guestClientDetails: GuestClientDetails) {
        guestClientDetailsDao.update(guestClientDetails)
    }

    fun insertGuestClientDetails(guestClientDetails: GuestClientDetails) {
        guestClientDetailsDao.insert(guestClientDetails)
    }

    fun updateGuestFcmToken(guestId: String, fcmId: String, isPrimary: Boolean): Single<Void> {
        return homeService.getGuestHomeApi().refreshGuestFcm(guestId, fcmId, false, isPrimary)
    }

    fun startGuestRegistration(): Single<StartIdentificationsReceiveParams> {
        return homeService.getGuestHomeApi().startRegistration()
    }

    fun changeGuestPassword(passDto: ChangePasswordV3DTO): Single<DataResponse> {
        return homeService.getGuestHomeApi().changePasswordGuest(passDto)
    }

    fun updateGuestPassword(passDto: ChangePasswordV3DTO): Single<DataResponse> {
        return homeService.getGuestHomeApi().updatePasswordGuest(passDto)
    }

    fun getLogoutGuest(): Single<DataResponse> = homeService.getGuestHomeApi().getLogoutGuest()

    fun getGuestNotifications(pageNo: Int, pageSize: Int): Single<MessageCenterList> {
        return homeService.getGuestHomeApi().getGuestNotifications(pageNo, pageSize)
    }

    fun getGuestUnreadNotificationCount(): Single<NotificationCount> {
        return homeService.getGuestHomeApi().getGuestUnreadNotificationCount()
    }

    fun setLanguagePreference(language: String): Single<DataResponse> {
        return homeService.getGuestHomeApi().setGuestLanguagePreference(
            LanguagePreference(language)
        )
    }

    fun htmlFIndCare(): Single<HtmlFindCareResponse> =
        homeService.getGuestHomeApi().getHtmlForFindDoc()

    fun openEnrollMember(member: OpenEnrollmentData): Single<Response<DocumentUploadSuccessEnroll>> =
        homeService.getGuestHomeApi().getOpenEnroll(member)

    fun getMissingRequiredDocuments(): Single<RootMissingDocumentResponse> =
        homeService.getGuestHomeApi().getMissingRequiredDocuments()

    fun getRequiredDocuments(): Single<RootDocumentResponse> =
        homeService.getGuestHomeApi().getRequiredDocuments()

    fun processLicense(file: MultipartBody.Part): Single<Response<ProcessLicenseResponse>> =
        homeService.getGuestHomeApi().processLicense(file)

    fun uploadOpenFile(
        filePart: MultipartBody.Part,
        fileName: String,
        enrollmentId: Long? = null
    ): Single<DocumentUploadSuccessEnroll> = homeService.getGuestHomeApi().uploadOpenFile(
        filePart,
        fileName,
        enrollmentId
    )
}
