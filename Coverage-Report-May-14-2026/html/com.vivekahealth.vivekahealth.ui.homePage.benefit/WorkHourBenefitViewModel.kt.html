    fun addStickyView(view: StickyViewHolder) {
        if (_stickyViews.value?.map { it.id }?.contains(view.id) == false) {
            _stickyViews.value = _stickyViews.value?.plus(view) ?: listOf(view)
        } else {
            _stickyViews.value = listOf(view)
        }
    }

    fun removeStickyView(view: StickyViewHolder) {
        _stickyViews.value = _stickyViews.value?.toMutableList()?.apply {
            view.view.translationY = 0f
            view.view.z = 0f
            remove(view)
        }?.toList()
    }

    data class StickyViewHolder(
        val id: Int,
        val view: View
    )

    val onStickyScrollChangeListener: View.OnScrollChangeListener =
        View.OnScrollChangeListener { scrollView, _, scrollY, _, _ ->
            kotlin.run {
                stickyViews.value?.let {
                    for (sticky in it) {
                        val index = it.indexOf(sticky)

                        val viewBound = Rect()
                        sticky.view.getDrawingRect(viewBound)
                        (scrollView as ViewGroup).offsetDescendantRectToMyCoords(
                            sticky.view,
                            viewBound
                        )

                        val viewHeightOffset = if (index == 0) {
                            viewBound.top
                        } else {
                            val previousSticky = it[index - 1]
                            val previousRect = Rect()
                            previousSticky.view.getDrawingRect(previousRect)
                            viewBound.top - previousRect.height()
                        }

                        if (scrollY >= viewHeightOffset) {
                            sticky.view.translationY = scrollY.toFloat() - viewHeightOffset
                            sticky.view.z = 100f
                        } else {
                            sticky.view.translationY = 0f
                            sticky.view.z = 0f
                        }
                    }
                }
            }
        }

    private fun changeTitleText(filter: String, value: WorkHourSummary) {
        when {
            filter == "REQUESTED_YEAR" -> {
                titleString = resources.getString(
                    R.string.currently_viewing_work_hours_for_s_api,
                    DateUtils.digitTOMonth(1, resources),
                    value.year.toString()
                )
            }

            filter == "LIFETIME" -> {
                titleString = resources.getString(
                    R.string.lifetime_starting_form_s_s,
                    DateUtils.digitTOMonth(value.month, resources),
                    value.year.toString()
                )
                totalWork = workHourList.sumOf { it.hoursWorked }
            }

            filter == "REQUESTED_YEAR_MONTH" -> {
                titleString =
                    DateUtils.digitTOMonthLong(value.month, resources) + " " + value.year.toString()
            }

            filter == "LAST_REPORTED" -> {
                titleString = resources.getString(
                    R.string.last_reported_month_s_s,
                    DateUtils.digitTOMonth(value.month, resources),
                    value.year.toString()
                )
            }

            filter == "ROLLING_YEAR" -> {
                titleString = resources.getString(
                    R.string.rolling_period_from_api,
                    DateUtils.digitTOMonth(value.month, resources),
                    value.year.toString()
                )
                val dropYear: Int = if (workHourList.size >= Constants.BENEFIT_ROLLING_PERIOD) {
                    workHourList.size - Constants.BENEFIT_ROLLING_PERIOD
                } else {
                    workHourList.size
                }
                val year = workHourList.drop(dropYear)
                if (year.isNotEmpty()) {
                    totalWork = year.sumOf { it.hoursWorked ?: 0.0 }
                    totalBanked = year.last().hoursBanked
                }
            }
        }
    }

    fun loadDataCalender() {
        var year: Int? = null
        var month: Int? = null
        month = if (currentSelectedMonth != -1) {
            currentSelectedMonth
        } else {
            null
        }
        year = if (currentYearValue != 0) {
            currentYear
        } else {
            null
        }
        compositeWorkLogDisposable.dispose()
        compositeWorkLogDisposable = CompositeDisposable()
        var workLogSingle = benefitWorkRepository.getWorkBenefitV2(
            preferences.getString("memberID", "").toString()
        )
        if (Constants.IS_OFFLINE) {
            if (preferences.contains("memberID")) {
                val workLog = benefitWorkRepository.getWorkLogFromDb(
                    preferences.getString("memberID", "").toString()
                )
                if (workLog != null) {
                    workLogSingle = Single.just(workLog)
                } else {
                    benefitLoading = false
                    _enableTab.value = Event(true)
                }
            }
        }

        // get work log V2
        compositeWorkLogDisposable.add(
            workLogSingle
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    benefitLoading = true
                    _benefitLoadingComplete.value = false
                }
                .subscribe({
                    if (!Constants.IS_OFFLINE) {
                        benefitWorkRepository.insertWorkLog(it, memberId)
                    }
                    workHourListV2 = it.worklogs
                    val workHourV2 =
                        workHourListV2.find { it.year == "$currentYear" && it.month == "$currentSelectedMonth" }
                    hideBenefits = if (workHourV2 == null) {
                        true
                    } else {
                        workHourV2.eligibilityStatus != "ELIGIBLE"
                    }
                    var currentYearItems =
                        workHourListV2.filter { item -> item.year.toInt() == DateUtils.getCurrentYear() }

                    if (!yearList.isNullOrEmpty()) {
                        yearList.clear()
                    }
                    for (year in workHourListV2) {
                        yearList.add(year.year.toInt())
                    }
                    benefitsLogos = it.benefitsLogos
                    var yearListDistinct = yearList.distinct()

                    var latestYear = yearListDistinct.maxOrNull() ?: DateUtils.getCurrentYear()
                    if (currentYear > latestYear) {
                        latestYear = currentYear
                    }
                    for (i in (yearListDistinct.firstOrNull() ?: DateUtils.getCurrentYear())..latestYear) {
                        if (!yearListDistinct.contains(i)) {
                            yearList.add(i)
                            for (m in 1..12) {
                                workHourListV2.add(
                                    WorkLog(
                                        year = i.toString(),
                                        month = m.toString(),
                                        addedItem = true
                                    )
                                )
                            }
                        }
                    }

                    // Full list of 12 months to compare a year's against for any missing
                    val fullMonths = List(12) {
                        "${it + 1}"
                    }

                    yearList.distinct().forEach { year ->
                        // Get all logs for the year
                        val months = workHourListV2.filter { it.year == "$year" }.map { it.month }

                        if (months != fullMonths) {
                            // The missing months
                            val remaining = fullMonths - months
                            remaining.forEach { month ->
                                workHourListV2 += WorkLog(
                                    year = "$year",
                                    month = month,
                                    addedItem = true
                                )
                            }
                        }
                    }

                    // Sort first by year, then by month
                    workHourListV2 = ArrayList(
                        workHourListV2.sortedWith(
                            compareBy({ it.year.toInt() }, { it.month.toInt() })
                        )
                    )

                    yearList.sort()
                    yearListDistinct = yearList.distinct()
                    _yearListGenerated.value = Event(yearListDistinct)

                    if (!yearList.isNullOrEmpty() && currentYearValue == 0) {
                        Timber.d(Calendar.getInstance().get(Calendar.YEAR).toString())

                        Timber.d("sourceYear:$selectedYear")
                        Timber.d("hereFirst")

                        if (yearList.contains(Calendar.getInstance().get(Calendar.YEAR))) {
                            if (selectedYear != 0) {
                                loadSheetYear(currentYear)
                            } else {
                                loadSheetYear(Calendar.getInstance().get(Calendar.YEAR))
                            }
                        } else {
                            Timber.d("hereLast")
                            loadSheetYear(yearList.last())
                        }
                    } else {
                        Timber.d("hereCurr")
                        loadSheetYear(currentYear)
                    }
                    calendarDataLoaded = true
                    showCalendarTutorialIfConditionMatched()

                    _enableTab.value = Event(true)
                }, {
                    Timber.d(it.toString())

                    benefitLoading = false
                    _enableTab.value = Event(true)
                })
        )
    }

    private fun mapWorkSummary() {
        val nonNull = workHourSummary.employerHourDTOs?.filter { item -> item.hoursWorked != null }
        totalWorkedHours = if (nonNull.isNullOrEmpty()) {
            null
        } else {
            nonNull.sumOf { item -> item.hoursWorked ?: 0.0 }
        }

        val contractList =
            workHourSummary.employerHourDTOs?.flatMap { item -> item.contractBasicInfoDTOList }
        val nonNullPremium = contractList?.filter { item -> item.amountReported != null }
        totalPremium = if (nonNullPremium.isNullOrEmpty()) {
            null
        } else {
            nonNullPremium.sumOf { item -> item.amountReported ?: 0.0 }
        }
        premiumBased = workHourSummary.memberEnrolledInPremiumContract

        workHourSummary.employerHourDTOs?.let { item ->
            employerCount = if (item.isNotEmpty()) {
                val employerWithContracts =
                    item.filter { item1 -> item1.contractBasicInfoDTOList.isNotEmpty() }
                (employerWithContracts.count().minus(1)).toString()
            } else {
                "0"
            }
        }
        workHourSummary.employerHourDTOs?.let {
            val employerWithContracts =
                it.filter { item1 -> item1.contractBasicInfoDTOList.isNotEmpty() }
            employerName = if (it.isNotEmpty() && employerWithContracts.isNotEmpty()) {
                it.last().employerName ?: ""
            } else {
                ""
            }
        }
        showReason =
            workHourSummary.employerHourDTOs.isNullOrEmpty() || totalWorkedHours == null || (totalPremium == null && workHourSummary.memberEnrolledInPremiumContract)
        employer = workHourSummary.employerHourDTOs?.filter { employerHourDTO ->
            if (employerHourDTO.contractBasicInfoDTOList.isNotEmpty()) {
                // Filter out EmployerHourDTO instances where any ContractBasicInfoDTO has a non-null amountExpected and amountReported
                employerHourDTO.contractBasicInfoDTOList.all { contractBasicInfoDTO ->
                    contractBasicInfoDTO.hoursExpected != null
                }
            } else {
                false // or any other value you want to return when contractBasicInfoDTOList is empty
            }
        } ?: emptyList()
        employerPrice = workHourSummary.employerHourDTOs?.filter { employerHourDTO ->
            if (employerHourDTO.contractBasicInfoDTOList.isNotEmpty()) {
                // Filter out EmployerHourDTO instances where any ContractBasicInfoDTO has a non-null amountExpected and amountReported
                employerHourDTO.contractBasicInfoDTOList.all { contractBasicInfoDTO ->
                    contractBasicInfoDTO.amountExpected != null && contractBasicInfoDTO.amountReported != null
                }
            } else {
                false // or any other value you want to return when contractBasicInfoDTOList is empty
            }
        } ?: emptyList()
        showHours = !employer.isNullOrEmpty()
        showPremium = !employerPrice.isNullOrEmpty()
        _summaryListGenerated.value = Event(Unit)
    }

            ) && !calendarTutorialShown && benefitDisclaimerShown && calendarDataLoaded
        ) {
            showingCalendarTutorial = true
            showingBenefitsTutorial = false
            _showCalendarTutorial.value = Event(Unit)
            calendarTutorialShown = true
            preferences.edit().putBoolean(Constants.SHOWN_CALENDAR_TUT_FOR_SESSION, true).apply()
        }
    }

    fun showBenefitsTutorialIfConditionMatched() {
        if (!preferences.getBoolean(
                Constants.SHOWN_BENEFITS_TUT_FOR_SESSION,
                false
            ) && !benefitsTutorialShown && !showingCalendarTutorial && eligible && benefitDisclaimerShown
        ) {
            showingCalendarTutorial = false
            showingBenefitsTutorial = true
            _showBenefitTutorial.value = Event(Unit)
            benefitsTutorialShown = true
            preferences.edit().putBoolean(Constants.SHOWN_BENEFITS_TUT_FOR_SESSION, true).apply()
        }
    }

    fun dontShowCalendarTutorial() {
        preferences.edit().putBoolean(memberId + "calendarTutorialShown", true).apply()
    }

    fun dontShowBenefitTutorial() {
        preferences.edit().putBoolean(memberId + "benefitsTutorialShown", true).apply()
    }

    fun resetCalendarTutorial() {
        _calendarTutorialsAdded.value = false
    }

    fun resetBenefitTutorial() {
        _benefitTutorialsAdded.value = false
    }

    fun shouldShowCalendarTutorial(): Boolean {
        return !preferences.getBoolean(memberId + "calendarTutorialShown", false)
    }

    fun shouldShowBenefitsTutorial(): Boolean {
        return !preferences.getBoolean(memberId + "benefitsTutorialShown", false)
    }

    private val _benefitLoadingComplete = MutableLiveData<Boolean>(false)
    val benefitLoadingComplete: LiveData<Boolean>
        get() = _benefitLoadingComplete

    var benefitDisclaimerShown: Boolean = false
    private var calendarDataLoaded: Boolean = false
    fun readyToShowCalendarTutorial() {
        benefitDisclaimerShown = true
        if (preferences.getBoolean(
                Constants.SHOWN_CALENDAR_TUT_FOR_SESSION,
                false
            )
        ) {
            showBenefitsTutorialIfConditionMatched()
            return
        }
        showCalendarTutorialIfConditionMatched()
    }
