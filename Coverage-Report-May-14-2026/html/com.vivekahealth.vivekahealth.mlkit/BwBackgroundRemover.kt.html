package com.vivekahealth.vivekahealth.mlkit

import android.graphics.Bitmap

/**
 * Standalone Black & White background remover.
 *
 * Algorithm — colour-aware binary split:
 *
 *   bgScore = luma × 0.6 + redExcess × 255 × 0.4
 *
 *   bgScore ≤ splitPoint  →  pure BLACK  (ink / dark text)
 *   bgScore >  splitPoint  →  pure WHITE  (background)
 *
 * Adding redness to the score means pinkish/reddish logo pixels score
 * higher than their luminance alone and are reliably mapped to white,
 * so background stripes cannot masquerade as text strokes.
 *
 * Usage:
 *   val result: Bitmap = BwBackgroundRemover.process(bitmap)
 *   val result: Bitmap = BwBackgroundRemover.process(bitmap, threshold = 0.35f)
 *
 * threshold  — split point as a fraction of 255.  Default 0.29 (splitPoint ≈ 74).
 *              Raise toward 0.50 if faint text is missed; lower if background bleeds in.
 *
 * No external dependencies. Pure CPU — no RenderScript, no ML Kit, no Coil.
 * Safe to copy into any Android project that targets minSdk 21+.
 */
object BwBackgroundRemover {

    private const val DEFAULT_THRESHOLD = 0.29f

    private const val BLACK = 0xFF000000.toInt()
    private const val WHITE = 0xFFFFFFFF.toInt()

    private const val LUMA_WEIGHT = 0.6f // luminance contribution to bgScore
    private const val RED_WEIGHT = 0.4f // redness contribution to bgScore

    // ── OCR letter → digit confusion table ───────────────────────────────────
    // One-way: only letter→digit substitutions.  Never digit→digit or digit→letter.
    // Covers the most common ML Kit OCR errors when reading printed digit sequences.
    //
    //  digit  |  common OCR misreads
    //  -------+-----------------------------------------------
    //    0    |  O o D
    //    1    |  I l i L |
    //    2    |  Z z
    //    3    |  E
    //    4    |  A h
    //    5    |  S s
    //    6    |  b G   ← user-confirmed: '6' often read as 'b'
    //    7    |  T Y
    //    8    |  B e
    //    9    |  g q
    val OCR_DIGIT_FIXES: Map<Char, Char> = mapOf(
        'O' to '0', 'o' to '0', 'D' to '0',
        'I' to '1', 'l' to '1', 'i' to '1', 'L' to '1', '|' to '1',
        'Z' to '2', 'z' to '2',
        'E' to '3',
        'A' to '4', 'h' to '4',
        'S' to '5', 's' to '5',
        'b' to '6', 'G' to '6',
        'T' to '7', 'Y' to '7',
        'B' to '8', 'e' to '8',
        'g' to '9', 'q' to '9'
    )

    // ── Public data classes ───────────────────────────────────────────────────

    /** A single OCR character substitution that was applied (letter → digit). */
    data class OcrFix(
        val position: Int,
        val original: Char,
        val replacement: Char
    ) {
        override fun toString() = "pos $position: '$original'→'$replacement'"
    }

    /**
     * Result of one step inside [processIterative].
     *
     * @param iteration         1-based index: 1 = original image; 2–5 = BW at 20%/30%/40%/50%
     * @param thresholdPct      0 = original (no BW); 20/30/40/50 = BW threshold percent
     * @param rawOcrText        Full raw OCR text returned at this step
     * @param rawCandidate      The SSN-like token found in OCR output before any fixing;
     *                          null if no 9-char alphanumeric token was found
     * @param ocrFixes          Letter→digit substitutions applied (empty when [foundDirect])
     * @param normalizedCandidate  [rawCandidate] after OCR fixes; null when found direct or absent
     * @param extractedSsn      9-digit SSN string if found; null if not detected at this step
     * @param foundDirect       SSN token was all-digits — no fixing needed
     * @param foundViaOcrFix    SSN found only after letter→digit normalization
     */
    data class IterationDetail(
        val iteration: Int,
        val thresholdPct: Int,
        val rawOcrText: String,
        val rawCandidate: String?,
        val ocrFixes: List<OcrFix>,
        val normalizedCandidate: String?,
        val extractedSsn: String?,
        val foundDirect: Boolean,
        val foundViaOcrFix: Boolean
    ) {
        /** Human-readable label for the step, e.g. "Original" or "BW 30%". */
        val stepLabel: String get() = if (thresholdPct == 0) "Original" else "BW $thresholdPct%"
    }

    /**
     * Aggregated result of [processIterative] across all steps.
     *
     * @param iterations        All steps that were actually run (1–5 entries)
     * @param finalSsn          First 9-digit SSN found; null if not detected in any step
     * @param foundAtThreshold  Threshold percent where SSN was found (0 = original); null if not found
     * @param foundAtIteration  1-based iteration index where SSN was found; null if not found
     */
    data class IterativeResult(
        val iterations: List<IterationDetail>,
        val finalSsn: String?,
        val foundAtThreshold: Int?,
        val foundAtIteration: Int?
    )

    // ── process ───────────────────────────────────────────────────────────────

    /**
     * Returns a new ARGB_8888 bitmap with every pixel converted to pure black or pure white.
     *
     * @param source     Input bitmap (not modified, not recycled).
     * @param threshold  Split sensitivity 0.0–1.0. Default [DEFAULT_THRESHOLD] = 0.29.
     */
    private fun process(source: Bitmap, threshold: Float = DEFAULT_THRESHOLD): Bitmap {
        val splitPoint = (threshold * 255f).coerceIn(10f, 245f)
        val pixels = source.copyPixels()

        for (i in pixels.indices) {
            val pixel = pixels[i]
            val r = (pixel shr 16) and 0xFF
            val g = (pixel shr 8) and 0xFF
            val b = pixel and 0xFF

            // Rec. 601 luma
            val luma = (0.299f * r + 0.587f * g + 0.114f * b)
            // How much redder R is than the average of G+B  (0 = neutral, ~1 = vivid red)
            val redExcess = ((r - (g + b) * 0.5f) / 255f).coerceIn(0f, 1f)
            val bgScore = luma * LUMA_WEIGHT + redExcess * 255f * RED_WEIGHT

            pixels[i] = if (bgScore <= splitPoint) BLACK else WHITE
        }

        return pixels.toBitmap(source.width, source.height)
    }

    // ── processIterative ──────────────────────────────────────────────────────

    /**
     * Iterative SSN extraction with progressive BW background removal.
     *
     * Step order (stops as soon as SSN is found):
     *   1. Original image — no BW processing (fastest; no image modification)
     *   2. BW at threshold 20%
     *   3. BW at threshold 30%
     *   4. BW at threshold 40%
     *   5. BW at threshold 50%
     *
     * At each step OCR letter→digit normalization is attempted on candidates
     * with 1–2 misread characters (e.g. '2b3-45-6789' → '263-45-6789').
     * Only letters are substituted; digit positions are never modified (one-way fix).
     *
     * All attempted steps are recorded in [IterativeResult.iterations] so the caller
     * can display the full trace.
     *
     * @param source  Input bitmap (not modified, not recycled by this function)
     * @param ocr     Suspend lambda that receives the bitmap for the current step and returns
     *                raw OCR text.  BW-processed bitmaps passed to [ocr] are recycled after
     *                [ocr] returns; the original [source] bitmap is never recycled.
     */
    suspend fun processIterative(
        source: Bitmap,
        ocr: suspend (Bitmap) -> String
    ): IterativeResult {
        val iterations = mutableListOf<IterationDetail>()
        var finalSsn: String? = null
        var foundAtThreshold: Int? = null
        var foundAtIteration: Int? = null

        // Build the step list: (thresholdPct=0 means use original, no BW processing)
        val steps = intArrayOf(0, 20, 30, 40, 50)

        for ((idx, thresholdPct) in steps.withIndex()) {
            // For step 0 use the original bitmap directly; do NOT recycle it.
            // For BW steps, create a new processed bitmap and recycle it after OCR.
            val bitmapForOcr: Bitmap
            val recycleBitmap: Boolean
            if (thresholdPct == 0) {
                bitmapForOcr = source
                recycleBitmap = false
            } else {
                bitmapForOcr = process(source, thresholdPct / 100f)
                recycleBitmap = true
            }

            val rawText = ocr(bitmapForOcr)
            if (recycleBitmap) bitmapForOcr.recycle()

            val extracted = extractSsnWithFixes(rawText)
            val ssn = extracted.directSsn ?: extracted.normalizedSsn

            iterations.add(
                IterationDetail(
                    iteration = idx + 1,
                    thresholdPct = thresholdPct,
                    rawOcrText = rawText,
                    rawCandidate = extracted.rawCandidate,
                    ocrFixes = extracted.fixes,
                    normalizedCandidate = extracted.normalizedSsn,
                    extractedSsn = ssn,
                    foundDirect = extracted.directSsn != null,
                    foundViaOcrFix = extracted.directSsn == null && extracted.normalizedSsn != null
                )
            )

            if (ssn != null) {
                finalSsn = ssn
                foundAtThreshold = thresholdPct
                foundAtIteration = idx + 1
                break // stop as soon as SSN is found
            }
        }

        return IterativeResult(
            iterations = iterations,
            finalSsn = finalSsn,
            foundAtThreshold = foundAtThreshold,
            foundAtIteration = foundAtIteration
        )
    }

    // ── OCR extraction + normalisation ───────────────────────────────────────

    // Private result carrier — not part of public API
    internal data class SsnExtractResult(
        val directSsn: String?,
        val normalizedSsn: String?,
        val fixes: List<OcrFix>,
        val rawCandidate: String?
    )

    /**
     * Scans [text] for a 9-character SSN token (pure digits, or digits with OCR letter errors).
     *
     * Strategy:
     *  1. Collect candidates: formatted (XXX-XX-XXXX allowing letters) and
     *     unformatted (9-char alphanumeric runs).
     *  2. For each candidate:
     *     a. All 9 chars are digits → direct SSN match (returned immediately).
     *     b. 1–2 chars are letters with known digit equivalents in [OCR_DIGIT_FIXES]
     *        → apply one-way letter→digit substitution and return the fixed SSN.
     *  3. Returns the first successful candidate; all fields null if nothing found.
     *
     * One-way constraint: letters are substituted for digits only.  Digits are never modified.
     */
    internal fun extractSsnWithFixes(text: String): SsnExtractResult {
        // (rawToken, strippedCore) — strippedCore is 9 alphanumeric chars, no dashes/spaces
        val candidates = mutableListOf<Pair<String, String>>()

        // 1. Formatted XXX-XX-XXXX  (dashes required so we know the field boundaries)
        Regex("""[0-9A-Za-z]{3}-[0-9A-Za-z]{2}-[0-9A-Za-z]{4}""")
            .findAll(text)
            .forEach { m ->
                val core = m.value.replace("-", "")
                if (core.length == 9) candidates.add(m.value to core)
            }

        // 2. Whitespace-delimited tokens — strip ALL non-alphanumeric chars (dashes, dots, colons…)
        //    Catches: "565-242083" (one dash), "565.24.2083" (dots), "565 242083" (space), etc.
        text.split(Regex("""[\s\n\r\t]+""")).forEach { token ->
            val core = token.replace(Regex("[^0-9A-Za-z]"), "")
            if (core.length == 9) {
                if (candidates.none { it.second == core }) candidates.add(token to core)
            }
        }

        // 2.5 Dash-containing alphanumeric runs not bounded by whitespace
        //     e.g. "SSN:565-242083." — regex extracts "565-242083", strips dash → "565242083"
        //     Handles any single-dash SSN variant regardless of surrounding punctuation.
        Regex("""[0-9A-Za-z][0-9A-Za-z\-]+[0-9A-Za-z]""")
            .findAll(text)
            .forEach { m ->
                if ('-' in m.value) {
                    val core = m.value.replace("-", "")
                    if (core.length == 9 && core.all { it.isLetterOrDigit() }) {
                        if (candidates.none { it.second == core }) candidates.add(m.value to core)
                    }
                }
            }

        // 3. Raw 9-char alphanumeric runs (catches cases embedded in longer text)
        Regex("""[0-9A-Za-z]{9}""").findAll(text).forEach { m ->
            if (candidates.none { it.second == m.value }) candidates.add(m.value to m.value)
        }

        for ((raw, core) in candidates) {
            if (core.length != 9) continue

            val letterCount = core.count { it.isLetter() }

            // ── Direct: all digits ────────────────────────────────────────────
            if (letterCount == 0) {
                return SsnExtractResult(
                    directSsn = core,
                    normalizedSsn = null,
                    fixes = emptyList(),
                    rawCandidate = raw
                )
            }

            // ── Attempt OCR fix: 1 or 2 letter positions ─────────────────────
            if (letterCount in 1..2) {
                val fixes = mutableListOf<OcrFix>()
                val normalized = core.toCharArray()
                var fixable = true

                for (i in core.indices) {
                    val c = core[i]
                    if (c.isLetter()) {
                        val digit = OCR_DIGIT_FIXES[c]
                        if (digit != null) {
                            fixes.add(OcrFix(position = i, original = c, replacement = digit))
                            normalized[i] = digit
                        } else {
                            fixable = false
                            break
                        }
                    }
                    // digit positions are left untouched — one-way fix only
                }

                if (fixable && normalized.all { it.isDigit() }) {
                    return SsnExtractResult(
                        directSsn = null,
                        normalizedSsn = String(normalized),
                        fixes = fixes,
                        rawCandidate = raw
                    )
                }
            }
        }

        return SsnExtractResult(
            directSsn = null,
            normalizedSsn = null,
            fixes = emptyList(),
            rawCandidate = null
        )
    }

    // ── Private helpers ───────────────────────────────────────────────────────

    private fun Bitmap.copyPixels(): IntArray {
        val px = IntArray(width * height)
        getPixels(px, 0, width, 0, 0, width, height)
        return px
    }

    private fun IntArray.toBitmap(width: Int, height: Int): Bitmap {
        val out = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)
        out.setPixels(this, 0, width, 0, 0, width, height)
        return out
    }
