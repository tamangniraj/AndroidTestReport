package com.vivekahealth.vivekahealth.mlkit
import android.content.Context
import android.graphics.Bitmap
import android.util.Log
import com.google.android.gms.tasks.Task
import com.google.mlkit.vision.common.InputImage
import com.google.mlkit.vision.face.Face
import com.google.mlkit.vision.face.FaceDetection
import com.google.mlkit.vision.face.FaceDetector
import com.google.mlkit.vision.face.FaceDetectorOptions
import com.google.mlkit.vision.face.FaceLandmark

object FaceStatus {
    const val PARTIAL_FACE = -4
    const val EMPTY_FACE = -3
    const val MULTIPLE_FACE = -2
    const val DIFFERENT_FACE = 0
    const val CLEAR = -1
    const val BLINK_DONE = 1
    const val MOVE_DONE = 2
    const val SMILE_DONE = 3
    const val CAPTURE = 4
    const val VERIFICATION_DONE = 5
}

/** Face Detector Demo.  */
class FaceDetectorProcessor(
    context: Context,
    detectorOptions: FaceDetectorOptions?,
    onStepCompleteDetected: OnStepCompleteDetected
) :
    VisionProcessorBase<List<Face>>(context) {

    private var stepCompleteDetected: OnStepCompleteDetected
    private val detector: FaceDetector

    init {
        val options = detectorOptions
            ?: FaceDetectorOptions.Builder()
                .setClassificationMode(FaceDetectorOptions.CLASSIFICATION_MODE_ALL)
                .setLandmarkMode(FaceDetectorOptions.LANDMARK_MODE_ALL)
                .enableTracking()
                .build()

        detector = FaceDetection.getClient(options)

        stepCompleteDetected = onStepCompleteDetected
    }

    override fun stop() {
        super.stop()
        detector.close()
    }

    override fun detectInImage(image: InputImage): Task<List<Face>> {
        return detector.process(image)
    }

    private var trackingId: Int? = null

    var lookLeft: Boolean = false
    var lookRight: Boolean = false
    var centered: Boolean = false

    var countSelfie = 0
    val THERESHOLD_COUNT = 7

    fun clearLast() {
        trackingId = null
        countSelfie = 0
        lookLeft = false
        lookRight = false
        centered = false
    }

    override fun onSuccessOriginal(
        faces: List<Face>,
        graphicOverlay: GraphicOverlay,
        originalImage: Bitmap?
    ) {
        if (faces.isEmpty()) {
            return
        }

        if (faces.size > 1) {
            stepCompleteDetected.onStep(FaceStatus.MULTIPLE_FACE, null, originalImage)
            return
        }

        val face = faces[0]

//        graphicOverlay.add(FaceGraphic(graphicOverlay, face))

        // check if the face is center and visible
        val direction = face.getLeftPercentage()
        val directionRight = face.getRightPercentage()
        val centerProgress = face.centerProgress()

        if (!centered) {
            stepCompleteDetected.onRotateHead(face, HeadDirection.CENTER, centerProgress)
        }

        if (face.isFullFace() && face.isCenterFace(originalImage) && face.isLookingCenter()) {
            // check if phone is

//            if (countSelfie < THERESHOLD_COUNT) {
// //                graphicOverlay.add(FaceGraphic(graphicOverlay, face))
//            }

            countSelfie++

            if (countSelfie == THERESHOLD_COUNT) {
                trackingId = face.trackingId
                stepCompleteDetected.onStep(FaceStatus.CAPTURE, face, originalImage)
                centered = true
                stepCompleteDetected.onRotateHead(face, HeadDirection.CENTER, 100f)
            }
        } else {
            // center your face
            stepCompleteDetected.onStep(FaceStatus.PARTIAL_FACE, null, originalImage)
        }

        if (countSelfie < THERESHOLD_COUNT) {
            // center your face
            stepCompleteDetected.onStep(FaceStatus.PARTIAL_FACE, null, originalImage)
            return
        }

        if (trackingId != face.trackingId) {
            stepCompleteDetected.onStep(FaceStatus.DIFFERENT_FACE, null, originalImage)
            return
        }

        if (!lookLeft) {
            stepCompleteDetected.onRotateHead(face, HeadDirection.LEFT, direction)
        }

        if (direction == 100f) {
            lookLeft = true
        }

        if (!lookRight) {
            stepCompleteDetected.onRotateHead(face, HeadDirection.RIGHT, directionRight)
        }

        if (directionRight == 100f) {
            lookRight = true
        }

        if (lookLeft && lookRight) {
            stepCompleteDetected.onStep(FaceStatus.VERIFICATION_DONE, face, originalImage)
        }
    }

    // Function to calculate how far the face is from the center (based on yaw and pitch)
    fun Face.centerProgress(
        yawThreshold: Float = 5f,
        pitchThreshold: Float = 5f,
        maxDeviation: Float = 25f
    ): Float {
        // Get the yaw (horizontal rotation around Y-axis) and pitch (vertical rotation around X-axis)
        val yaw = headEulerAngleY
        val pitch = headEulerAngleX

        // Check if both yaw and pitch are within their respective thresholds
        val isCenteredYaw = yaw in -yawThreshold..yawThreshold
        val isCenteredPitch = pitch in -pitchThreshold..pitchThreshold

        // If both yaw and pitch are within thresholds, return 100% (fully centered)
        if (isCenteredYaw && isCenteredPitch) {
            return 99f
        }

        // Calculate how far the yaw and pitch are from the center
        val yawDeviation = Math.abs(yaw) - yawThreshold
        val pitchDeviation = Math.abs(pitch) - pitchThreshold

        // Calculate percentage based on how far yaw and pitch deviate from the thresholds
        val yawPercentage = if (yawDeviation > 0) {
            (yawDeviation / maxDeviation) * 99f
        } else {
            0f
        }

        val pitchPercentage = if (pitchDeviation > 0) {
            (pitchDeviation / maxDeviation) * 99f
        } else {
            0f
        }

        // The total deviation is the maximum of yaw or pitch deviation percentage
        val totalDeviation = Math.max(yawPercentage, pitchPercentage)

        // Return the inverse of the deviation (100% is centered, 0% is fully turned)
        return (100f - totalDeviation).coerceIn(0f, 99f)
    }

    // Function to check if the face is centered in the image
    fun Face.isLookingCenter(yawThreshold: Float = 15f, pitchThreshold: Float = 15f): Boolean {
        // Get Euler angles (yaw, pitch)
        val yaw = headEulerAngleY
        val pitch = headEulerAngleX

        // Check if the yaw (rotation around Y-axis) and pitch (rotation around X-axis) are within thresholds
        return yaw in -yawThreshold..yawThreshold && pitch in -pitchThreshold..pitchThreshold
    }

    fun Face.isCenterFace(original: Bitmap?): Boolean {
        original?.let {
            val faceCenterX = this.boundingBox.centerX().toFloat()
            val faceCenterY = this.boundingBox.centerY().toFloat()

            val imageCenterX = original.width / 2f
            val imageCenterY = original.height / 2f

            val threshold = 30f
            val isCentered = kotlin.math.abs(faceCenterX - imageCenterX) < threshold &&
                kotlin.math.abs(faceCenterY - imageCenterY) < threshold
            return isCentered
        }

        return false
    }

    fun Face?.getLeftPercentage(): Float {
        if (this == null) return 0f

        val yaw = this.headEulerAngleY // Yaw (left/right head movement)

        // If yaw is less than 0 (head looking left), return percentage based on how far left
        return if (yaw < 0) {
            val percentage = ((-yaw) / 40 * 100).coerceIn(0f, 100f) // Normalize to 0-100%
            percentage
        } else {
            0f // If head is looking right or centered, return 0
        }
    }

    fun Face?.getRightPercentage(): Float {
        if (this == null) return 0f

        val yaw = this.headEulerAngleY // Yaw (left/right head movement)

        // If yaw is greater than 0 (head looking right), return percentage based on how far right
        return if (yaw > 0) {
            val percentage = ((yaw) / 40 * 100).coerceIn(0f, 100f) // Normalize to 0-100%
            percentage
        } else {
            0f // If head is looking left or centered, return 0
        }
    }

    fun Face.isFullFace(): Boolean {
        // Get face landmarks (eyes, nose, etc.)
        val landmarks = allLandmarks
        val leftEye = landmarks.find { it.landmarkType == FaceLandmark.LEFT_EYE }
        val rightEye = landmarks.find { it.landmarkType == FaceLandmark.RIGHT_EYE }
        val noseBase = landmarks.find { it.landmarkType == FaceLandmark.NOSE_BASE }
        val mouthLeft = landmarks.find { it.landmarkType == FaceLandmark.MOUTH_LEFT }
        val mouthRight = landmarks.find { it.landmarkType == FaceLandmark.MOUTH_RIGHT }

        // Check if key landmarks are present (this ensures the face is not occluded)
        if (leftEye == null || rightEye == null || noseBase == null || mouthLeft == null || mouthRight == null) {
            return false
        }

        // Check if Euler Y angle is near 0 (face looking straight)
        val eulerY = headEulerAngleY
        if (Math.abs(eulerY) > 15) { // 15-degree tolerance for tilt
            return false
        }

        // The face is considered full and straight if all landmarks are visible and Euler Y angle is minimal
        return true
    }

    override fun onFailure(e: Exception) {
        Log.e(TAG, "Face detection failed $e")
    }

    companion object {
        private const val TAG = "FaceDetectorProcessor"
    }
}
