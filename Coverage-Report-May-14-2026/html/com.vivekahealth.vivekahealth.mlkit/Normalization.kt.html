package com.vivekahealth.vivekahealth.mlkit

import kotlin.text.iterator

data class NormalizationResult(
    val normalized: String,
    val wasModified: Boolean,
    val modifications: List<String> = emptyList()
)

class IdNormalizer {
    companion object {
        // Valid prefixes - only these should trigger normalization
        private val VALID_PREFIXES = setOf(
            "ASF", "FUN", "TLO", "XYB", "GLW",
            "EGA", "FFI", "PTH", "QCH", "QEW"
        )

        // OCR character confusion mappings
        private val CHAR_TO_DIGIT = mapOf(
            'O' to '0',
            'o' to '0',
            'I' to '1',
            'l' to '1',
            '|' to '1',
            'Z' to '2',
            'z' to '2',
            'G' to '6',
            'g' to '6',
            'S' to '5',
            's' to '5',
            'B' to '8',
            'D' to '0'
        )
    }

    /**
     * Normalizes an extracted ID with OCR corrections
     * Only processes IDs that match valid prefix patterns or APU suffix pattern
     */
    fun normalize(extractedId: String): NormalizationResult {
        val modifications = mutableListOf<String>()
        val normalized = extractedId.trim().uppercase()

        // Step 1: Check if it ends with APU
        val hasAPU = normalized.endsWith("APU")

        if (hasAPU) {
            // Pattern: numeric + APU (no prefix)
            return normalizeApuFormat(normalized, modifications)
        } else {
            // Pattern: PREFIX + numeric (no APU)
            // Only process if it matches one of our valid prefixes
            return normalizePrefixFormat(normalized, extractedId, modifications)
        }
    }

    /**
     * Normalize IDs with APU suffix (no prefix)
     * Format: digits + APU
     */
    private fun normalizeApuFormat(extractedId: String, modifications: MutableList<String>): NormalizationResult {
        // Remove APU suffix
        val withoutAPU = extractedId.dropLast(3)

        // Normalize the numeric portion
        val normalizedNumeric = fixNumericPortion(withoutAPU, modifications)

        // Validate it's all digits and correct length
        if (normalizedNumeric.length !in 4..20) {
            modifications.add(
                "Warning: Numeric portion length ${normalizedNumeric.length} not in range 4-20"
            )
        }

        val finalId = normalizedNumeric + "APU"
        val wasModified = finalId != extractedId

        return NormalizationResult(finalId, wasModified, modifications)
    }

    /**
     * Normalize IDs with prefix (no APU)
     * Format: PREFIX + digits
     * Only processes if prefix matches one of the valid prefixes (with OCR tolerance)
     */
    private fun normalizePrefixFormat(
        normalized: String,
        originalId: String,
        modifications: MutableList<String>
    ): NormalizationResult {
        // Step 1: Try to extract potential prefix (first 3 chars)
        if (normalized.length < 7) { // At least 3 char prefix + 4 digits minimum
            return NormalizationResult(
                originalId,
                false,
                listOf("Info: ID too short to match pattern")
            )
        }

        val potentialPrefix = normalized.take(3)

        // Step 2: Check if this could be a valid prefix (with OCR errors)
        val correctedPrefix = findMatchingPrefix(potentialPrefix)

        if (correctedPrefix == null) {
            // Not matching any valid prefix, return as-is (don't normalize)
            return NormalizationResult(
                originalId,
                false,
                listOf("Info: Prefix '$potentialPrefix' does not match any valid prefix")
            )
        }

        // Step 3: We found a matching prefix, proceed with normalization
        if (correctedPrefix != potentialPrefix) {
            modifications.add("Prefix: $potentialPrefix → $correctedPrefix")
        }

        // Step 4: Extract numeric portion
        val numericPart = normalized.drop(3)

        // Step 5: Normalize the numeric portion (fix OCR confusions)
        // QEW prefix: M, W, A are valid mid-ID characters (not OCR errors)
        val allowedAlpha = if (correctedPrefix == "QEW") setOf('M', 'W', 'A') else emptySet()
        val normalizedNumeric = fixNumericPortion(numericPart, modifications, allowedAlpha)

        // Step 6: Validate numeric portion length
        if (normalizedNumeric.length !in 4..20) {
            modifications.add(
                "Warning: Numeric portion length ${normalizedNumeric.length} not in range 4-20"
            )
        }

        // Step 7: Reconstruct the ID
        val finalId = correctedPrefix + normalizedNumeric

        val wasModified = finalId != originalId
        return NormalizationResult(finalId, wasModified, modifications)
    }

    /**
     * Find matching valid prefix considering OCR errors
     * Returns null if no match found
     */
    private fun findMatchingPrefix(extractedPrefix: String): String? {
        if (extractedPrefix.length != 3) return null

        // First, check for exact match
        if (VALID_PREFIXES.contains(extractedPrefix)) {
            return extractedPrefix
        }

        // Check each valid prefix for potential OCR matches
        for (validPrefix in VALID_PREFIXES) {
            val corrected = attemptPrefixCorrection(extractedPrefix, validPrefix)
            if (corrected != null) {
                return corrected
            }
        }

        return null
    }

    /**
     * Attempt to correct prefix by replacing likely OCR errors
     * Returns the valid prefix if correction is successful, null otherwise
     */
    private fun attemptPrefixCorrection(extracted: String, target: String): String? {
        if (extracted.length != 3 || target.length != 3) return null

        var differences = 0
        val potentialFixes = mutableListOf<Pair<Int, Char>>()

        for (i in 0..2) {
            if (extracted[i] != target[i]) {
                differences++

                // Check if this is a likely OCR confusion
                val isLikelyOcrError = when (i) {
                    2 -> {
                        // Last character might be O/I confused with 0/1
                        (extracted[i] == '0' && target[i] == 'O') ||
                            (extracted[i] == 'O' && target[i] == '0') ||
                            (extracted[i] == '1' && target[i] == 'I') ||
                            (extracted[i] == 'I' && target[i] == '1') ||
                            (extracted[i] == 'l' && target[i] == 'I') ||
                            (extracted[i] == 'L' && target[i] == 'I')
                    }
                    else -> {
                        // First two characters: check for common confusions
                        (extracted[i] == '0' && target[i] == 'O') ||
                            (extracted[i] == '1' && target[i] == 'I') ||
                            (extracted[i] == 'l' && target[i] == 'I')
                    }
                }

                if (isLikelyOcrError) {
                    potentialFixes.add(Pair(i, target[i]))
                }
            }
        }

        // Only accept if we have 1-2 differences and all are OCR-explainable
        if (differences in 1..2 && potentialFixes.size == differences) {
            return target
        }

        return null
    }

    /**
     * Fix OCR issues in the numeric portion.
     * [allowedAlpha] contains alpha characters that are valid for this ID format and should be kept as-is.
     */
    private fun fixNumericPortion(
        numericPart: String,
        modifications: MutableList<String>,
        allowedAlpha: Set<Char> = emptySet()
    ): String {
        val fixed = StringBuilder()

        for (char in numericPart) {
            when {
                char.isDigit() -> fixed.append(char)
                allowedAlpha.contains(char) -> fixed.append(char)
                CHAR_TO_DIGIT.containsKey(char) -> {
                    val replacement = CHAR_TO_DIGIT[char]!!
                    modifications.add("Numeric: $char → $replacement")
                    fixed.append(replacement)
                }
                else -> {
                    // Unknown character in numeric portion
                    modifications.add(
                        "Warning: Unexpected char '$char' in numeric portion, skipping"
                    )
                    // Skip invalid characters
                }
            }
        }

        return fixed.toString()
    }
}
