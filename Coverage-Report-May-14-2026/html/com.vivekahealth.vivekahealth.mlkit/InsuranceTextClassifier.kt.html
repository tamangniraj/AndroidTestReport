package com.vivekahealth.vivekahealth.mlkit

import android.content.Context
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import org.tensorflow.lite.Interpreter
import timber.log.Timber
import java.io.FileInputStream
import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.MappedByteBuffer
import java.nio.channels.FileChannel

data class TextClassificationResult(
    val label: String, // "Insurance" or "Other"
    val confidence: Float // 0.0 - 1.0 (always the winning class confidence)
)

class InsuranceTextClassifier(private val context: Context) {

    private var interpreter: Interpreter? = null
    private var wordIndex: Map<String, Int> = emptyMap()
    private var maxSequenceLength: Int = 200
    private var oovIndex: Int = 1

    init {
        loadTokenizer()
        loadModel()
    }

    private fun loadTokenizer() {
        val json = context.assets.open("insurance_other/tokenizer_config.json")
            .bufferedReader().use { it.readText() }

        val type = object : TypeToken<Map<String, Any>>() {}.type
        val config: Map<String, Any> = Gson().fromJson(json, type)

        // Gson parses word_index values as Double → cast to Int
        @Suppress("UNCHECKED_CAST")
        val rawWordIndex = config["word_index"] as? Map<String, Double> ?: emptyMap()
        wordIndex = rawWordIndex.mapValues { it.value.toInt() }

        maxSequenceLength = (config["max_sequence_length"] as? Double)?.toInt() ?: 200
        oovIndex = wordIndex[config["oov_token"] as? String ?: "<OOV>"] ?: 1
    }

    private fun loadModel() {
        val modelBuffer = loadModelFile("insurance_other/insurance_classifier_quant.tflite")
        val options = Interpreter.Options().apply {
            numThreads = 4
        }
        interpreter = Interpreter(modelBuffer, options)
    }

    private fun loadModelFile(modelPath: String): MappedByteBuffer {
        val assetFileDescriptor = context.assets.openFd(modelPath)
        val inputStream = FileInputStream(assetFileDescriptor.fileDescriptor)
        val fileChannel = inputStream.channel
        return fileChannel.map(
            FileChannel.MapMode.READ_ONLY,
            assetFileDescriptor.startOffset,
            assetFileDescriptor.declaredLength
        )
    }

    private fun tokenize(text: String): FloatArray {
        val cleaned = text.lowercase()
            .replace(Regex("[^a-z0-9\\s]"), "")
            .trim()

        val words = cleaned.split(Regex("\\s+")).filter { it.isNotEmpty() }

        val tokens = FloatArray(maxSequenceLength) // zero-padded
        val length = minOf(words.size, maxSequenceLength)
        for (i in 0 until length) {
            tokens[i] = (wordIndex[words[i]] ?: oovIndex).toFloat()
        }
        return tokens
    }

    fun classify(ocrText: String): TextClassificationResult? {
        val interpreter = this.interpreter ?: return null
        if (ocrText.isBlank()) return null

        return try {
            val tokens = tokenize(ocrText)

            val inputBuffer = ByteBuffer.allocateDirect(maxSequenceLength * 4)
            inputBuffer.order(ByteOrder.nativeOrder())
            for (token in tokens) {
                inputBuffer.putFloat(token)
            }

            val outputArray = Array(1) { FloatArray(1) }
            interpreter.run(inputBuffer, outputArray)

            val score = outputArray[0][0]
            if (score > 0.5f) {
                TextClassificationResult("Insurance", score)
            } else {
                TextClassificationResult("Other", 1f - score)
            }
        } catch (e: Exception) {
            Timber.e(e, "Text classification failed")
            null
        }
    }

    fun close() {
        interpreter?.close()
    }
}
