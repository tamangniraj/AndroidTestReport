package com.vivekahealth.vivekahealth.mlkit

import android.content.Context
import android.graphics.Bitmap
import org.tensorflow.lite.Interpreter
import org.tensorflow.lite.DataType
import timber.log.Timber
import java.io.FileInputStream
import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.MappedByteBuffer
import java.nio.channels.FileChannel
import kotlin.math.min

class DocumentClassifier(private val context: Context) {

    private var interpreter: Interpreter? = null
    private var inputHeight = 0
    private var inputWidth = 0
    private var numChannels = 0
    private var numClasses = 0
    private lateinit var inputDataType: DataType

    private val labels = listOf("Insurance", "SSN", "Other")

    fun loadModel() {
        val modelBuffer = loadModelFile("insurance_ssn_other/document_classifier_quantized.tflite")
        val options = Interpreter.Options().apply {
            numThreads = 4
        }
        interpreter = Interpreter(modelBuffer, options).also { interp ->
            val inputTensor = interp.getInputTensor(0)
            val inputShape = inputTensor.shape() // [1, 224, 224, 3]
            inputDataType = inputTensor.dataType() // UINT8
            inputHeight = inputShape[1] // 224
            inputWidth = inputShape[2] // 224
            numChannels = inputShape[3] // 3

            val outputTensor = interp.getOutputTensor(0)
            numClasses = outputTensor.shape()[1] // 3
        }
    }

    private fun loadModelFile(modelPath: String): MappedByteBuffer {
        val assetFileDescriptor = context.assets.openFd(modelPath)
        val inputStream = FileInputStream(assetFileDescriptor.fileDescriptor)
        val fileChannel = inputStream.channel
        return fileChannel.map(
            FileChannel.MapMode.READ_ONLY,
            assetFileDescriptor.startOffset,
            assetFileDescriptor.declaredLength
        )
    }

    private fun centerCropToSquare(bitmap: Bitmap): Bitmap {
        val size = min(bitmap.width, bitmap.height)
        val x = (bitmap.width - size) / 2
        val y = (bitmap.height - size) / 2
        return Bitmap.createBitmap(bitmap, x, y, size, size)
    }

    private fun convertBitmapToByteBuffer(bitmap: Bitmap): ByteBuffer {
        val bytesPerChannel = if (inputDataType == DataType.UINT8) 1 else 4
        val byteBuffer = ByteBuffer.allocateDirect(
            bytesPerChannel * inputHeight * inputWidth * numChannels
        )
        byteBuffer.order(ByteOrder.nativeOrder())

        val pixels = IntArray(inputHeight * inputWidth)
        bitmap.getPixels(pixels, 0, inputWidth, 0, 0, inputWidth, inputHeight)

        for (pixel in pixels) {
            val r = (pixel shr 16 and 0xFF)
            val g = (pixel shr 8 and 0xFF)
            val b = (pixel and 0xFF)

            if (inputDataType == DataType.UINT8) {
                byteBuffer.put(r.toByte())
                byteBuffer.put(g.toByte())
                byteBuffer.put(b.toByte())
            } else {
                byteBuffer.putFloat(r / 255.0f)
                byteBuffer.putFloat(g / 255.0f)
                byteBuffer.putFloat(b / 255.0f)
            }
        }
        return byteBuffer
    }

    fun classify(bitmap: Bitmap): List<ClassificationResult> {
        val interpreter = this.interpreter ?: return emptyList()
        val croppedBitmap = centerCropToSquare(bitmap)
        val resizedBitmap = Bitmap.createScaledBitmap(croppedBitmap, inputWidth, inputHeight, true)
        val inputBuffer = convertBitmapToByteBuffer(resizedBitmap)

        return try {
            if (inputDataType == DataType.UINT8) {
                val outputArray = Array(1) { ByteArray(numClasses) }
                interpreter.run(inputBuffer, outputArray)

                val results = mutableListOf<ClassificationResult>()
                for (i in 0 until numClasses) {
                    val confidence = (outputArray[0][i].toInt() and 0xFF) / 255.0f
                    results.add(
                        ClassificationResult(i, labels.getOrElse(i) { "Unknown" }, confidence)
                    )
                }
                results.sortedByDescending { it.confidence }
            } else {
                val outputArray = Array(1) { FloatArray(numClasses) }
                interpreter.run(inputBuffer, outputArray)

                val results = mutableListOf<ClassificationResult>()
                for (i in 0 until numClasses) {
                    results.add(
                        ClassificationResult(
                            i,
                            labels.getOrElse(i) { "Unknown" },
                            outputArray[0][i]
                        )
                    )
                }
                results.sortedByDescending { it.confidence }
            }
        } catch (e: Exception) {
            Timber.e(e, "Classification failed")
            emptyList()
        }
    }

    fun close() {
        interpreter?.close()
    }

    data class ClassificationResult(
        val index: Int, // 0=Insurance, 1=SSN, 2=Other
        val label: String, // "Insurance", "SSN", or "Other"
        val confidence: Float // 0.0 to 1.0
    )
}
