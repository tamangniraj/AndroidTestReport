    /**
     * Calculate similarity between two strings (Levenshtein-based)
     */
    fun calculateSimilarity(str1: String, str2: String): Double {
        val s1 = str1.lowercase()
        val s2 = str2.lowercase()

        if (s1 == s2) return 1.0
        if (s1.isEmpty() || s2.isEmpty()) return 0.0

        val maxLen = maxOf(s1.length, s2.length)
        val distance = levenshteinDistanceSingleRow(s1, s2)
        return 1.0 - (distance.toDouble() / maxLen)
    }

    // Single-row Levenshtein for optimal memory usage
    private fun levenshteinDistanceSingleRow(s1: String, s2: String): Int {
        val prev = IntArray(s2.length + 1) { it }
        val curr = IntArray(s2.length + 1)

        for (i in 1..s1.length) {
            curr[0] = i
            for (j in 1..s2.length) {
                val cost = if (s1[i - 1] == s2[j - 1]) 0 else 1
                curr[j] = minOf(
                    prev[j] + 1, // deletion
                    curr[j - 1] + 1, // insertion
                    prev[j - 1] + cost
                ) // substitution
            }
            for (k in prev.indices) prev[k] = curr[k]
        }

        return curr[s2.length]
    }

    /**
     * Fuzzy-fix OCR-mangled words by matching against known insurance labels.
     * Uses similarity ratio (1 - dist/maxLen) with a 0.70 threshold so heavily
     * mangled words like "lderecation" → "Identification" still get caught.
     */
    private fun fuzzyFixKnownLabels(text: String): String {
        val wordPattern = Regex("\\b[A-Za-z]{4,}\\b")
        return wordPattern.replace(text) { match ->
            val word = match.value
            val wordLower = word.lowercase()

            // Skip if already an exact match (case-insensitive)
            if (knownLabels.any { it.lowercase() == wordLower }) {
                return@replace knownLabels.first { it.lowercase() == wordLower }
            }

            var bestLabel: String? = null
            var bestSimilarity = 0.0

            for (label in knownLabels) {
                val maxLen = maxOf(word.length, label.length)
                // Skip if lengths are too different (>40% difference)
                if (abs(word.length - label.length) > maxLen * 0.4) continue

                val dist = levenshteinDistanceSingleRow(wordLower, label.lowercase())
                val similarity = 1.0 - dist.toDouble() / maxLen

                if (similarity > bestSimilarity && similarity >= 0.60) {
                    bestSimilarity = similarity
                    bestLabel = label
                }
            }

            bestLabel ?: word
        }
    }

    /**
     * Clean and normalize OCR text
     */
    fun cleanOcrText(text: String): String {
        if (text.isEmpty()) return ""

        var cleaned = text

        // Step 1: Remove exclusion phrases
        cleaned = removeExclusionPhrases(cleaned)

        // Replace escaped newlines
        cleaned = cleaned.replace("\\n", "\n")

        // Normalize whitespace (except newlines)
        cleaned = cleaned.replace(Regex("[^\\S\n]+"), " ")

        // Fix common OCR character substitutions
        val ocrFixes = mapOf(
            Regex("(?<=\\d)O(?=\\d)") to "0",
            Regex("\\bl(?=\\s*[A-Z])") to "I",
            Regex("(?<=[A-Z])l(?=[A-Z])") to "I",
            Regex("\\bldentification\\b") to "Identification",
            Regex("\\b[Il]dentification\\b") to "Identification",
            Regex("\\brn\\b") to "m",
            Regex("\\bvv") to "w"
        )

        for ((pattern, replacement) in ocrFixes) {
            cleaned = pattern.replace(cleaned, replacement)
        }

        // Fuzzy fix known insurance labels
        cleaned = fuzzyFixKnownLabels(cleaned)

        // Remove excessive punctuation
        cleaned = cleaned.replace(Regex("[^\\w\\s\\-:/.,#()&@+$]"), "")

        // Normalize spaces
        cleaned = cleaned.replace(Regex("[ \\t]+"), " ")

        // Normalize newlines
        cleaned = cleaned.replace(Regex("\n\\s*\n+"), "\n")

        // Clean each line
        val lines = cleaned.split("\n")
        val cleanedLines = mutableListOf<String>()

        for (line in lines) {
            var lineClean = line.trim()

            // Skip standalone "I" or "i"
            if (lineClean == "I" || lineClean == "i") continue

            // Remove "I " or "i " prefix (left border artifact)
            if (lineClean.startsWith("I ")) {
                lineClean = lineClean.substring(2).trim()
            } else if (lineClean.startsWith("i ")) {
                lineClean = lineClean.substring(2).trim()
            }

            if (lineClean.isNotEmpty()) {
                cleanedLines.add(lineClean)
            }
        }

        return cleanedLines.joinToString("\n").trim()
    }

    /**
     * Fix Personal Choice name split pattern
     */
    private fun fixPersonalChoiceNameSplit(text: String): String {
        if (text.isEmpty()) return text

        val lines = text.split("\n").toMutableList()
        val fixedLines = mutableListOf<String>()
        var i = 0

        while (i < lines.size) {
            val currentLine = lines[i].trim()
            val currentLower = currentLine.lowercase()

            if (currentLower.contains("independence") ||
                currentLower.contains("personal choice") ||
                currentLower.contains("personal choic") ||
                currentLower.contains("indenpdendenc")
            ) {
                fixedLines.add(currentLine)
                i++

                // Check if next line is also header
                if (i < lines.size) {
                    val nextLine = lines[i].trim()
                    val nextLower = nextLine.lowercase()

                    if (nextLower.contains("independence") ||
                        nextLower.contains("personal choice") ||
                        nextLower.contains("personal choic")
                    ) {
                        fixedLines.add(nextLine)
                        i++
                    }
                }

                // Look for name lines and Member ID
                if (i + 2 < lines.size) {
                    var namePart1 = lines[i].trim()
                    var namePart2 = lines[i + 1].trim()
                    val possibleId = lines[i + 2].trim()

                    // Clean left border artifacts
                    if (namePart1 == "I" || namePart1 == "i") {
                        namePart1 = ""
                    } else if (namePart1.startsWith("i ")) {
                        namePart1 = namePart1.substring(2).trim()
                    } else if (namePart1.startsWith("I ")) namePart1 = namePart1.substring(2).trim()

                    if (namePart2 == "I" || namePart2 == "i") {
                        namePart2 = ""
                    } else if (namePart2.startsWith("i ")) {
                        namePart2 = namePart2.substring(2).trim()
                    } else if (namePart2.startsWith("I ")) namePart2 = namePart2.substring(2).trim()

                    // Check if possibleId is a Member ID line
                    val isMemberIdLine = possibleId.contains(
                        Regex("(?i)(member\\s*id|subscriber\\s*id|id)\\s*[:]\\s*[A-Z0-9]")
                    ) ||
                        possibleId.matches(Regex("^[A-Z0-9]{8,15}$"))

                    if (isMemberIdLine) {
                        val mergedName = "$namePart1 $namePart2".trim()
                        fixedLines.add(mergedName)
                        fixedLines.add(possibleId)
                        i += 3
                    } else {
                        fixedLines.add(namePart1)
                        i++
                    }
                }
            } else {
                fixedLines.add(currentLine)
                i++
            }
        }

        return fixedLines.joinToString("\n")
    }

    private fun mergeLabelLines(text: String): String {
        if (text.isEmpty()) return text

        val lines = text.split("\n")
        val mergedLines = mutableListOf<String>()
        var i = 0

        while (i < lines.size) {
            val currentLine = lines[i].trim()

            if (currentLine.endsWith(":") && i + 1 < lines.size) {
                val nextLine = lines[i + 1].trim()

                if (nextLine.isNotEmpty() && !nextLine.endsWith(":")) {

                }
            } else {
                if (currentLine.isNotEmpty()) {
                    mergedLines.add(currentLine)
                }
                i++
            }
        }

        return mergedLines.joinToString("\n")
    }

    /**
     * Merge split name lines (consecutive ALL CAPS single words)
     */
    private fun mergeSplitNameLines(text: String): String {
        if (text.isEmpty()) return text

        // Skip if there's already a name label
        if (text.contains(
                Regex("(?i)(?:Name|Member|Subscriber|Patient|Cardholder|Beneficiary)\\s*:")
            )
        ) {
            return text
        }

        val nonNameTerms = setOf(
            "plan", "hlt", "pcp", "spec", "inn", "oon", "ind", "fam", "ded",
            "oop", "max", "bin", "pcn", "grp", "rxid", "group", "member",
            "subscriber", "patient", "cardholder", "identification", "id",

            "emergency", "urgent", "hospital", "medical", "dental", "vision",
            "pharmacy", "prescription", "drug", "program", "benefits",

            "visit", "er", "lab", "xray", "imaging", "therapy", "mental",
            "behavioral", "substance", "abuse", "maternity", "pediatric",
            "adult", "senior", "family", "individual", "single", "employee",
            "dependent", "spouse", "child", "children", "coverage", "covered",
            "excluded", "limitation", "exclusion", "preexisting", "condition"
        )

        val lines = text.split("\n")
        val mergedLines = mutableListOf<String>()
        var i = 0

        while (i < lines.size) {
            val currentLine = lines[i].trim()

            // Check if current line is single ALL CAPS word
            if (currentLine.matches(Regex("^[A-Z]{2,15}$")) &&
                currentLine.lowercase() !in nonNameTerms &&
                i + 1 < lines.size
            ) {
                val nextLine = lines[i + 1].trim()

                // Check if next line is also single ALL CAPS word
                if (nextLine.matches(Regex("^[A-Z]{2,15}$")) &&
                    nextLine.lowercase() !in nonNameTerms
                ) {
                    // Check if followed by insurance data
                    var isFollowedByInsuranceData = false
                    if (i + 2 < lines.size) {
                        val thirdLine = lines[i + 2].trim().lowercase()
                        if (thirdLine.contains(
                                Regex("(plan|pcp|spec|ded|oop|inn|oon|grp|bin|pcn|id|group|member)")
                            ) ||
                            thirdLine.matches(Regex("^[a-z]{2,5}\\s+\\d.*")) ||
                            thirdLine.matches(Regex("^\\d.*"))
                        ) {
                            isFollowedByInsuranceData = true
                        }
                    } else {
                        isFollowedByInsuranceData = true
                    }

                    if (isFollowedByInsuranceData) {
                        mergedLines.add("$currentLine $nextLine")
                        i += 2
                        continue
                    }
                }
            }

            if (currentLine.isNotEmpty()) {
                mergedLines.add(currentLine)
            }
            i++
        }

        return mergedLines.joinToString("\n")
    }

    /**
     * Extract dependent names to exclude from primary member detection
     */
    private fun extractDependentNames(text: String): Set<String> {
        val dependentNames = mutableSetOf<String>()

        val dependentPattern = Regex(
            """Dependents?(?:\(s\))?[\s:]+(.+?)(?=\n(?:Group|Plan|PPO|HMO|EPO|BIN|PCN|Member|Subscriber|$)|\Z)""",
            setOf(RegexOption.IGNORE_CASE, RegexOption.DOT_MATCHES_ALL)
        )

        val match = dependentPattern.find(text)
        if (match != null) {
            val dependentText = match.groupValues[1]
            for (line in dependentText.split("\n")) {
                val lineTrimmed = line.trim()
                // Check if line looks like a name (ALL CAPS, 2+ words)
                if (lineTrimmed.matches(Regex("^[A-Z]{2,15}(?:\\s+[A-Z]{2,15})+$"))) {
                    dependentNames.add(lineTrimmed.uppercase())
                }
                // Also check comma-separated names
                if (lineTrimmed.contains(",")) {
                    for (part in lineTrimmed.split(",")) {
                        val partTrimmed = part.trim()
                        if (partTrimmed.matches(Regex("^[A-Z]{2,15}(?:\\s+[A-Z]{2,15})+$"))) {
                            dependentNames.add(partTrimmed.uppercase())
                        }
                    }
                }
            }
        }

        // Exclude names after Child/Spouse labels
        val childSpousePattern = Regex(
            """(?:Child|Spouse)[\s:]+([A-Z][A-Za-z\-]+(?:\s+[A-Z][A-Za-z\-]+)+)""",
            RegexOption.IGNORE_CASE
        )
        for (m in childSpousePattern.findAll(text)) {
            dependentNames.add(m.groupValues[1].uppercase().trim())
        }

        return dependentNames
    }

    /**
     * Remove plan names from a line to isolate member name
     */
    private fun removePlanNamesFromLine(line: String): String {
        var cleaned = line

        for (pattern in planNamePatterns) {
            cleaned = cleaned.replace(Regex(pattern, RegexOption.IGNORE_CASE), "")
        }

        for (phrase in OCR_EXCLUSION_PHRASES) {
            if (cleaned.lowercase().contains(phrase.lowercase())) {
                val startIdx = cleaned.lowercase().indexOf(phrase.lowercase())
                if (startIdx != -1) {
                    cleaned = cleaned.substring(0, startIdx) + cleaned.substring(
                        startIdx + phrase.length
                    )
                }
            }
        }

        return cleaned.split("\\s+".toRegex()).filter { it.isNotEmpty() }.joinToString(" ")
    }

    /**
     * Extract name near Member ID / Identification Number label
     */
    private fun extractNameNearMemberIdLabel(text: String): List<Pair<String, Double>> {
        val names = mutableListOf<Pair<String, Double>>()
        var foundPrimary = false

        val lines = text.split("\n")

        for (i in lines.indices) {
            val lineStripped = lines[i].trim()
            val lineLower = lineStripped.lowercase()

            // Pattern 1: Name appears BEFORE "Identification Number" or "Member ID" on next line
            if (i + 1 < lines.size) {
                val nextLine = lines[i + 1].trim().lowercase()
                if (nextLine.contains("identification") ||
                    nextLine.contains("member id") ||
                    nextLine.contains("subscriber id")
                ) {
                    val cleanedLine = removePlanNamesFromLine(lineStripped)
                    val nameMatch = Regex("^([A-Z]{2,15}(?:\\s+[A-Z]\\.?)?\\s+[A-Z]{2,15})").find(
                        cleanedLine
                    )

                    if (nameMatch != null) {
                        val name = nameMatch.groupValues[1].trim()
                        if (isValidName(name)) {
                            names.add(name to 0.98)
                            foundPrimary = true
                        }
                    }
                }
            }

            // Pattern 2: Current line contains ID label, check for name on NEXT line(s)
            val isIdLine = lineLower.contains(
                Regex(
                    "(medical.*id|mh.*id|map.*id|rx.*id|member\\s*id|subscriber\\s*id|identification)"
                )
            )

            if (isIdLine && !foundPrimary) {
                for (j in i + 1 until minOf(i + 5, lines.size)) {
                    val nextLine = lines[j].trim()

                    if (nextLine.isEmpty()) continue
                    if (nextLine.contains(
                            Regex("(?i)(id\\s*:|#|deductible|co-?pay|network|bin|pcn|grp)")
                        )
                    ) {
                        break
                    }

                    val nameMatch = Regex(
                        "^([A-Z]{2,15}(?:\\s+[A-Z]\\.?)?\\s+[A-Z]{2,15}(?:\\s+[A-Z]{2,15})?)$"
                    ).find(nextLine)
                    if (nameMatch != null) {
                        val name = nameMatch.groupValues[1].trim()
                        if (isValidName(name)) {
                            names.add(name to 0.99)
                            foundPrimary = true
                            break
                        }
                    }
                }
            }
        }

        return names
    }

    /**
     * Extract names using regex patterns
     */
    fun extractNamesRegex(text: String): List<Pair<String, Double>> {
        val names = mutableListOf<Pair<String, Double>>()
        val dependentNames = extractDependentNames(text)

        // Try to extract name near Member ID label (highest priority)
        val namesNearId = extractNameNearMemberIdLabel(text)
        for ((name, conf) in namesNearId) {
            if (name.uppercase() !in dependentNames) {
                names.add(name to conf)
            }
        }

        // Try labeled patterns (higher confidence)
        for (regex in nameLabeledRegexes) {
            val matches = regex.findAll(text)
            for (match in matches) {
                val name = cleanName(
                    match.groupValues.drop(1).firstOrNull { it.isNotEmpty() } ?: continue
                )

                if (name.uppercase() in dependentNames) continue
                if (isValidName(name)) {
                    names.add(name to 0.95)
                }
            }
        }

        // Try unlabeled patterns (lower confidence)
        for (regex in nameUnlabeledRegexes) {
            val matches = regex.findAll(text)
            for (match in matches) {
                val name = cleanName(
                    match.groupValues.drop(1).firstOrNull { it.isNotEmpty() } ?: continue
                )

                if (name.uppercase() in dependentNames) continue
                if (isValidName(name)) {
                    names.add(name to 0.5)
                }
            }
        }

        return names
    }

    /**
     * Validate if a string is likely a person's name
     */
    private fun isValidName(name: String): Boolean {
        if (name.isEmpty() || name.length < 3) return false

        // Clean the name
        var cleanedName = name.replace(Regex("[\n\r]+"), " ")
        cleanedName = cleanedName.replace(Regex("\\s+"), " ").trim()

        if (cleanedName.length < 3) return false

        val nameLower = cleanedName.lowercase()
        val words = nameLower.split(" ")

        // Exact match exclusions
        val exactExclusions = setOf(
            "ind ded", "fam ded", "ind oop", "fam oop",
            "oop max", "ind max", "fam max",
            "in network", "out network",
            "inn oon",
            "personal choice", "personal choic",
            "independence personal", "indenpdendenc",
            "number", "issuer", "prescription", "member", "id", "subscriber",
            "patient", "cardholder", "identification", "memberid"
        )

        if (nameLower in exactExclusions) return false

        // Must have at least 2 words
        if (words.size < 2) return false

        // Must have letters
        if (cleanedName.count { it.isLetter() } < 4) return false

        // Should not be mostly numbers
        if (cleanedName.count { it.isDigit() } > cleanedName.length / 4) return false

        // Additional exclusions
        val additionalExclusions = setOf(
            "member", "subscriber", "patient", "cardholder", "group",
            "plan", "type", "tvpe", "id", "number", "no", "local", "demo",
            "eligible", "eligibility", "elcible", "updaled", "updated",
            "effective", "valid", "expires", "bin", "pcn", "grp", "rx",
            "insurance", "benelils", "adminislered", "administered",
            "benefits", "medlcal", "medical", "dental", "vision",
            "laboratorles", "laboratories", "lab", "account", "issuer",
            "yes"
        )

        val allExclusions = nameExclusions + insuranceTerms + additionalExclusions
        val validSuffixes = setOf("jr", "sr", "iii", "ii", "iv", "v", "jr.", "sr.")

        for (word in words) {
            if (word in allExclusions && word !in validSuffixes) {
                return false
            }
        }

        // Filter if name contains insurance terms
        for (term in insuranceTerms) {
            if (term in nameLower) return false
        }

        // Reasonable length
        if (words.size > 5 || cleanedName.length > 50) return false

        // Each word should look like a name part
        for ((index, word) in cleanedName.split(" ").withIndex()) {
            val wordLower = word.lowercase().trimEnd('.')

            // Skip middle initials
            if (word.length <= 2 && word.endsWith(".")) continue

            // Allow suffixes at the end
            if (index == words.size - 1 && wordLower in validSuffixes) continue

            // Word should start with uppercase
            if (!word[0].isUpperCase()) return false

            // Word should be mostly letters
            if (word.count { it.isLetter() } < word.length * 0.8) return false
        }

        return true
    }

    /**
     * Clean extracted name string
     */
    private fun cleanName(name: String): String {
        return name.replace(Regex("[\n\r]+"), " ")
            .replace(Regex("\\s+"), " ")
            .trim()
    }

    /**
     * Scan [text] for alphanumeric tokens that could be OCR-mangled IDs and
     * normalize them through [IdNormalizer].  Only tokens whose first-3-char
     * prefix matches a known prefix in [IdNormalizer] are touched, so
     * regular words are never altered.
     *
     * e.g. "ASFO02765994" → "ASF002765994" (O→0 in numeric portion)
     */
    private fun normalizeIdTokensInText(text: String): String {
        val normalizer = IdNormalizer()
        return Regex("\\b([A-Z][A-Z0-9]{7,17})\\b").replace(text) { mr ->
            val token = mr.groupValues[1]
            val result = normalizer.normalize(token)
            if (result.wasModified) result.normalized else token
        }
    }

    /**
     * Extract member/subscriber IDs using regex
     */
    fun extractMemberIds(text: String): List<Pair<String, Double>> {
        // Check for special "no ID" patterns
        val noIdPatterns = listOf(
            """Call\s+NVA\s+for\s+your\s+ID""",
            """ID\s*[:\s]*Use\s+your\s+SSN""",
            """Use\s+your\s+SSN""",
            """Contact.*for.*ID""",
            """Call.*for.*ID"""
        )

        for (pattern in noIdPatterns) {
            if (text.contains(Regex(pattern, RegexOption.IGNORE_CASE))) {
                return listOf("N/A" to 1.0)
            }
        }

        // Normalize OCR-mangled tokens that match a known prefix (e.g. ASFO→ASF0)
        // so the regex patterns below can match them correctly.
        val normalizedText = normalizeIdTokensInText(text)

        val ids = mutableListOf<Pair<String, Double>>()
        val foundIds = mutableSetOf<String>()
        val idsToExclude = mutableSetOf<String>()

        // Exclude Rx IDs (handles labels like "Rx/DEN/VIS ID:")
        val rxIdPattern = Regex(
            """(?:Rx(?:/DEN)?(?:/VIS)?)\s*ID[\s:]+([A-Z0-9]{1,15})""",
            RegexOption.IGNORE_CASE
        )
        for (match in rxIdPattern.findAll(normalizedText)) {
            idsToExclude.add(match.groupValues[1].replace("\\s".toRegex(), ""))
        }

        // Exclude Sponsor Numbers
        val sponsorPattern = Regex(
            """Sponsor\s*(?:No|Number|#)?[\s:]*([A-Z0-9]{3,20})""",
            RegexOption.IGNORE_CASE
        )
        for (match in sponsorPattern.findAll(normalizedText)) {
            idsToExclude.add(match.groupValues[1].replace("\\s".toRegex(), ""))
        }

        // Exclude Issuer Numbers
        val issuerPattern = Regex(
            """I?ssuer\s*(?:No|Number|#)?[\s:]+(\d{3,20})""",
            RegexOption.IGNORE_CASE
        )
        for (match in issuerPattern.findAll(normalizedText)) {
            idsToExclude.add(match.groupValues[1].replace("\\s".toRegex(), ""))
        }
        // Exclude Issuer with parenthesized number and trailing number e.g. "Issuer (80840) 9140860054"
        val issuerParenPattern = Regex(
            """(?i)I?ssuer\s*(?:No|Number|#)?[\s:]*\((\d{3,20})\)[\s:]*(\d{3,20})"""
        )
        for (match in issuerParenPattern.findAll(normalizedText)) {
            idsToExclude.add(match.groupValues[1])
            idsToExclude.add(match.groupValues[2])
        }

        // Exclude Group ID / Group Number values
        val groupPattern = Regex(
            """(?:Group\s*(?:Name|No|Number|#|ID)?|Grp|GRP)[\s:]+([A-Z0-9][A-Z0-9\-]{2,19})""",
            RegexOption.IGNORE_CASE
        )
        for (match in groupPattern.findAll(normalizedText)) {
            idsToExclude.add(match.groupValues[1].replace("\\s".toRegex(), ""))
        }

        // Exclude currency amounts e.g. "$2000", "$200", "$500"
        val currencyPattern = Regex("""\$(\d+(?:[.,]\d+)?)""")
        for (match in currencyPattern.findAll(normalizedText)) {
            idsToExclude.add(match.groupValues[1].replace(",", "").replace(".", ""))
        }

        // Exclude phone numbers on lines containing "call" or "phone"
        val phoneLinePattern = Regex(
            """^.*(?:call|phone|dial|contact|fax).*?(\(?\d{3}\)?[\s\-.]?\d{3}[\s\-.]?\d{4}).*$""",
            setOf(RegexOption.IGNORE_CASE, RegexOption.MULTILINE)
        )
        for (match in phoneLinePattern.findAll(normalizedText)) {
            val phoneDigits = match.groupValues[1].replace(Regex("[^\\d]"), "")
            idsToExclude.add(phoneDigits)
            // Also exclude with dashes in case it appears that way
            idsToExclude.add(match.groupValues[1].replace(Regex("[\\s.()]"), ""))
        }

        for ((index, regex) in idRegexes.withIndex()) {
            val matches = regex.findAll(normalizedText)
            for (match in matches) {
                val matchValue =
                    match.groupValues.drop(1).firstOrNull { it.isNotEmpty() } ?: continue
                val memberId = matchValue.replace("\\s".toRegex(), "")

                if (memberId in idsToExclude) continue

                if (isValidId(memberId) && memberId !in foundIds) {
                    val confidence = if (index < 3) 0.95 else 0.7
                    ids.add(memberId to confidence)
                    foundIds.add(memberId)
                }
            }
        }

        // Only apply fallback if no IDs found yet
        if (ids.isEmpty()) {
            for (regex in fallbackIdRegexes) {
                for (match in regex.findAll(normalizedText)) {
                    val extractedId = match.groups[2]?.value // <-- use group 2, not 1
                    if (!extractedId.isNullOrEmpty() && isValidId(extractedId) && extractedId !in idsToExclude) {
                        ids.add(Pair(extractedId, 1.0))
                    }
                }
            }
        }

        return ids
    }

    /**
     * Validate if a string is likely a member ID
     */
    private fun isValidId(idStr: String): Boolean {
        if (idStr.isEmpty() || idStr.length < 6) return false

        val invalidIdWords = setOf(
            "number", "member", "issuer", "prescription", "subscriber",
            "patient", "cardholder", "identification", "memberid",
            "benefits", "insurance", "healthcare", "medical", "dental",
            "vision", "pharmacy", "account", "sponsor", "group",
            "eligible", "eligibility", "effective", "coverage",
            "network", "provider", "service", "program", "plan",
            "electronic", "payer", "facilitator", "administrator"
        )

        if (idStr.lowercase() in invalidIdWords) return false

        // Not a year
        if (idStr.matches(Regex("^(19|20)\\d{2}$"))) return false

        // Not a phone number (US format: NXX-NXX-XXXX where N=2-9)
        val cleanId = idStr.replace(Regex("[-\\s.]"), "")
        if (cleanId.matches(Regex("^[2-9]\\d{2}[2-9]\\d{6}$"))) return false

        // Not too long
        if (idStr.length > 20) return false

        // Must contain at least one digit
        if (!idStr.any { it.isDigit() }) return false

        return true
    }

    /**
     * Extract dates using regex patterns
     */
    fun extractDates(text: String): List<Pair<String, Double>> {
        val dates = mutableListOf<Pair<String, Double>>()
        val foundDates = mutableSetOf<String>()

        for ((index, regex) in dateRegexes.withIndex()) {
            val matches = regex.findAll(text)
            for (match in matches) {
                val parts = match.groupValues.drop(1).filter { it.isNotEmpty() }
                val dateStr = parts.joinToString("/")

                val normalized = normalizeDate(dateStr)
                if (normalized != null && normalized !in foundDates) {
                    val confidence = if (index < 2) 0.95 else 0.8
                    dates.add(normalized to confidence)
                    foundDates.add(normalized)
                }
            }
        }

        return dates
    }

    /**
     * Normalize date to readable format
     */
    private fun normalizeDate(dateStr: String): String? {
        if (dateStr.isEmpty() || dateStr.length < 6) return null

        return dateStr.replace("\\s+".toRegex(), " ")
            .trim()
            .replace("-", "/")
    }

    /**
     * Build merged text from vision text blocks by grouping lines that share
     * the same vertical position. Handles three layout patterns:
     *
     * 1. Key and value on the SAME line in nearby blocks → join them:
     *    Block["Member ID"]  Block["ABC123"]  →  "Member ID ABC123"
     *
     * 2. Keys on one row, values on the NEXT row → pair by X-proximity:
     *    Row1: "Sponsor #"   "Cardmember #"
     *    Row2: "4279000001"  "XYBC9JEOCLIORFTB"
     *    →  "Sponsor # 4279000001"
     *    →  "Cardmember # XYBC9JEOCLIORFTB"
     *
     * 3. Two-panel cards (left/right columns from different blocks) → keep separate:
     *    Left["Member Name: Lloyd Bates"]  Right["Claims Fax: (888) 965-4011"]
     *    →  "Member Name: Lloyd Bates"
     *    →  "Claims Fax: (888) 965-4011"
     */
    fun buildSameLineText(visionText: Text): String {
        data class LineInfo(
            val text: String,
            val yCenter: Float,
            val xLeft: Int,
            val xRight: Int,
            val height: Int,
            val blockIdx: Int
        )

        val allLines = mutableListOf<LineInfo>()

        for ((blockIdx, block) in visionText.textBlocks.withIndex()) {
            for (line in block.lines) {
                val box = line.boundingBox ?: continue
                allLines.add(
                    LineInfo(
                        text = line.text,
                        yCenter = (box.top + box.bottom) / 2f,
                        xLeft = box.left,
                        xRight = box.right,
                        height = box.bottom - box.top,
                        blockIdx = blockIdx
                    )
                )
            }
        }

        if (allLines.isEmpty()) return visionText.text

        // Sort by Y center
        allLines.sortBy { it.yCenter }

        // Group lines with similar Y center (tolerance = half of average line height)
        val avgHeight = allLines.map { it.height }.average()
        val yTolerance = (avgHeight * 0.5).toFloat()
        // Column gap: items from different blocks with gap > this are separate panels
        val columnGapThreshold = (avgHeight * 2).toInt()

        val yGroups = mutableListOf<MutableList<LineInfo>>()
        var currentGroup = mutableListOf(allLines[0])

        for (idx in 1 until allLines.size) {
            val current = allLines[idx]
            val groupYCenter = currentGroup.map { it.yCenter }.average().toFloat()

            if (abs(current.yCenter - groupYCenter) <= yTolerance) {
                currentGroup.add(current)
            } else {
                yGroups.add(currentGroup)
                currentGroup = mutableListOf(current)
            }
        }
        yGroups.add(currentGroup)

        // Sort each group left-to-right
        yGroups.forEach { it.sortBy { line -> line.xLeft } }

        // Split a Y-group into column sub-groups.
        // A column boundary exists when consecutive items are from DIFFERENT
        // blocks AND have a large horizontal gap (separate panels).
        fun splitIntoColumns(group: List<LineInfo>): List<List<LineInfo>> {
            if (group.size <= 1) return listOf(group)
            val columns = mutableListOf<MutableList<LineInfo>>()
            var col = mutableListOf(group[0])
            for (j in 1 until group.size) {
                val prev = group[j - 1]
                val curr = group[j]
                val gap = curr.xLeft - prev.xRight
                if (prev.blockIdx != curr.blockIdx && gap > columnGapThreshold) {
                    columns.add(col)
                    col = mutableListOf(curr)
                } else {
                    col.add(curr)
                }
            }
            columns.add(col)
            return columns
        }

        // Process groups
        val result = mutableListOf<String>()
        var i = 0
        while (i < yGroups.size) {
            val group = yGroups[i]
            val columns = splitIntoColumns(group)

            if (columns.size >= 2) {
                // Multiple columns detected (e.g. two-panel card)
                val allColumnsAreLabels = columns.all { col ->
                    col.all { isLabelText(it.text) }
                }

                if (allColumnsAreLabels && i + 1 < yGroups.size) {
                    // All columns are labels → pair with values from next row
                    val nextGroup = yGroups[i + 1]
                    val allLabels = columns.flatten()
                    val allValues = splitIntoColumns(nextGroup).flatten()
                    val usedValues = mutableSetOf<Int>()

                    for (label in allLabels) {
                        var closestIdx = -1
                        var closestDist = Int.MAX_VALUE
                        for (j in allValues.indices) {
                            if (j in usedValues) continue
                            val dist = abs(label.xLeft - allValues[j].xLeft)
                            if (dist < closestDist) {
                                closestDist = dist
                                closestIdx = j
                            }
                        }
                        if (closestIdx != -1) {
                            result.add("${label.text} ${allValues[closestIdx].text}")
                            usedValues.add(closestIdx)
                        } else {
                            result.add(label.text)
                        }
                    }
                    for (j in allValues.indices) {
                        if (j !in usedValues) {
                            result.add(allValues[j].text)
                        }
                    }
                    i += 2
                } else {
                    // Mixed content → output each column as a separate line
                    for (col in columns) {
                        result.add(col.joinToString(" ") { it.text })
                    }
                    i++
                }
            } else {
                // Single column → join same-line items with space
                result.add(group.joinToString(" ") { it.text })
                i++
            }
        }

        // Split lines that contain multiple key:value pairs from cross-panel merging
        // e.g. "Lloyd Bates Claims Fax: (888) 965-4011" → two separate lines
        val finalResult = result.flatMap { splitAtKeyBoundaries(it) }
        return finalResult.joinToString("\n")
    }

    /**
     * Split a line at label boundaries. When cross-panel merging produces
     * lines like "Lloyd Bates Claims Fax: (888) 965-4011", this splits into
     * ["Lloyd Bates", "Claims Fax: (888) 965-4011"].
     *
     * Only splits when a recognized label keyword (followed by : or #)
     * appears in the MIDDLE of a line with content before it.
     */
    private val keyBoundaryPattern = Regex(
        "(?i)\\b(?:" +
            "Member\\s*(?:Name|ID|Number|#)|" +
            "Subscriber\\s*(?:Name|ID)|" +
            "Patient\\s*(?:Name|ID)|" +
            "Cardholder\\s*(?:Name|ID)|" +
            "Cardmember\\s*#?|" +
            "Group\\s*(?:ID|No|Number|Name|#)|" +
            "Plan\\s*(?:Type|Year)|" +
            "Effective\\s*Date|" +
            "Deductible|Coinsurance|Co-?pay|" +
            "Rx\\s*(?:BIN|PCN|GRP|ID)|RxBIN|RxPCN|RxGRP|RxID|" +
            "Issuer|Sponsor\\s*(?:No|Number|#)?|" +
            "Electronic\\s*Payer\\s*ID|" +
            "Claims\\s*(?:Fax|Address|Phone)|" +
            "Eligibility|Eligible|" +
            "Identification\\s*(?:Number|No)|" +
            "(?<!Group\\s)Name|" +
            "BIN|PCN|GRP|DOB|Updated|Network" +
            ")\\s*[:#]"
    )

    private fun splitAtKeyBoundaries(line: String): List<String> {
        val matches = keyBoundaryPattern.findAll(line).toList()
        if (matches.isEmpty()) return listOf(line)

        // Collect split points: positions of keys that have content before them
        val splitPoints = mutableListOf<Int>()
        for (match in matches) {
            val pos = match.range.first
            if (pos > 0 && line.substring(0, pos).trim().isNotEmpty()) {
                splitPoints.add(pos)
            }
        }

        if (splitPoints.isEmpty()) return listOf(line)

        val result = mutableListOf<String>()
        var lastPos = 0
        for (pos in splitPoints) {
            val segment = line.substring(lastPos, pos).trim()
            if (segment.isNotEmpty()) result.add(segment)
            lastPos = pos
        }
        val remaining = line.substring(lastPos).trim()
        if (remaining.isNotEmpty()) result.add(remaining)

        return result
    }

    /**
     * Check if a text fragment looks like a label/key (not a value).
     * Labels typically end with #, :, or match known field names.
     */
    private fun isLabelText(text: String): Boolean {
        val trimmed = text.trim()
        // Ends with common label delimiters
        if (trimmed.endsWith("#") || trimmed.endsWith(":")) return true
        // Matches known insurance card field labels
        return trimmed.matches(
            Regex(
                "(?i)^(Sponsor|Cardmember|Member|Subscriber|Patient|Cardholder|Group|" +
                    "Name|ID|Identification|Issuer|Plan|Rx\\s*BIN|Rx\\s*PCN|Rx\\s*GRP|" +
                    "Rx\\s*ID|BIN|PCN|GRP|Eligible|Effective|Updated|DOB|Date)\\s*[#:.]?\\s*$"
            )
        )
    }

    fun extractFromDocument(visionText: Text): ExtractedData {

     */
    fun extractFromText(rawText: String): ExtractedData =
        extractFromDocument(mapOf("full_text" to rawText))

    /**
     * Extract all insurance card details from a single OCR document
     */
    fun extractFromDocument(ocrDoc: Map<String, Any?>): ExtractedData {
        val text = extractTextFromOcr(ocrDoc)

        // Extract names using regex only (no spacy in Kotlin)
        val namesRegex = extractNamesRegex(text)
        val combined = namesRegex.sortedByDescending { it.second }

//
//        // Combine and deduplicate by confidence
        val allNames = mutableListOf<String>()
        val seenNames = mutableSetOf<String>()

        for ((name, _) in combined) {
            val nameNorm = name.lowercase().trim()
            if (nameNorm !in seenNames) {
                allNames.add(name)
                seenNames.add(nameNorm)
            }
        }

        val ids = extractMemberIds(text).sortedByDescending { it.second }.map { it.first }

        val phone = extractPhone(text)

        return ExtractedData(
            memberIds = ids,
            rawText = text,
            names = allNames,
            phone = phone
        )
    }
