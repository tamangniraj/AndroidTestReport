package com.vivekahealth.vivekahealth.mlkit

import android.content.Context
import android.graphics.Canvas
import android.graphics.Paint
import android.graphics.RectF
import android.util.AttributeSet
import android.view.View

class ArcProgressBar @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
    defStyleAttr: Int = 0
) : View(context, attrs, defStyleAttr) {

    private val rect = RectF() // RectF to draw arcs

    val colorLeft = 0xFFEBD301.toInt()
    val colorRight = 0xFFEBD301.toInt()

    val padding = 15f
    val arcWidth = 25f

    // Paints for the background and foreground arcs
    private val backgroundPaint = Paint().apply {
        color = 0xFFE0E0E0.toInt() // Light Gray for background
        style = Paint.Style.STROKE
        strokeWidth = arcWidth
        isAntiAlias = true
    }

    private val arcPaintBottomRight = Paint().apply {
        color = colorRight // Blue for top progress
        style = Paint.Style.STROKE
        strokeWidth = arcWidth
        isAntiAlias = true
    }

    private val arcPaintTopRight = Paint().apply {
        color = colorRight // Red for bottom progress
        style = Paint.Style.STROKE
        strokeWidth = arcWidth
        isAntiAlias = true
    }

    private val arcPaintTopLeft = Paint().apply {
        color = colorLeft // Green for left progress
        style = Paint.Style.STROKE
        strokeWidth = arcWidth
        isAntiAlias = true
    }

    private val arcPaintBottomLeft = Paint().apply {
        color = colorLeft // Yellow for right progress
        style = Paint.Style.STROKE
        strokeWidth = arcWidth
        isAntiAlias = true
    }

    private val arcPaintTopBottom = Paint().apply {
        color = colorLeft // Green for left progress
        style = Paint.Style.STROKE
        strokeWidth = arcWidth
        isAntiAlias = true
    }

    private var progressLeftTop = 0f
    private var progressLeftBottom = 0f

    private var progressRightTop = 0f
    private var progressRightBottom = 0f

    private var progressTopBottom = 0f

    var gap = 22.5f

    var padding_between_arc = 2f

    // Set progress values for each arc (in degrees)
    fun setProgress(p1: Float, p2: Float, p3: Float, p4: Float, p5: Float) {
        progressLeftTop = (p1 / 100f) * (45f + gap)
        progressLeftBottom = (p2 / 100f) * (45f + gap)

        progressRightTop = (p3 / 100f) * (45f + gap)
        progressRightBottom = (p4 / 100f) * (45f + gap)

        progressTopBottom = (p5 / 100f) * (gap - padding_between_arc)

        updateColorOfArc(p1, p2, p3, p4, p5)

        invalidate() // Redraw the view when progress is updated
    }

    fun updateColorOfArc(p1: Float, p2: Float, p3: Float, p4: Float, p5: Float) {
        if (p1 == 100f && p2 == 100f) {
            arcPaintTopLeft.color = 0xAB5DBA3A.toInt()
            arcPaintBottomLeft.color = 0xAB5DBA3A.toInt()
        } else {
            arcPaintTopLeft.color = 0xFFEBD301.toInt()
            arcPaintBottomLeft.color = 0xFFEBD301.toInt()
        }

        if (p3 == 100f && p4 == 100f) {
            arcPaintTopRight.color = 0xAB5DBA3A.toInt()
            arcPaintBottomRight.color = 0xAB5DBA3A.toInt()
        } else {
            arcPaintTopRight.color = 0xFFEBD301.toInt()
            arcPaintBottomRight.color = 0xFFEBD301.toInt()
        }

        if (p5 == 100f) {
            arcPaintTopBottom.color = 0xAB5DBA3A.toInt()
        } else {
            arcPaintTopBottom.color = 0xFFEBD301.toInt()
        }
    }

    override fun onSizeChanged(w: Int, h: Int, oldw: Int, oldh: Int) {
        super.onSizeChanged(w, h, oldw, oldh)
        // Define the RectF to draw arcs, ensuring the arcs fit within the view
        val padding = padding // Padding around the arcs
        rect.set(padding, padding, w.toFloat() - padding, h.toFloat() - padding)
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)

        // background
        drawBackgroundArc(canvas)

        drawLeftArcProgress(canvas)

        drawRightArcProgress(canvas)

        drawTopBottomArcProgress(canvas)
    }

    private fun drawLeftArcProgress(canvas: Canvas) {
        canvas.drawArc(rect, ((90f * 2) + 45f + gap), -progressLeftTop, false, arcPaintTopLeft)
        canvas.drawArc(rect, (90f + gap), progressLeftBottom, false, arcPaintBottomLeft)
    }

    private fun drawRightArcProgress(canvas: Canvas) {
        canvas.drawArc(rect, ((90f * 3) + gap), progressRightTop, false, arcPaintTopRight)
        canvas.drawArc(rect, (90f - gap), -progressRightBottom, false, arcPaintBottomRight)
    }

    private fun drawTopBottomArcProgress(canvas: Canvas) {
        // progress top Bottom
        canvas.drawArc(rect, (90f), -progressTopBottom, false, arcPaintTopBottom)
        canvas.drawArc(rect, (90f), progressTopBottom, false, arcPaintTopBottom)

        canvas.drawArc(rect, (90 * 3f), -progressTopBottom, false, arcPaintTopBottom)
        canvas.drawArc(rect, (90 * 3f), progressTopBottom, false, arcPaintTopBottom)
    }

    private fun drawBackgroundArc(canvas: Canvas) {
        // Draw background arcs with gaps
        canvas.drawArc(rect, (90f * 3) + gap, (90f + 45f), false, backgroundPaint) // Top Left Background
        canvas.drawArc(rect, (90f + gap), 90f + 45f, false, backgroundPaint) // Top Right Background

        // top - background
        canvas.drawArc(
            rect,
            (((90f * 3) - gap + padding_between_arc)),
            45f - (2 * padding_between_arc),
            false,
            backgroundPaint
        ) // Top Right Background

        // bottom - background
        canvas.drawArc(
            rect,
            ((90f - gap + padding_between_arc)),
            45f - (2 * padding_between_arc),
            false,
            backgroundPaint
        ) // Top Right Background
    }
}
