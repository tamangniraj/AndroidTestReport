package com.vivekahealth.vivekahealth.data.remote.home.workBenefit.benefitPerMonth

import com.vivekahealth.vivekahealth.utils.DateUtils

data class Member(
    val eligible: Boolean?,
    val exDependent: Boolean = false,
    val firstName: String?,
    val inEligibilityReason: List<String>? = listOf(),
    val lastName: String?,
    val middleName: String?,
    val relationship: Int?,
    val relationshipDesc: String?,
    val selfieAvailable: Boolean,
    val memberId: Long,
    val terminationDate: String?
) {
    /**
     * Return terminated date as a formatted string.
     */
    fun getFormattedDate(): String {
        return if (terminationDate == null) {
            ""
        } else {
            return DateUtils.dateToName(DateUtils.stringServerDateToDate(terminationDate))
        }
    }
}
