class ZipAddViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    var uploadRepository: GuestUploadRepository

) : ViewModel(), Observable {

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack

    private val _navigateToNext = MutableLiveData<Event<Unit>>()
    val navigateToNext: LiveData<Event<Unit>>
        get() = _navigateToNext

    var zipExtension: String = ""
    var zipFirst: String = ""

    var gson = Gson()
    private lateinit var job: Job
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    var isFull: Boolean = false

    private val _zipChanged = MutableLiveData<Event<String>>()
    val zipChanged: LiveData<Event<String>>
        get() = _zipChanged

    var zip: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.zip, null)
        }

    @get:Bindable
    val zipTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
            _zipChanged.value = Event(s.toString())
        }
    }

    private val _apartmentChanged = MutableLiveData<Event<String>>()
    val apartmentChanged: LiveData<Event<String>>
        get() = _apartmentChanged
    private val _addressChanged = MutableLiveData<Event<String>>()
    val addressChanged: LiveData<Event<String>>
        get() = _addressChanged

    private val _stateChanged = MutableLiveData<Event<String>>()
    val stateChanged: LiveData<Event<String>>
        get() = _stateChanged

    private val _cityChanged = MutableLiveData<Event<String>>()

    val cityChanged: LiveData<Event<String>>
        get() = _cityChanged

    var addressStreet: String = ""
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.addressStreet, null)
        }

    var enableButton: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.enableButton, null)
        }

    var state: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.state, null)
        }

    var apartment: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.apartment, null)
        }

    var city: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.city, null)
        }

    @get:Bindable
    val addressTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _addressChanged.value = Event(s.toString())
        }
    }

    @get:Bindable
    val apartmentTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

            _apartmentChanged.value = Event(s.toString())
        }

    @get:Bindable
    val cityTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _cityChanged.value = Event(s.toString())
        }
    }

    val showError: LiveData<APIResponse<DataResponse>>
        get() = _showError
    private val _showError = MutableLiveData<APIResponse<DataResponse>>()

    private val _uploadState = MutableLiveData<Event<UploadState>>()
    val uploadState: LiveData<Event<UploadState>> get() = _uploadState

    private val _showAlternateAppAvailable = MutableLiveData<Event<DataResponse>>()
    val showAlternateAppAvailable: LiveData<Event<DataResponse>> get() = _showAlternateAppAvailable

    private var identifyDisposable: Disposable? = null

    fun addPreviousZip(zipVal: String?) {
        zipVal?.let {
            val zips: List<String> = zipVal.split("-")
            if (zips[0].length == 5) {
                zipFirst = zips[0]
            }
            if (zips.size > 1) {
                zipExtension = zips[1]
            }

            zip = zipVal.replace("-", "")
            _zipChanged.value = Event(it)
        }
    }

    fun updateZip(
        zip: String,
        token: String,
        currentFullAddress: CurrentFullAddress?
    ) {
        viewModelScope.launch {
            compositeDisposable.add(

                uploadRepository.updateDetails(
                    token,
                    UploadDetailsSendParams(
                        additionalVerification = AdditionalVerification(
                            currentZip = zip.replace("-", ""),
                            currentFullAddress = currentFullAddress
                        )
                    )
                ).subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _uploadState.value = Event(UploadState.Loading(LoadingState.UPLOAD))
                    }
                    .subscribe({ result ->

                        if (result.isSuccessful) {
                            identify(token)
                        } else {
                            val errorResponse: GenericErrorResponse = gson.fromJson(
                                result.errorBody()?.string(),
                                GenericErrorResponse::class.java
                            )

                            _uploadState.value = Event(
                                UploadState.Error(errorResponse.getErrorMessage())
                            )
                        }
                    }, { error ->
                        // Handling errors explicitly
                        _uploadState.value = Event(
                            UploadState.Error(error.message ?: "Something went wrong.")
                        )
                    })
            )
        }
    }

    fun setSelectedState(changedState: String) {
        _stateChanged.value = Event(changedState)
        state = changedState
    }

    fun resetState() {
        _uploadState.value = Event(UploadState.Idle)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    private var debounceJob: Job? = null
    private var isDebouncing = false
    private val debounceInterval = 500L // Set your debounce interval in milliseconds

    fun onNextClicked() {
        if (isDebouncing) return // Ignore clicks during debounce period

        _navigateToNext.value = Event(Unit) // Process the first click immediately
        isDebouncing = true // Start debounce

        debounceJob?.cancel()
        debounceJob = viewModelScope.launch {
            delay(debounceInterval) // Wait for the debounce interval
            isDebouncing = false // Reset debounce state
        }
    }

    fun onPrevClicked() {
        _navigateToBack.value = Event(Unit)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }

    fun checkEnableButton() {
        if (isFull) {
            enableButton = AddressChangedValidator.isAddressReadyGuest(
                addressChanged = addressChanged.value?.peekContent(),
                cityChanged = cityChanged.value?.peekContent(),
                state = stateChanged.value?.peekContent(),
                zipChanged = zipChanged.value?.peekContent()
            )
        } else {
            enableButton = AddressChangedValidator.isValidZip(zipChanged.value?.peekContent())
        }
    }

    fun verify(token: String) {
        viewModelFactory {
            compositeDisposable.add(
                uploadRepository.verify(token)
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _uploadState.value = Event(UploadState.Loading(LoadingState.VERIFY))
                    }.subscribe({
                        if (it.isSuccessful) {
                            identify(token)
                        } else {
                            val errorResponse: GenericErrorResponse = gson.fromJson(
                                it.errorBody()?.string(),
                                GenericErrorResponse::class.java
                            )
                            _uploadState.value = Event(
                                UploadState.Error(errorResponse.getErrorMessage())
                            )
                        }
                    }, {
                            error ->

                        _uploadState.value = Event(
                            UploadState.Error(error)
                        )
                    })
            )
        }
    }

    fun identify(token: String) {
        viewModelFactory {
            identifyDisposable?.let {
                compositeDisposable.delete(it)
            }
            identifyDisposable = uploadRepository.identify(token)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _uploadState.value = Event(UploadState.Loading(LoadingState.IDENTIFY))
                }.subscribe({
                    if (it.isSuccessful) {
                        it.body()?.let { responseBody ->
                            when (responseBody.code) {
                                200 -> _uploadState.value = Event(UploadState.Success())
                                else -> handleError(responseBody.error)
                            }
                        }
                    } else {
                        if (it.code() == 403) {
                            _uploadState.value = Event(UploadState.Error(""))
                            handleAlternateAppAvailable(it.errorBody())
                        } else {
                            handleErrorResponse(it.errorBody())
                        }
                    }
                }, {
                        error ->
                    _uploadState.value = Event(
                        UploadState.Error(error)
                    )
                })
            compositeDisposable.add(identifyDisposable!!)
        }
    }

    private fun handleError(error: ErrorDetail?) {
        when (error?.error) {
            "SENT_TO_CUSTOMER_SUPPORT" -> _uploadState.value = Event(UploadState.CustomerSupport)
            "NO_MATCH" -> {
                when {
                    error.requireCurrentZip -> _uploadState.value = Event(UploadState.NoMatchAddZip)
                    error.requireCurrentAddress -> _uploadState.value = Event(
                        UploadState.NoMatchAddAddress
                    )
                    else -> _uploadState.value = Event(UploadState.Error(error.errorDescription))
                }
            }
            "MULTIPLE_MATCH" -> {
                when {
                    error.requireCurrentZip -> _uploadState.value = Event(UploadState.AddZip)
                    error.requireCurrentAddress -> _uploadState.value = Event(
                        UploadState.AddAddress
                    )

                    else -> _uploadState.value = Event(UploadState.Error(error.errorDescription))
                }
            }
            else -> _uploadState.value = Event(UploadState.Error(error?.errorDescription ?: ""))
        }
    }

    private fun handleErrorResponse(errorBody: ResponseBody?) {
        try {
            val errorResponse: IdentifyErrorResponse = gson.fromJson(
                errorBody?.string(),
                IdentifyErrorResponse::class.java
            )
            _uploadState.value = Event(
                UploadState.Error(
                    errorResponse.error?.errorDescription ?: errorResponse.message ?: ""
                )
            )
        } catch (e: Exception) {
            _uploadState.value = Event(UploadState.Error("HTTP Status 404 – Not Found"))
        }
    }

    private fun handleAlternateAppAvailable(errorBody: ResponseBody?) {
        val errorResponse: DataResponse? = gson.fromJson(
            errorBody?.string(),
            DataResponse::class.java
        )
        errorResponse?.let {
            _showAlternateAppAvailable.value = Event(it)
        }
    }
}
