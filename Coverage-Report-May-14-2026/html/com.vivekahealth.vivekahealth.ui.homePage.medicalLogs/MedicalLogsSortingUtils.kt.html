package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs

import com.vivekahealth.vivekahealth.data.remote.home.ClaimsQuickView
import com.vivekahealth.vivekahealth.data.remote.home.DiagnosisHistoryQuickView
import com.vivekahealth.vivekahealth.data.remote.home.ImmunizationQuickView
import com.vivekahealth.vivekahealth.data.remote.home.LabResultQuickView
import com.vivekahealth.vivekahealth.data.remote.home.PrescriptionQuickView
import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.MedicalLogsAll
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.MedicalLogsSortingUtils.MEDICAL_LOG_CLAIMS
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.MedicalLogsSortingUtils.MEDICAL_LOG_DIAGNOSIS
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.MedicalLogsSortingUtils.MEDICAL_LOG_LAB
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.MedicalLogsSortingUtils.MEDICAL_LOG_PRESCRIPTION
import com.vivekahealth.vivekahealth.utils.DateUtils
import timber.log.Timber
import java.util.Date

object MedicalLogsSortingUtils {

    // this is used in this class only
    const val MEDICAL_LOG_CLAIMS = 1
    const val MEDICAL_LOG_DIAGNOSIS = 2
    const val MEDICAL_LOG_PRESCRIPTION = 3
    const val MEDICAL_LOG_LAB = 4

    /**
     * Sort and group ALL medical logs (diagnosis, prescriptions, lab, claims)
     */
    fun sortAllMedicalLogs(logsData: MedicalLogsAll, sortAction: String): List<Any> {
        val isDescending = sortAction.equals("desc", true)

        // Collect all medical items
        val allMedicalItems = collectAllMedicalItems(logsData)

        // Sort, group, and flatten in one operation
        return sortGroupAndFlatten(allMedicalItems, isDescending)
    }

    /**
     * Sort and group PRESCRIPTIONS only with specific sort criteria
     * Jun 17, 2025
     */
    fun sortPrescriptionsOnly(
        logsData: MedicalLogsAll,
        sortAction: String,
        sortBy: String
    ): List<Any> {
        val prescriptionList = logsData.prescriptionList ?: return emptyList()

        val isDescending = sortAction.equals("desc", true)
        val useActionDate = sortBy.equals("recentlyUpdated", true)

        // Create and collect prescription items
        val prescriptionItems = prescriptionList.mapNotNull { prescription ->
            val dateString = prescription.dateOfService

            extractDate(dateString) { DateUtils.stringServerDateToDate(it) }?.let { date ->
                MedicalItem(prescription, date, MEDICAL_LOG_PRESCRIPTION, prescription.drugName)
            }
        }

        // Sort, group, and flatten in one operation
        return sortGroupAndFlatten(
            prescriptionItems,
            isDescending,
            isPrescriptionsOnly = true,
            sortBy = sortBy
        )
    }

    fun sortClaimsOnly(logsData: MedicalLogsAll, sortAction: String, sortBy: String): List<Any> {
        val claimsList = logsData.claimsList ?: return emptyList()

        val isDescending = sortAction.equals("desc", true)
        val useActionDate = sortBy.equals("recentlyUpdated", true)

        // Create and collect claim items
        val claimItems = claimsList.mapNotNull { claim ->
            val dateString = if (useActionDate) claim.actionDate else claim.visitedDate

            extractDate(dateString) { DateUtils.stringServerDateToDate(it) }?.let { date ->
                MedicalItem(claim, date, MEDICAL_LOG_CLAIMS, claim.providerDisplayName ?: "")
            }
        }

        // Sort, group, and flatten in one operation
        return sortGroupAndFlatten(claimItems, isDescending, isClaimsOnly = true, sortBy = sortBy)
    }

    fun sortDiagnosisOnly(logsData: MedicalLogsAll): List<DiagnosisHistoryQuickView> {
        return logsData.diagnosisList ?: emptyList()
    }
    fun sortImmunizationsOnly(logsData: MedicalLogsAll): List<ImmunizationQuickView> {
        return logsData.immunizationList ?: emptyList()
    }

    // Combined sort, group, and flatten operation
    private fun sortGroupAndFlatten(
        items: List<MedicalItem>,
        isDescending: Boolean,
        isClaimsOnly: Boolean = false,
        isPrescriptionsOnly: Boolean = false,
        sortBy: String? = null
    ): List<Any> {
        if (items.isEmpty()) return emptyList()

        // Group by appropriate date key based on context
        val groupedByDate = items.groupBy { item ->
            when {
                isClaimsOnly -> getClaimDateGroupKey(item.data, sortBy)
                isPrescriptionsOnly -> getPrescriptionDateGroupKey(item.data, sortBy)
                else -> getAllDateGroupKey(item.data, item.data.getLogViewType())
            }
        }.filterKeys { it != null }

        val sortedGroups = when {
            isPrescriptionsOnly && sortBy.equals("drugName") -> {
                // Keys are Strings when sorting by drug name
                if (isDescending) {
                    groupedByDate.toSortedMap(compareByDescending { it as String })
                } else {
                    groupedByDate.toSortedMap(compareBy { it as String })
                }
            }

            else -> {
                // Keys are Dates for all other cases
                if (isDescending) {
                    groupedByDate.toSortedMap(compareByDescending { it as Date })
                } else {
                    groupedByDate.toSortedMap(compareBy { it as Date })
                }
            }
        }

        // Create final flattened list with headers
        // Main sorting logic
        return sortedGroups.flatMap { (dateKey, groupItems) ->

            val sortedGroupItems = if (isPrescriptionsOnly && sortBy == "drugName") {
                groupItems.sortedBy { (it.data as PrescriptionQuickView).drugName }
            } else if (isDescending) {
                groupItems.sortedWith(
                    compareByDescending<MedicalItem> { it.data.getSortingDate() } // First sort by date descending
                        .thenBy { it.data.getLogViewType() } // Then sort by viewType ascending
                )
            } else {
                groupItems.sortedWith(
                    compareBy<MedicalItem> { it.data.getSortingDate() } // First sort by date descending
                        .thenBy { it.data.getLogViewType() } // Then sort by viewType ascending
                )
            }

            // Date header followed by items
            arrayListOf<Any>(dateKey!!) + sortedGroupItems.map { it.data }
        }
    }

    // Private helper functions

    private fun collectAllMedicalItems(logsData: MedicalLogsAll): List<MedicalItem> {
        val allMedicalItems = mutableListOf<MedicalItem>()

        // Diagnosis
        logsData.diagnosisList?.forEach { diagnosis ->
            extractDate(diagnosis.diagnosisDate) { DateUtils.stringDateToDate(it) }?.let { date ->
                allMedicalItems.add(
                    MedicalItem(
                        diagnosis.copy(sortingDate = date),
                        date,
                        MEDICAL_LOG_DIAGNOSIS,
                        diagnosis.icd10Code.description ?: ""
                    )
                )
            }
        }

        // Prescriptions
        logsData.prescriptionList?.forEach { prescription ->
            extractDate(prescription.dateOfService) { DateUtils.stringServerDateToDate(it) }?.let { date ->
                allMedicalItems.add(
                    MedicalItem(
                        prescription.copy(sortingDate = date),
                        date,
                        MEDICAL_LOG_PRESCRIPTION,
                        prescription.drugName
                    )
                )
            }
        }

        // Lab Results
        logsData.labInfoList?.forEach { lab ->
            extractDate(lab.latestReadingDateTime) { DateUtils.labDateStringToDate(it) }?.let { date ->
                allMedicalItems.add(
                    MedicalItem(lab.copy(sortingDate = date), date, MEDICAL_LOG_LAB, lab.loincCode)
                )
            }
        }

        // Claims (using visitedDate for all medical logs view)
        logsData.claimsList?.forEach { claim ->
            claim.visitedDate?.let { visitedDate ->
                extractDate(visitedDate) { DateUtils.stringServerDateToDate(it) }?.let { date ->
                    allMedicalItems.add(
                        MedicalItem(
                            claim.copy(sortingDate = date),
                            date,
                            MEDICAL_LOG_CLAIMS,
                            claim.providerDisplayName ?: ""
                        )
                    )
                }
            }
        }

        return allMedicalItems
    }

    // Date group key for ALL medical logs view
    private fun getAllDateGroupKey(data: Any, viewType: Int): Date? {
        return try {
            when (viewType) {
                MEDICAL_LOG_DIAGNOSIS -> (data as? DiagnosisHistoryQuickView)?.let { diagnosis ->
                    DateUtils.groupStringToDate(
                        DateUtils.getMonthServerDateSlashedToMMMM(diagnosis.diagnosisDate)
                    )
                }

                MEDICAL_LOG_PRESCRIPTION -> (data as? PrescriptionQuickView)?.let { prescription ->
                    DateUtils.groupStringToDate(
                        DateUtils.getDashToMMMMPrescription(prescription.dateOfService)
                    )
                }

                MEDICAL_LOG_LAB -> (data as? LabResultQuickView)?.let { lab ->
                    DateUtils.groupStringToDate(
                        DateUtils.getDashMMMMLab(lab.latestReadingDateTime)
                    )
                }

                MEDICAL_LOG_CLAIMS -> (data as? ClaimsQuickView)?.let { claim ->
                    DateUtils.groupStringToDate(
                        DateUtils.getDashToMMMMClaim(claim.visitedDate.toString())
                    )
                }

                else -> null
            }
        } catch (e: Exception) {
            Timber.e("Error extracting ALL date group key: ${e.message}")
            null
        }
    }

    // Date group key for PRESCRIPTIONS ONLY view
    private fun getPrescriptionDateGroupKey(data: Any, sortBy: String?): Any? {
        return try {
            (data as? PrescriptionQuickView)?.let { prescription ->

                if (sortBy.equals("drugName")) {
                    prescription.drugName.first().uppercaseChar().toString()
                } else {
                    val dateString = prescription.dateOfService
                    // Different grouping logic for prescriptions only
                    dateString?.let {
                        DateUtils.stringServerDateToDate(
                            it
                        )
                    }
                }
            }
        } catch (e: Exception) {
            Timber.e("Error extracting PRESCRIPTIONS date group key: ${e.message}")
            null
        }
    }

    private fun getClaimDateGroupKey(data: Any, sortBy: String?): Date? {
        return try {
            (data as? ClaimsQuickView)?.let { claim ->
                val dateString = if (sortBy?.equals("recentlyUpdated", true) == true) {
                    claim.actionDate
                } else {
                    claim.visitedDate
                }

                // Different grouping logic for claims only
                dateString?.let {
                    DateUtils.groupStringToDate(
                        DateUtils.getDashToMMMMClaim(it)
                    )
                }
            }
        } catch (e: Exception) {
            Timber.e("Error extracting CLAIMS date group key: ${e.message}")
            null
        }
    }

    private fun extractDate(dateString: String?, dateParser: (String) -> Date?): Date? {
        return try {
            dateString?.let { dateParser(it) }
        } catch (e: Exception) {
            Timber.e("Error parsing date '$dateString': ${e.message}")
            null
        }
    }
}

// Internal data class for processing
data class MedicalItem(
    val data: Any,
    val date: Date,
    val viewType: Int,
    val title: String
)

fun Any.getSortingDate(): Date? = when (this) {
    is ClaimsQuickView -> this.sortingDate
    is DiagnosisHistoryQuickView -> this.sortingDate
    is LabResultQuickView -> this.sortingDate
    is PrescriptionQuickView -> this.sortingDate
    else -> null
}

fun Any.getLogViewType(): Int = when (this) {
    is DiagnosisHistoryQuickView -> MEDICAL_LOG_DIAGNOSIS
    is ClaimsQuickView -> MEDICAL_LOG_CLAIMS
    is PrescriptionQuickView -> MEDICAL_LOG_PRESCRIPTION
    is LabResultQuickView -> MEDICAL_LOG_LAB
    else -> -1 // or throw an exception if you prefer
}
