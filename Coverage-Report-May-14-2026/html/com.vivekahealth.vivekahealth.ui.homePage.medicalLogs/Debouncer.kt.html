package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs

class ParameterizedDebouncer<T> {
    private var lastParams: T? = null
    private var lastExecutionTime = 0L

    fun debounce(params: T, delayMs: Long = 500L, action: (T) -> Unit) {
        val currentTime = System.currentTimeMillis()
        if (lastParams != params || currentTime - lastExecutionTime >= delayMs) {
            lastParams = params
            lastExecutionTime = currentTime
            action(params)
        }
    }
}

// Extension for parameterized debouncing
fun <T, P> T.debouncedWith(): ParameterizedDebouncer<P> = ParameterizedDebouncer()
