package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs

import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.MedicalLogsAll

class MedicalLogAllDataSaver {
    // Store the saved data
    private var savedData: MedicalLogsAll? = null

    fun saveData(newData: MedicalLogsAll, shouldAppend: Boolean): MedicalLogsAll {
        return if (shouldAppend && savedData != null) {
            val combined = appendData(savedData!!, newData)
            savedData = combined
            combined
        } else {
            savedData = newData
            newData
        }
    }

    fun getSavedData(): MedicalLogsAll? = savedData

    fun clearSavedData() {
        savedData = null
    }

    fun hasSavedData(): Boolean {
        return savedData?.let { !isDataEmpty(it) } ?: false
    }
    private fun isDataEmpty(data: MedicalLogsAll): Boolean {
        return data.claimsList.isNullOrEmpty() &&
            data.diagnosisList.isNullOrEmpty() &&
            data.labInfoList.isNullOrEmpty() &&
            data.prescriptionList.isNullOrEmpty()
    }

    private fun appendData(existing: MedicalLogsAll, newData: MedicalLogsAll): MedicalLogsAll {
        return MedicalLogsAll(
            claimsList = (existing.claimsList.orEmpty() + newData.claimsList.orEmpty()),
            diagnosisList = (existing.diagnosisList.orEmpty() + newData.diagnosisList.orEmpty()),
            labInfoList = (existing.labInfoList.orEmpty() + newData.labInfoList.orEmpty()),
            prescriptionList = (existing.prescriptionList.orEmpty() + newData.prescriptionList.orEmpty()),
            paging = newData.paging // Use latest paging info
        )
    }
}
