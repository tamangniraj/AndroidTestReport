package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs

import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.utils.Event
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.combine

class TabSwitchViewModel : ViewModel() {
    // LiveData for backward compatibility
    private val _currentPosition = MutableLiveData(0)
    val currentPosition: LiveData<Int> = _currentPosition

    // StateFlow for current position
    private val _currentPositionFlow = MutableStateFlow(0)
    val currentPositionFlow: StateFlow<Int> = _currentPositionFlow.asStateFlow()

    // StateFlow for member ID
    private val _memberId = MutableStateFlow("")
    val memberId: StateFlow<String> = _memberId.asStateFlow()

    // StateFlow for access flag
    private val _hasAccess = MutableStateFlow(false)
    val hasAccess: StateFlow<Boolean> = _hasAccess.asStateFlow()

    // Combined state flow for all three
    val combinedState = combine(
        _currentPositionFlow,
        _memberId
    ) { position, memberId ->
        TabState(position, memberId)
    }

    fun updatePosition(position: Int) {
        _currentPosition.value = position
        _currentPositionFlow.value = position
    }

    fun updateMemberIdOnlyIfNull(memberId: String) {
        if (_memberId.value.isNullOrEmpty()) {
            _memberId.value = memberId
        }
    }

    fun updateMemberId(memberId: String) {
        _memberId.value = memberId
    }

    fun updateHasAccess(hasAccess: Boolean) {
        _hasAccess.value = hasAccess
    }

    private val _reconnect = MutableLiveData<Event<Unit>>()
    val reconnect: LiveData<Event<Unit>> = _reconnect

    fun reconnect() {
        _currentPosition.value = currentPosition.value
        _reconnect.value = Event(Unit)
    }
}

data class TabState(
    val currentPosition: Int,
    val memberId: String
)
