    fun setSelectedMember(memberInfo: MemberInfo, showPicker: Boolean = true) {
        val index = bubbleMemberList.indexOf(memberInfo)

        for (item in bubbleMemberList) {
            if (bubbleMemberList.indexOf(item) == index) {
                bubbleMemberList[index].apply {
                    isSelected = true
                }
            } else {
                bubbleMemberList[bubbleMemberList.indexOf(item)].apply {
                    isSelected = false
                }
            }
        }

        // This method checks if the user is eligible or not and views is shown accordingly
        checkAccess(memberInfo, showPicker)
    }

    fun checkNewAvailable() {
        // Get the current member ID
        val memberId = preferences.getString("memberID", "") ?: ""

        // Create a unique key for this user
        val userSpecificKey = "${memberId}_$PREF_IMMUNIZATION_TAB_CLICKED"

        // Check if this user has clicked the immunization tab before
        val hasClickedImmunizationTab = preferences.getBoolean(userSpecificKey, false)

        // Display "NEW" indicator only if this user hasn't clicked the immunization tab yet
        displayNew = !hasClickedImmunizationTab
    }

    /**
     * Hide the "NEW" indicator when the user clicks the immunization tab.
     * This saves the state to SharedPreferences per user (memberID).
     */
    fun hideNew() {
        // Get the current member ID
        val memberId = preferences.getString("memberID", "") ?: ""

        // Create a unique key for this user
        val userSpecificKey = "${memberId}_$PREF_IMMUNIZATION_TAB_CLICKED"

        // Save to SharedPreferences that this user has clicked the immunization tab
        preferences.edit().putBoolean(userSpecificKey, true).apply()

        // Hide the "NEW" indicator
        displayNew = false
    }

    fun checkAccess(
        memberInfo: MemberInfo,
        showPicker: Boolean
    ) {
        if (bubbleMemberList.any { memberInfo.selfId == it.selfId }) {
            val member = bubbleMemberList.find { it.selfId == memberInfo.selfId }
            val index = bubbleMemberList.indexOf(member ?: 0)
            val selectedMember = bubbleMemberList[index]
            val selectedMemberInfoList =
                memberListArray.filter { it.selfId == selectedMember.selfId }

            if (selectedMemberInfoList.isEmpty() && memberInfo.selfId.toLong() > 0) {
                noData = false
                var page = ""
                if (currentPage.equals(resources.getString(R.string.all), true)) {
                    page = resources.getString(R.string.medical_logs)
                } else {
                    page = currentPage
                }
                reason = resources.getString(
                    R.string.access_to_s_s_s_s_is_restricted_by_the_member,
                    (selectedMember.lastName ?: "") + ",",
                    selectedMember.firstName ?: "",
                    resources.getString(R.string.medical_logs)
                )
                _hasAccess.value = Event(false)
                return
            } else {
                reason = ""
                noData = false
                //            getMessage(selectedMember)
                if (!memberInfo.hasAvatar.equals(
                        "true",
                        true
                    ) && !memberInfo.avatarInReview && memberInfo.selfieMandatorySwitch && !memberInfo.memberRelationship.toString()
                        .equals(
                                "insured",
                                true
                            ) && !memberInfo.memberAdditionInReview && memberInfo.selfId.toLong() > 0 && !memberInfo.demographicUpdatePending
                ) {
                    if (showPicker) {
                        _showPicker.value = Event(Unit)
                    }
                    _hasAccess.value = Event(false)
                    return
                }
                _hasAccess.value = Event(true)
            }
        }
    }

    fun getMemberList(type: String) {
        compositeDisposable.add(
            medicalLogsRepository.getMemberList(
                preferences.getString("memberID", "").toString(),
                type
            ).subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                loading = true
            }.subscribe({ it ->

                memberListArray = it.valueArrayList ?: arrayListOf()

                if (currentMember.selfId.isNotBlank() && bubbleMemberList.isNotEmpty()) {
                    checkAccess(currentMember, false)
                }
                loading = false
            }, {
                if (it is HttpException) {
                    _serverError.value = Event(
                        NetworkMessage.getErrorMessage(
                            it.response()?.errorBody()
                        )
                    )
                }
                loading = false
            })
        )
    }

    fun getMemberFromDatabaseAndList() {
        bubbleMemberList.clear()
        memberInfo.insured?.let { it1 ->
            bubbleMemberList.add(it1)
            bubbleMemberList.add(memberInfo)

            bubbleMemberList[0].apply { }
        } ?: run {
            bubbleMemberList.add(0, memberInfo)
            bubbleMemberList[0].apply {
                isSelected = true
            }
        }

        memberInfo.dependentsArrayList?.sortByDescending {
            DateUtils.stringServerDateToDate(
                it.birthDate.toString()
            )
        }
        memberInfo.dependentsArrayList?.reverse()
        memberInfo.dependentsArrayList?.let { it1 ->

            for (member in it1) {
                if (StringUtils.equalsIgnoreCase(member.memberRelationship, "Spouse")) {
                    bubbleMemberList.add(1, member)
                } else {
                    bubbleMemberList.add(member)
                }
            }

            var grouped = bubbleMemberList.groupBy { it.memberRelationship }
            grouped.forEach {
                it.value.sortedByDescending {
                    DateUtils.stringServerDateToDate(
                        it.birthDate.toString()
                    )
                }
                it.value.reversed()
            }

            bubbleMemberList.clear()

            grouped.forEach {
                if (it.key.equals("Insured", true)) {
                    bubbleMemberList.add(0, it.value[0])
                } else if (it.key.equals("Spouse", true)) {
                    if (grouped.containsKey("Insured")) {
                        bubbleMemberList.add(1, it.value[0])
                    } else {
                        bubbleMemberList.add(0, it.value[0])
                    }
                } else if (it.key.equals("Child", true)) {
                    val dateGroup = it.value.groupBy { date ->
                        DateUtils.stringServerDateToDate(
                            date.birthDate.toString()
                        )
                    }

                    dateGroup.forEach { item ->
                        val sorted = item.value.sortedBy { item1 -> item1.firstName }
                        bubbleMemberList.addAll(sorted)
                    }
                }
            }

            val tempOthers: ArrayList<MemberInfo> = arrayListOf()
            grouped.forEach {
                if (!it.key.equals("Insured", true) && !it.key.equals(
                        "Spouse",
                        true
                    ) && !it.key.equals("Child", true)
                ) {
                    tempOthers.addAll(it.value)
                }
            }

            val dateGroup = tempOthers.groupBy { date ->
                DateUtils.stringServerDateToDate(
                    date.birthDate.toString()
                )
            }
            dateGroup.forEach { item ->
                val sorted = item.value.sortedBy { item1 -> item1.firstName }
                bubbleMemberList.addAll(sorted)
            }
        }

        eligibleBubbleMemberList.clear()

        for (item in bubbleMemberList) {
            item.isSelected = item.selfId == memberInfo.selfId
        }
        memberInfo.isInsured =
            memberInfo.insured == null || memberInfo.relationDesc.equals("spouse", true)
//        bubbleMemberList.add(
//            MemberInfo(
//                isInsured = memberInfo.insured == null || memberInfo.relationDesc.equals(
//                    "spouse", true
//                )
//            )
//        )
        listForHomeBubbleAdapter = bubbleMemberList.toList()
        _bubbleListGenerated.value = Event(listForHomeBubbleAdapter)
    }
