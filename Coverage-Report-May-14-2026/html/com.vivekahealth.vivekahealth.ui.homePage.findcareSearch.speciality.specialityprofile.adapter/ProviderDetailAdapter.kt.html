package com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityprofile.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.databinding.SingleRowDocDetailBinding
import com.vivekahealth.vivekahealth.data.remote.home.findCare.doctorModelsNew.Value

enum class DetailType {
    EIN, NPI, GENDER, PRI_TAXO_CAT, PRI_TAXO, SPECIALTIES, BOARD_CERT
}

class ProviderDetailAdapter(var provider: Value, var detailsList: ArrayList<DetailType>) :
    RecyclerView.Adapter<ProviderDetailAdapter.DocDetailViewHolder>() {
    lateinit var docDetailBinding: SingleRowDocDetailBinding

    inner class DocDetailViewHolder(val binding: SingleRowDocDetailBinding) :
        RecyclerView.ViewHolder(binding.root) {
        fun setData(provider: Value, detailType: DetailType) {
            when (detailType) {
                DetailType.EIN -> {
                    binding.detailKey = "EIN"
                    binding.detailValue = provider.ein
                }
                DetailType.NPI -> {
                    binding.detailKey = "NPI"
                    binding.detailValue = provider.npi
                }
                DetailType.GENDER -> {
                    binding.detailKey = binding.view.context.getString(R.string.gender)
                    binding.detailValue = provider.gender
                }
                DetailType.PRI_TAXO_CAT -> {
                    binding.detailKey = binding.view.context.getString(
                        R.string.primary_taxonomy_category
                    )
                    binding.detailValue = provider.primaryTaxonomyCategory
                }
                DetailType.PRI_TAXO -> {
                    binding.detailKey = binding.view.context.getString(R.string.primary_taxonomy)
                    binding.detailValue = provider.primaryTaxonomy
                }
                DetailType.SPECIALTIES -> {
                    binding.detailKey = binding.view.context.getString(R.string.specialties)
                    binding.detailValue = provider.specialities?.joinToString(", ")
                }
                DetailType.BOARD_CERT -> {
                    binding.detailKey = binding.view.context.getString(R.string.board_certification)
                    binding.detailValue = provider.boardCertified
                }
            }
        }
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): DocDetailViewHolder {
        docDetailBinding =
            SingleRowDocDetailBinding.inflate(LayoutInflater.from(parent.context), parent, false)
        return DocDetailViewHolder(docDetailBinding)
    }

    override fun onBindViewHolder(holder: DocDetailViewHolder, position: Int) {
        val itemPosition = holder.layoutPosition
        holder.setData(provider, detailsList[itemPosition])
    }

    override fun getItemCount(): Int {
        return detailsList.size
    }
}
