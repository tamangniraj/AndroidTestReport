    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.register, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.register, ""),
                        RegisterUser::class.java
                    )
                }
            },
            {
                firstName = user.firstName.toString()
                lastName = user.lastName.toString()
                if (overrideOldRegistration.isBlank()) {
                    user.overrideOldRegistration?.let { overrideOldRegistration = it }
                }
                _phoneFirstPrev.value = Event(user.firstName.toString())
                _phoneLastPrev.value = Event(user.lastName.toString())
            }
        )
    }

    fun saveName() {
        user.firstName = firstName.trim()
        user.lastName = lastName.trim()
        user.overrideOldRegistration = overrideOldRegistration
        user.memberId = memberId
        if (!memberId.isNullOrBlank()) {
            user.email = email
        }

        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.register, regString)
        regEdit.apply()
    }
    fun removeReg() {
        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.Registration, "")
        regEdit.apply()
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {

    override fun onCleared() {
