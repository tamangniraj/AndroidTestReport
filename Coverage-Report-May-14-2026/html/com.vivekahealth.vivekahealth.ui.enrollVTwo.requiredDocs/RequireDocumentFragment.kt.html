open class RequireDocumentFragment : BaseFragment(), Injectable {

    protected lateinit var binding: FragmentChildRequireDocumentBinding

    protected lateinit var viewModel: RequiredDocsViewModel

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory
    private lateinit var cropImageLauncher: ActivityResultLauncher<CropImageContractOptions>
    private var uriForFile: Uri? = null
    private lateinit var requestCameraPermissionLauncher: ActivityResultLauncher<String>
    private lateinit var cameraLauncher: ActivityResultLauncher<Intent>
    private lateinit var galleryLauncher: ActivityResultLauncher<Intent>

    /**
     * Replaceable in tests — set before clicking a doc card.
     * When non-null, [showPicker] calls [ImagePickerClient.pick] instead of showing the
     * real picker bottom sheet. Same pattern as DocumentScanFragment.scannerClient.
     */
    internal var imagePickerClient: ImagePickerClient? = null

    protected val sharedViewModel: SharedViewModel by activityViewModels()

    val args: RequireDocumentFragmentArgs by navArgs()

    companion object {
        fun newInstance() = RequireDocumentFragment()
    }

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = FragmentChildRequireDocumentBinding.inflate(inflater, container, false)
        return binding.root
    }

    val adapter = DocAdapterV2(
        onClick = {

            if (it.disable != true) {
                viewModel.currentItem = it
                viewModel.currentPerson = it.person!!
                viewModel.currentDoc = it.fileName.toString()
                showPicker()
            }
        }
    )
    var data: DependentCheckItem? = null
    protected var relationType: Int = -1

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        requestCameraPermissionLauncher = registerForActivityResult(
            ActivityResultContracts.RequestPermission()
        ) { isGranted ->
            if (isGranted) {
                launchCamera()
            }
        }

        cameraLauncher = registerForActivityResult(
            ActivityResultContracts.StartActivityForResult()
        ) { result ->
            if (result.resultCode == Activity.RESULT_OK) {
                uriForFile?.let { uri ->
                    startCrop(uri)
                }
            }
        }

        galleryLauncher = registerForActivityResult(
            ActivityResultContracts.StartActivityForResult()
        ) { result ->
            if (result.resultCode == Activity.RESULT_OK) {
                result.data?.data?.let { uri ->
                    startCrop(uri)
                }
            }
        }

        cropImageLauncher = registerForActivityResult(CropImageContract()) { result ->

            if (result.isSuccessful) {
                result.uriContent?.let { croppedUri ->
                    openUpload(croppedUri)
                }
            } else if (result !is CropImage.CancelledResult) {
                val exception = result.error
                exception?.let {
                    android.widget.Toast.makeText(
                        requireContext(),
                        "Crop failed: ${it.message}",
                        android.widget.Toast.LENGTH_SHORT
                    ).show()
                }
            }
        }
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            RequiredDocsViewModel::class.java
        )

        binding.lifecycleOwner = viewLifecycleOwner
        binding.viewModel = this@RequireDocumentFragment.viewModel

        relationType = args.relationType
        binding.relationType = relationType

        viewModel.navigateToNext.observe(
            viewLifecycleOwner,
            EventObserver {
                markDependentComplete()

                if (relationType == RelationshipType.PRIMARY.code) {
                    navigateForPrimary()
                } else {
                    navigateForDependent()
                }
            }
        )

        sharedViewModel.enrollmentData.observe(
            viewLifecycleOwner,
            Observer {
                if (relationType == RelationshipType.CHILD.code) {
                    data = sharedViewModel.enrollmentData.value?.findChildByIndex(
                        sharedViewModel.index
                    )
                } else if (relationType == RelationshipType.SPOUSE.code) {
                    data = sharedViewModel.enrollmentData.value?.findSpouse()
                } else if (relationType == RelationshipType.PRIMARY.code) {
                    data = sharedViewModel.enrollmentData.value?.findPrimary()
                }

                viewModel.firstNameSpouse = data?.firstName ?: ""

                data?.let { dt ->

                    val childAgeInYear = DateUtils.getDateDiff(
                        SimpleDateFormat("yyyy-MM-dd", Locale.getDefault()),
                        data?.dob ?: "",
                        DateUtils.getCurrentDateServer()
                    )

                    Timber.d("Boolean:1 ${dt.previousInsurance} ")

                    val docs = sharedViewModel.findRequiredDocument(
                        relationType,
                        dt.previousInsurance,
                        dt.spouseInsured
                    ).map {
                        val isMinorChild = data?.type == DependentType.CHILD && childAgeInYear < 1

                        val hasValidSSN = relationType == RelationshipType.CHILD.code &&
                            !data?.ssn.isNullOrEmpty() &&
                            ActivityUtils.checkSsnValidity(data?.ssn ?: "")

                        val optionalForChild = (isMinorChild && !hasValidSSN)

                        ListDoc(
                            uploaded = containsId(dt, it.documentType) || it.uploaded,
                            person = Person(
                                firstName = dt.firstName,
                                lastName = dt.lastName,
                                ssnOptional = (!it.required) || optionalForChild
                            ),
                            fileName = it.documentName,
                            id = it.documentType,
                            optional = (!it.required) || optionalForChild,
                            disable = it.uploaded,
                            documentName = it.translatedDocumentName
                        )
                    }

                    // if all the item uploaded then next enable
                    setEnableNextButton(docs)
                    adapter.submitList(docs)
                }
            }
        )

        viewModel.navigateToBack.observe(
            viewLifecycleOwner,
            EventObserver {
                findNavController().navigateUp()
            }
        )

        val animator: RecyclerView.ItemAnimator? = binding.docRecyclerView.itemAnimator
        if (animator is SimpleItemAnimator) {
            (animator as SimpleItemAnimator).supportsChangeAnimations = false
        }
        binding.docRecyclerView.adapter = adapter

        viewModel.navigateToUpload.observe(viewLifecycleOwner) {
            when (it.status) {
                Status.LOADING -> {
                    showUploadLoading(getString(R.string.uploading_document))
                }

                Status.SUCCESS -> {
//                    viewModel.setDocumentName(it.data?.id.toString(), it.data?.fileName.toString())
                    sharedViewModel.updateUploadStatus(
                        data,
                        viewModel.currentItem,
                        it.data?.id.toString()
                    )

                    dismissUploadLoading()
                }

                Status.ERROR -> {
//                    sharedViewModel.updateUploadStatus(
//                        data,
//                        viewModel.currentItem,
//                        "file $relationType ${viewModel.currentItem.docName}"
//                    )

                    dismissUploadLoading()
                    showOkMessageDialog(requireContext(), it.error ?: "Something went wrong")

                    // Timber.d(it.error)
                }
            }
        }

        if (relationType == RelationshipType.PRIMARY.code) {
            sharedViewModel.updateTitle(getString(R.string.document_upload))
        }
    }

    open fun setEnableNextButton(docs: List<ListDoc>) {
        binding.nextEnable = docs.filter { !it.optional }.all { it.uploaded }
    }

    open fun markDependentComplete() {
        when (relationType) {
            RelationshipType.CHILD.code -> sharedViewModel.updateChildCompleteRequireDocument()
            RelationshipType.SPOUSE.code -> sharedViewModel.updateSpouseCompleteRequireDocument()
            RelationshipType.PRIMARY.code -> sharedViewModel.updatePrimaryCompleteRequireDocument()
        }
    }

    private fun showPicker() {
        // Tests inject an imagePickerClient that calls onImageReady immediately,
        // bypassing the real picker bottom sheet entirely — same pattern as scannerClient.
        imagePickerClient?.let { client ->
            imagePickerClient = null // reset so subsequent calls use the real picker
            client.pick { uri -> openUpload(uri) }
            return
        }

        val dialog: BottomSheetDialogFragment = PickerSheetFragment.newInstance()
        val bundle = Bundle()
        bundle.putBoolean("home", true)
        dialog.arguments = bundle
        (dialog as PickerSheetFragment).setCallback(object : PickerButtonListener {
            override fun onCamera() {
                requestCameraPermission()
                dialog.dismiss()
            }

            override fun onLibrary() {
                val intent = Intent(Intent.ACTION_PICK).apply {
                    type = "image/*"
                }
                galleryLauncher.launch(intent)
                dialog.dismiss()
            }

            override fun onFileLibrary() {
            }

            override fun onRemoveDependent() {
            }

            override fun onRemove() {
            }

            override fun onClose() {
                dialog.dismiss()
            }
        })

        dialog.show(childFragmentManager, "tag")
    }

    private fun requestCameraPermission() {
        when {
            ContextCompat.checkSelfPermission(
                requireContext(),
                Manifest.permission.CAMERA
            ) == PackageManager.PERMISSION_GRANTED -> {
                launchCamera()
            }

            shouldShowRequestPermissionRationale(Manifest.permission.CAMERA) -> {
                Timber.tag(TAG).d("requestCameraPermission - Camera Permission NOT Granted")
                requestCameraPermissionLauncher.launch(Manifest.permission.CAMERA)
            }

            else -> {
                requestCameraPermissionLauncher.launch(Manifest.permission.CAMERA)
            }
        }
    }

    private fun launchCamera() {
        val takePictureIntent = Intent(MediaStore.ACTION_IMAGE_CAPTURE)
        try {
            val imageFile = HelperUtils.createImageFile(requireActivity())
            uriForFile = FileProvider.getUriForFile(
                requireActivity(),
                requireContext().packageName + ".provider",
                imageFile
            )
            takePictureIntent.putExtra(MediaStore.EXTRA_OUTPUT, uriForFile)
            takePictureIntent.putExtra(
                MediaStore.EXTRA_SCREEN_ORIENTATION,
                ActivityInfo.SCREEN_ORIENTATION_PORTRAIT
            )
            cameraLauncher.launch(takePictureIntent)
        } catch (e: java.io.IOException) {
            e.printStackTrace()
        }
    }

    private fun startCrop(uri: Uri) {
        cropImageLauncher.launch(
            CropImageContractOptions(
                uri = uri,
                cropImageOptions = CropImageOptions(
                    guidelines = CropImageView.Guidelines.ON,
                    imageSourceIncludeCamera = false
                )
            )
        )
    }

    // Old onActivityResult removed - now using Activity Result API with registered launchers

    fun openUpload(item: Uri?) {
        val file = File(item?.path.toString())
        val originalFileName = file.name

        val imageStream: InputStream? =
            item?.let { activity?.contentResolver?.openInputStream(it) }

        val docBitmap = BitmapFactory.decodeStream(imageStream)

        // Build the full-quality multipart body for upload
        viewModel.filePart = HelperUtils.buildImageBodyPart(
            originalFileName,
            docBitmap,
            requireContext()
        )

        // Scale down for preview only — passing a full-resolution JPEG byte array through
        // Fragment.arguments crosses a Binder transaction and crashes with
        // "Could not copy bitmap to parcel blob" on large camera images.
        val previewStream = ByteArrayOutputStream()
        val maxPreviewDim = 800
        val scaleRatio = minOf(
            maxPreviewDim.toFloat() / docBitmap.width,
            maxPreviewDim.toFloat() / docBitmap.height,
            1f
        )
        val previewBitmap = if (scaleRatio < 1f) {
            Bitmap.createScaledBitmap(
                docBitmap,
                (docBitmap.width * scaleRatio).toInt().coerceAtLeast(1),
                (docBitmap.height * scaleRatio).toInt().coerceAtLeast(1),
                true
            )
        } else {
            docBitmap
        }
        previewBitmap.compress(Bitmap.CompressFormat.JPEG, 80, previewStream)
        if (previewBitmap !== docBitmap) previewBitmap.recycle()
        val byteArray = previewStream.toByteArray()

        val dialog: DialogFragment = UploadDocumentFragment.newInstance()
        val bundle = Bundle()
        bundle.putByteArray("BitMap", byteArray)

        bundle.putString("DocumentName", viewModel.currentDoc)

        bundle.putString(
            "PersonName",
            viewModel.currentPerson.firstName + " " + viewModel.currentPerson.lastName
        )
        dialog.arguments = bundle
        (dialog as UploadDocumentFragment).setCallback(object : UploadDocCallback {
            override fun onActionClick() {
                showPicker()
                dialog.dismiss()
            }

            override fun onUploadClick() {
                callUploadAPI()
                dialog.dismiss()
            }
        })

        dialog.show(childFragmentManager, "tag")
    }

    fun containsId(item: DependentCheckItem, targetId: Int): Boolean {
        return item.uploadedList?.any { it.fileTypeId == targetId } == true
    }

    open fun callUploadAPI() {
        viewModel.uploadPicture(viewModel.currentItem)
    }

    open fun navigateForPrimary() {
        findNavController().navigate(R.id.action_require_document_to_legal_fragment)
    }

    open fun navigateForDependent() {
        findNavController().popBackStack(R.id.reviewDependentDetails, false)
    }

    open fun showUploadLoading(message: String) {
        (activity as EnrollmentV2Activity).showLoading(message)
    }

    open fun dismissUploadLoading() {
        (activity as EnrollmentV2Activity).dismissLoading()
    }
}

fun flattenJson(obj: Any?, prefix: String = ""): Map<String, String> {
    val flatMap = mutableMapOf<String, String>()

    when (obj) {
        is Map<*, *> -> {
            obj.forEach { (key, value) ->
                val fullKey = if (prefix.isEmpty()) "$key" else "$prefix.$key"
                flatMap.putAll(flattenJson(value, fullKey))
            }
        }

        is List<*> -> {
            obj.forEachIndexed { index, value ->
                val fullKey = "$prefix[$index]"
                flatMap.putAll(flattenJson(value, fullKey))
            }
        }

        else -> {
            flatMap[prefix] = obj?.toString() ?: "null"
        }
    }

    return flatMap
}
