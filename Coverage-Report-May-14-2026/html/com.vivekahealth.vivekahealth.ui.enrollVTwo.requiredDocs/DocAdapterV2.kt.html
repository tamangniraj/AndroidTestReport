package com.vivekahealth.vivekahealth.ui.enrollVTwo.requiredDocs

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.local.enroll.ListDoc
import com.vivekahealth.vivekahealth.databinding.ListRequiredDocumentItemBinding

class DocAdapterV2
(private var onClick: (ListDoc) -> Unit) : ListAdapter<ListDoc, RecyclerView.ViewHolder>(
    DocDiffCallback()
) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return ViewHolder(
            ListRequiredDocumentItemBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as ViewHolder).binding.person = item.person

        (holder as ViewHolder).binding.docUploaded = item.uploaded
        (holder as ViewHolder).binding.docName = item.documentName

        (holder as ViewHolder).binding.docCard.setOnClickListener {
            onClick(item)
        }
    }

    class ViewHolder(
        val binding: ListRequiredDocumentItemBinding,
        private val onClick: (ListDoc) -> Unit
    ) : RecyclerView.ViewHolder(binding.root)

    private class DocDiffCallback : DiffUtil.ItemCallback<ListDoc>() {
        override fun areItemsTheSame(oldItem: ListDoc, newItem: ListDoc): Boolean {
            return oldItem == newItem
        }

        override fun areContentsTheSame(oldItem: ListDoc, newItem: ListDoc): Boolean {
            return oldItem.docName == newItem.docName
        }
    }
}
