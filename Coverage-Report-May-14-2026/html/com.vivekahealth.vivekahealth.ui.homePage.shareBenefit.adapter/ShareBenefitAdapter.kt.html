package com.vivekahealth.vivekahealth.ui.homePage.shareBenefit.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.databinding.SingleDependentViewBinding
import timber.log.Timber

class ShareBenefitAdapter(
    private var onClick: (MemberInfo) -> Unit,
    private var onError: (MemberInfo) -> Unit
) : ListAdapter<MemberInfo, RecyclerView.ViewHolder>(
    BenefitBubbleDiffCallback()
) {

    var selectedPosition: Int = 0

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return MemberViewHolder(
            SingleDependentViewBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick,
            selectedPosition
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val member = getItem(position)

        (holder as MemberViewHolder).binding.setClickListener {
            if ((member.hasAvatar.equals("true", true) || !member.selfieMandatorySwitch) && !member.avatarInReview && !member.memberAdditionInReview && !member.birthDate.isNullOrBlank() && !member.gender.isNullOrBlank() && !member.gender.equals(
                    "U",
                    true
                ) && !member.ssn.isNullOrBlank()
            ) {
                notifyItemChanged(selectedPosition)

//
//            holder.binding.imageView.setImageResource(R.drawable.button_green)

                Timber.d(itemCount.toString())
                onClick(member)

                selectedPosition = position

                notifyItemChanged(position)
            } else {
                notifyItemChanged(selectedPosition)
                selectedPosition = position
                notifyItemChanged(position)
                onError(member)
            }
        }

//        holder.binding.ssnButton.isSelected = member.isSelected

        holder.binding.member = member
    }
    class MemberViewHolder(
        val binding: SingleDependentViewBinding,
        private val onClick: (MemberInfo) -> Unit,
        private var selectedPosition: Int
    ) : RecyclerView.ViewHolder(binding.root)
}

private class BenefitBubbleDiffCallback : DiffUtil.ItemCallback<MemberInfo>() {

    override fun areItemsTheSame(oldItem: MemberInfo, newItem: MemberInfo): Boolean {
        return oldItem.isSelected == newItem.isSelected
    }

    override fun areContentsTheSame(oldItem: MemberInfo, newItem: MemberInfo): Boolean {
        return oldItem == newItem
    }
}
