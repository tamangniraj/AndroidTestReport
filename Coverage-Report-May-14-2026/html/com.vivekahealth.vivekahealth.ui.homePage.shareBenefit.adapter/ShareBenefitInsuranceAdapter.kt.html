package com.vivekahealth.vivekahealth.ui.homePage.shareBenefit.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.local.home.CardList
import com.vivekahealth.vivekahealth.databinding.CardsListItemBinding

class ShareBenefitInsuranceAdapter(private var onClick: (Int, CardList) -> Unit) : ListAdapter<CardList, RecyclerView.ViewHolder>(
    BenefitInsuranceDiffCallback()
) {

    var selectedPosition: Int = 0

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return MemberViewHolder(
            CardsListItemBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick,
            selectedPosition
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as MemberViewHolder).binding.setClickListener {
            if (holder.binding.checkBox.isChecked) {
                item.selected = true
                onClick(position, item)
            } else {
                item.selected = false
                onClick(position, item)
            }
        }
        holder.binding.item = item
    }
    class MemberViewHolder(
        val binding: CardsListItemBinding,
        private val onClick: (Int, CardList) -> Unit,
        private var selectedPosition: Int
    ) : RecyclerView.ViewHolder(binding.root)
}

private class BenefitInsuranceDiffCallback : DiffUtil.ItemCallback<CardList>() {

    override fun areItemsTheSame(oldItem: CardList, newItem: CardList): Boolean {
        return oldItem.selected == newItem.selected
    }

    override fun areContentsTheSame(oldItem: CardList, newItem: CardList): Boolean {
        return oldItem == newItem
    }
}
