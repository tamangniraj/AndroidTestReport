package com.vivekahealth.vivekahealth.ui.openEnrollment.di

import com.vivekahealth.vivekahealth.di.ActivityScope
import com.vivekahealth.vivekahealth.ui.guestRegister.email.di.GuestEmailModule
import com.vivekahealth.vivekahealth.ui.guestRegister.otp.di.GuestOtpModule
import com.vivekahealth.vivekahealth.ui.guestRegister.password.di.GuestPasswordModule
import com.vivekahealth.vivekahealth.ui.guestVerify.documents.di.GuestDocumentScanModule
import com.vivekahealth.vivekahealth.ui.guestVerify.documents.di.ZipAddModule
import com.vivekahealth.vivekahealth.ui.guestVerify.selfie.di.SelfieCaptureModule
import com.vivekahealth.vivekahealth.ui.guestHome.di.GuestHomeModule
import com.vivekahealth.vivekahealth.ui.guestRegister.phone.di.GuestPhoneModule
import com.vivekahealth.vivekahealth.ui.homePage.accountsDetails.pickerBottomSheet.di.PickerBottomSheetModule
import com.vivekahealth.vivekahealth.ui.openEnrollment.OpenEnrollmentActivity
import dagger.Module
import dagger.android.ContributesAndroidInjector

@Module
abstract class OpenEnrollmentModule {

    @ActivityScope
    @ContributesAndroidInjector(
        modules = [
            GuestPasswordModule::class,
            GuestPhoneModule::class,
            GuestOtpModule::class,
            GuestEmailModule::class,
            GuestDocumentScanModule::class,
            FundVerificationModule::class,
            SelfieCaptureModule::class,
            ZipAddModule::class,
            GuestHomeModule::class,
            PickerBottomSheetModule::class
        ]
    )
    abstract fun contributesOpenEnrollmentActivity(): OpenEnrollmentActivity
}
