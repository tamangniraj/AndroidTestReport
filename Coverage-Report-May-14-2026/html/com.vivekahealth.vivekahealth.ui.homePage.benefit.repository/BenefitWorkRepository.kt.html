package com.vivekahealth.vivekahealth.ui.homePage.benefit.repository

/**
 * Repository that handles User objects.
 */
@Singleton
class BenefitWorkRepository @Inject constructor(
    private val appExecutors: AppExecutors,
    private val benefitWorkService: BenefitWorkService,
    private val insuranceWithCardDataDao: InsuranceWithCardCardDataDao,
    private val workLogV2Dao: WorkBenefitV2Dao
) {

    fun insertInsurance(insuranceDataModel: InsuranceDataModel) {
        var insuranceDataNewModel = InsuranceDataNewModel(
            selfId = insuranceDataModel.selfId.toLong(),
            updatedDate = insuranceDataModel.updatedDate
        )

        var list = ArrayList<CardsInsurance>()

        insuranceDataModel.values.forEachIndexed {
                index, it ->
            list.add(
                CardsInsurance(
                    selfId = insuranceDataModel.selfId.toLong(),
                    front = it.front,
                    back = it.back,
                    benefitType = it.benefitType,
                    indexOrder = index
                )
            )
        }

        insuranceWithCardDataDao.insert(insuranceDataNewModel)
        insuranceWithCardDataDao.insertAll(*list.toTypedArray())
    }

    fun getInsuranceFromDb(selfId: String): InsuranceDataModel? {
        val data = insuranceWithCardDataDao.getInsuranceCardNewLiveData(selfId)

        data?.let {
                it ->
            val cards = it.cards.map {
                ValueInsurance(it.front, it.back, it.benefitType)
            }
            return InsuranceDataModel(
                selfId = it.data.selfId.toString(),
                updatedDate = it.data.updatedDate,
                values = cards.sortedByDescending { it.benefitType }
            )
        }
        return null
    }

    fun getInsuranceFromDbLiveData(selfId: String): LiveData<InsuranceDataModel?> {
        val mediator = MediatorLiveData<InsuranceDataModel?>()
        val metadataLiveData = insuranceWithCardDataDao.getInsuranceMetadataLiveData(selfId)
        val cardsLiveData = insuranceWithCardDataDao.getCardDataOrderedLiveData(selfId)

        var metadata: InsuranceDataNewModel? = null
        var cards: List<CardsInsurance>? = null

        val combineAndEmit = {
            when {
                metadata != null && cards != null -> {
                    mediator.value = InsuranceDataModel(
                        selfId = metadata!!.selfId.toString(),
                        updatedDate = metadata!!.updatedDate,
                        values = cards!!.map {
                            ValueInsurance(it.front, it.back, it.benefitType)
                        }
                    )
                }
                metadata == null -> mediator.value = null
            }
        }

        mediator.addSource(metadataLiveData) {
            metadata = it
            combineAndEmit()
        }

        mediator.addSource(cardsLiveData) {
            cards = it
            combineAndEmit()
        }

        return mediator
    }

    fun insertWorkLog(workBenefitV2: WorkBenefitV2?, selfId: String) {
        val workBenefitV2Entity = WorkBenefitV2Entity(selfId, workBenefitV2)
        workLogV2Dao.insert(workBenefitV2Entity)
    }

    fun getWorkLogFromDb(selfId: String): WorkBenefitV2? {
        val data = workLogV2Dao.getWorkLogLiveData(selfId)
        data?.let { entity ->
            entity.workBenefitV2.let {
                return it
            }
        }
        return null
    }

    fun getWorkBenefitOld(memberId: String): Single<ArrayList<WorkHour>> = benefitWorkService.getBenefitApi().getWorkHourOld(
        memberId
    )
    fun getWorkBenefit(year: Int?, month: Int?): Single<ArrayList<WorkHourV2>> = benefitWorkService.getBenefitApi().getWorkHour(
        year,
        month
    )
    fun getWorkBenefitV2(selfId: String): Single<WorkBenefitV2> = benefitWorkService.getBenefitApi().getWorkHourV2(
        selfId
    )

    fun getMemberBenefits(selfId: String, year: Int, month: Int): Single<MemberBenefitPerMonth> = benefitWorkService.getBenefitApi().getMemberBenefits(
        selfId,
        year,
        month
    )

    fun getContractDetails(selfId: String): Single<ContractDetails> = benefitWorkService.getBenefitApi().getContractDetails(
        selfId
    )
    fun getPerMonthBenefit(selfId: String, year: Int, month: Int): Single<WorkBenefit> = benefitWorkService.getBenefitApi().getBenefitPerMonth(
        selfId,
        year,
        month
    )

    fun getContractDetails(
        selfId: String,
        year: Int,
        month: Int,
        benefitId: String,
        contractId: String
    ): Single<ContractBenefitDetails> = benefitWorkService.getBenefitApi().getMemberEligibilityDetails(
        selfId,
        year,
        month,
        benefitId,
        contractId
    )
    fun getInsuranceCard(selfId: String, cardTypes: String, includeHtmlCards: Boolean? = true): Single<InsuranceDataModel> = benefitWorkService.getBenefitApi().getInsuranceCard(
        selfId,
        cardTypes,
        includeHtmlCards
    )

    fun getBarInfo(selfId: String, family: Boolean, year: Int, month: Int): Single<BarModel> = benefitWorkService.getBenefitApi().getBarData(
        selfId,
        family,
        year,
        month
    )

    fun getEligibilityReason(memberId: String, year: Int, month: Int): Single<EligibilityReason> = benefitWorkService.getBenefitApi().getEligibilityReason(
        memberId,
        year,
        month
    )

    fun getWorkSummary(selfId: String, filterBy: String, year: Int, month: Int): Single<WorkHourSummary> = benefitWorkService.getBenefitApi().getWorkHoursSummary(
        selfId,
        filterBy,
        year,
        month
    )
}
