package com.vivekahealth.vivekahealth.data.remote.home.findCare.doctorModelsNew

data class PracticeAddress(
    val addressStatus: String? = "",
    val checkEnrollment: Boolean = false,
    val city: String? = "",
    val country: String? = "",
    val current: String? = null,
    val latitude: String? = null,
    val links: List<String?>? = null,
    val longitude: String? = "",
    val originalAddress: String? = "",
    val postalCode: String? = "",
    val selfId: String? = "",
    val state: String? = "",
    val street1: String? = "",
    val street2: String? = "",
    val type: String? = "",
    val zip: String? = "",
    val zipExtension: String? = ""
) {
    fun secondStreet(): String {
        return if (!street2.isNullOrEmpty()) {
            "$street2, "
        } else {
            ""
        }
    }
    var address = street1?.let { "$it, " } + secondStreet() + city?.let { "$it, " } + state?.let { "$it, " } + postalCode
}
