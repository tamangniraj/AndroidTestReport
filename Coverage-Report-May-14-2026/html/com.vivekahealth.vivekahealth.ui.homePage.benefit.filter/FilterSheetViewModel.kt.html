package com.vivekahealth.vivekahealth.ui.homePage.benefit.filter

import android.content.SharedPreferences
import android.content.res.Resources
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.data.local.home.FilterData
import com.vivekahealth.vivekahealth.utils.DateUtils
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.disposables.CompositeDisposable
import javax.inject.Inject

class FilterSheetViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val resources: Resources
) :
    ViewModel(),
    Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    var filter: String = "REQUESTED_YEAR"

    private val _navigateToFilter = MutableLiveData<Event<FilterData>>()
    val navigateToFilter: LiveData<Event<FilterData>>
        get() = _navigateToFilter

    private val _navigateToSubmit = MutableLiveData<Event<Unit>>()
    val navigateToSubmit: LiveData<Event<Unit>>
        get() = _navigateToSubmit
    private val _navigateToCurrent = MutableLiveData<Event<Unit>>()
    val navigateToCurrent: LiveData<Event<Unit>>
        get() = _navigateToCurrent

    private val _navigateToSelected = MutableLiveData<Event<Unit>>()
    val navigateToSelected: LiveData<Event<Unit>>
        get() = _navigateToSelected

    private val _navigateToReported = MutableLiveData<Event<Unit>>()
    val navigateToReported: LiveData<Event<Unit>>
        get() = _navigateToReported
    private val _navigateToRolling = MutableLiveData<Event<Unit>>()
    val navigateToRolling: LiveData<Event<Unit>>
        get() = _navigateToRolling
    private val _navigateToLife = MutableLiveData<Event<Unit>>()
    val navigateToLife: LiveData<Event<Unit>>
        get() = _navigateToLife

    private val _navigateToClose = MutableLiveData<Event<Unit>>()
    val navigateToClose: LiveData<Event<Unit>>
        get() = _navigateToClose

    var rollingYear: String = resources.getString(
        R.string.rolling_period,
        DateUtils.monthYear(DateUtils.getRollingYear() ?: "")
    )
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.rollingYear, null)
        }

    val current: String = DateUtils.getCurrentYear().toString()
    var currentYear: String = resources.getString(R.string.current_year, current)
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.currentYear, null)
        }

    var totalBanked: Double = 0.0
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.totalBanked, null)
        }

    fun loadSetting() {
        if (filter.equals("REQUESTED_YEAR", true)) {
            _navigateToCurrent.value = Event(Unit)
        } else if (filter.equals("LIFETIME", true)) {
            _navigateToLife.value = Event(Unit)
        } else if (filter.equals("ROLLING_YEAR", true)) {
            _navigateToRolling.value = Event(Unit)
        } else if (filter.equals("REQUESTED_YEAR_MONTH", true)) {
            _navigateToSelected.value = Event(Unit)
        } else if (filter.equals("LAST_REPORTED", true)) {
            _navigateToReported.value = Event(Unit)
        }
    }

    fun rollingPeriod() {
        filter = "ROLLING_YEAR"
    }
    fun lifeTime() {
        filter = "LIFETIME"
    }

    fun currentYear() {
        filter = "REQUESTED_YEAR"
    }
    fun selectedMonth() {
        filter = "REQUESTED_YEAR_MONTH"
    }

    fun lastReportedMonth() {
        filter = "LAST_REPORTED"
    }

    fun apply() {
        _navigateToFilter.value = Event(FilterData(filter))
    }
    fun close() {
        _navigateToClose.value = Event(Unit)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
