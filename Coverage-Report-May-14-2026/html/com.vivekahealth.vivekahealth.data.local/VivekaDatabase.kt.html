/**
 * Main database VivekaDataBase.
 */
@Database(
    entities = [
        MemberInfo::class,
        InsuranceCardHome::class,
        RecentActivityHome::class,
        ClientDetailResponse::class,
        InsuranceDataNewModel::class,
        CardsInsurance::class,
        WorkBenefitV2Entity::class,
        GuestUserInfo::class,
        GuestClientDetails::class,
        SummaryImpactEntity::class,
        WorkHistoryEntity::class,
        PensionConfigEntity::class
    ],
    version = 39,
    exportSchema = false
)

@TypeConverters(VivekaTypeConverters::class)

abstract class VivekaDatabase : RoomDatabase() {
    abstract fun memberInfoDao(): MemberInfoDao
    abstract fun HomeBenefitDao(): HomeBenefitDao
    abstract fun RecentCardsDao(): RecentCardsDao
    abstract fun insuranceWithCardDataDao(): InsuranceWithCardCardDataDao
    abstract fun ClientDetailsDao(): ClientDetailsDao
    abstract fun workBenefitV2Dao(): WorkBenefitV2Dao
    abstract fun guestInfoDao(): GuestInfoDao
    abstract fun guestClientDetailsDao(): GuestClientDetailsDao
    abstract fun summaryImpactDao(): SummaryImpactDao
    abstract fun workHistoryDao(): WorkHistoryDao
    abstract fun pensionConfigDao(): PensionConfigDao

    companion object {

        // For Singleton instantiation
        @Volatile
        private var INSTANCE: VivekaDatabase? = null
        fun getDatabase(context: Context): VivekaDatabase {
            // if the INSTANCE is not null, then return it,
            // if it is, then create the database
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    VivekaDatabase::class.java,
                    "viveka_db.db"
                ).allowMainThreadQueries()
                    .addMigrations(MIGRATION_26_28)
                    .fallbackToDestructiveMigration().build()
                INSTANCE = instance
                // return instance
                instance
            }
        }

        var MIGRATION_26_28 = object : Migration(26, 28) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL(
                    "CREATE TABLE IF NOT EXISTS insurance_data_new (" +
                        "selfId INTEGER NOT NULL PRIMARY KEY, " +
                        "updatedDate TEXT)"
                )

                db.execSQL(
                    "CREATE TABLE IF NOT EXISTS card_data (" +
                        "selfId INTEGER NOT NULL, " +
                        "front TEXT, " +
                        "back TEXT, " +
                        "benefitType TEXT NOT NULL, " +
                        "PRIMARY KEY (selfId, benefitType))"
                )

                try {
                    val cursor = db.query("SELECT * FROM insurance_data")
                    while (cursor.moveToNext()) {
                        val selfId = cursor.getString(cursor.getColumnIndexOrThrow("selfId")).toLong()
                        val updatedDate = cursor.getString(
                            cursor.getColumnIndexOrThrow("updatedDate")
                        )
                        val cardsStrings = cursor.getString(cursor.getColumnIndexOrThrow("values"))
                        val cards = stringToValues(cardsStrings)

                        val list = ArrayList<CardsInsurance>()
                        cards?.forEachIndexed { index, it ->

                            list.add(
                                CardsInsurance(
                                    selfId = selfId,
                                    front = it.front,
                                    back = it.back,
                                    benefitType = it.benefitType,
                                    indexOrder = index
                                )
                            )
                        }

                        db.execSQL(
                            "INSERT INTO insurance_data_new (selfId, updatedDate) VALUES (?, ?)",
                            arrayOf(selfId, updatedDate)
                        )

                        list.forEach {
                            db.execSQL(
                                "INSERT INTO card_data (selfId, front, back, benefitType) VALUES (?, ?, ?, ?)",
                                arrayOf(selfId, it.front, it.back, it.benefitType)
                            )
                        }
                    }
                    cursor.close()

                    db.execSQL(
                        "DROP TABLE insurance_data"
                    )
                } catch (exp: Exception) {
                    exp.printStackTrace()
                }
            }
        }
    }
}

fun stringToValues(data: String?): List<ValueInsurance>? {
    var gson = Gson()

    if (data == null) {
        return null
    }
    val listType = object : TypeToken<List<ValueInsurance>>() {}.type
    return gson.fromJson(data, listType)
}
