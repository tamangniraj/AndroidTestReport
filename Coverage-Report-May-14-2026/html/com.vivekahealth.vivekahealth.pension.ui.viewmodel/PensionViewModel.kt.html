package com.vivekahealth.vivekahealth.pension.ui.viewmodel

import android.content.Context
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.google.firebase.analytics.FirebaseAnalytics
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.PensionErrorResponse
import com.vivekahealth.vivekahealth.pension.data.model.AvailableOptions
import com.vivekahealth.vivekahealth.pension.data.model.RetirementOption
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.flatMapLatest
import kotlinx.coroutines.flow.flowOf
import kotlinx.coroutines.launch
import com.vivekahealth.vivekahealth.pension.data.repository.PensionRepository
import com.vivekahealth.vivekahealth.pension.data.model.SummaryImpactData
import com.vivekahealth.vivekahealth.pension.data.model.SurvivorOptionConfig
import com.vivekahealth.vivekahealth.pension.data.model.UserData
import com.vivekahealth.vivekahealth.pension.data.model.WorkHistorySummary
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import timber.log.Timber
import java.io.File
import java.io.FileOutputStream
import javax.inject.Inject

class PensionViewModel @Inject constructor(
    private val repository: PensionRepository,
    private val analytics: FirebaseAnalytics
) : ViewModel() {

    private val compositeDisposable = CompositeDisposable()

    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading.asStateFlow()

    private val _error = MutableStateFlow<String?>(null)
    val error: StateFlow<String?> = _error.asStateFlow()

    // Reactive userData that resets when member ID changes
    private val _userData = MutableStateFlow(UserData())
    val userData: StateFlow<UserData> = _userData.asStateFlow()

    val hasPensionData: StateFlow<Boolean> = repository.hasPensionData

    private val _currentMemberId = MutableStateFlow<String?>(null)
    private val _summaryImpactDataFromDb = MutableStateFlow<SummaryImpactData?>(null)
    val summaryImpactDataFromDb: StateFlow<SummaryImpactData?> = _summaryImpactDataFromDb.asStateFlow()

    // Work History state
    private val _workHistoryFromDb = MutableStateFlow<WorkHistorySummary?>(null)
    val workHistoryFromDb: StateFlow<WorkHistorySummary?> = _workHistoryFromDb.asStateFlow()

    private val _isLoadingWorkHistory = MutableStateFlow(false)
    val isLoadingWorkHistory: StateFlow<Boolean> = _isLoadingWorkHistory.asStateFlow()

    private val _workHistoryError = MutableStateFlow<String?>(null)
    val workHistoryError: StateFlow<String?> = _workHistoryError.asStateFlow()

    // Pension Config state
    private val _availableOptions = MutableStateFlow<AvailableOptions?>(
        null
    )
    val survivorOptionsFromDb: StateFlow<AvailableOptions?> =
        _availableOptions.asStateFlow()

    private val _isLoadingConfig = MutableStateFlow(false)
    val isLoadingConfig: StateFlow<Boolean> = _isLoadingConfig.asStateFlow()

    private val _configError = MutableStateFlow<String?>(null)
    val configError: StateFlow<String?> = _configError.asStateFlow()

    // Pension Estimate state
    private val _pensionEstimate = MutableStateFlow<com.vivekahealth.vivekahealth.pension.data.model.PensionEstimateResponse?>(
        null
    )
    val pensionEstimate: StateFlow<com.vivekahealth.vivekahealth.pension.data.model.PensionEstimateResponse?> =
        _pensionEstimate.asStateFlow()

    private val _isLoadingEstimate = MutableStateFlow(false)
    val isLoadingEstimate: StateFlow<Boolean> = _isLoadingEstimate.asStateFlow()

    private val _estimateError = MutableStateFlow<String?>(null)
    val estimateError: StateFlow<String?> = _estimateError.asStateFlow()

    // Debounce job for estimate API calls
    private var estimateJob: kotlinx.coroutines.Job? = null

    // Download Pension Application state
    private val _isDownloadingApplication = MutableStateFlow(false)
    val isDownloadingApplication: StateFlow<Boolean> = _isDownloadingApplication.asStateFlow()

    private val _downloadError = MutableStateFlow<String?>(null)
    val downloadError: StateFlow<String?> = _downloadError.asStateFlow()

    private val _downloadedPdfFile = MutableStateFlow<File?>(null)
    val downloadedPdfFile: StateFlow<File?> = _downloadedPdfFile.asStateFlow()

    init {
        // Clear userData when member ID changes to a different member
        viewModelScope.launch {
            var previousMemberId: String? = null
            _currentMemberId.collect { newMemberId ->
                if (newMemberId != null && previousMemberId != null && newMemberId != previousMemberId) {
                    // Different member detected - reset userData
                    _userData.value = UserData()
                    repository.clearPensionData()
                }
                previousMemberId = newMemberId
            }
        }

        // Observe database changes for current member using flatMapLatest to switch between members
        viewModelScope.launch {
            _currentMemberId
                .flatMapLatest { memberId ->
                    if (memberId != null) {
                        repository.getSummaryImpactData(memberId)
                    } else {
                        flowOf(null)
                    }
                }
                .collect { data ->
                    _summaryImpactDataFromDb.value = data
                }
        }

        // Observe work history from database
        viewModelScope.launch {
            _currentMemberId
                .flatMapLatest { memberId ->
                    if (memberId != null) {
                        repository.getWorkHistoryFromDb(memberId)
                    } else {
                        flowOf(null)
                    }
                }
                .collect { data ->
                    _workHistoryFromDb.value = data
                }
        }

        // Observe pension config from database
        viewModelScope.launch {
            _currentMemberId
                .flatMapLatest { memberId ->
                    if (memberId != null) {
                        repository.getPensionConfigFromDb(memberId)
                    } else {
                        flowOf(null)
                    }
                }
                .collect { data ->
                    _availableOptions.value = data
                }
        }
    }

    fun setCurrentMemberId(memberId: String) {
        _currentMemberId.value = memberId
    }

    fun fetchPensionData(memberId: String) {
        setCurrentMemberId(memberId)
        viewModelScope.launch {
            compositeDisposable.add(
                repository.getPensionMemberData(memberId)
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _isLoading.value = true
                        _error.value = null
                    }
                    .subscribe({ response ->
                        _isLoading.value = false

                        if (response.isSuccessful) {
                            // Update local userData with pension member response
                            _userData.value = _userData.value.copy(
                                pensionMemberResponse = response.body()
                            )

                            response.body()?.let {
                                repository.updatePensionData(it)
                            }
                            // Also update repository to keep hasPensionData in sync

                            // Save SummaryImpactData to database
                            viewModelScope.launch {
                                val summaryImpactData = userData.value.summaryImpactData
                                repository.saveSummaryImpactData(memberId, summaryImpactData)
                            }
                        } else {
                            if (response.code() == 403) {
                                viewModelScope.launch {
                                    repository.deleteSummaryImpactData(memberId)
                                }
                            }
                        }

                        Timber.d("Pension data fetched successfully for member: $memberId")
                    }, { throwable ->
                        _isLoading.value = false
                        _error.value = throwable.message
                        Timber.e(throwable, "Error fetching pension data for member: $memberId")
                    })
            )
        }
    }

    fun updateRetirementAge(retirementAge: RetirementOption) {
        _userData.value = _userData.value.copy(
            selectedRetirementAge = retirementAge
        )
    }

    fun updateExpectedHours(hours: String) {
        _userData.value = _userData.value.copy(
            expectedHours = hours
        )
    }

    fun updateSurvivorBenefit(benefit: SurvivorOptionConfig?) {
        _userData.value = _userData.value.copy(
            selectedSurvivorBenefit = benefit
        )
    }

    fun clearPensionData() {
        _userData.value = UserData()
        repository.clearPensionData()
    }

    // Work History methods
    fun fetchWorkHistory(memberId: String) {
        setCurrentMemberId(memberId)
        viewModelScope.launch {
            // Check if data exists in database first
            val existingData = repository.getWorkHistoryFromDbOnce(memberId)
            val hasExistingData = existingData != null

            compositeDisposable.add(
                repository.getWorkHistorySummary(memberId)
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        // Only show loading if no data exists in database
                        if (!hasExistingData) {
                            _isLoadingWorkHistory.value = true
                        }
                        _workHistoryError.value = null
                    }
                    .subscribe({ response ->
                        // Save to database - the StateFlow will automatically update via the database Flow
                        viewModelScope.launch {
                            repository.saveWorkHistory(memberId, response)
                            // Loading will be automatically hidden when the UI detects the updated data from StateFlow
                            _isLoadingWorkHistory.value = false
                            Timber.d(
                                "Work history fetched and saved successfully for member: $memberId"
                            )
                        }
                    }, { throwable ->
                        _isLoadingWorkHistory.value = false
                        _workHistoryError.value = throwable.message
                        Timber.e(throwable, "Error fetching work history for member: $memberId")
                    })
            )
        }
    }

    fun clearWorkHistoryError() {
        _workHistoryError.value = null
    }

    // Pension Config methods
    fun fetchPensionConfig(memberId: String) {
        setCurrentMemberId(memberId)
        viewModelScope.launch {
            // Check if data exists in database first
            val existingData = repository.getPensionConfigFromDbOnce(memberId)
            val hasExistingData = existingData != null

            compositeDisposable.add(
                repository.getPensionConfig(memberId)
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        // Only show loading if no data exists in database
                        if (!hasExistingData) {
                            _isLoadingConfig.value = true
                        }
                        _configError.value = null
                    }
                    .subscribe({ response ->
                        // Save survivor options to database
                        viewModelScope.launch {
                            response.availableOptions?.let { options ->
                                repository.savePensionConfig(memberId, options)
                                _isLoadingConfig.value = false
                                Timber.d(
                                    "Pension config fetched and saved successfully for member: $memberId"
                                )
                            } ?: run {
                                _isLoadingConfig.value = false
                                Timber.w(
                                    "No survivor options in config response for member: $memberId"
                                )
                            }
                        }
                    }, { throwable ->
                        _isLoadingConfig.value = false
                        _configError.value = throwable.message
                        Timber.e(throwable, "Error fetching pension config for member: $memberId")
                    })
            )
        }
    }

    fun clearConfigError() {
        _configError.value = null
    }

    fun clearRunning() {
        estimateJob?.cancel()
        compositeDisposable.clear()
    }

    // Pension Estimate methods
    fun fetchPensionEstimate(
        memberId: String,
        retirementOption: RetirementOption?,
        survivorOptionCode: String?,
        expectedHoursPerYear: Double?,
        maritalStatus: String?
    ) {
        setCurrentMemberId(memberId)

        // Cancel previous job if exists
        estimateJob?.cancel()

        // Launch new job with debounce
        estimateJob = viewModelScope.launch {
            // Debounce delay
            kotlinx.coroutines.delay(500)

            val request = com.vivekahealth.vivekahealth.pension.data.model.PensionEstimateRequest(
                retirementAge = retirementOption?.age,
                pensionType = retirementOption?.pensionType,
                survivorOption = survivorOptionCode,
                projectionsEnabled = true,
                expectedHoursPerYear = expectedHoursPerYear,
                maritalStatus = maritalStatus
            )

            compositeDisposable.add(
                repository.getPensionEstimate(memberId, request)
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _isLoadingEstimate.value = true
                        _estimateError.value = null
                    }
                    .subscribe({ response ->

                        if (response.isSuccessful) {
                            _pensionEstimate.value = response.body()
                        } else {
                            val errorResponse: PensionErrorResponse = Gson().fromJson(
                                response.errorBody()?.string(),
                                PensionErrorResponse::class.java
                            )
                            errorResponse.message?.let {
                                _estimateError.value = it
                            }
                        }

                        _isLoadingEstimate.value = false

                        Timber.d("Pension estimate fetched successfully for member: $memberId")
                    }, { throwable ->
                        _isLoadingEstimate.value = false
                        _estimateError.value = throwable.message
                        Timber.e(throwable, "Error fetching pension estimate for member: $memberId")
                    })
            )
        }
    }

    fun clearEstimateError() {
        _estimateError.value = null
    }

    // Download Pension Application
    fun downloadPensionApplication(context: Context, memberId: String) {
        setCurrentMemberId(memberId)
        compositeDisposable.add(
            repository.downloadPensionApplication(memberId)
                .subscribeOn(Schedulers.io())
                .map { responseBody ->
                    try {
                        // Save the file to cache directory
                        val file = File(context.cacheDir, "Pension Application.pdf")
                        val inputStream = responseBody.byteStream()
                        val outputStream = FileOutputStream(file)

                        inputStream.use { input ->
                            outputStream.use { output ->
                                input.copyTo(output)
                            }
                        }
                        file
                    } catch (e: Exception) {
                        _isDownloadingApplication.value = false
                        _downloadError.value = e.message ?: "Failed to save downloaded file"
                        Timber.e(e, "Error saving downloaded file for member: $memberId")
                        null
                    }
                }
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _isDownloadingApplication.value = true
                    _downloadError.value = null
                    _downloadedPdfFile.value = null
                }
                .subscribe({ file ->
                    file?.let {
                        _downloadedPdfFile.value = file
                        _isDownloadingApplication.value = false
                        Timber.d(
                            "Pension application downloaded successfully for member: $memberId"
                        )
                    }
                }, { throwable ->
                    _isDownloadingApplication.value = false
                    _downloadError.value = throwable.message
                    Timber.e(
                        throwable,
                        "Error downloading pension application for member: $memberId"
                    )
                })
        )
    }

    fun clearDownloadError() {
        _downloadError.value = null
    }

    fun clearDownloadedPdfFile() {
        _downloadedPdfFile.value = null
    }

    fun logEvent(event: String) {
        analytics.logEvent(event, null)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
