package com.vivekahealth.vivekahealth.ui.homePage.informationSharing.datePreference

import android.content.SharedPreferences
import android.content.res.Resources
import android.text.Editable
import android.text.TextWatcher
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MediatorLiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.local.home.PreferenceData
import com.vivekahealth.vivekahealth.data.local.home.QuickItem
import com.vivekahealth.vivekahealth.utils.DateUtils
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.disposables.CompositeDisposable
import javax.inject.Inject

class DatePreferenceSheetViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val resources: Resources
) :
    ViewModel(),
    Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    var filter: String = "REQUESTED_YEAR"

    private val _navigateToSubmit = MutableLiveData<Event<Unit>>()
    val navigateToSubmit: LiveData<Event<Unit>>
        get() = _navigateToSubmit

    private val _navigateToClose = MutableLiveData<Event<Unit>>()
    val navigateToClose: LiveData<Event<Unit>>
        get() = _navigateToClose

    val current: String = DateUtils.getCurrentYear().toString()

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }

    /**
     * new field
     */

    var tempSelectionMap = HashMap<Int, List<QuickItem>?>()

    var errorMessage: String = ""

    var showError: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showError, null)
        }

    fun apply() {
        _navigateToApply.value = Event(
            PreferenceData(
                startDateComplete.value?.peekContent()?.replace(" ", ""),
                endDateComplete.value?.peekContent()?.replace(" ", ""),
                diagnois = tempSelectionMap[0]?.let { ArrayList(it) },
                prescreption = tempSelectionMap[1]?.let { ArrayList(it) }

            )
        )
    }
    fun close() {
        _navigateToClose.value = Event(Unit)
    }

    private val _navigateToApply = MutableLiveData<Event<PreferenceData>>()
    val navigateToApply: LiveData<Event<PreferenceData>>
        get() = _navigateToApply

    private val _startDateComplete = MutableLiveData<Event<String>>()
    val startDateComplete: LiveData<Event<String>>
        get() = _startDateComplete

    private val _endDateComplete = MutableLiveData<Event<String>>()
    val endDateComplete: LiveData<Event<String>>
        get() = _endDateComplete

    val combineStartEndDate = MediatorLiveData<Pair<String?, String?>>()

    init {
        combineStartEndDate.addSource(startDateComplete) { startDate ->
            val endDate = endDateComplete.value
            combineStartEndDate.value = Pair(startDate.peekContent(), endDate?.peekContent())
        }
        combineStartEndDate.addSource(endDateComplete) { endDate ->
            val startDate = startDateComplete.value
            combineStartEndDate.value = Pair(startDate?.peekContent(), endDate?.peekContent())
        }
    }

    @get:Bindable
    val startDateTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _startDateComplete.value = Event(s.toString())
        }
    }

    fun previousStartDate(date: String) {
        _startDateComplete.value = Event(date)
    }

    fun previousEndDate(date: String) {
        _endDateComplete.value = Event(date)
    }

    @get:Bindable
    val endDateTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _endDateComplete.value = Event(s.toString())
        }
    }
}
