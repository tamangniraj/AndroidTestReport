    private var memberInfoList = homeRepository.getMemberFromDb().asLiveData()

    var showSearch: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showSearch, null)
        }

    var data: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.data, null)
        }
    var reason: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.reason, null)
        }
    var redDot: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.redDot, null)
        }

    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    private val _navigateToSearch = MutableLiveData<Event<Unit>>()
    val navigateToSearch: LiveData<Event<Unit>>
        get() = _navigateToSearch

    var currentSearch = ""
    var sentSelected = false
    var receiveSelected = false
    var openSelected = false
    var unOpenSelected = false
    var currentViewed = ""
    var currentSentReceived = ""
    var currentSharedMemberId = ""
    var currentCategoryId = ""
    var currentSortOrder = "desc"
    var currentDocumentType = ""

    var dependentMemberList: ArrayList<MemberInfo> = ArrayList()
    var bubbleMemberList: ArrayList<MemberInfo> = ArrayList()

    private val _categoryListGenerated = MutableLiveData<Event<List<Dto>>>()
    val categoryListGenerated: LiveData<Event<List<Dto>>>
        get() = _categoryListGenerated
    private val _categoryListGeneratedUpdate = MutableLiveData<Event<List<Dto>>>()
    val categoryListGeneratedUpdate: LiveData<Event<List<Dto>>>
        get() = _categoryListGeneratedUpdate
    private val _docListGenerated = MutableLiveData<Event<DocumentList>>()
    val docListGenerated: LiveData<Event<DocumentList>>
        get() = _docListGenerated

    private val _pendingDocument = MutableLiveData<Event<List<PendingRequestDTO>>>()
    val pendingDocument: LiveData<Event<List<PendingRequestDTO>>>
        get() = _pendingDocument
    private val _navigateToSort = MutableLiveData<Event<Unit>>()
    val navigateToSort: LiveData<Event<Unit>>
        get() = _navigateToSort
    var sort = "desc"
    var docTypeArgs = ""
    var list: List<Dto> = listOf()

    private val _navigateToFilter = MutableLiveData<Event<Unit>>()
    val navigateToFilter: LiveData<Event<Unit>>
        get() = _navigateToFilter

    private val _navigateToSubmit = MutableLiveData<Event<Unit>>()
    val navigateToSubmit: LiveData<Event<Unit>>
        get() = _navigateToSubmit

    init {
        viewModelScope.launch {
            if (homeRepository.getMemberSid(preferences.getString("memberID", "").toString())
                .firstOrNull() == null
            ) {
                compositeDisposable.add(
                    homeRepository.getMember()
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .subscribe({
                            if (!Constants.IS_OFFLINE) {
                                homeRepository.insertMember(it)
                            }
                        }, {
                        })
                )
            }
        }
        getLogView()
        getDocumentPending()
    }

    fun getCategory(clientId: String) {
        compositeDisposable.add(
            documentRepository.getType(clientId)
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                    loading = true
                }.subscribe({ typeList ->
                    Timber.d(typeList.toString())

                    list = typeList.dtos.distinctBy { it.documentCategory }
                    var _list = list.toMutableList()
                    _list.add(0, Dto("All", null, "All", -1, -1, -1, isSelected = true))

                    list = _list
                    var temp: Dto? = null
                    if (docTypeArgs.isNotBlank()) {
                        temp = list.find { it.documentType.equals(docTypeArgs, true) }
                        currentCategoryId = temp?.documentCategoryId.toString() ?: ""
                        list.forEach {
                            it.isSelected = it.documentCategoryId?.equals(temp?.documentCategoryId) == true
                        }
                    }
                    loading = false
                    _categoryListGenerated.value = Event(list)
                }, {
                    Timber.d(it.toString())
                    loading = false
                })
        )
    }

    fun catListSelected(dto: Dto) {
        var temp: Dto? = null
        temp = list.find { it.documentCategoryId?.equals(dto.documentCategoryId) == true }
        list.forEach {
            it.isSelected = it.documentCategoryId?.equals(temp?.documentCategoryId) == true
        }

        if (list.find { it.isSelected } == null) {
            list.firstOrNull()?.isSelected = true
        }

        _categoryListGeneratedUpdate.value = Event(list)
    }

    fun loadMemberList(memberInfo: MemberInfo) {
        dependentMemberList.clear()
        bubbleMemberList.clear()
        memberInfo.isSelected = false

        memberInfo.insured?.let { it1 ->
            bubbleMemberList.add(it1)
            bubbleMemberList.add(memberInfo)
        } ?: run {
            bubbleMemberList.add(0, memberInfo)
        }
        memberInfo.dependentsArrayList?.let { it1 ->

            for (member in it1) {
                if (StringUtils.equalsIgnoreCase(member.memberRelationship, "Spouse")) {
                    bubbleMemberList.add(1, member)
                } else {
                    if (!member.hideProfileFromHome) {
                        bubbleMemberList.add(member)
                    }
                }
            }

            var grouped = bubbleMemberList.groupBy { it.memberRelationship }
            grouped.forEach {
                it.value.sortedByDescending { item ->
                    DateUtils.stringServerDateToDate(
                        item.birthDate.toString()
                    )
                }
                it.value.reversed()
            }

            bubbleMemberList.clear()

            grouped.forEach {
                if (it.key.equals("Insured", true)) {
                    bubbleMemberList.add(0, it.value[0])
                } else if (it.key.equals("Spouse", true)) {
                    if (grouped.containsKey("Insured")) {
                        bubbleMemberList.add(1, it.value[0])
                    } else {
                        bubbleMemberList.add(0, it.value[0])
                    }
                } else if (it.key.equals("Child", true)) {
                    val dateGroup = it.value.groupBy { date ->
                        DateUtils.stringServerDateToDate(
                            date.birthDate.toString()
                        )
                    }

                    dateGroup.forEach { item ->
                        val sorted = item.value.sortedBy { item1 -> item1.firstName }
                        bubbleMemberList.addAll(sorted)
                    }
                }
            }
            val tempOthers: ArrayList<MemberInfo> = arrayListOf()
            grouped.forEach {
                if (!it.key.equals("Insured", true) && !it.key.equals(
                        "Spouse",
                        true
                    ) && !it.key.equals("Child", true)
                ) {
                    tempOthers.addAll(it.value)
                }
            }

            val dateGroup = tempOthers.groupBy { date ->
                DateUtils.stringServerDateToDate(
                    date.birthDate.toString()
                )
            }
            dateGroup.forEach { item ->
                val sorted = item.value.sortedBy { item1 -> item1.firstName }
                bubbleMemberList.addAll(sorted)
            }
        }

        dependentMemberList = bubbleMemberList
    }

    fun selectedMember() {
        val slist: ArrayList<String> = ArrayList()
        var ssmem = ""
        for (member in dependentMemberList) {
            if (member.isSelected) {
                slist.add(member.selfId)
            }
        }

        currentSharedMemberId = if (slist.isNotEmpty()) {
            slist.joinToString().lowercase()
        } else {
            ""
        }
    }

    fun sort() {
        _navigateToSort.value = Event(Unit)
    }

    fun filter() {
        _navigateToFilter.value = Event(Unit)
    }

    fun gotoSubmit() {
        if (!loading) {
            _navigateToSubmit.value = Event(Unit)
        }
    }
    fun getLogView() {
        compositeDisposable.add(
            analyticsRepository.getLogPageView(
                preferences.getString(
                    "memberID",
                    ""
                ).toString(),
                PageType.DOCUMENTS_LIST_SCREEN.pageName
            )
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                }.subscribe({
                }, {
                })
        )
    }
    fun getDocuments(categoryId: String, sort: String) {
        compositeDisposable.add(
            documentRepository.getDocumentList(
                preferences.getString("memberID", "").toString(),
                currentSharedMemberId,
                currentDocumentType,
                categoryId,
                currentViewed,
                createdDateSort = currentSortOrder,
                currentSearch
            )
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                    loading = true
                }.subscribe({
                    Timber.d(it.toString())
                    data = it.values.isNullOrEmpty()
                    loading = false
                    _docListGenerated.value = Event(it)
                }, {
                    Timber.d(it.toString())
                    loading = false
                })
        )
    }

    fun getDocumentsSearch(search: String) {
        currentSearch = search
        compositeDisposable.add(
            documentRepository.getDocumentListSearched(
                preferences.getString("memberID", "").toString(),
                currentSharedMemberId,
                currentDocumentType,
                currentCategoryId,
                currentViewed,
                createdDateSort = currentSortOrder,
                search
            )
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                    loading = true
                }.subscribe({
                    Timber.d(it.toString())
                    _docListGenerated.value = Event(it)
                    data = it.values.isNullOrEmpty()

                    loading = false
                }, {
                    Timber.d(it.toString())
                    loading = false
                })
        )
    }

    fun getDocumentPending() {
        compositeDisposable.add(
            documentRepository.getPendingDocument(
                preferences.getString(
                    "memberID",
                    ""
                ).toString()
            )
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                    loading = true
                }.subscribe({
                    Timber.d(it.toString())
                    if (it.pendingRequestDTOList != null) {
                        pendingRequestDTO = it.pendingRequestDTOList
                        _pendingDocument.value = Event(it.pendingRequestDTOList)
                    }
                    loading = false
                }, {
                    Timber.d(it.toString())
                    loading = false
                })
        )
    }

    fun closeSearch() {
        showSearch = false

        getDocumentsSearch("")
        _navigateToSearch.value = Event(Unit)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
