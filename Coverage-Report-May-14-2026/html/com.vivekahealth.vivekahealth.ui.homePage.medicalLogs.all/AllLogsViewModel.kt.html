package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.all

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.MedicalLogsAll
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.LoadParams
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.MedicalLogAllDataSaver
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.MedicalLogsFragment.Companion.PAGE_SIZE
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.MedicalLogsSortingUtils
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.ParameterizedDebouncer
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.repository.MedicalLogsRepository
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import javax.inject.Inject
import timber.log.Timber

class AllLogsViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val medicalLogsRepository: MedicalLogsRepository,
    private val homeRepository: HomeRepository
) : ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _navigateToSort = MutableLiveData<Event<Unit>>()
    val navigateToSort: LiveData<Event<Unit>>
        get() = _navigateToSort
    var sort = "desc"

    var currentMemberId: String = preferences.getString("memberID", "").toString()

    var selectedMemberId: String = ""

    var title: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.title, null)
        }
    private val _navigateToFilter = MutableLiveData<Event<Unit>>()
    val navigateToFilter: LiveData<Event<Unit>>
        get() = _navigateToFilter

    var data: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.data, null)
        }
    var reason: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.reason, null)
        }
    var labSize: Int = 0
    var presSize: Int = 0

    var type: String = "all"
    private val _processedLogs = MutableLiveData<List<Any>>()
    val processedLogs: LiveData<List<Any>> = _processedLogs

    var currentPage = 1
    private val pageSize = PAGE_SIZE
    var hasMorePages = false
    var isLoadedFirstPage = false

    private val _uiState = MutableLiveData<MedicalLogsUiState>()
    val uiState: LiveData<MedicalLogsUiState> = _uiState

    var selectedMember = ""

    var shouldAppendAllData: Boolean = false
    val dataSaver = MedicalLogAllDataSaver()

    fun onMemberIdChanged(newMemberId: String) {
        sort = "desc" // Reset sort order to default
        selectedMemberId = newMemberId
        loadDataWithDebounce(newMemberId, isRefresh = true)
    }

    private val loadDebouncer = ParameterizedDebouncer<LoadParams>()

    fun loadDataWithDebounce(sharedId: String, isRefresh: Boolean = false) {
        loadDebouncer.debounce(LoadParams(sharedId, isRefresh)) { params ->
            loadMedicalLogs(params.sharedIds, params.isRefresh)
        }
    }

    private fun loadMedicalLogs(selected: String, isRefresh: Boolean = false) {
        selectedMember = if (selected.isNullOrBlank() || selected.equals("null", true)) {
            currentMemberId
        } else {
            selected
        }
        // do not call api for the member id whose id is negative
        if (
            selectedMember.isNotEmpty() && selectedMember.toLong() < 0
        ) {
            return
        }

        if (isRefresh) {
            currentPage = 1
            _processedLogs.value = emptyList()
            compositeDisposable.clear()
            data = false
        }

        compositeDisposable.add(
            medicalLogsRepository.getMedicalLogs(
                currentMemberId,
                type,
                selectedMember,
                sort,
                "",
                pageSize,
                currentPage
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _uiState.value = MedicalLogsUiState.LoadingMore
                }
                .subscribe({
                    isLoadedFirstPage = true

                    if (currentPage == 1) {
                        it?.let {
                            data = it.claimsList.isNullOrEmpty() and it.diagnosisList.isNullOrEmpty() and it.labInfoList.isNullOrEmpty() and it.prescriptionList.isNullOrEmpty()
                        }
                    }

                    handleSuccessResponse(it)
                }, {
                    Timber.d(it.toString())
                    _uiState.value = MedicalLogsUiState.Error(
                        it.message ?: "Unknown error occurred"
                    )
                })
        )
    }

    private fun handleSuccessResponse(response: MedicalLogsAll) {
        // Update pagination info
        response.paging?.let { paging ->
            hasMorePages = paging.hasNextPage
            currentPage = paging.currentPage
        }

        // Check if data is empty
        val isEmpty = isDataEmpty(response)

        if (isEmpty) {
            _uiState.value = MedicalLogsUiState.Empty
        } else {
            _uiState.value = MedicalLogsUiState.Success(response)
            val processedData = dataSaver.saveData(response, shouldAppendAllData)
            _processedLogs.value = MedicalLogsSortingUtils.sortAllMedicalLogs(
                processedData,
                sort
            )
        }
    }

    private fun isDataEmpty(data: MedicalLogsAll): Boolean {
        return data.claimsList.isNullOrEmpty() &&
            data.diagnosisList.isNullOrEmpty() &&
            data.labInfoList.isNullOrEmpty() &&
            data.prescriptionList.isNullOrEmpty()
    }

    fun hasMorePages(): Boolean {
        return hasMorePages
    }

    fun loadNextPage() {
        if (_uiState.value !is MedicalLogsUiState.LoadingMore && _uiState.value !is MedicalLogsUiState.Loading && hasMorePages) {
            currentPage++
            loadMedicalLogs(selectedMemberId)
        }
    }

    fun sort() {
        _navigateToSort.value = Event(Unit)
    }
    fun filter() {
        _navigateToFilter.value = Event(Unit)
    }
    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    private val _tutorialsAdded = MutableLiveData(false)
    val tutorialsAdded: LiveData<Boolean>
        get() = _tutorialsAdded

    private val _showTutorial = MutableLiveData<Event<Unit>>()
    val showTutorial: LiveData<Event<Unit>>
        get() = _showTutorial

    var tutorialShown = false
    private var addedMedicalLogTutorial = false

    fun addMedicalLogTutorials() {
        addedMedicalLogTutorial = true
        updateTutorialAdditionStatus()
    }

    private fun updateTutorialAdditionStatus() {
        _tutorialsAdded.value = addedMedicalLogTutorial
    }

    fun showTutorialIfConditionMatched() {
        if (!preferences.getBoolean(
                Constants.SHOWN_MEDICAL_TUT_FOR_SESSION,
                false
            ) && !tutorialShown && isLoadedFirstPage
        ) {
            _showTutorial.value = Event(Unit)
            tutorialShown = true
            preferences.edit().putBoolean(Constants.SHOWN_MEDICAL_TUT_FOR_SESSION, true).apply()
        }
    }

    fun dontShowTutorial() {
        preferences.edit().putBoolean(currentMemberId + "medicalLogTutorialShown", true).apply()
    }

    fun resetTutorial() {
        _tutorialsAdded.value = false
    }

    fun shouldShowTutorial(): Boolean {
        return !preferences.getBoolean(currentMemberId + "medicalLogTutorialShown", false)
    }
}

// UI State sealed class for better state management
sealed class MedicalLogsUiState {
    object Loading : MedicalLogsUiState()
    object LoadingMore : MedicalLogsUiState()
    object Empty : MedicalLogsUiState()
    data class Success(val data: MedicalLogsAll) : MedicalLogsUiState()
    data class Error(val message: String) : MedicalLogsUiState()
}
