package com.vivekahealth.vivekahealth.ui.guestVerify.repository

import com.vivekahealth.vivekahealth.data.remote.guest.AdditionalRequirementResponse
import com.vivekahealth.vivekahealth.data.remote.guest.DocumentUploadResponse
import com.vivekahealth.vivekahealth.data.remote.guest.IdentifyErrorResponse
import com.vivekahealth.vivekahealth.data.remote.guest.UploadDetailsSendParams
import com.vivekahealth.vivekahealth.data.remote.guest.UploadResponse
import com.vivekahealth.vivekahealth.ui.guestVerify.documents.DocumentType
import com.vivekahealth.vivekahealth.ui.guestVerify.service.GuestUploadService
import com.vivekahealth.vivekahealth.utils.thread.AppExecutors
import io.reactivex.Single
import okhttp3.MultipartBody
import retrofit2.Response
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class GuestUploadRepository @Inject constructor(
    private val appExecutors: AppExecutors,
    private val guestUploadService: GuestUploadService
) {

    fun scanBarCode(
        sessionId: String,
        filePart: MultipartBody.Part,
        documentType: DocumentType
    ): Single<Response<DocumentUploadResponse>> {
        val id = when (documentType) {
            DocumentType.DL_FRONT -> 103
            DocumentType.DL_BACK -> 104
            DocumentType.SELFIE -> 105
            DocumentType.SELFIE_MANUAL -> 106
            DocumentType.SSN -> 71
            DocumentType.INSURANCE_CARD -> 60
            else -> 0
        }

        return guestUploadService.getGuestHomeApi().scanBarcode(
            sessionId,
            filePart,
            id
        )
    }

    fun uploadDoc(
        sessionId: String,
        filePart: MultipartBody.Part,
        documentType: DocumentType
    ): Single<DocumentUploadResponse> {
        val id = when (documentType) {
            DocumentType.DL_FRONT -> 103
            DocumentType.DL_BACK -> 104
            DocumentType.SELFIE -> 105
            DocumentType.SELFIE_MANUAL -> 106
            DocumentType.SSN, DocumentType.SSN_ONLY -> 71
            DocumentType.INSURANCE_CARD -> 60
            else -> 0
        }

        return guestUploadService.getGuestHomeApi().uploadImage(
            sessionId,
            filePart,
            id
        )
    }

    fun updateDetails(
        sessionId: String,
        uploadDetails: UploadDetailsSendParams
    ): Single<Response<UploadResponse>> = guestUploadService.getGuestHomeApi().uploadDetails(
        sessionId,
        uploadDetails
    )

    fun verify(sessionId: String): Single<Response<UploadResponse>> =
        guestUploadService.getGuestHomeApi().verify(
            sessionId
        )

    fun identify(
        sessionId: String,
        reRegister: Boolean? = null
    ): Single<Response<IdentifyErrorResponse>> = guestUploadService.getGuestHomeApi().identify(
        sessionId,
        reRegister
    )

    fun additionalRequirements(sessionId: String): Single<Response<AdditionalRequirementResponse>> =
        guestUploadService.getGuestHomeApi().additionalRequirements(
            sessionId
        )

    fun processDocument(sessionId: String): Single<Response<UploadResponse>> =
        guestUploadService.getGuestHomeApi().processDocument(
            sessionId
        )

    fun validateRegisteredEmail(
        sessionId: String,
        email: String
    ): Single<Response<IdentifyErrorResponse>> =
        guestUploadService.getGuestHomeApi().validateRegisteredEmail(
            sessionId,
            email
        )

    fun logFailedDocumentAttempt(
        sessionId: String,
        documentType: Int
    ): Single<Response<IdentifyErrorResponse>> = guestUploadService.getGuestHomeApi().logFailedAttempt(
        sessionId,
        documentType
    )
}
