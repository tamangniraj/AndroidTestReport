package com.vivekahealth.vivekahealth.ui.homePage.guardianNurses.di
import android.content.Context
import android.content.SharedPreferences
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.ui.homePage.guardianNurses.GuardianNursesFragment
import com.vivekahealth.vivekahealth.ui.homePage.service.HomeService
import dagger.Module
import dagger.Provides
import dagger.android.ContributesAndroidInjector
import okhttp3.OkHttpClient

@Module
abstract class GuardianNursesModule {

    @ContributesAndroidInjector
    abstract fun contributesGuardianNursesFragment(): GuardianNursesFragment

    companion object {
        @JvmStatic
        @Provides
        fun provideGuardianNursesFragmentService(
            gson: Gson,
            builder: OkHttpClient.Builder,
            sharedPreferences: SharedPreferences,
            context: Context
        ): HomeService = HomeService(
            gson = gson,
            builder = builder,
            preferences = sharedPreferences,
            context = context
        )
    }
}
