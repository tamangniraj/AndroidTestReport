package com.vivekahealth.vivekahealth.ui.homePage.findcareprofileactivity

import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import com.vivekahealth.vivekahealth.ui.homePage.findCareSelectSpecialityActivity.BaseViewModel
import com.vivekahealth.vivekahealth.ui.homePage.findCareSelectSpecialityActivity.RetrofitClient
import com.vivekahealth.vivekahealth.ui.homePage.findCareSelectSpecialityActivity.SelectSpecialityVerifyNavigator
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityprofile.responce.ProcedureList
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityprofile.responce.Procedures
import java.util.ArrayList
import retrofit2.Call
import retrofit2.Callback
import retrofit2.Response

class FindAllProfileActivityViewModel : BaseViewModel<SelectSpecialityVerifyNavigator>() {
    private var mutableSignup: MutableLiveData<Procedures>? = null
    var ValueList1: ArrayList<ProcedureList> = ArrayList()
    var tempListFlat: ArrayList<Any> = arrayListOf()
    fun getProcedureList(npi: String): LiveData<Procedures> {
        mutableSignup = MutableLiveData()
        getProcederLists(npi)
        return mutableSignup!!
    }
    private fun getProcederLists(npi: String) {
        val call: Call<Procedures> = RetrofitClient.getApi()!!.getProcedersList(npi)
        call.enqueue(object : Callback<Procedures> {
            override fun onResponse(
                call: Call<Procedures>,
                response: Response<Procedures>
            ) {
                try {
                    mutableSignup?.value = response.body()
                } catch (e: Exception) {
                }
            }

            override fun onFailure(call: Call<Procedures>, t: Throwable) {
                getNavigator()?.onError("Email or Password Invalid")
                //  Log.d(TAG, "onFailure: ${t.localizedMessage}")
            }
        })
    }
}
