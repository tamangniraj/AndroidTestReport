package com.vivekahealth.vivekahealth.ui.register.step8.repository

import com.vivekahealth.vivekahealth.data.remote.register.RegisterUser
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.register.step8.service.RegisterService
import io.reactivex.Single
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Repository that handles User objects.
 */
@Singleton
class RegisterRepository @Inject constructor(
    private val registerService: RegisterService
) {

    fun reSendEmail(email: String): Single<DataResponse> = registerService.getRegisterApi().registerService(
        email,
        ""
    )

    fun registerUser(registerUser: RegisterUser): Single<DataResponse> = registerService.getRegisterApi().registerService(
        registerUser
    )

    fun verifyRegisterUser(registerUser: RegisterUser): Single<DataResponse> = registerService.getRegisterApi().registerVerifyService(
        registerUser
    )

    fun verifySsn(registerUser: RegisterUser): Single<DataResponse> = registerService.getRegisterApi().registerVerifySsnService(
        registerUser
    )
}
