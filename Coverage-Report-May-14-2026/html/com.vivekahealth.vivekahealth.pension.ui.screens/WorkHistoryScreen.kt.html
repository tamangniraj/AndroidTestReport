package com.vivekahealth.vivekahealth.pension.ui.screens

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun WorkHistoryScreen(
    onBackPressed: () -> Unit,
    viewModel: PensionViewModel,
    memberId: String,
    isOffline: Boolean = false
) {
    var selectedTabIndex by remember { mutableIntStateOf(0) }
    val workHistory by viewModel.workHistoryFromDb.collectAsStateWithLifecycle()
    val isLoading by viewModel.isLoadingWorkHistory.collectAsStateWithLifecycle()
    val error by viewModel.workHistoryError.collectAsStateWithLifecycle()
    var previousOfflineState by remember { mutableStateOf(isOffline) }

    // Log screen view
    LaunchedEffect(Unit) {
        viewModel.logEvent(FirebaseTags.PENSION_WORK_HISTORY_SCREEN_VIEW)
    }

    // Fetch work history on screen load
    LaunchedEffect(memberId) {
        viewModel.fetchWorkHistory(memberId)
    }

    // Fetch fresh data when coming back online
    LaunchedEffect(isOffline) {
        if (previousOfflineState && !isOffline && memberId.isNotEmpty()) {
            // Changed from offline to online - fetch fresh data
            viewModel.fetchWorkHistory(memberId)
        }
        previousOfflineState = isOffline
    }

    val plans = workHistory?.plans ?: emptyList()
    OfflineScaffold(
        isOffline = isOffline,
        topBar = {
            TopAppBar(
                title = {
                    Box(
                        modifier = Modifier.fillMaxWidth()
                    ) {
                        IconButton(
                            onClick = onBackPressed,
                            modifier = Modifier
                                .align(Alignment.CenterStart)
                                .tagged("Work History Back Button")
                        ) {
                            Icon(
                                painter = painterResource(R.drawable.outline_arrow_back_ios_24),
                                contentDescription = "Back",
                                tint = Color.Black
                            )
                        }
                        Text(
                            text = stringResource(R.string.work_history_details),
                            color = Color.Black,
                            fontSize = 18.sp,
                            fontWeight = FontWeight.Medium,
                            textAlign = TextAlign.Center,
                            modifier = Modifier
                                .align(Alignment.Center)
                                .tagged("Work History Title")
                        )
                    }
                },
                colors = TopAppBarDefaults.topAppBarColors(
                    containerColor = Color.White
                )
            )
        },
        containerColor = Color(0xFFF8F8F8)
    ) { paddingValues ->
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(paddingValues)
        ) {
            if (plans.isNotEmpty()) {
                if (plans.size <= 3) {
                    PrimaryTabRow(
                        selectedTabIndex = selectedTabIndex,
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(top = 1.dp),
                        containerColor = Color.White,
                        contentColor = Color.Black,
                        indicator = {},
                        divider = {
                            HorizontalDivider(
                                color = Color(0xFFE0E0E0),
                                thickness = 1.dp
                            )
                        }
                    ) {
                        plans.forEachIndexed { index, plan ->
                            Tab(
                                selected = selectedTabIndex == index,
                                onClick = {
                                    viewModel.logEvent(FirebaseTags.PENSION_WORK_HISTORY_TAB_CLICK)
                                    selectedTabIndex = index
                                },
                                modifier = Modifier
                                    .weight(1f)
                                    .padding(vertical = 8.dp)
                                    .tagged("Work History Tab ${plan.planName}")
                            ) {
                                Box(
                                    modifier = Modifier.fillMaxWidth(),
                                    contentAlignment = Alignment.Center
                                ) {
                                    Box(
                                        Modifier
                                            .background(
                                                if (selectedTabIndex == index) {
                                                    Color(0x102196F3)
                                                } else {
                                                    Color(
                                                        0x00666666
                                                    )
                                                },
                                                RoundedCornerShape(8.dp)
                                            )
                                            .padding(horizontal = 16.dp, vertical = 8.dp)
                                    ) {
                                        Text(
                                            text = plan.planRules.planShortName,
                                            fontSize = 14.sp,
                                            fontWeight = if (selectedTabIndex == index) FontWeight.SemiBold else FontWeight.Normal,
                                            color = if (selectedTabIndex == index) {
                                                colorResource(R.color.iconColor)
                                            } else {
                                                Color(
                                                    0xFF666666
                                                )
                                            },
                                            textAlign = TextAlign.Center
                                        )
                                    }
                                }
                            }
                        }
                    }
                } else {
                    PrimaryScrollableTabRow(
                        selectedTabIndex = selectedTabIndex,
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(top = 1.dp),
                        containerColor = Color.White,
                        contentColor = Color.Black,
                        indicator = {},
                        divider = {
                            HorizontalDivider(
                                color = Color(0xFFE0E0E0),
                                thickness = 1.dp
                            )
                        }
                    ) {
                        plans.forEachIndexed { index, plan ->
                            Tab(
                                selected = selectedTabIndex == index,
                                onClick = {
                                    viewModel.logEvent(FirebaseTags.PENSION_WORK_HISTORY_TAB_CLICK)
                                    selectedTabIndex = index
                                },
                                modifier = Modifier
                                    .padding(horizontal = 8.dp, vertical = 8.dp)
                                    .tagged("Work History Scrollable Tab ${plan.planName}")
                            ) {
                                Box(
                                    Modifier
                                        .background(
                                            if (selectedTabIndex == index) {
                                                Color(0x102196F3)
                                            } else {
                                                Color(
                                                    0x00666666
                                                )
                                            },
                                            RoundedCornerShape(8.dp)
                                        )
                                        .padding(horizontal = 16.dp, vertical = 8.dp)
                                ) {
                                    Text(
                                        text = plan.planName,
                                        fontSize = 14.sp,
                                        fontWeight = if (selectedTabIndex == index) FontWeight.SemiBold else FontWeight.Normal,
                                        color = if (selectedTabIndex == index) {
                                            colorResource(R.color.iconColor)
                                        } else {
                                            Color(
                                                0xFF666666
                                            )
                                        }
                                    )
                                }
                            }
                        }
                    }
                }

                WorkHistoryTabContent(
                    selectedPlan = plans.getOrNull(selectedTabIndex),
                    modifier = Modifier.fillMaxSize(),
                    isLoading = isLoading,
                    viewModel = viewModel
                )
            } else if (isLoading) {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = androidx.compose.ui.Alignment.Center
                ) {
                    CircularProgressIndicator()
                }
            } else if (error != null) {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = androidx.compose.ui.Alignment.Center
                ) {
                    Text(
                        text = error ?: "Unknown error",
                        color = colorResource(R.color.TextGrey),
                        textAlign = TextAlign.Center,
                        modifier = Modifier.padding(16.dp)
                    )
                }
            } else {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = androidx.compose.ui.Alignment.Center
                ) {
                    Text(
                        text = stringResource(R.string.no_data),
                        color = colorResource(R.color.TextGrey),
                        textAlign = TextAlign.Center,
                        modifier = Modifier.padding(16.dp)
                    )
                }
            }
        }
    }
}

@Composable
fun WorkHistoryTabContent(
    selectedPlan: com.vivekahealth.vivekahealth.pension.data.model.PlanDetail?,
    modifier: Modifier = Modifier,
    viewModel: PensionViewModel,
    isLoading: Boolean
) {
    val scrollState = rememberScrollState()

    Box(
        modifier = modifier
            .fillMaxSize()
            .padding(24.dp)
    ) {
        when {
            selectedPlan != null -> {
                Column(
                    modifier = Modifier
                        .fillMaxSize()
                        .padding(4.dp)
                        .verticalScroll(scrollState),
                    verticalArrangement = Arrangement.spacedBy(16.dp)
                ) {
                    // display the plan
                    selectedPlan.yearGroups.forEach {
                        PensionHistoryCard(it)
                    }

                    WhyCreditsAreLostExpandable(
                        scrollState = scrollState,
                        rules = selectedPlan.planRules,
                        viewModel = viewModel
                    )
                }
            }
            isLoading -> {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = androidx.compose.ui.Alignment.Center
                ) {
                    CircularProgressIndicator()
                }
            }
        }
    }
}
