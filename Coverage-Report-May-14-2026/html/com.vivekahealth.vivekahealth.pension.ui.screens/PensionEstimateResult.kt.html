@Composable
fun PensionEstimateContent(
    pensionData: PensionEstimateResponse,
    pensionType: String?,
    onBackClick: () -> Unit,
    onCloseClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    Column(
        modifier = modifier
            .fillMaxSize()
            .background(Color(0xFFF5F5F5))
    ) {
        // Header with navigation
        PensionEstimateHeader(
            onBackClick = onBackClick,
            onCloseClick = onCloseClick
        )

        // Scrollable content
        LazyColumn(
            modifier = Modifier.fillMaxSize(),
            verticalArrangement = Arrangement.spacedBy(16.dp)
        ) {
            item {
                // Pension Benefit Card
                PensionBenefitCard(
                    monthlyBenefit = pensionData.survivorBenefitCalculation?.memberMonthlyBenefit ?: 0.0,
                    spouseBenefit = pensionData.survivorBenefitCalculation?.survivorMonthlyBenefit ?: 0.0,
                    startingAge = pensionData.selectedOptions?.retirementAge ?: 0,
                    selectedOptionCode = pensionData.survivorBenefitCalculation?.selectedOptionCode
                        ?: ""
                )
            }

            item {
                // Calculation Breakdown
                CalculationBreakdownCard(pensionData = pensionData, pensionType)
            }

            item {
                // Add some bottom padding
                Spacer(modifier = Modifier.height(32.dp))
            }
        }
    }
}

@Composable
private fun PensionEstimateHeader(
    onBackClick: () -> Unit,
    onCloseClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    Surface(
        modifier = modifier
            .fillMaxWidth()
            .windowInsetsPadding(WindowInsets.statusBars)
            .tagged("Pension Estimate Result Header"),
        color = Color.White,
        shadowElevation = 2.dp
    ) {
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp)
        ) {
            IconButton(
                onClick = onBackClick,
                modifier = Modifier
                    .align(Alignment.CenterStart)
                    .tagged("Back Button")
            ) {
                Icon(
                    imageVector = Icons.AutoMirrored.Filled.ArrowBack,
                    contentDescription = "Back",
                    tint = Color.Black
                )
            }

            Text(
                text = stringResource(R.string.your_pension_estimate),
                style = MaterialTheme.typography.titleMedium,
                fontWeight = FontWeight.Medium,
                color = Color.Black,
                modifier = Modifier
                    .align(Alignment.Center)
                    .padding(horizontal = 56.dp)
                    .tagged("Pension Estimate Result Title")
            )

            IconButton(
                onClick = onCloseClick,
                modifier = Modifier
                    .align(Alignment.CenterEnd)
                    .tagged("Close Button Result")
            ) {
                Icon(
                    imageVector = Icons.Default.Close,
                    contentDescription = "Close",
                    tint = Color.Black
                )
            }
        }
    }
}

@Composable
private fun PensionBenefitCard(
    monthlyBenefit: Double,
    startingAge: Int?,
    spouseBenefit: Double,
    selectedOptionCode: String,
    modifier: Modifier = Modifier
) {
    Box(
        modifier = modifier
            .fillMaxWidth()
            .padding(top = 12.dp, bottom = 8.dp, start = 12.dp, end = 12.dp)
            .background(
                brush = Brush.verticalGradient(
                    colors = listOf(
                        colorResource(R.color.gradient_home_end),
                        colorResource(R.color.gradient_home_start)
                    )
                ),
                shape = RoundedCornerShape(12.dp)
            )
            .tagged("Pension Benefit Card")
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(24.dp),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Spacer(modifier = Modifier.height(16.dp))

            // Monthly Pension Benefit
            Text(
                text = stringResource(R.string.monthly_pension_benefit),
                style = MaterialTheme.typography.titleMedium,
                color = Color.White,
                fontWeight = FontWeight.Medium,
                modifier = Modifier.tagged("Monthly Pension Benefit Label")
            )

            Text(
                text = formatAmountValue(monthlyBenefit),
                style = MaterialTheme.typography.displaySmall,
                color = Color.White,
                fontWeight = FontWeight.Bold,
                modifier = Modifier
                    .padding(vertical = 8.dp)
                    .tagged("Monthly Pension Benefit Value")
            )

            Text(
                text = stringResource(R.string.starting_at_age, startingAge ?: 0),
                style = MaterialTheme.typography.bodyMedium,
                color = Color.White.copy(alpha = 0.9f)
            )

            Spacer(modifier = Modifier.height(24.dp))

            if (selectedOptionCode != "SINGLE_LIFE") {
                // Divider
                HorizontalDivider(
                    color = Color.White.copy(alpha = 0.3f),
                    thickness = 1.dp
                )

                Spacer(modifier = Modifier.height(24.dp))

                // Spouse Survivor Benefit
                Text(
                    text = stringResource(R.string.spouse_survivor_benefit),
                    style = MaterialTheme.typography.titleMedium,
                    color = Color.White,
                    fontWeight = FontWeight.Medium,
                    modifier = Modifier.tagged("Spouse Survivor Benefit Label")
                )

                Text(
                    text = formatAmountValue(spouseBenefit),
                    style = MaterialTheme.typography.displaySmall,
                    color = Color.White,
                    fontWeight = FontWeight.Bold,
                    modifier = Modifier
                        .padding(top = 8.dp)
                        .tagged("Spouse Survivor Benefit Value")
                )
            }
        }
    }
}

@SuppressLint("DefaultLocale")
@Composable
private fun CalculationBreakdownCard(
    pensionData: PensionEstimateResponse,
    pensionType: String?,
    modifier: Modifier = Modifier
) {
    Box(
        modifier
            .background(Color.White)
            .tagged("Calculation Breakdown Card")
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(20.dp)
        ) {
            Text(
                text = stringResource(R.string.calculation_breakdown),
                style = MaterialTheme.typography.titleMedium,
                fontWeight = FontWeight.SemiBold,
                color = Color.Black,
                modifier = Modifier.padding(bottom = 16.dp)
            )

            BreakdownRow(
                stringResource(R.string.current_credits),
                formatCreditValue(pensionData.benefitBreakdown?.actualCredits ?: 0.0),
                type = "Current"
            )
            BreakdownRow(
                stringResource(R.string.forfeited_credits),
                "-${
                formatCreditValue(
                    pensionData.forfeitureImpact?.totalCreditsForfeit ?: 0.0
                )
                }",
                Color(0xFFE53E3E),
                type = "Forfited"
            )
            BreakdownRow(
                stringResource(R.string.future_credits),
                "+${
                formatCreditValue(
                    pensionData.projectionSummary?.additionalCreditsExpected ?: 0.0
                )
                }",
                Color(0xFF38A169),
                type = "Future"
            )

            Spacer(modifier = Modifier.height(16.dp))
            Divider(color = Color(0xFFE0E0E0))
            Spacer(modifier = Modifier.height(16.dp))

            BreakdownRow(
                stringResource(R.string.total_projected_credits),
                formatCreditValue(
                    pensionData.retirementProjection?.totalCreditsAtRetirement ?: 0.0
                ),
                fontWeight = FontWeight.Bold,
                type = "Total"
            )

            BreakdownRow(
                stringResource(R.string.pension_type),
                pensionType ?: "",
                Color(0xFF38A169),
                type = "Type"
            )

            BreakdownRow(
                stringResource(R.string.survivor_option),
                pensionData.survivorBenefitCalculation?.selectedOptionName ?: "",
                type = "Option"
            )

            pensionData.forfeitureImpact?.impactDescription?.let { message ->
                Spacer(Modifier.height(16.dp))
                ImpactMessageCard(message = message)
            }

            Spacer(Modifier.height(16.dp))

            pensionData.notes?.let {
                if (it.isNotEmpty()) {
                    DisclaimerCard(pensionData.notes.joinToString("\n") ?: "")
                    Spacer(Modifier.height(32.dp))
                }
            }
        }
    }
}

@Composable
private fun BreakdownRow(
    label: String,
    value: String,
    valueColor: Color = Color.Black,
    fontWeight: FontWeight = FontWeight.Normal,
    modifier: Modifier = Modifier,
    type: String
) {
    Row(
        modifier = modifier
            .fillMaxWidth()
            .padding(vertical = 4.dp),
        horizontalArrangement = Arrangement.SpaceBetween,
        verticalAlignment = Alignment.Top
    ) {
        Text(
            text = label,
            style = MaterialTheme.typography.bodyMedium,
            color = Color.Gray,
            modifier = Modifier.weight(1f)
        )
        Text(
            text = value,
            style = MaterialTheme.typography.bodyMedium,
            fontWeight = fontWeight,
            color = valueColor,
            textAlign = TextAlign.End,
            modifier = Modifier.tagged("BreakDown Value $type")
        )
    }
}
