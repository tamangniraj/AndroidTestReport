package com.vivekahealth.vivekahealth.pension.ui.screens

import android.content.Intent
import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.res.painterResource
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.text.style.TextAlign
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import androidx.core.content.FileProvider
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import com.vivekahealth.vivekahealth.FirebaseTags
import com.vivekahealth.vivekahealth.pension.ui.component.PrimaryButton
import com.vivekahealth.vivekahealth.pension.ui.component.SecondaryButton
import com.vivekahealth.vivekahealth.pension.ui.component.SummaryImpactCard
import com.vivekahealth.vivekahealth.pension.ui.component.getTealTheme
import com.vivekahealth.vivekahealth.pension.ui.tagged
import com.vivekahealth.vivekahealth.pension.ui.viewmodel.PensionViewModel
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.ui.component.OfflineScaffold

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun PensionScreen(
    onBackPressed: () -> Unit,
    onGetPensionEstimate: () -> Unit,
    onViewDetailedHistory: () -> Unit,
    viewModel: PensionViewModel,
    isOffline: Boolean = false,
    context: android.content.Context,
    memberId: String
) {
    val summaryData by viewModel.summaryImpactDataFromDb.collectAsStateWithLifecycle()
    val isDownloadingApplication by viewModel.isDownloadingApplication.collectAsStateWithLifecycle()
    val downloadedPdfFile by viewModel.downloadedPdfFile.collectAsStateWithLifecycle()
    val downloadError by viewModel.downloadError.collectAsStateWithLifecycle()

    var showDownloadErrorDialog by remember { mutableStateOf(false) }
    var errorMessage by remember { mutableStateOf("") }

    // Log screen view
    LaunchedEffect(Unit) {
        viewModel.logEvent(FirebaseTags.PENSION_SCREEN_VIEW)
    }

    // Handle PDF viewing when download is successful
    LaunchedEffect(downloadedPdfFile) {
        downloadedPdfFile?.let { file ->
            try {
                val uri = FileProvider.getUriForFile(
                    context,
                    "${context.packageName}.provider",
                    file
                )
                val intent = Intent(Intent.ACTION_VIEW).apply {
                    setDataAndType(uri, "application/pdf")
                    flags = Intent.FLAG_GRANT_READ_URI_PERMISSION or Intent.FLAG_ACTIVITY_NEW_TASK
                }
                context.startActivity(intent)
                viewModel.clearDownloadedPdfFile()
            } catch (e: Exception) {
                // Handle error - no PDF viewer available
                timber.log.Timber.e(e, "Error opening PDF")
            }
        }
    }

    // Show error if download fails
    LaunchedEffect(downloadError) {
        downloadError?.let { error ->
            // Show error message
            timber.log.Timber.e("Download error: $error")
            errorMessage = error
            showDownloadErrorDialog = true
            viewModel.clearDownloadError()
        }
    }

    OfflineScaffold(
        isOffline = isOffline,
        topBar = {
            TopAppBar(
                title = {
                    Box(
                        modifier = Modifier.fillMaxWidth()
                    ) {
                        IconButton(
                            onClick = onBackPressed,
                            modifier = Modifier
                                .align(Alignment.CenterStart)
                                .tagged("Pension Details Back Button")
                        ) {
                            Icon(
                                painter = painterResource(R.drawable.outline_arrow_back_ios_24),
                                contentDescription = "Back",
                                tint = Color.Black
                            )
                        }
                        Text(
                            text = stringResource(R.string.pension_details),
                            color = Color.Black,
                            fontSize = 18.sp,
                            fontWeight = FontWeight.Medium,
                            textAlign = TextAlign.Center,
                            modifier = Modifier
                                .align(Alignment.Center)
                                .tagged("Pension Details Title")
                        )
                    }
                },
                colors = TopAppBarDefaults.topAppBarColors(
                    containerColor = Color.White
                )
            )
        },
        containerColor = Color(0xFFFFFFFF)
    ) { paddingValues ->
        LazyColumn(
            modifier = Modifier
                .fillMaxSize()
                .padding(paddingValues)
        ) {
            // Summary Card
            summaryData?.let { data ->
                item {
                    Row(
                        modifier = Modifier
                            .fillMaxWidth()
                            .background(Color(0xFFF8F8F8))
                            .padding(24.dp, 24.dp, 24.dp, 24.dp)
                    ) {
                        SummaryImpactCard(
                            data = data,
                            theme = getTealTheme(),
                            isOffline = isOffline,
                            modifier = Modifier.padding(0.dp)
                        )
                    }
                }
            }

            // Buttons
            item {
                Column(
                    modifier = Modifier
                        .padding(24.dp)
                        .background(Color.White),
                    verticalArrangement = Arrangement.spacedBy(16.dp)
                ) {
                    Spacer(modifier = Modifier.height(0.dp))

                    PrimaryButton(
                        text = stringResource(R.string.get_pension_estimate),
                        onClick = {
                            viewModel.logEvent(FirebaseTags.PENSION_GET_ESTIMATE_CLICK)
                            onGetPensionEstimate()
                        }
                    )

                    SecondaryButton(
                        text = stringResource(R.string.view_detailed_history),
                        onClick = {
                            viewModel.logEvent(FirebaseTags.PENSION_VIEW_HISTORY)
                            onViewDetailedHistory()
                        }
                    )

                    if (summaryData?.isEligibleForPensionBenefits == true) {
                        SecondaryButton(
                            text = stringResource(R.string.download_pension_application),
                            enabled = !isDownloadingApplication,
                            onClick = {
                                if (!isDownloadingApplication) {
                                    viewModel.logEvent(FirebaseTags.PENSION_DOWNLOAD_APPLICATION)
                                    viewModel.downloadPensionApplication(context, memberId)
                                }
                            },
                            isLoading = isDownloadingApplication
                        )
                    }
                }
            }
        }

        // Download Error Dialog
        if (showDownloadErrorDialog) {
            AlertDialog(
                onDismissRequest = { showDownloadErrorDialog = false },
                text = {
                    Text(text = errorMessage)
                },
                confirmButton = {
                    TextButton(
                        onClick = { showDownloadErrorDialog = false }
                    ) {
                        Text(stringResource(R.string.ok))
                    }
                },
                containerColor = Color.White,
                shape = RoundedCornerShape(4.dp)
            )
        }
    }
}
