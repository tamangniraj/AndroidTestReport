@Composable
fun PensionEstimateScreen(
    memberId: String,
    onClosePressed: () -> Unit,
    viewModel: PensionViewModel,
    isOffline: Boolean = false
) {
    val context = LocalContext.current
    val focusManager = LocalFocusManager.current
    val userData by viewModel.userData.collectAsStateWithLifecycle()
    val availableOptions by viewModel.survivorOptionsFromDb.collectAsStateWithLifecycle()

    val isLoadingEstimates by viewModel.isLoadingEstimate.collectAsStateWithLifecycle()

    var showOfflineErrorDialog by remember { mutableStateOf(false) }
    var showNotEligibleDialog by remember { mutableStateOf(false) }
    var showSpousalConsentDialog by remember { mutableStateOf(false) }
    var pendingSurvivorOption by remember {
        mutableStateOf<com.vivekahealth.vivekahealth.pension.data.model.SurvivorOptionConfig?>(
            null
        )
    }
    var previousOfflineState by remember { mutableStateOf(isOffline) }

    // Log screen view
    LaunchedEffect(Unit) {
        viewModel.logEvent(com.vivekahealth.vivekahealth.FirebaseTags.PENSION_ESTIMATE_SCREEN_VIEW)
    }

    // Set default values when first loaded
    LaunchedEffect(availableOptions) {
        if (userData.selectedSurvivorBenefit == null && availableOptions?.survivorOptions != null) {
            viewModel.updateSurvivorBenefit(
                availableOptions?.survivorOptions?.findLast { it.isDefault == true }
            )
        }

        if (userData.expectedHours == null && availableOptions?.expectedHours != null) {
            viewModel.updateExpectedHours(availableOptions?.expectedHours?.expectedHours.toString())
        }

        if (userData.selectedRetirementAge == null && availableOptions?.retirementOptions != null) {
            availableOptions?.retirementOptions?.findLast { it.isDefault == true }?.let {
                viewModel.updateRetirementAge(it)
            }
        }
    }

    val pensionEstimate by viewModel.pensionEstimate.collectAsStateWithLifecycle()

    val selectedRetirementAge = userData.selectedRetirementAge
    val expectedHours = userData.expectedHours
    val selectedBenefit = userData.selectedSurvivorBenefit

    // Validation logic
    val isValidExpectedHours = expectedHours?.isNotEmpty() == true &&
        expectedHours?.toDoubleOrNull() != null &&
        ((expectedHours?.toDoubleOrNull() ?: 0.0) >= 0.0)
    val isCalculateEnabled = isValidExpectedHours && !isLoadingEstimates
    val min = availableOptions?.expectedHours?.minHours
    val max = availableOptions?.expectedHours?.maxHours
    // Fetch pension estimate when retirement age, survivor option, or expected hours change
    LaunchedEffect(
        selectedRetirementAge,
        selectedBenefit?.optionCode,
        expectedHours
    ) {
        val maritalStatus = userData.pensionMemberResponse?.maritalStatus
        val survivorOptionCode = selectedBenefit?.optionCode
        val expectedHoursPerYear = expectedHours?.toDoubleOrNull()

        val hasError = expectedHours?.toDoubleOrNull()?.let { hours ->
            (min == null || hours >= min) && (max == null || hours <= max)
        } != true
        // Only call API if we have all required data
        if (!hasError) {
            viewModel.fetchPensionEstimate(
                memberId = memberId,
                retirementOption = selectedRetirementAge,
                survivorOptionCode = survivorOptionCode,
                expectedHoursPerYear = expectedHoursPerYear,
                maritalStatus = maritalStatus
            )
        } else {
            viewModel.clearRunning()
        }
    }

    LaunchedEffect(isOffline) {
        if (previousOfflineState && !isOffline && memberId.isNotEmpty()) {
            // Changed from offline to online - fetch fresh data
            val maritalStatus = userData.pensionMemberResponse?.maritalStatus
            val survivorOptionCode = selectedBenefit?.optionCode
            val expectedHoursPerYear = expectedHours?.toDoubleOrNull()

            // Only call API if we have all required data

            val hasError = expectedHours?.toDoubleOrNull()?.let { hours ->
                (min == null || hours >= min) && (max == null || hours <= max)
            } != true

            if (!hasError) {
                viewModel.fetchPensionEstimate(
                    memberId = memberId,
                    retirementOption = selectedRetirementAge,
                    survivorOptionCode = survivorOptionCode,
                    expectedHoursPerYear = expectedHoursPerYear,
                    maritalStatus = maritalStatus
                )
            }
        }
        previousOfflineState = isOffline
    }

    OfflineScaffold(
        isOffline = isOffline,
        containerColor = Color(0xFFF8F8F8),
        contentWindowInsets = WindowInsets(0, 0, 0, 0),
        topBar = {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .background(color = Color.White)
                    .windowInsetsPadding(WindowInsets.statusBars)
            ) {
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(12.dp)
                        .tagged("Pension Estimate Header")
                ) {
                    IconButton(
                        onClick = onClosePressed,
                        modifier = Modifier
                            .align(Alignment.CenterStart)
                            .tagged("Close Button")
                    ) {
                        Icon(
                            painter = painterResource(R.drawable.ic_baseline_close_24),
                            contentDescription = "Close",
                            tint = Color.Black,
                            modifier = Modifier.padding(8.dp)
                        )
                    }

                    Text(
                        text = stringResource(
                            R.string.pension_estimate
                        ),
                        color = Color.Black,
                        fontSize = 18.sp,
                        fontWeight = FontWeight.Medium,
                        textAlign = TextAlign.Center,
                        modifier = Modifier
                            .align(Alignment.Center)
                            .padding(horizontal = 56.dp)
                            .tagged("Pension Estimate Title")
                    )
                }
                HorizontalDivider(color = Color(0xFFf8f8f8), thickness = 2.dp)
            }
        }
    ) { paddingValues ->

        LazyColumn(
            modifier = Modifier
                .padding(paddingValues)
                .windowInsetsPadding(WindowInsets.ime)
                .fillMaxSize()
        ) {
            item {
                PersonalInformationInputCard(
                    selectedRetirementAge = selectedRetirementAge ?: availableOptions?.retirementOptions?.findLast { it.isDefault == true },
                    retirementOption = availableOptions?.retirementOptions ?: emptyList(),
                    expectedHours = expectedHours,
                    currentHours = formatCreditValue(
                        availableOptions?.expectedHours?.currentYearHours ?: 0.0
                    ),
                    editableExpectedHour = availableOptions?.expectedHours?.isEditable,
                    min = availableOptions?.expectedHours?.minHours,
                    max = availableOptions?.expectedHours?.maxHours,
                    onRetirementAgeChange = {
                        viewModel.updateRetirementAge(it)
                    },
                    onExpectedHoursChange = {
                        viewModel.updateExpectedHours(it)
                    },
                    viewModel = viewModel
                )
            }

            item {
                Spacer(modifier = Modifier.height(8.dp))
            }

            item {
                SurvivorBenefitsSelector(
                    selectedBenefit = selectedBenefit,
                    survivorOptions = availableOptions?.survivorOptions,
                    onBenefitSelected = { selectedOption ->
                        // Check if we need to show spousal consent dialog
                        if (selectedOption.requiresSpouseConsent == true) {
                            // Store pending selection and show dialog
                            pendingSurvivorOption = selectedOption
                            showSpousalConsentDialog = true
                        } else {
                            // No dialog needed, update immediately
                            viewModel.updateSurvivorBenefit(selectedOption)
                        }
                    },
                    viewModel = viewModel
                )
            }

            item {
                Spacer(modifier = Modifier.height(8.dp))
            }

            item {
                pensionEstimate?.projectionSummary?.let {
                    ProjectionSummaryCard(
                        yearsToRetirement = pensionEstimate?.projectionSummary?.yearsToRetirement,
                        additionalCreditsExpected = pensionEstimate?.projectionSummary?.additionalCreditsExpected,
                        totalProjectedCredits = pensionEstimate?.projectionSummary?.totalProjectedCredits,
                        infoMessage = pensionEstimate?.projectionSummary?.projectionSummaryNote,
                        errorMessage = pensionEstimate?.forfeitureImpact?.impactDescription ?: "",
                        isCalculateEnabled = isCalculateEnabled,
                        isLoading = isLoadingEstimates,
                        onCalculateClick = {
//                        // Get current language
                        },
                        viewModel = viewModel
                    )
                }
            }

            item {
                Spacer(modifier = Modifier.height(12.dp))

                PrimaryButton(
                    modifier = Modifier.padding(horizontal = 16.dp),
                    text = stringResource(R.string.calculate),
                    onClick = {
                        // Dismiss keyboard first
                        focusManager.clearFocus()

                        viewModel?.logEvent(FirebaseTags.PENSION_CALCULATE_CLICK)

                        // Check eligibility first
                        if (selectedRetirementAge?.isEligible == false) {
                            showNotEligibleDialog = true
                        } else {
                            pensionEstimate?.let {
                                val currentLanguage = LanguageUtils.getCurrentLanguageCode(context)
                                // Create intent and launch activity

                                val intent = PensionResultComposeActivity.createIntent(
                                    context = context,
                                    pensionData = it,
                                    pensionType = formatPensionType(
                                        selectedRetirementAge?.pensionType
                                    ),
                                    languageCode = currentLanguage
                                )
                                context.startActivity(intent)
                            } ?: run {
                                if (isOffline) {
                                    showOfflineErrorDialog = true
                                }
                            }
                        }
                    },
                    enabled = isCalculateEnabled,
                    isLoading = isLoadingEstimates
                )
            }

            item {
                Spacer(modifier = Modifier.height(48.dp))
            }
        }

        // Offline Error Dialog
        if (showOfflineErrorDialog) {
            AlertDialog(
                onDismissRequest = { showOfflineErrorDialog = false },
                text = {
                    Text(text = stringResource(R.string.no_internet))
                },
                confirmButton = {
                    TextButton(
                        onClick = { showOfflineErrorDialog = false }
                    ) {
                        Text(stringResource(R.string.ok))
                    }
                },
                containerColor = Color.White,
                shape = RoundedCornerShape(4.dp)

            )
        }

        // Not Eligible Dialog
        if (showNotEligibleDialog) {
            NotEligibleDialog(
                selectedRetirementAge = selectedRetirementAge,
                eligibleOption = availableOptions?.retirementOptions?.find { it.isEligible == true },
                onDismiss = { showNotEligibleDialog = false }
            )
        }

        // Spousal Consent Dialog
        if (showSpousalConsentDialog) {
            SpousalConsentDialog(
                pendingSurvivorOption,
                onDismiss = {
                    showSpousalConsentDialog = false
                    // Don't update viewModel - keep previous selection
                    pendingSurvivorOption = null
                },
                onContinue = {
                    showSpousalConsentDialog = false
                    // Update viewModel with pending selection
                    pendingSurvivorOption?.let { viewModel.updateSurvivorBenefit(it) }
                    pendingSurvivorOption = null
                }
            )
        }
    }
}

fun formatPensionType(pensionType: String?): String {
    return pensionType
        ?.replace("_", " ")
        ?.lowercase()
        ?.split(" ")
        ?.joinToString(" ") { it.replaceFirstChar { char -> char.uppercase() } }
        ?: ""
}

private fun isUserMarried(
    maritalStatuses: List<com.vivekahealth.vivekahealth.pension.data.model.MaritalStatus>?
): Boolean {
    return maritalStatuses
        ?.firstOrNull { it.isDefault == true && it.statusCode == "MARRIED" } != null
}

@Composable
private fun RequirementsList(requirementsText: String) {
    val requirements = remember(requirementsText) {
        requirementsText
            .split(Regex("[\\n,]"))
            .map { it.trim().removePrefix("•").removePrefix("-").trim() }
            .filter { it.isNotBlank() }
    }

    Column(modifier = Modifier.fillMaxWidth()) {
        requirements.forEach { requirement ->
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(vertical = 4.dp),
                verticalAlignment = Alignment.Top
            ) {
                Text(
                    text = "• ",
                    fontSize = 14.sp,
                    color = Color.Black
                )
                Text(
                    text = requirement,
                    fontSize = 14.sp,
                    color = Color.Black,
                    modifier = Modifier.weight(1f)
                )
            }
        }
    }
}

@Composable
private fun NotEligibleDialog(
    selectedRetirementAge: com.vivekahealth.vivekahealth.pension.data.model.RetirementOption?,
    eligibleOption: com.vivekahealth.vivekahealth.pension.data.model.RetirementOption ?,
    onDismiss: () -> Unit
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        confirmButton = {},
        text = {
            Column(
                modifier = Modifier.fillMaxWidth(),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                // Warning Icon
                Icon(
                    painter = painterResource(R.drawable.alert),
                    contentDescription = "Warning",
                    modifier = Modifier.size(80.dp),
                    tint = Color.Unspecified
                )

                Spacer(modifier = Modifier.height(16.dp))

                // Title
                Text(
                    text = stringResource(
                        R.string.not_eligible_title
                    ),
                    fontSize = 20.sp,
                    fontWeight = FontWeight.Bold,
                    color = Color.Black,
                    textAlign = TextAlign.Center
                )

                Spacer(modifier = Modifier.height(8.dp))

                val formatted = formatPensionType(selectedRetirementAge?.pensionType ?: "")
                // Description
                Text(
                    text = stringResource(R.string.not_eligible_description, formatted),
                    fontSize = 14.sp,
                    color = colorResource(R.color.TextGrey),
                    textAlign = TextAlign.Center
                )

                Spacer(modifier = Modifier.height(24.dp))

                // Requirements Section
                Column(
                    modifier = Modifier.fillMaxWidth()
                        .background(
                            colorResource(R.color.background_reg),
                            RoundedCornerShape(4.dp)
                        ).padding(8.dp),
                    horizontalAlignment = Alignment.Start
                ) {
                    Text(
                        text = stringResource(R.string.required_section_title),
                        fontSize = 14.sp,
                        fontWeight = FontWeight.SemiBold,
                        color = Color.Black
                    )

                    Spacer(modifier = Modifier.height(8.dp))

                    selectedRetirementAge?.requirements?.let { requirements ->
                        RequirementsList(requirements)
                    }
                }

                Spacer(modifier = Modifier.height(24.dp))

                // Deferred Pension Option Section
                eligibleOption?.pensionType?.let { pensionType ->
                    Column(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalAlignment = Alignment.Start
                    ) {
                        Text(
                            text = stringResource(R.string.you_may_qualify_for),
                            fontSize = 14.sp,
                            color = colorResource(R.color.TextGrey)
                        )

                        Spacer(modifier = Modifier.height(4.dp))

                        val formattedType = formatPensionType(pensionType)
                        Text(
                            text = formattedType,
                            fontSize = 16.sp,
                            fontWeight = FontWeight.SemiBold,
                            color = Color.Black
                        )
                    }
                }

                Spacer(modifier = Modifier.height(24.dp))

                // OK Button
                PrimaryButton(
                    text = stringResource(R.string.ok),
                    onClick = onDismiss,
                    modifier = Modifier.fillMaxWidth()
                )
            }
        },
        containerColor = Color.White,
        shape = RoundedCornerShape(16.dp),
        properties = DialogProperties(usePlatformDefaultWidth = false),
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 16.dp).tagged("Not Eligible Dialog")
    )
}

@Composable
private fun SpousalConsentDialog(
    survivorOptionConfig: SurvivorOptionConfig?,
    onDismiss: () -> Unit,
    onContinue: () -> Unit
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        confirmButton = {},
        text = {
            Column(
                modifier = Modifier.fillMaxWidth(),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                // Alert Icon (same as NotEligibleDialog)
                Icon(
                    painter = painterResource(R.drawable.alert),
                    contentDescription = "Alert",
                    modifier = Modifier.size(80.dp),
                    tint = Color.Unspecified
                )

                Spacer(modifier = Modifier.height(16.dp))

                // Title
                Text(
                    text = survivorOptionConfig?.spouseConsentInfo?.heading ?: stringResource(
                        R.string.spousal_consent_required
                    ),
                    fontSize = 20.sp,
                    fontWeight = FontWeight.Bold,
                    color = Color.Black,
                    textAlign = TextAlign.Center
                )

                Spacer(modifier = Modifier.height(8.dp))

                HorizontalDivider()

                Spacer(modifier = Modifier.height(8.dp))

                // Required Documents Section
                Column(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(8.dp),
                    horizontalAlignment = Alignment.Start
                ) {
                    Text(
                        text = stringResource(R.string.required_documents),
                        fontSize = 14.sp,
                        fontWeight = FontWeight.SemiBold,
                        color = Color.Black
                    )

                    Spacer(modifier = Modifier.height(8.dp))

                    survivorOptionConfig?.spouseConsentInfo?.requiredDocuments?.forEach {
                        Row(
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(vertical = 4.dp),
                            verticalAlignment = Alignment.CenterVertically
                        ) {
                            Icon(
                                painter = painterResource(R.drawable.ic_checked),
                                contentDescription = "Check",
                                modifier = Modifier.size(20.dp),
                                tint = colorResource(R.color.primaryLight)
                            )
                            Spacer(modifier = Modifier.width(8.dp))
                            Text(
                                text = it.name ?: "",
                                fontSize = 14.sp,
                                color = Color.Black
                            )
                        }
                    }
                }

                Spacer(modifier = Modifier.height(16.dp))

                // Important Note Box
                Column(
                    Modifier
                        .fillMaxWidth()
                        .background(
                            colorResource(R.color.primaryUltraLight),
                            shape = RoundedCornerShape(8.dp)
                        )
                        .border(
                            1.dp,
                            colorResource(R.color.primaryLight),
                            shape = RoundedCornerShape(8.dp)
                        )
                        .padding(12.dp)
                ) {
                    Text(
                        text = buildAnnotatedString {
                            withStyle(style = SpanStyle(fontWeight = FontWeight.Bold)) {
                                append(stringResource(R.string.important))
                                append(" ")
                            }
                            append(survivorOptionConfig?.spouseConsentInfo?.note ?: "")
                        },
                        fontSize = 13.sp,
                        color = Color.Black, // Darker blue text
                        lineHeight = 18.sp
                    )
                }

                Spacer(modifier = Modifier.height(24.dp))

                // Buttons Row
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.spacedBy(12.dp)
                ) {
                    // Cancel Button
                    TextButton(
                        onClick = onDismiss,
                        modifier = Modifier.weight(1f)
                    ) {
                        Text(
                            text = stringResource(R.string.cancel),
                            fontSize = 16.sp,
                            fontWeight = FontWeight.Medium,
                            color = Color(0xFF16bdfc)
                        )
                    }

                    // Continue Button
                    PrimaryButton(
                        text = stringResource(R.string.continue_button),
                        onClick = onContinue,
                        padding = 8.dp,
                        modifier = Modifier.weight(1f)
                    )
                }
            }
        },
        containerColor = Color.White,
        shape = RoundedCornerShape(16.dp),
        properties = DialogProperties(usePlatformDefaultWidth = false),
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 16.dp).tagged("Spousal Consent Dialog")
    )
}
