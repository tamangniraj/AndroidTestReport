    fun documentList(selectedRel: String) {
        if (docList.any { it.uploaded }) {
            docList = docList.map {
                it.copy(optional = ssnOptional)
            }.toMutableList()
            return
        }
        docList.clear()

        if (selectedRel.equals("Spouse", true)) {
            if (isSsnRequired) {
                docList.add(
                    DocumentAddDependent(false, DocumentType.Ssn, "", "")
                )
            }
            docList.add(
                DocumentAddDependent(
                    false,
                    DocumentType.MarriageCertificate,
                    "",
                    ""
                )
            )
        } else {
            if (isSsnRequired) {
                docList.add(
                    DocumentAddDependent(false, DocumentType.Ssn, "", "", optional = ssnOptional)
                )
            }
            docList.add(
                DocumentAddDependent(
                    false,
                    DocumentType.BirtCertificate,
                    "",
                    "",
                    optional = ssnOptional
                )
            )
        }
    }

    fun setDocumentName(id: String, fileName: String) {
        when {
            StringUtils.containsIgnoreCase(
                fileName,
                DocumentType.Ssn.FileName.sanitize()
            ) -> {
                ssnCardFileId = id
            }
            StringUtils.containsIgnoreCase(
                fileName,
                DocumentType.BirtCertificate.FileName.sanitize()
            ) -> {
                birthCertificateFileId = id
            }
            StringUtils.containsIgnoreCase(
                fileName,
                DocumentType.MarriageCertificate.FileName.sanitize()
            ) -> {
                marriageCertificateFileId = id
            }
        }
        for (item in docList) {
            if (fileName.contains(
                    docList[docList.indexOf(item)].docName!!.FileName.sanitize(),
                    true
                )
            ) {
                docList[docList.indexOf(item)].uploaded = true
            } else if (fileName.contains(
                    docList[docList.indexOf(item)].docName!!.FileName.sanitize(),
                    true
                )
            ) {
                docList[docList.indexOf(item)].uploaded = true
            }
        }
    }

    fun addDependent() {
        var relation = ""
        if (isDemographicUpdate) {
            relation = memberInfo.memberRelationship ?: ""
        } else {
            relation = dependent.relationship ?: ""
        }
        compositeDisposable.add(
            homeRepository.postAddDependent(
                preferences.getString("memberID", "").toString(),
                AddDependent(firstNameChild.trim(), lastNameChild.trim(), dateOfBirthChild, genderChild, ssnRaw, birthCertificateFileId, ssnCardFileId, nearMatchedSelfId, relation, dateOfMarriage = marriageDate, marriageCertificateFileId = marriageCertificateFileId, healthInsuranceAvailable = spouseInsured, provideInsurance = provideInsuranceSpouse, selfEmployed = childEmployed, employerName = employerName),
                actionType,
                isDemographicUpdate
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _addDependent.value = APIResponse<ResponseAddDependent>().loading()
                }
                .subscribe({
                    _addDependent.value = APIResponse<ResponseAddDependent>().success(it)
                }, {
                    _addDependent.value = APIResponse<ResponseAddDependent>().error(
                        it.localizedMessage
                    )
                })
        )
    }

    fun nextState(): Int {
        val firstViewPair = viewSequence.first {
            it.second
        }
        val lastPair = viewSequence.last() {
            it.second
        }
        val listOfView = viewSequence.filter { it.second }
        if (listOfView.count() == 1) {
            return -1
        } else {
            listOfView.forEachIndexed { index, item ->
                if (item.first == currentUpdateState.first) {
                    if (lastPair.first == currentUpdateState.first) {
                        return -1
                    } else {
                        currentUpdateState = listOfView[index + 1]
                        return listOfView[index + 1].first
                    }
                }
            }
        }
        return -2
    }
    fun prevState(): Int {
        val firstViewPair = viewSequence.first {
            it.second
        }
        val lastPair = viewSequence.last() {
            it.second
        }
        val listOfView = viewSequence.filter { it.second }
        if (listOfView.count() == 1) {
            return -1
        } else {
            listOfView.forEachIndexed { index, item ->
                if (item.first == currentUpdateState.first) {
                    if (firstViewPair.first == currentUpdateState.first) {
                        return -1
                    } else {
                        currentUpdateState = listOfView[index - 1]
                        return listOfView[index - 1].first
                    }
                }
            }
        }
        return -2
    }
