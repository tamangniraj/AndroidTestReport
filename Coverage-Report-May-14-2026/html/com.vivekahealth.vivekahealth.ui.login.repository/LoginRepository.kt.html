package com.vivekahealth.vivekahealth.ui.login.repository

import com.vivekahealth.vivekahealth.BuildConfig
import com.vivekahealth.vivekahealth.data.remote.login.AuthToken
import com.vivekahealth.vivekahealth.data.remote.login.ChangePasswordDTO
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.login.service.LoginService
import io.reactivex.Single
import javax.inject.Inject
import javax.inject.Singleton
import retrofit2.Response

/**
 * Repository that handles User objects.
 */
@Singleton
class LoginRepository @Inject constructor(
    private val loginService: LoginService
) {

    fun getPassword(userName: String): Single<DataResponse> =
        loginService.getLoginApi().forgotPasswordV2(userName)

    fun changePassword(changePasswordDTO: ChangePasswordDTO): Single<DataResponse> = loginService.getLoginApi().changePassword(
        changePasswordDTO
    )

    fun getLogin(userName: String, password: String, tempUUID: String, isBiometricLogin: Boolean): Single<Response<AuthToken>> = loginService.getLoginApi().getLogin(
        BuildConfig.LOGIN_CLIENT_ID,
        BuildConfig.LOGIN_CLIENT_SECRET,
        "password",
        userName,
        password,
        tempUUID,
        isBiometricLogin
    )

    fun getMagicLogin(tempUUID: String): Single<Response<AuthToken>> = loginService.getLoginApi().getMagicLogin(
        BuildConfig.MAGIC_LOGIN_CLIENT_ID,
        BuildConfig.MAGIC_LOGIN_CLIENT_SECRET,
        "magic_link",
        tempUUID
    )

    fun getGuestToken(): Single<Response<AuthToken>> = loginService.getLoginApi().getGuestToken(
        BuildConfig.GUEST_LOGIN_CLIENT_ID,
        BuildConfig.GUEST_LOGIN_CLIENT_SECRET,
        "client_credentials"
    )

    fun getCompleteRegistrationToken(
        username: String,
        password: String
    ): Single<Response<AuthToken>> = loginService.getLoginApi().getLogin(
        BuildConfig.LOGIN_CLIENT_ID,
        BuildConfig.LOGIN_CLIENT_SECRET,
        "complete_registration_grant_type",
        username,
        password,
        "",
        false
    )

    fun getOpenRegistrationToken(
        username: String,
        password: String
    ): Single<Response<AuthToken>> = loginService.getLoginApi().getLogin(
        BuildConfig.LOGIN_CLIENT_ID,
        BuildConfig.LOGIN_CLIENT_SECRET,
        "password",
        username,
        password,
        "",
        false
    )
}
