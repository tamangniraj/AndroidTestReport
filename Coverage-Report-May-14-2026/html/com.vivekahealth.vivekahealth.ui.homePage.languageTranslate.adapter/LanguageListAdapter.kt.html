package com.vivekahealth.vivekahealth.ui.homePage.languageTranslate.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.LanguageList
import com.vivekahealth.vivekahealth.databinding.LanguageListBinding
import java.util.*

class LanguageListAdapter(private var onClick: (LanguageList) -> Unit) : ListAdapter<LanguageList, RecyclerView.ViewHolder>(
    LanguageListDiffCallback()
) {

    var selectedPosition: Int = -1
    var lastSelected: Int = -1
    var count: Int = 0
    var currentYearPos: Int = 0
    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return TypeViewHolder(
            LanguageListBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick,
            selectedPosition
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as TypeViewHolder).binding.setClickListener {
            onClick(item)
        }
        holder.binding.language = item
    }

    class TypeViewHolder(
        val binding: LanguageListBinding,
        private val onClick: (LanguageList) -> Unit,
        private var selectedPosition: Int
    ) : RecyclerView.ViewHolder(binding.root)
}

private class LanguageListDiffCallback : DiffUtil.ItemCallback<LanguageList>() {

    override fun areItemsTheSame(oldItem: LanguageList, newItem: LanguageList): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: LanguageList, newItem: LanguageList): Boolean {
        return oldItem.name == newItem.name
    }
}
