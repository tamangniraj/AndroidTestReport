    fun sortList(procedureList: ArrayList<Procedure>): ArrayList<Procedure> {
        if (sortby.equals("cost", true)) {
            if (sortAction.equals("asc", true)) {
                val sortedList = procedureList.sortedWith(compareBy { it.minPrice })
                procedureList.clear()
                procedureList.addAll(sortedList)
            } else if (sortAction.equals("desc", true)) {
                val sortedList = procedureList.sortedByDescending { it.maxPrice }
                procedureList.clear()
                procedureList.addAll(sortedList)
            }
        } else if (sortby.equals("code", true)) {
            if (sortAction.equals("asc", true)) {
                val sortedList = procedureList.sortedBy { it.code }
                procedureList.clear()
                procedureList.addAll(sortedList)
            } else if (sortAction.equals("desc", true)) {
                val sortedList = procedureList.sortedByDescending { it.code }
                procedureList.clear()
                procedureList.addAll(sortedList)
            }
        } else if (sortby.equals("relevance", true)) {
            val sortedList = procedureList.sortedBy { it.index }
            procedureList.clear()
            procedureList.addAll(sortedList)
        }

        return procedureList
    }

    private fun selectSpecialityAll(post: String) {
        val call: Call<SpecialityAll> = RetrofitClient.getApi()!!.getAllAllSpeciality(post)
        call.enqueue(object : Callback<SpecialityAll> {
            override fun onResponse(
                call: Call<SpecialityAll>,
                response: Response<SpecialityAll>
            ) {
                try {
                    mutableSignup?.value = response.body()
                } catch (e: Exception) {
                }
            }
            override fun onFailure(call: Call<SpecialityAll>, t: Throwable) { }
        })
    }

    fun getProcedureList(search: String, provider_type: String, within: String) {
        var radiu: String
        if (within.equals("No limits", ignoreCase = true)) {
            radiu = ""
        } else {
            radiu = within
        }
        loading = true
        compositeDisposable.add(
            findCareSearchRepository.getProcedureList(
                search,
                memberID,
                preferences.getString(
                    Constants.LATITUDE,
                    ""
                ).toString(),
                preferences.getString(Constants.LONGITUDE, "").toString(),
                provider_type,
                radiu
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->
                    eligibleValueList = ValueList
                    listForDoctorAdapter = eligibleValueList.toList()
                    countDoc = it.doctors.size
                    countProcedure = it.procedures.size
                    countSpeciality = it.specialties.size

                    showMapBtn = it.doctors.isNotEmpty()

                    _logsListGenerated.value = Event(it)
                }, {
                    loading = false
                })
        )
    }

    // heavly used
    fun getSpecialityList() {
        compositeDisposable.clear()

        val displayPrice = false

        compositeDisposable.add(
            findCareSearchRepository.getProcedureList(
                filterOption.searchText.toString(),
                memberID,
                filterOption.current?.latitude.toString(),
                filterOption.current?.longitude.toString(),
                filterOption.providerType.toString(),
                filterOption.radius.toString(),
                displayPrice = displayPrice
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->

                    // if display price is
                    if (!displayPrice) {
                        it.procedures.forEachIndexed { index, procedure ->
                            procedure.priceLoading = true
                            procedure.index = index
                        }
                    }

                    loading = false

                    countDoc = it.doctors.size
                    countProcedure = it.procedures.size
                    countSpeciality = it.specialties.size

                    if (firstOpen) {
                        firstOpen = false
                        var larger = largestList()

                        if (larger == 0) {
                            _statusLiveData.value = Event("Speciality")
                        } else if (larger == 1) {
                            _statusLiveData.value = Event("Procedure")
                        } else {
                            _statusLiveData.value = Event("Doctor")
                        }
                    }

                    _logsListGenerated.value = Event(it)

                    if (!displayPrice) {
                        getPriceFromBillingCode(it)
                    }

                    // only call if display price is false
                }, {
                    loading = false
                })
        )
    }

    fun getPriceFromBillingCode(specialties: Specialties) {
        var codes = specialties.procedures.map {
            it.code
        }

        compositeDisposable.add(
            findCareSearchRepository.getPriceFromBillingList(
                codes,
                filterOption.current?.latitude.toString(),
                filterOption.current?.longitude.toString(),
                filterOption.radius.toString()
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                }.subscribe({
                    val scheduleMap = it.billingCodeFeeScheduleDtoList.associateBy { dto -> dto.billingCode }

                    // Update each procedure
                    specialties.procedures.forEach { procedure ->
                        scheduleMap[procedure.code]?.let { dto ->
                            procedure.price = dto.amount
                            procedure.priceSource = dto.priceSource
                            procedure.minPrice = dto.minAmount
                            procedure.maxPrice = dto.maxAmount
                            procedure.occurrence = dto.occurrence?.toLong() ?: 0L

                            procedure.priceLoading = false
                        } ?: kotlin.run {
                            procedure.priceLoading = false
                        }
                        // Re-post the updated value wrapped in Event again
                    }

                    val sorted = sortList(specialties.procedures)
                    specialties.procedures = sorted

                    _logsListGenerated.value = Event(specialties)
                }, {
                    // on error handle the result to hide loading
                    specialties.procedures.forEach { procedure ->
                        procedure.priceLoading = false
                    }

                    val sorted = sortList(specialties.procedures)
                    specialties.procedures = sorted

                    _logsListGenerated.value = Event(specialties)
                })
        )
    }
