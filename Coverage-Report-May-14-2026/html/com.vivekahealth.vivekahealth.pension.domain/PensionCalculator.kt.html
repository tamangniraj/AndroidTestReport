package com.vivekahealth.vivekahealth.pension.domain

import com.vivekahealth.vivekahealth.pension.data.model.PensionMemberResponse
import com.vivekahealth.vivekahealth.pension.data.model.SummaryImpactData
import com.vivekahealth.vivekahealth.utils.DateUtils
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class PensionCalculator @Inject constructor() {

    companion object {
        const val EXPECTED_HOURS_BASE = 1500.0
    }

    /**
     * Calculates summary impact data from pension data
     */
    fun mapSummaryImpactData(pensionMemberResponse: PensionMemberResponse?): SummaryImpactData {
        var totalCredit = 0.0
        var creditKept = 0.0
        var creditsLost = 0.0

        pensionMemberResponse?.let {
            totalCredit = it.totalCredits
            creditsLost = it.totalForfeited
            creditKept = it.totalCredits - it.totalForfeited
        }

        // hard coded value for credits kept and credit lost,
        // no proper logic found for this
        return SummaryImpactData(
            totalCreditsEarned = totalCredit,
            creditsKept = creditKept,
            creditsLost = creditsLost,
            date = DateUtils.getSystemDateTimeAsString() ?: "",
            isEligibleForPensionBenefits = pensionMemberResponse?.isEligibleForPensionBenefits,
            showPensionDetails = pensionMemberResponse?.showPensionDetails
        )
    }
}
