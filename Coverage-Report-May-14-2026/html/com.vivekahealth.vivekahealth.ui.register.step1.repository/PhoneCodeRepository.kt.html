package com.vivekahealth.vivekahealth.ui.register.step1.repository

import com.vivekahealth.vivekahealth.data.remote.guest.RequestOtpSendParams
import com.vivekahealth.vivekahealth.data.remote.guest.ResponseOtpReceiveParams
import com.vivekahealth.vivekahealth.data.remote.guest.Source
import com.vivekahealth.vivekahealth.data.remote.guest.VerifyOtpReceiveParams
import com.vivekahealth.vivekahealth.data.remote.guest.VerifyOtpSendParams
import com.vivekahealth.vivekahealth.data.remote.register.TempCode
import com.vivekahealth.vivekahealth.ui.register.step1.service.PhoneService
import io.reactivex.Single
import retrofit2.Response
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Repository that handles User objects.
 */
@Singleton
class PhoneCodeRepository @Inject constructor(
    private val phoneService: PhoneService
) {

    fun getCode(phone: String, countryCode: String, channel: String): Single<TempCode> = phoneService.getCodeApi().getCode(
        phone,
        countryCode,
        true,
        channel
    )

    fun requestOTPV2(
        to: String,
        countryCode: String?,
        channel: String,
        captchaToken: String?,
        registrationId: String,
        isCheckConsent: Boolean?,
        source: Source?
    ): Single<Response<ResponseOtpReceiveParams>> {
        return phoneService.getCodeApi().requestOTP(
            RequestOtpSendParams(
                to = to,
                countryCode = countryCode,
                channel = if (channel == "PHONE") 1 else if (channel == "VOICE") 2 else 3,
                registrationId = registrationId,
                captchaToken = captchaToken,
                smsConsent = isCheckConsent,
                source = source?.value
            )
        )
    }

    fun verifyCode(phone: String, tempUuid: String, Code: String, expirationTime: String): Single<TempCode> = phoneService.getCodeApi().verifyCode(
        phone,
        tempUuid,
        Code,
        expirationTime
    )

    fun verifyCodeV2(
        id: String,
        registrationId: String?,
        attemptId: String,
        code: String,
        channel: String
    ): Single<Response<VerifyOtpReceiveParams>> {
        return phoneService.getCodeApi().verifyOTP(
            VerifyOtpSendParams(
                registrationId = registrationId,
                attemptId = attemptId,
                to = id,
                otpCode = code,
                channel = if (channel == "PHONE") 1 else if (channel == "VOICE") 2 else 3
            )
        )
    }
}
