package com.vivekahealth.vivekahealth.ui.register.step3

import android.content.SharedPreferences
import android.text.Editable
import android.text.TextWatcher
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.login.AuthToken
import com.vivekahealth.vivekahealth.data.remote.register.RegisterUser
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.login.repository.LoginRepository
import com.vivekahealth.vivekahealth.ui.register.step3.repository.EmailRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Coroutines
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.remote.APILoginResponse
import com.vivekahealth.vivekahealth.utils.remote.APIResponse
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import javax.inject.Inject
import kotlinx.coroutines.Job
import org.apache.commons.lang3.StringUtils
import retrofit2.Response

class StepEmailViewModel@Inject constructor(
    private val preferences: SharedPreferences,
    private val emailRepository: EmailRepository,
    private val loginRepository: LoginRepository
) : ViewModel(), Observable {

    var gson = Gson()
    var user = RegisterUser()
    private lateinit var job: Job
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _logInSuccess = MutableLiveData<APILoginResponse<Response<AuthToken>>>()
    val logInSuccess: LiveData<APILoginResponse<Response<AuthToken>>>
        get() = _logInSuccess
    private val _logInFail = MutableLiveData<Event<String>>()
    val logInFail: LiveData<Event<String>>
        get() = _logInFail
    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack
    private val _navigateToPassword = MutableLiveData<APIResponse<DataResponse>>()
    val navigateToPassword: LiveData<APIResponse<DataResponse>>
        get() = _navigateToPassword

    private val _emailChanged = MutableLiveData<Event<String>>()

    val emailChanged: LiveData<Event<String>>
        get() = _emailChanged

    val showError: LiveData<APIResponse<DataResponse>>
        get() = _showError
    private val _showError = MutableLiveData<APIResponse<DataResponse>>()
    private val _phoneEmailPrev = MutableLiveData<Event<String>>()

    val phoneEmailPrev: LiveData<Event<String>>
        get() = _phoneEmailPrev

    var email: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.email, null)
        }
    fun onPrevClicked() {
        _navigateToBack.value = Event(Unit)
    }

    var ssnOrMember: Boolean = true
    var firstName: String = ""
    var lastName: String = ""
    var dob: String = ""
    var password: String = ""
    var passwordConfirm: String = ""
    var gender: String = ""
    var mobile: String = ""
    var usernameType: String = ""
    var countryCode: String = ""
    var electronicDocuments: String = ""
    var overrideOldRegistration: String = ""
    var resendVerification: String = ""
    var tempUUID: String = ""

    @get:Bindable
    val emailTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _emailChanged.value = Event(s.toString())
        }
    }

    fun loginClicked(password: String) {
        compositeDisposable.add(

            loginRepository.getLogin(
                userName = email,
                password = password,
                user.tempUUID.toString(),
                false
            ).subscribeOn(
                Schedulers.io()
            )
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _logInSuccess.value = APILoginResponse<Response<AuthToken>>().loading()
                }
                .subscribe({
                    _logInSuccess.value = APILoginResponse<Response<AuthToken>>().success(it)
                }, {
                    _logInFail.value = Event(it.message.toString())
                })
        )
    }

    fun onNextClicked() {
        // bypass email verification if overriding old email
        if (user.overrideOldRegistration.equals("true")) {
            _navigateToPassword.value = APIResponse<DataResponse>().success(
                DataResponse("NOT_REGISTERED", "200", "", null, "", "NOT_REGISTERED", "", null)
            )
            return
        }

        compositeDisposable.add(
            emailRepository.verifyEmail(email)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _navigateToPassword.value = APIResponse<DataResponse>().loading()
                }
                .subscribe({
                    saveEmail()
                    _navigateToPassword.value = APIResponse<DataResponse>().success(it)
                }, {
                    _showError.value =
                        APIResponse<DataResponse>().error(it.toString())
                })
        )
    }

    fun saveTokens(data: AuthToken) {
        preferences.edit().putString(Constants.ACCESS_TOKEN, data.access_token).apply()
        preferences.edit().putString(Constants.TOKEN_TYPE, data.token_type).apply()
        preferences.edit().putString(Constants.EXPIRES_IN, data.expires_in).apply()
        preferences.edit().putString(Constants.SCOPE, data.scope).apply()
    }

    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.register, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.register, ""),
                        RegisterUser::class.java
                    )
                }
            },
            {
                email = user.email.toString()
                _emailChanged.value = Event(user.email.toString())
            }
        )
    }

    fun saveEmail() {
        user.email = email
        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.register, regString)
        regEdit.apply()
    }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
