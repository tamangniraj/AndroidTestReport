    fun resetTimer(time: Long) {
        timer?.cancel()
        timer = object : CountDownTimer(
            time,
            ONE_SECOND
        ) {
            override fun onTick(millisUntilFinished: Long) {
                currentTime.value = Event(millisUntilFinished / ONE_SECOND)

                if (channel == Constants.CHANNEL_PHONE && phoneOtpState.get() != OtpState.CALL_CLICKED) {
                    if ((millisUntilFinished / 1000) == (time - 15000) / 1000) {
                        phoneOtpState.set(OtpState.FIFTEEN_SEC_ELAPSED)
                    }
                }
            }

            override fun onFinish() {
                errorOrSuccess = true
                showTimeOut = true

                resend = true

                // clear text field
                code = ""

                error = verificationCodeExpired
                showErrorTv = true

                allowBack = true
                currentTime.value = Event(DONE)

                if (channel == Constants.CHANNEL_PHONE) {
                    onUpdateOtpOption(OtpState.TIMER_ENDED)
                }
            }
        }

        timer?.start()
    }

        this.countDown = countDown
        this.isConsentChecked = isConsent

        this.triggerCaptchaInNextRequest = triggerCaptcha ?: false

    fun onCallVerify() {
        compositeDisposable.add(
            phoneCodeRepository.verifyCodeV2(
                if (channel == Constants.CHANNEL_PHONE || channel == Constants.CHANNEL_VOICE) phone else email,
                registrationId = registrationId,
                attemptId = UUID.randomUUID().toString(),
                code = code,
                channel = lastChannel
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _navigateToNextScreen.value = APIResponse<VerifyOtpReceiveParams>().loading()
                }
                .subscribe({
                    if (it.isSuccessful) {
                        currentTime.value = Event(0)

                        it.body()?.let {
                                body ->
                            allowBack = true
                            phoneOtpState.set(OtpState.EMPTY)
                            if (body.status == Constants.STATUS_SUCCESS) {
                                preferences.edit().putBoolean(Constants.ShowErrorMessage, false).apply()
                                preferences.edit().putString(Constants.errorMessage, "").apply()
                            }
                            _navigateToNextScreen.value = APIResponse<VerifyOtpReceiveParams>().success(
                                body
                            )
                        }
                    } else {
                        val errorResponse: GenericErrorResponse = gson.fromJson(
                            it.errorBody()?.string(),
                            GenericErrorResponse::class.java
                        )

                        _showError.value =
                            APIResponse<ResponseOtpReceiveParams>().error(
                                errorResponse.getErrorMessage()
                            )
                    }
                }, {
                    _showError.value =
                        APIResponse<ResponseOtpReceiveParams>().error(it)
                })
        )
    }

    var lastChannel = Constants.CHANNEL_PHONE

    fun onResend(channel: String) {
        val lastState = phoneOtpState.get()
        lastChannel = channel

        executeWithOptionalCaptcha(triggerCaptchaInNextRequest) {
                token ->
            compositeDisposable.add(
                phoneCodeRepository.requestOTPV2(
                    if (channel == Constants.CHANNEL_PHONE || channel == Constants.CHANNEL_VOICE) phone else email,
                    cCode,
                    channel,
                    token,
                    registrationId,
                    isConsentChecked,
                    source = source
                )
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _navigateToResend.value = APIResponse<ResponseOtpReceiveParams>().loading()
                    }
                    .subscribe({
                        if (it.isSuccessful) {
                            timer?.cancel()

                            it.body()?.let {
                                it?.responseData?.firstOrNull()?.let {
                                        responseData ->

                                    triggerCaptchaInNextRequest = responseData.triggerCaptchaInNextRequest ?: false

                                    responseData.registrationId?.let { rid ->
                                        registrationId = rid
                                    }

                                    val regEdit = preferences.edit()
                                    regEdit.putLong(
                                        Constants.ExpirationTime,
                                        responseData.otpExpiry.toLong() * 1000
                                    )

                                    resetTimer(responseData.otpExpiry.toLong() * 1000)

                                    regEdit.putBoolean(
                                        Constants.ShowErrorMessage,
                                        responseData.status == "ERROR"
                                    )

                                    regEdit.putString(
                                        Constants.errorMessage,
                                        responseData.message.toString()
                                    )
                                    regEdit.apply()
                                }

                                _navigateToResend.value = APIResponse<ResponseOtpReceiveParams>().success(
                                    it
                                )
                            }
                        } else {
                            val errorResponse: GenericErrorResponse = gson.fromJson(
                                it.errorBody()?.string(),
                                GenericErrorResponse::class.java
                            )

                            resetTimer(
                                (errorResponse.otherDetails?.remainingTime?.toLong() ?: 0) * 1000
                            )

                            errorOrSuccess = true
                            resend = false
                            error = verificationCodeExpired
                            showErrorTv = true

                            error = errorResponse.getErrorMessage()
                            _showError.value =
                                APIResponse<ResponseOtpReceiveParams>().error(
                                    errorResponse.getErrorMessage()
                                )

                            if (errorResponse.errorType == "SESSION_LIMIT_EXCEEDED") {
                                phoneOtpState.set(OtpState.EMPTY)
                            }
                        }
                    }, {
                        phoneOtpState.set(lastState)
                        _showError.value =
                            APIResponse<ResponseOtpReceiveParams>().error(it)
                    })
            )
        }
    }

    var verificationCodeExpired = resources.getString(R.string.verification_code_has_expired)

    fun loadData(text: String) {
        verificationCodeExpired = text
        showErrorMessage = preferences.getBoolean(Constants.ShowErrorMessage, false)
        errorMessage = preferences.getString(Constants.errorMessage, "").toString()

        if (!StringUtils.equals(preferences.getString(Constants.register, ""), "")) {
            user = gson.fromJson(
                preferences.getString(Constants.register, ""),
                RegisterUser::class.java
            )
        }

        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.register, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.register, ""),
                        RegisterUser::class.java
                    )
                }
            },
            {
                tempUuid = user.tempUUID.toString()
                cCode = user.countryCode.toString()
                phone = user.mobile.toString()

                email = user.email.toString()

                COUNTDOWN_TIME =
                    preferences.getLong(Constants.ExpirationTime, 300000L)

                _completeData.value = Event(Unit)
            }
        )
    }

    fun onResendClicked() {

        errorOrSuccess = false

        }

    fun onCallClicked() {
        resend = false
        errorOrSuccess = false
        showTimeOut = false
        error = ""
        showErrorTv = false
        allowBack = false
        code = ""
        onResend(Constants.CHANNEL_VOICE)

        if (channel == Constants.CHANNEL_PHONE) {
            phoneOtpState.set(OtpState.CALL_CLICKED)
        }
    }

    private fun executeWithOptionalCaptcha(
        requestCaptcha: Boolean,
        onResult: (token: String?) -> Unit
    ) {
        if (requestCaptcha) {
            viewModelScope.launch {
                val result =
                    recaptchaClient
                        .execute(RecaptchaAction.SIGNUP, timeout = 10000L)
                onResult(result.getOrNull())
            }
        } else {
            onResult(null)
        }
    }

    fun onPrevClicked() {
        _navigateToBack1.value = Event(Unit)

enum class OtpState {
    EMPTY,
    INIT, // Start timer, show initial message
    FIFTEEN_SEC_ELAPSED, // Timer running, show "didn't receive code?" clickable
    SHOW_OPTIONS, // User clicked "didn't receive", show call/sms options
    TIMER_ENDED, // Timer expired, show error + options
    CALL_CLICKED // Call option selected, show status + SMS option only
}
