    fun loadMemberList(memberInfo: MemberInfo) {
        bubbleMemberList.clear()
        memberInfo.isSelected = false

        loading = true
        memberInfo.insured?.let { it1 ->
            bubbleMemberList.add(it1)
            bubbleMemberList.add(memberInfo)
        } ?: run {
            bubbleMemberList.add(0, memberInfo)
        }

        memberInfo.dependentsArrayList?.sortByDescending {
            DateUtils.stringServerDateToDate(
                it.birthDate.toString()
            )
        }

        memberInfo.dependentsArrayList?.reverse()

        memberInfo.dependentsArrayList?.let { it1 ->
            for (member in it1) {
//                if (!member.memberAdditionInReview) {
                if (StringUtils.equalsIgnoreCase(member.memberRelationship, "Spouse")) {
                    bubbleMemberList.add(1, member)
                } else {
                    if (!member.hideProfileFromHome) {
                        bubbleMemberList.add(member)
                    }
                }
//                }
            }

            var grouped = bubbleMemberList.groupBy { it.memberRelationship }
            grouped.forEach {
                it.value.sortedByDescending { item ->
                    DateUtils.stringServerDateToDate(
                        item.birthDate.toString()
                    )
                }
                it.value.reversed()
            }

            bubbleMemberList.clear()

            grouped.forEach {
                if (it.key.equals("Insured", true)) {
                    bubbleMemberList.add(0, it.value[0])
                } else if (it.key.equals("Spouse", true)) {
                    if (grouped.containsKey("Insured")) {
                        bubbleMemberList.add(1, it.value[0])
                    } else {
                        bubbleMemberList.add(0, it.value[0])
                    }
                } else if (it.key.equals("Child", true)) {
                    val dateGroup = it.value.groupBy { date ->
                        DateUtils.stringServerDateToDate(
                            date.birthDate.toString()
                        )
                    }

                    dateGroup.forEach { item ->
                        val sorted = item.value.sortedBy { item1 -> item1.firstName }
                        bubbleMemberList.addAll(sorted)
                    }
                }
            }
            val tempOthers: ArrayList<MemberInfo> = arrayListOf()
            grouped.forEach {
                if (!it.key.equals("Insured", true) && !it.key.equals(
                        "Spouse",
                        true
                    ) && !it.key.equals("Child", true)
                ) {
                    tempOthers.addAll(it.value)
                }
            }

            val dateGroup = tempOthers.groupBy { date ->
                DateUtils.stringServerDateToDate(
                    date.birthDate.toString()
                )
            }
            dateGroup.forEach { item ->
                val sorted = item.value.sortedBy { item1 -> item1.firstName }
                bubbleMemberList.addAll(sorted)
            }
        }
        loading = false
        _bubbleListGenerated.value = Event(bubbleMemberList.toList())
    }

    fun getCategory(clientId: String) {
        docTypeList.clear()
        loading = true

        compositeDisposable.add(
            documentRepository.getType(clientId).subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                    loading = true
                }.subscribe({ typeList ->
                    Timber.d(typeList.toString())

                    var list = typeList.dtos
                    typeList.dtos?.let { it1 ->
                        for (doc in it1) {
                            docTypeList.add(doc)
                        }
                    }
                    docTypeList.sortBy { it.documentType.lowercase() }
                    _typeListGenerated.value = Event(docTypeList.toList())
                    loading = false
                }, {
                    Timber.d(it.toString())
                    loading = false
                })
        )
    }

    fun setSelectedDoc(docType: Dto) {
        val index = docTypeList.indexOf(docType)

        for (item in docTypeList) {
            if (docTypeList.indexOf(item) == index) {
                docTypeList[index].apply {
                    isSelected = !(docTypeList[index].isSelected)
                }
            } else {
                docTypeList[docTypeList.indexOf(item)].apply {
                    isSelected = false
                }
            }
        }

        // set doc title
        docTypeTitle = resources.getString(R.string.select_type)
        docCategoryTitle = resources.getString(R.string.document_category)
        for (doc in docTypeList) {
            if (doc.isSelected) {
                docTypeTitle = doc.documentType
                docCategoryTitle = doc.documentCategory
                docName = doc.documentType
                selectedDocType = doc
            }
        }
        docTypeList.sortBy { it.documentType.lowercase() }
        updateDocumentName()
        _typeListGenerated.value = Event(docTypeList.toList())
    }

    fun updateDocumentName() {
        if (memberName.isNotBlank()) {
            if (memberName != resources.getString(R.string.select_person)) {
                docName = if (selectedDocType == null) {
                    memberName
                } else {
                    memberName + "'s " + selectedDocType?.documentType
                }
            }
        }
    }

    fun selectedReferenceMember(memberInfo: MemberInfo) {
        selectedReferenceMember = memberInfo
        val index = bubbleMemberList.indexOf(memberInfo)

        for (item in bubbleMemberList) {
            if (bubbleMemberList.indexOf(item) == index) {
                bubbleMemberList[index].apply {
                    isSelected = !(bubbleMemberList[index].isSelected)
                }
            } else {
                bubbleMemberList[bubbleMemberList.indexOf(item)].apply {
                    isSelected = false
                }
            }
        }

        // set reference name
        memberName = resources.getString(R.string.select_person)
        for (mem in bubbleMemberList) {
            if (mem.isSelected) {
                memberName = mem.lastName.toString() + ", " + mem.firstName.toString()
            }
        }
        updateDocumentName()
        _bubbleListGenerated.value = Event(bubbleMemberList.toList())
    }

    private fun uploadDocument(
        documentName: String,
        docTypeID: Int,
        docCategoryId: Int,
        referenceId: Int,
        longitude: Double?,
        latitude: Double?
    ) {
        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    documentRepository.uploadDoc(
                        memberInfo.selfId,
                        filePart,
                        docTypeID,
                        docCategoryId,
                        documentName,
                        referenceId,
                        longitude,
                        latitude,
                        ""
                    ).subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                        loading = true
                        _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().loading()
                    }.subscribe({
                        crossDocumentButton()
                        loading = false
                        _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().success(
                            it
                        )
                    }, {
                        loading = false
                        _navigateToUpload.value =
                            APIResponse<DocumentUploadSuccessEnroll>().error(it.localizedMessage!!)
                    })
                )
            }
        )
    }

    fun resetSubmitView() {
        hasDocument = false
        selectedFile = null

        for (i in 0 until bubbleMemberList.count()) {
            bubbleMemberList[i].isSelected = false
        }

        _bubbleListGenerated.value = Event(bubbleMemberList.toList())

        for (i in 0 until docTypeList.count()) {
            docTypeList[i].isSelected = false
        }

        _typeListGenerated.value = Event(docTypeList.toList())

        // set doc title
        docTypeTitle = resources.getString(R.string.select_type)
        docCategoryTitle = resources.getString(R.string.document_category)
        docName = ""
        memberName = resources.getString(R.string.select_person)

        enableUploadButton()
        enableAddButton()
    }

    fun enableAddButton() {
        addButtonEnabled =

    }

    fun enableUploadButton() {
        uploadEnabled =
            if ((docTypeTitle != resources.getString(R.string.select_type)) && memberName != resources.getString(
                    R.string.select_person
                )
            ) {
                hasDocument
            } else {
                false
            }
    }

    fun String.capitalize(): String {
        return this.replaceFirstChar {
            if (it.isLowerCase()) {
                it.titlecase(Locale.getDefault())
            } else {
                it.toString()
            }
        }
    }
