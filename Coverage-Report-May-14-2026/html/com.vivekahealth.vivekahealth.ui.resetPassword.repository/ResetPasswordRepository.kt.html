package com.vivekahealth.vivekahealth.ui.resetPassword.repository

import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.data.remote.resetPassword.ChangePasswordV3DTO
import com.vivekahealth.vivekahealth.ui.resetPassword.service.ResetPasswordService
import io.reactivex.Single
import javax.inject.Inject

class ResetPasswordRepository @Inject constructor(
    private val resetPasswordService: ResetPasswordService
) {

    fun getForgotPasswordV2(email: String): Single<DataResponse> =
        resetPasswordService.getResetPasswordApi().forgotPasswordV2(email)

    fun getForgotPassword(email: String): Single<DataResponse> =
        resetPasswordService.getResetPasswordApi().forgotPasswordV3(email)

    fun changePassword(changePasswordDTO: ChangePasswordV3DTO): Single<DataResponse> = resetPasswordService.getResetPasswordApi().changePassword(
        changePasswordDTO
    )
}
