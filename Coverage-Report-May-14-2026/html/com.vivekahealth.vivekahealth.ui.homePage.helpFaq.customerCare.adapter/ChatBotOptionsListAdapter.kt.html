package com.vivekahealth.vivekahealth.ui.homePage.helpFaq.customerCare.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.about.Option
import com.vivekahealth.vivekahealth.databinding.ChipListItemBinding

class ChatBotOptionsListAdapter(
    private var onClick: (Int) -> Unit,
    var canInteract: Boolean,
    var actionsTaken: List<String>?
) : ListAdapter<com.vivekahealth.vivekahealth.data.remote.about.Option, RecyclerView.ViewHolder>(
    ChatBotOptionsDiffCallback()
) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return BotViewHolder(
            ChipListItemBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as BotViewHolder).binding.setClickListener {
            if (canInteract) {
                onClick(position)
            }
        }

        if (!actionsTaken.isNullOrEmpty()) {
            if (actionsTaken?.contains(item.text) == true) {
                holder.binding.actionsTaken = item.text
            } else {
                if (actionsTaken?.contains(item.id) == true) {
                    holder.binding.actionsTaken = item.text
                }
            }
        } else {
            holder.binding.actionsTaken = ""
        }
        holder.binding.canInteract = canInteract
        holder.binding.data = item
    }

    class BotViewHolder(
        val binding: ChipListItemBinding,
        private val onClick: (Int) -> Unit
    ) : RecyclerView.ViewHolder(binding.root) {

        //        var toggle = binding.clicked
        init {

//            binding.clicked = adapterPosition == count
        }
    }
}

private class ChatBotOptionsDiffCallback : DiffUtil.ItemCallback<com.vivekahealth.vivekahealth.data.remote.about.Option>() {
    override fun areItemsTheSame(oldItem: Option, newItem: Option): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: Option, newItem: Option): Boolean {
        return oldItem == newItem
    }
}
