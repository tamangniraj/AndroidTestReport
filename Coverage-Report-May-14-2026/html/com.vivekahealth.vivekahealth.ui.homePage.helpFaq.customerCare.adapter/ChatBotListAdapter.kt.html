package com.vivekahealth.vivekahealth.ui.homePage.helpFaq.customerCare.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.about.Result
import com.vivekahealth.vivekahealth.databinding.ChatButtonsItemBinding

class ChatBotListAdapter(
    private var onClick: (com.vivekahealth.vivekahealth.data.remote.about.Result) -> Unit,
    private var onClickChips: (com.vivekahealth.vivekahealth.data.remote.about.Option) -> Unit,
    var canInteract: Boolean,
    var actionsTaken: List<String>?
) : ListAdapter<com.vivekahealth.vivekahealth.data.remote.about.Result, RecyclerView.ViewHolder>(
    ChatBotDiffCallback()
) {

    private val sColumnWidth = 120 // assume cell width of 120dp

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return BotViewHolder(
            ChatButtonsItemBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick,
            onClickChips
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as BotViewHolder).binding.setClickListener {
            onClick(item)
        }
        var adapter = ChatBotOptionsListAdapter(onClick = {
            if (canInteract) {
                var value = item.options?.get(it)
                value?.let { opt ->
                    onClickChips(opt)
                }
            }
        }, canInteract, actionsTaken)
        holder.binding.chipList.adapter = adapter

        if (!actionsTaken.isNullOrEmpty()) {
            if (item.id.isNullOrBlank()) {
                if (actionsTaken?.contains(item.text) == true) {
                    holder.binding.actionsTaken = item.text
                } else {
                    if (actionsTaken?.contains(item.id) == true) {
                        holder.binding.actionsTaken = item.text
                    }
                }
            }
        } else {
            holder.binding.actionsTaken = ""
        }
        holder.binding.canInteract = canInteract
        adapter.submitList(item.options)

        holder.binding.data = item
    }

    class BotViewHolder(
        val binding: ChatButtonsItemBinding,
        private val onClick: (com.vivekahealth.vivekahealth.data.remote.about.Result) -> Unit,
        private val onClickChips: (com.vivekahealth.vivekahealth.data.remote.about.Option) -> Unit
    ) : RecyclerView.ViewHolder(binding.root) {

        //        var toggle = binding.clicked
        init {

//            binding.clicked = adapterPosition == count
        }
    }
}

private class ChatBotDiffCallback : DiffUtil.ItemCallback<com.vivekahealth.vivekahealth.data.remote.about.Result>() {
    override fun areItemsTheSame(oldItem: Result, newItem: Result): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: Result, newItem: Result): Boolean {
        return oldItem == newItem
    }
}
