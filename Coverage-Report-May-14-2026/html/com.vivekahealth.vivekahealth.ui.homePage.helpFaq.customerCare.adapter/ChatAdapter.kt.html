package com.vivekahealth.vivekahealth.ui.homePage.helpFaq.customerCare.adapter

import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.local.home.ChatDataModel
import com.vivekahealth.vivekahealth.data.remote.about.ChatBotMessage
import com.vivekahealth.vivekahealth.databinding.*
import com.vivekahealth.vivekahealth.utils.DateUtils

class ChatAdapter(
    private var onClick: (com.vivekahealth.vivekahealth.data.remote.about.Result, Long) -> Unit,
    private var onClickChips: (com.vivekahealth.vivekahealth.data.remote.about.Option, Long) -> Unit
) :
    ListAdapter<ChatDataModel, RecyclerView.ViewHolder>(ChatDiffCallback()) {
    var gson = Gson()
    var overAllTranslated: Boolean = false
    var languageToggled: ArrayList<Int> = arrayListOf()
    private var lastSelectedViewHolder: RecyclerView.ViewHolder? = null
    companion object {
        const val CustomerCare = 1
        const val User = 0
    }

    private inner class ViewUserViewHolder(
        val binding: SingleChatClientBinding
    ) : RecyclerView.ViewHolder(binding.root) {
        fun bind(position: Int) {
        }
    }

    private inner class ViewAdminViewHolder(
        val binding: SingleChatCustomerCareBinding,
        private val onClick: (com.vivekahealth.vivekahealth.data.remote.about.Result, Long) -> Unit,
        private val onClickChips: (com.vivekahealth.vivekahealth.data.remote.about.Option, Long) -> Unit
    ) : RecyclerView.ViewHolder(binding.root) {
        fun bind(position: Int) {
        }
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        when (viewType) {
            User -> {
                return ViewUserViewHolder(
                    SingleChatClientBinding.inflate(
                        LayoutInflater.from(parent.context),
                        parent,
                        false
                    )
                )
            }
            CustomerCare -> {
                return ViewAdminViewHolder(
                    SingleChatCustomerCareBinding.inflate(
                        LayoutInflater.from(parent.context),
                        parent,
                        false
                    ),
                    onClick,
                    onClickChips
                )
            }

            else -> return ViewUserViewHolder(
                SingleChatClientBinding.inflate(
                    LayoutInflater.from(parent.context),
                    parent,
                    false
                )
            )
        }
    }

    fun translated(translate: Boolean) {
        overAllTranslated = translate
        languageToggled.clear()
        if (translate) {
            val indices = currentList.indices.toList()
            languageToggled.addAll(indices)
        }
        notifyDataSetChanged()
    }

    fun hideLoadingForLastSelected() {
        lastSelectedViewHolder?.let { holder ->
            if (holder is ViewUserViewHolder) {
                holder.binding.layoutLoading.root.visibility = View.GONE
            }
        }
    }
    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        var item = getItem(position)

        var time = DateUtils.chatDate(item.createdOn)

        if (position == 0 && item.userId.isNullOrEmpty()) {
            (holder as ViewUserViewHolder).binding.layoutLoading.root.visibility = View.VISIBLE
            lastSelectedViewHolder = holder
        } else {
            hideLoadingForLastSelected()
        }

        if (item.userId.isNullOrEmpty()) {
            (holder as ViewUserViewHolder).binding.chat = item
            (holder as ViewUserViewHolder).binding.time = time
        } else {
            (holder as ViewAdminViewHolder).binding.chat = item
            (holder as ViewAdminViewHolder).binding.time = time
            if (item.autoReply && item.messageContentType.equals("application/json", true)) {
                holder.binding.textContentCCare.visibility = View.GONE
                holder.binding.botList.visibility = View.VISIBLE
                var action = item.actionsTaken?.let { it }
                var adapter = ChatBotListAdapter(onClick = {
                    onClick(it, item.selfId ?: 0)
                }, onClickChips = { option ->
                        onClickChips(option, item.selfId ?: 0)
                    }, canInteract = item.canInteract, actionsTaken = action)

                var messageToggled = item.translatedMessage
                messageToggled = if (overAllTranslated) {
                    item.originalMessage
                } else {
                    item.translatedMessage
                }
                var value = gson.fromJson(messageToggled, ChatBotMessage::class.java)
                holder.binding.botList.adapter = adapter
                adapter.submitList(value.results)
                holder.binding.overallTranslated = overAllTranslated
                holder.binding.showTranslated = overAllTranslated
                holder.binding.setClickListener {
                    if (languageToggled.contains(position)) {
                        languageToggled.remove(position)
                        holder.binding.showTranslated = false
                        val valueItem = gson.fromJson(
                            item.translatedMessage,
                            ChatBotMessage::class.java
                        )
                        holder.binding.botList.adapter = adapter
                        adapter.submitList(valueItem.results)
                    } else {
                        languageToggled.add(position)
                        holder.binding.showTranslated = true
                        val valueItem = gson.fromJson(
                            item.originalMessage,
                            ChatBotMessage::class.java
                        )
                        holder.binding.botList.adapter = adapter

                        adapter.submitList(valueItem.results)
                    }
                }
            } else {
                holder.binding.overallTranslated = overAllTranslated
                holder.binding.showTranslated = overAllTranslated
                holder.binding.setClickListener {
                    if (languageToggled.contains(position)) {
                        languageToggled.remove(position)
                        holder.binding.showTranslated = false
                    } else {
                        languageToggled.add(position)
                        holder.binding.showTranslated = true
                    }
                }
                holder.binding.botList.visibility = View.GONE
                holder.binding.textContentCCare.visibility = View.VISIBLE
            }
        }
    }

    override fun getItemViewType(position: Int): Int {
        var item = getItem(position)

        return if (item.userId.isNullOrEmpty()) {
            User
        } else {
            CustomerCare
        }
    }
}

private class ChatDiffCallback : DiffUtil.ItemCallback<ChatDataModel>() {

    override fun areItemsTheSame(oldItem: ChatDataModel, newItem: ChatDataModel): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: ChatDataModel, newItem: ChatDataModel): Boolean {
        return oldItem.selfId == newItem.selfId
    }
}
