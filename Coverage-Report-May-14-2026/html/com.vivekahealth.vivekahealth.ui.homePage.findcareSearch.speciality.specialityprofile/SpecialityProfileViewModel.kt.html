    fun filter(text: String) {
        if (text.isNotBlank()) {
            var found = feeScheduleList.filter {
                it.codeValue.lowercase().contains(text) || it.codeLongDescription?.lowercase()?.contains(
                    text
                ) ?: false || it.codeShortDescription?.lowercase()?.contains(
                    text
                ) ?: false
            }
            tempList.clear()
            tempList.addAll(found)
            data = tempList.isEmpty()
            filterOn = true
            sortTemp()
            _logsListGeneratedSearch.value = Event(tempList)
        } else {
            filterOn = false
            data = feeScheduleList.isEmpty()
            sortMian()
            _logsListGenerated.value = Event(feeScheduleList)
        }
    }

    fun listUpdate(mainList: List<FeeSchedule>, changed: Int, isChecked: Boolean, price: String) {
        if (changed < 0 || changed >= mainList.size) {
            return // or handle the error
        }
        val index = changed
        val value = mainList[index]
        value.checked = isChecked
        value.showCheckBox = false

        if (price.isNotBlank()) {
            value.allowedAmount = price.toDouble()
        }
        if (value.amountChanged) {
            if (!isChecked) {
                value.allowedAmount = 0.0
            }
        }
        if (isChecked) {
            selectedItemsList.add(value)
        } else {
            selectedItemsList.removeAll { it.codeValue == value.codeValue }
        }

        var originalIndex = feeScheduleList.indexOf(
            feeScheduleList.find { it.codeValue == value.codeValue }
        )
//        var item = selectedScheduleList.find { it.codeValue == value.codeValue }
//        item?.let {
//            selectedScheduleList.remove(
//                selectedScheduleList.find { it.codeValue == value.codeValue }
//            )
//        }

        val updatedItems = feeScheduleList.toMutableList().apply { this[originalIndex] = value }
        val updatedItemsView = mainList.toMutableList().apply { this[index] = value }

        feeScheduleList.clear()
        feeScheduleList.addAll(updatedItems)

        recalculatePrice(value)
        _updateList.value = Event(updatedItemsView)
    }

    fun listUpdateSelected(
        mainList: List<FeeSchedule>,
        changed: FeeSchedule,
        isChecked: Boolean,
        price: String

        val index = mainList.indexOf(mainList.find { it.codeValue == changed.codeValue })

        val originalIndex =
            feeScheduleList.indexOf(feeScheduleList.find { it.codeValue == changed.codeValue })

            selectedItemsList.removeAll { it.codeValue == changed.codeValue }
        }

        if (originalIndex != -1) {
            val updatedItems =
                feeScheduleList.toMutableList().apply { this[originalIndex] = changed }
            val updatedItemsView = mainList.toMutableList().apply { this[index] = changed }
            feeScheduleList.clear()
            feeScheduleList.addAll(updatedItems)
            _updateList.value = Event(updatedItems)
        }
        recalculatePrice(changed)
    }

    fun listUpdatePrice(
        mainList: List<FeeSchedule>,
        price: String,
        item: FeeSchedule,
        changed: Int,
        isChecked: Boolean
    ) {
        var index = changed

        var value = item
        var originalIndex = feeScheduleList.indexOf(
            feeScheduleList.find { it.codeValue == item.codeValue }
        )
        if (price.isNotBlank()) {
            value.allowedAmount = price.toDouble()
        }
        value.checked = isChecked && price.isNotBlank()
        value.amountChanged = true

        if (!isChecked) {
            value.allowedAmount = 0.0
            value.amountChanged = false
        }

        if (isChecked) {
            selectedItemsList.add(value)
        } else {
            selectedItemsList.removeAll { it.codeValue == value.codeValue }
        }
        val updatedItems = feeScheduleList.toMutableList().apply { this[originalIndex] = value }
        val updatedItemsView = mainList.toMutableList().apply { this[index] = value }

        feeScheduleList.clear()
        feeScheduleList.addAll(updatedItems)

        recalculatePrice(item)
        _updateList.value = Event(updatedItemsView)
    }

    fun recalculatePrice(value: FeeSchedule?) {
        val mergedList = selectedItemsList.distinctBy { it.codeValue }

//        val mergedList = (selectedScheduleList.filter { it.checked } + selectedItemsList + feeScheduleList.filter { it.checked }).distinctBy { it.codeValue }
        totalPrice = (mergedList.sumOf { it.allowedAmount ?: 0.0 })
        val totalSelectedOld = totalSelected
        totalSelected = mergedList.count()
        if (totalSelected > totalSelectedOld && value?.allowedAmount != null && value.allowedAmount != 0.0) {
            added()
        }
        var v1 = mergedList.count { it.allowedAmount == null }
        var v2 = mergedList.count { it.allowedAmount == 0.0 }
        val totalNonZero = mergedList.filter { (it.allowedAmount == null || it.allowedAmount == 0.0) }
        if (mergedList.any { it.allowedAmount == null } || mergedList.any { it.allowedAmount == 0.0 }) {
            outOf = true
            outOfText = resources.getString(
                R.string.s_out_of_s_procedures,
                (mergedList.count() - totalNonZero.count()).toString(),
                mergedList.count().toString()
            )
        } else {
            outOf = false
        }
        getRecommendedProcedure(docValue.npi, docValue.ein)
    }
    fun sortMian() {
        if (sortby.equals("cost", true)) {
            if (sortAction.equals("asc", true)) {
                val sortedList = feeScheduleList.sortedWith(compareBy { it.allowedAmount })
                feeScheduleList.clear()
                feeScheduleList.addAll(sortedList)
            } else if (sortAction.equals("desc", true)) {
                val sortedList = feeScheduleList.sortedByDescending { it.allowedAmount }
                feeScheduleList.clear()
                feeScheduleList.addAll(sortedList)
            }
        } else if (sortby.equals("name", true)) {
            if (sortAction.equals("asc", true)) {
                val sortedList = feeScheduleList.sortedBy { it.codeValue }
                feeScheduleList.clear()
                feeScheduleList.addAll(sortedList)
            } else if (sortAction.equals("desc", true)) {
                val sortedList = feeScheduleList.sortedByDescending { it.codeValue }
                feeScheduleList.clear()
                feeScheduleList.addAll(sortedList)
            }
        } else if (sortby.equals("relevance", true)) {
            val sortedList = feeScheduleList.sortedByDescending { it.occurrence }
            feeScheduleList.clear()
            feeScheduleList.addAll(sortedList)
        }
    }

    fun getProvidersList(npi: String, ein: String, search: Boolean) {
        compositeDisposable.add(
            specialityRepository.getProcedureList(
                npi,
                ein,
                page,
                pageSize,
                sortby,
                sortAction,
                searchText,
                codeType,
                codeValue
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->
                    // data =  it.doctorList.isNotEmpty()
//                    Value.isSelected = true
                    //   ValueList = it.values
                    loading = false
                    page++
                    eligibleValueList = ValueList
                    if (search) {
                        feeScheduleList.clear()
                    }

                    feeScheduleList.addAll(it.feeSchedule)

                    var tempList: ArrayList<FeeSchedule> = arrayListOf()

                    val commonObjects = selectedItemsList.filter { obj1 ->
                        feeScheduleList.any { obj2 -> obj1.codeValue == obj2.codeValue }
                    }

                    commonObjects.forEach { it.checked = true }

                    feeScheduleList.replaceAll { obj1 ->
                        commonObjects.find { obj2 -> obj1.codeValue == obj2.codeValue } ?: obj1
                    }

                    tempList.addAll(feeScheduleList)
//                    tempList.forEach { item ->
// //                        item.showCheckBox = true
//                    }

                    // Find the elements that match the predicate in both lists
                    data = feeScheduleList.isEmpty()
                    listForDoctorAdapter = eligibleValueList.toList()
                    isLastPage = it.paging?.totalPages == page

                    if (search) {
                        _logsListGeneratedSearch.value = Event(tempList)
                    } else {
                        _logsListGenerated.value = Event(tempList)
                    }
                }, {
                    data = false

                    loading = false
                })
        )
    }

    fun getRecommendedProcedure(npi: String?, ein: String?) {
        var selectedList: List<String> = listOf()

        selectedList = if (filterOn) {
            tempList.filter { it.checked }.map { it.codeValue }
        } else {
            feeScheduleList.filter { it.checked }.map { it.codeValue }
        }
        val list = selectedScheduleList.map { it.codeValue }
        val listSelected = selectedItemsList.map { it.codeValue }

        selectedList = (selectedList + list + listSelected).distinctBy { it }
        selectedList.distinct()

//        selectedList = selectedItemsList.map { it.codeValue }.distinct()
//        totalSelected = selectedList.count()
        compositeDisposable.add(
            specialityRepository.getRecommendedProcedureList(
                RecommendedRequest(
                    ein = ein,
                    npi = npi,
                    latitude = lat,
                    longitude = long,
                    primaryTaxonomy = primaryTaxonomy,
                    radius = radius,
                    selectedProcedures = selectedList
                )
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({
                    loading = false

                    recommendedList.clear()
                    it.results?.let { item ->
                        recommendedList.addAll(item)
                        recommendedList = recommendedList
                    }

                    showNoData = totalSelected == 0
                    _recommendedProcedureList.value = Event(it.results)
                }, {
                    data = false

                    showNoData = totalSelected == 0
                    loading = false
                })
        )
    }
