package com.vivekahealth.vivekahealth.ui.enrollVTwo.start

import android.content.Context
import android.os.Bundle
import android.os.SystemClock
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.lifecycle.ViewModelProvider
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.data.local.enroll.DocumentItem
import com.vivekahealth.vivekahealth.data.local.enroll.DocumentItemPlus
import com.vivekahealth.vivekahealth.data.local.enroll.RelationshipType
import com.vivekahealth.vivekahealth.data.local.enroll.RootDocumentResponse
import com.vivekahealth.vivekahealth.databinding.StartEnrollmentDialogBinding

import com.vivekahealth.vivekahealth.ui.BaseDialogFragment
import com.vivekahealth.vivekahealth.utils.remote.Status
import javax.inject.Inject

class StartEnrollmentDialog : BaseDialogFragment(), Injectable {
    companion object {

        private const val ARG_MEMBER_ID = "arg_member_id"
        private const val ARG_IS_REJECTED = "arg_rejected"

        fun newInstance(memberId: String, isRejected: Boolean): StartEnrollmentDialog {
            val dialog = StartEnrollmentDialog()
            val args = Bundle()
            args.putString(ARG_MEMBER_ID, memberId)
            args.putBoolean(ARG_IS_REJECTED, isRejected)
            dialog.arguments = args
            return dialog
        }
    }

    private val memberId: String? by lazy {
        arguments?.getString(ARG_MEMBER_ID)
    }

    private val isRejected: Boolean? by lazy {
        arguments?.getBoolean(ARG_IS_REJECTED)
    }

    private lateinit var binding: StartEnrollmentDialogBinding

    private var startEnrollmentListener: StartEnrollmentListener? = null

    private val adapter = RequiredDocumentAdapter()

    fun setCallback(verifyIdentityListener: StartEnrollmentListener) {
        this.startEnrollmentListener = verifyIdentityListener
    }

    private lateinit var viewModel: StartEnrollmentViewModel

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory

    private lateinit var rootDocumentResponse: RootDocumentResponse

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = StartEnrollmentDialogBinding.inflate(inflater, container, false)

        return binding.root
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            StartEnrollmentViewModel::class.java
        )

        memberId?.let {
            viewModel.startEnrollment(it)
        }

        binding.textViewTitle.text = if (isRejected == true) {
            getString(R.string.your_enrollment_was_not_approved)
        } else {
            getString(R.string.almost_there)
        }

        binding.tvText.visibility = if (isRejected == true) View.VISIBLE else View.GONE
        binding.tvText.text = if (isRejected == true) {
            getString(R.string.there_was_issue_with_the_documents_you_submitted)
        } else {
            getString(R.string.almost_there)
        }

        binding.txtSecondDesc.visibility = if (isRejected == true) View.VISIBLE else View.GONE
        binding.txtSecondDesc.text = if (isRejected == true) {
            getString(R.string.to_continue_please_review_your)
        } else {
            getString(R.string.insturction_enrooment)
        }

        viewModel.startSession.observe(viewLifecycleOwner) {
            when (it.status) {
                Status.LOADING -> {
                    binding.proceed.isEnabled = false
                    binding.progressBar.visibility = View.VISIBLE
                }

                Status.SUCCESS -> {
                    binding.proceed.isEnabled = true
                    binding.progressBar.visibility = View.GONE
                    val combinedDocuments = mutableListOf<DocumentItemPlus>()
                    val requiredDocuments = it.data?.documentList ?: emptyList()
                    val predicate: (DocumentItem) -> Boolean =
                        { document -> document.documentType == 103 || document.documentType == 104 }
                    val primary =
                        requiredDocuments.find { it.relationshipType == RelationshipType.PRIMARY.code }
                    val spouse =
                        requiredDocuments.find { it.relationshipType == RelationshipType.SPOUSE.code }
                    val child =
                        requiredDocuments.find { it.relationshipType == RelationshipType.CHILD.code }

                    if (primary != null) {
                        val primaryDocuments =
                            primary.documents.filterNot { it.uploaded }.toMutableList()
                        // If primary has any of driver's license front or driver's license back
                        if (primaryDocuments.any(predicate)) {
                            primaryDocuments.removeAll(predicate)
                            primaryDocuments += DocumentItem(
                                42,
                                false,
                                true,
                                "Driver's License",
                                getString(R.string.drivers_license)
                            )
                        }
                        combinedDocuments += primaryDocuments.map {
                            it.toDocumentPlus(
                                RelationshipType.PRIMARY
                            )
                        }
                    }
                    if (spouse != null) {
                        val spouseDocuments =
                            spouse.documents.filterNot { it.uploaded }.toMutableList()
                        // If spouse has any of driver's license front or driver's license back
                        if (spouseDocuments.any(predicate)) {
                            spouseDocuments.removeAll(predicate)
                            spouseDocuments += DocumentItem(
                                42,
                                false,
                                true,
                                "Driver's License",
                                getString(R.string.drivers_license)
                            )
                        }
                        combinedDocuments += spouseDocuments.map {
                            it.toDocumentPlus(
                                RelationshipType.SPOUSE
                            )
                        }
                    }
                    if (child != null) {
                        // Exclude already-uploaded documents
                        combinedDocuments += child.documents.filterNot { it.uploaded }.map {
                            it.toDocumentPlus(
                                RelationshipType.CHILD
                            )
                        }
                    }

                    // Group required documents by document type
                    val grouped = combinedDocuments.groupBy { it.translatedDocumentName }

                    val data = grouped.map {
                        val iconResId = when (it.value[0].documentType) {
                            71 -> R.drawable.ssn_card
                            42 -> R.drawable.drivers_license
                            50 -> R.drawable.wedding_rings
                            51 -> R.drawable.child_dependent
                            else -> R.drawable.document_ic
                        }
                        RequiredDocumentRowData(
                            documentName = it.key,
                            iconResId = iconResId,
                            dependentsText = getName(it.value, requireContext())
                        )
                    }
                    binding.gatherTheseDocuments.text = getString(
                        R.string.gather_these_documents,
                        grouped.size
                    )
                    binding.documentCards.adapter = adapter
                    adapter.submitList(data)
                    rootDocumentResponse = it.data ?: RootDocumentResponse(emptyList())

                    binding.documentCards.visibility = if (isRejected == true) {
                        View.GONE
                    } else {
                        View.VISIBLE
                    }
                    binding.youCanTakePhotos.visibility = if (isRejected == true) {
                        View.GONE
                    } else {
                        View.VISIBLE
                    }
                    binding.gatherTheseDocuments.visibility = if (isRejected == true) {
                        View.GONE
                    } else {
                        View.VISIBLE
                    }
                }

                Status.ERROR -> {
                    binding.proceed.isEnabled = true
                    binding.progressBar.visibility = View.GONE

                    val err = it.error.equals(
                        "Unable to proceed. Please check your internet connection and try again later."
                    )
                    if (err) {
                        startEnrollmentListener?.onError(
                            "You are offline. Some features may not be available."
                        )
                    } else {
                        startEnrollmentListener?.onError(it.error ?: "")
                    }
                }
            }
        }

        binding.proceed.setOnClickListenerDebounced {
            startEnrollmentListener?.onProceed(rootDocumentResponse)
        }
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setStyle(STYLE_NORMAL, R.style.CustomBottomSheetDialogTheme)
        isCancelable = false
    }

    override fun onStart() {
        super.onStart()
        dialog?.window?.setLayout(
            ViewGroup.LayoutParams.MATCH_PARENT,
            ViewGroup.LayoutParams.WRAP_CONTENT
        )
    }
}

fun View.setOnClickListenerDebounced(debounceTime: Long = 600L, action: (View) -> Unit) {
    this.setOnClickListener(object : View.OnClickListener {
        private var lastClickTime: Long = 0

        override fun onClick(v: View) {
            if (SystemClock.elapsedRealtime() - lastClickTime < debounceTime) {
                return
            }
            lastClickTime = SystemClock.elapsedRealtime()
            action(v)
        }
    })
}

fun getName(values: List<DocumentItemPlus>, context: Context): String {
    val relationshipTypes = values.map { it.relationshipType }

    return when (relationshipTypes) {
        listOf(RelationshipType.PRIMARY, RelationshipType.SPOUSE) -> {
            context.getString(R.string.for_you_and_your_spouse)
        }

        listOf(RelationshipType.SPOUSE, RelationshipType.CHILD) -> {
            context.getString(R.string.for_your_spouse_and_each_child)
        }

        listOf(RelationshipType.PRIMARY, RelationshipType.CHILD) -> context.getString(
            R.string.for_you_and_each_child
        )

        listOf(RelationshipType.PRIMARY) -> context.getString(R.string.for_you)

        listOf(RelationshipType.SPOUSE) -> {
            context.getString(R.string.if_you_re_adding_a_spouse)
        }

        listOf(RelationshipType.CHILD) -> {
            context.getString(R.string.for_each_child_dependent)
        }

        listOf(
            RelationshipType.PRIMARY,
            RelationshipType.SPOUSE,
            RelationshipType.CHILD
        ) -> {
            context.getString(R.string.for_you_your_spouse_and_each_child)
        }

        else -> {
            ""
        }
    }
}
