package com.vivekahealth.vivekahealth.ui.login.successReg

import android.content.Intent
import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.lifecycle.ViewModelProvider
import androidx.navigation.fragment.findNavController
import androidx.navigation.fragment.navArgs
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.databinding.FragmentRegistrationSuccessLoginBinding
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseFragment
import com.vivekahealth.vivekahealth.ui.login.LoginActivity
import com.vivekahealth.vivekahealth.ui.register.step3.StepEmailFragmentDirections
import com.vivekahealth.vivekahealth.utils.*
import com.vivekahealth.vivekahealth.utils.remote.Status
import javax.inject.Inject
import org.apache.commons.lang3.StringUtils
import timber.log.Timber

class StepRegSuccessFragmentLogin : BaseFragment(), Injectable {

    companion object {
        fun newInstance() = StepRegSuccessFragmentLogin()
    }

    private lateinit var viewModel: StepRegSuccessLoginViewModel

    private lateinit var binding: FragmentRegistrationSuccessLoginBinding

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory
    private val args: StepRegSuccessFragmentLoginArgs by navArgs()

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = FragmentRegistrationSuccessLoginBinding.inflate(inflater, container, false)
        return binding.root
    }

    var isFromLogin = false

    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            StepRegSuccessLoginViewModel::class.java
        )
        binding.viewModel = viewModel

//        viewModel.loadData()
        viewModel.email = args.email.toString()
        binding.email.setText(args.email)

        isFromLogin = args.fromLogin

        Timber.d(args.email)
        binding.email.isEnabled = args.editEmail

        if (!args.editEmail || isFromLogin) {
            binding.update.visibility = View.GONE
        } else {
            binding.update.visibility = View.VISIBLE
        }
        binding.email.isEnabled = false

//        val spannable = SpannableString(binding.textResend.text)
//
//        spannable.setSpan(
//            ForegroundColorSpan(resources.getColor(R.color.blue_btn,resources.newTheme())),
//
//            53, spannable.length,
//            Spannable.SPAN_EXCLUSIVE_EXCLUSIVE)
//        spannable.setSpan(
//            StyleSpan(BOLD),
//            53, spannable.length,
//            Spannable.SPAN_EXCLUSIVE_EXCLUSIVE)

        viewModel.navigateToLogin.observe(
            viewLifecycleOwner,
            EventObserver {
                viewModel.removeReg()

                if (isFromLogin) {
                    findNavController().popBackStack()
                } else {
                    val intent = Intent(requireContext(), LoginActivity::class.java)
                    startActivity(intent)
                    requireActivity().finish()
                }
            }
        )

        viewModel.changeEnable.observe(
            viewLifecycleOwner,
            EventObserver {
                if (viewModel.cancel) {
                    viewModel.email = viewModel.originalEmail
                    binding.email.isEnabled = false
                    binding.update.text = getString(R.string.update)
                    viewModel.cancel = false
                    viewModel.Done = false
                    viewModel.updateEnabled = true
                } else {
                    binding.email.isEnabled = viewModel.updateEnabled
                }
            }
        )
        viewModel.emailChanged.observe(
            viewLifecycleOwner,
            EventObserver { email ->

                if (ActivityUtils.isEmailValid(email)) {
                    if (!email.equals(viewModel.originalEmail, ignoreCase = true)) {
                        binding.update.text = getString(R.string.done)
                        viewModel.email = email
                        binding.showError = false
                        viewModel.Done = true
                        viewModel.cancel = false
                    } else if (viewModel.count == 0 && email.equals(
                            viewModel.originalEmail,
                            ignoreCase = true
                        )
                    ) {
                        Timber.d(viewModel.count.toString())
                        binding.email.isEnabled = false
                        binding.update.text = getString(R.string.update)
                        viewModel.cancel = false
                        viewModel.Done = false
                    } else {
                        binding.update.text = getString(R.string.cancel)
                        viewModel.Done = false
                        viewModel.cancel = true
                        binding.showError = false
                    }

//                if(viewModel.firstTextChange) {
//                }
                } else {
                    binding.showError = true

                    viewModel.Done = false
                    viewModel.cancel = true

                    binding.update.text = getString(R.string.cancel)

                    binding.error.text = getString(R.string.invalid_email_format)
//               HelperUtils.showOkMessageDialog(requireContext(), "Invalid email format. Please enter a valid format e.g. user@example.com")
                }
            }
        )

        viewModel.navigateToSuccess.observe(viewLifecycleOwner) {
            when (it.status) {
                Status.LOADING -> {
                    Timber.d(getString(R.string.loading))
                    (this.requireActivity() as LoginActivity).showLoading()
                }
                Status.SUCCESS -> {
                    (this.requireActivity() as LoginActivity).dismissLoading()

                    if (it.data?.code == "200" && StringUtils.isEmpty(it.data?.action)) {
                        binding.email.isEnabled = false
                        binding.update.text = getString(R.string.update)
                        viewModel.cancel = false
                        viewModel.Done = false
                        viewModel.updateEnabled = true

//                        dialog(it.data!!.message, "OK", "", "")

//                        HelperUtils.showOkMessageDialog(requireContext(), it.data!!.message)
                    } else {
                        HelperUtils.showOkMessageDialog(requireContext(), it.data!!.reason)
                    }
                }
                Status.ERROR -> {
                    (this.requireActivity() as LoginActivity).dismissLoading()

//
//                    val intent = Intent(requireContext(), EnrollActivity::class.java)
//                    startActivity(intent)
                    // Timber.d(it.error)
                }
            }
        }

        viewModel.navigateToResend.observe(viewLifecycleOwner) {
            when (it.status) {
                Status.LOADING -> {
                    Timber.d(getString(R.string.loading))
                    (this.requireActivity() as LoginActivity).showLoading()
                }
                Status.SUCCESS -> {
                    (this.requireActivity() as LoginActivity).dismissLoading()

                    if (it.data?.code == "200") {
                        HelperUtils.showOkMessageDialog(requireContext(), it.data!!.message)
                    } else {
                        HelperUtils.showOkMessageDialog(requireContext(), it.data!!.message)
                    }
                }
                Status.ERROR -> {
                    (this.requireActivity() as LoginActivity).dismissLoading()

//
//                    val intent = Intent(requireContext(), EnrollActivity::class.java)
//                    startActivity(intent)
                    // Timber.d(it.error)
                }
            }
        }

        viewModel.navigateToReRegister.observe(viewLifecycleOwner) {
            when (it.status) {
                Status.LOADING -> {
                    Timber.d(getString(R.string.loading))
                    (activity as LoginActivity).showLoading()
                }
                Status.SUCCESS -> {
                    (activity as LoginActivity).dismissLoading()

                    when {
                        it.data?.reason.equals("REGISTERED_AND_VERIFIED", true) -> {
                            DialogUtils(
                                requireActivity(),
                                object : DialogListener {
                                    override fun onNeutralClicked() {
                                        val directions =
                                            StepEmailFragmentDirections.actionEmailFragmentToForgetPasswordFragment(
                                                viewModel.email,
                                                "0"
                                            )
                                        findNavController().navigate(directions)
                                    }

                                    override fun onPositiveClicked() {
                                        val intent = Intent(
                                            requireContext(),
                                            LoginActivity::class.java
                                        )
                                        startActivity(intent)
                                        requireActivity().finish()
                                    }

                                    override fun onNegativeClicked() {
                                    }
                                }
                            ).dialogWithView(
                                getString(R.string.log_in),
                                getString(R.string.reset_password),
                                "",
                                R.layout.layout_reg_green_txt
                            )
                        }
                        it.data?.reason.equals("REGISTERED_BUT_NOT_VERIFIED", true) -> {
                            val direction =
                                StepEmailFragmentDirections.actionEmailFragmentToSuccessFragment(
                                    false
                                )

                            HelperUtils.showOkMessageDialog(
                                requireContext(),
                                getString(R.string.you_have_already_successfully_registered)
                            )
                        }
                        it.data?.reason.equals("REGISTERED_BUT_BENEFIT_NOT_YET_ACTIVE", true) -> {
                            HelperUtils.showOkMessageDialog(
                                requireContext(),
                                getString(R.string.benefit_not_active)
                            )
                        }
                        it.data?.reason.equals("NOT_REGISTERED", true) -> {
                            viewModel.reRegister()
                        }
                    }

//                   (this.requireActivity() as LoginActivity).navigationFragmentManager.addFragmentToBackStack(StepPasswordFragment.newInstance(viewModel.mobile,viewModel.countryCode,viewModel.tempUUID,viewModel.email))
                }
                Status.ERROR -> {
                    // Timber.d(it.error)
                }
            }
        }
    }

    fun dialog(content: String, positive: String, neutral: String, negative: String) {
        DialogUtils(
            requireActivity(),
            object : DialogListener {
                override fun onNeutralClicked() {
                }

                override fun onPositiveClicked() {
                }

                override fun onNegativeClicked() {
                }
            }
        ).dialog(
            content,
            positive,
            neutral,
            negative
        )
    }
}
