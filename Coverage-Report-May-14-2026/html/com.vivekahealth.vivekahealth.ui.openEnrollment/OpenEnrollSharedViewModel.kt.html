package com.vivekahealth.vivekahealth.ui.openEnrollment

import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.remote.guest.GuestRegisterUser
import javax.inject.Inject

class OpenEnrollSharedViewModel @Inject constructor() : ViewModel() {

    var guestRegisterUser: GuestRegisterUser? = null

    var dlData: DLData? = null
    var ssnValue: String? = null

    var employmentData: EmploymentData? = null

    var selectedPlan: String? = null

    var guestTokenUpdated: Boolean = false

    data class EmploymentData(
        val employerName: String = "",
        val planOption: String = "",
        val jobTitle: String = "",
        val dateOfHire: String = ""
    )

    data class DLData(
        val firstName: String = "",
        val lastName: String = "",
        val dob: String = "",
        val gender: String = "",
        val streetAddress: String = "",
        val apt: String = "",
        val city: String = "",
        val state: String = "",
        val zip: String = ""
    )
}
