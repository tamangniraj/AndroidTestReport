package com.vivekahealth.vivekahealth.ui.openEnrollment

import com.vivekahealth.vivekahealth.data.remote.guest.GuestUserInfo
import com.vivekahealth.vivekahealth.ui.openEnrollment.OpenEnrollmentActivity.Companion.ENROLLMENT_TYPE_VALUE_COMPLETED
import com.vivekahealth.vivekahealth.ui.openEnrollment.OpenEnrollmentActivity.Companion.ENROLLMENT_TYPE_VALUE_MISSING_DOCUMENT
import com.vivekahealth.vivekahealth.ui.openEnrollment.OpenEnrollmentActivity.Companion.ENROLLMENT_TYPE_VALUE_RESUME
import java.text.SimpleDateFormat
import java.util.Locale

fun formatEnrollmentDate(dateStr: String?, displayPattern: String = "MM/dd/yyyy"): String {
    if (dateStr.isNullOrBlank()) return ""
    return try {
        val date = SimpleDateFormat("yyyy-MM-dd", Locale.getDefault()).parse(dateStr)
        if (date != null) {
            SimpleDateFormat(displayPattern, Locale.getDefault()).format(date)
        } else {
            dateStr
        }
    } catch (e: Exception) {
        dateStr
    }
}

fun enrollmentTypeForUser(user: GuestUserInfo): Int = when {
    user.isEnrolled == true && user.pendingIdentityVerification -> ENROLLMENT_TYPE_VALUE_MISSING_DOCUMENT
    user.isEnrolled == true -> ENROLLMENT_TYPE_VALUE_COMPLETED
    else -> ENROLLMENT_TYPE_VALUE_RESUME
}

fun maskSsn(ssn: String?): String {
    if (ssn.isNullOrBlank()) return ""
    val clean = ssn.replace("-", "")
    return if (clean.length >= 4) "***-**-${clean.takeLast(4)}" else ssn
}
