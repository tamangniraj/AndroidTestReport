package com.vivekahealth.vivekahealth.ui.openEnrollment

import androidx.annotation.StringRes
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.utils.DateUtils
import java.text.SimpleDateFormat
import java.util.Locale

/**
 * Pure validation logic for every date field in the open enrollment flow.
 * Returns a string resource ID (null = valid) so fragments can call getString()
 * and tests can assert directly — no duplication, no Android Context needed here.
 */
object OpenEnrollmentDateValidator {

    @StringRes
    fun memberDobError(
        display: String,
        dateOfHireServer: String = "",
        marriageDateServer: String = ""
    ): Int? {
        if (!isFormatValid(display)) return R.string.date_of_birth_is_not_valid
        if (isFuture(display)) return R.string.date_of_birth_is_not_valid
        if (!DateUtils.isUserMature(display)) return R.string.error_member_must_be_16
        val dobServer = DateUtils.toAPIDate(display) ?: return R.string.date_of_birth_is_not_valid
        if (dateOfHireServer.isNotBlank() && DateUtils.ageAtDate(dobServer, dateOfHireServer) < 16) {
            return R.string.error_date_of_hire_is_invalid_for_updated_dob
        }
        if (marriageDateServer.isNotBlank() && DateUtils.ageAtDate(dobServer, marriageDateServer) < 16) {
            return R.string.error_marriage_date_is_invalid_for_updated_dob
        }
        return null
    }

    @StringRes
    fun dateOfHireError(display: String, memberDobServer: String = ""): Int? {
        if (!isFormatValid(display)) return R.string.error_date_of_hire_invalid
        if (isFuture(display)) return R.string.error_date_of_hire_future
        if (memberDobServer.isNotBlank()) {
            val hireServer = DateUtils.toAPIDate(display) ?: ""
            if (hireServer.isNotBlank() && DateUtils.ageAtDate(memberDobServer, hireServer) < 16) {
                return R.string.error_member_must_be_16_at_hire
            }
        }
        return null
    }

    @StringRes
    fun spouseDobError(display: String, marriageDateServer: String = ""): Int? {
        if (!isFormatValid(display)) return R.string.date_of_birth_is_not_valid
        if (isFuture(display)) return R.string.date_of_birth_is_not_valid
        if (!DateUtils.isUserMature(display)) return R.string.error_spouse_must_be_16
        if (marriageDateServer.isNotBlank()) {
            val spouseDobServer = DateUtils.toAPIDate(display) ?: ""
            if (DateUtils.ageAtDate(spouseDobServer, marriageDateServer) < 16) {
                return R.string.error_spouse_must_be_16_at_marriage
            }
        }
        return null
    }

    @StringRes
    fun marriageDateError(
        display: String,
        spouseDobServer: String = "",
        memberDobServer: String = ""
    ): Int? {
        if (!isFormatValid(display)) return R.string.marriage_date_is_not_valid
        if (isFuture(display)) return R.string.marriage_date_is_not_valid
        val marriageServer = DateUtils.toAPIDate(display) ?: return R.string.marriage_date_is_not_valid
        if (spouseDobServer.isNotBlank()) {
            if (DateUtils.ageAtDate(spouseDobServer, marriageServer) < 0) {
                return R.string.error_marriage_date_before_dob
            }
            if (DateUtils.ageAtDate(spouseDobServer, marriageServer) < 16) {
                return R.string.error_spouse_must_be_16_at_marriage
            }
        }
        if (memberDobServer.isNotBlank() && DateUtils.ageAtDate(memberDobServer, marriageServer) < 16) {
            return R.string.error_member_must_be_16_at_marriage
        }
        return null
    }

    @StringRes
    fun childDobError(display: String): Int? {
        if (!isFormatValid(display)) return R.string.date_of_birth_is_not_valid
        if (isFuture(display)) return R.string.date_is_not_valid
        return null
    }

    private fun isFormatValid(display: String): Boolean = try {
        SimpleDateFormat("MM/dd/yyyy", Locale.US).also { it.isLenient = false }.parse(display) != null
    } catch (_: Exception) { false }

    private fun isFuture(display: String): Boolean = try {
        DateUtils.isFutureMonth(DateUtils.stringDateToDate(display))
    } catch (_: Exception) { false }
}
