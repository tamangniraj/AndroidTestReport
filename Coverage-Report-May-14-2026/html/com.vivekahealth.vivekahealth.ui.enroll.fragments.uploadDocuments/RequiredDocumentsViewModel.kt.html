    fun setDocumentName(id: String, fileName: String) {
        job = Coroutines.ioToMainThread(
            {
                when {
                    currentPerson == user.subscriber -> {
                        when {
                            currentItem.id == 3 -> {
                                user.subscriber!!.ssnCardFileId = id
                            }
                            currentItem.id == 1 -> {
                                user.subscriber!!.licenseFrontFileId = id
                            }
                            currentItem.id == 2 -> {
                                user.subscriber!!.licenseBackFileId = id
                            }
                            currentItem.id == 4 -> {
                                user.subscriber!!.continueCoverageFileId = id
                                user.subscriber!!.continueCoverageFileName = fileName
                            }
                        }
                    }
                    currentPerson == user.spouse -> {
                        when {
                            currentItem.id == 3 -> {
                                user.spouse!!.ssnCardFileId = id
                            }
                            currentItem.id == 1 -> {
                                user.spouse!!.licenseFrontFileId = id
                            }
                            currentItem.id == 2 -> {
                                user.spouse!!.licenseBackFileId = id
                            }
                            currentItem.id == 5 -> {
                                user.spouse!!.marriageCertificateFileId = id
                                user.spouse!!.marriageCertificateFileName = fileName
                            }
                            fileName.contains(
                                resources.getString(R.string.front_of_insurance),
                                true
                            ) -> {
                                user.spouse!!.spouseInsuranceFrontViewFileId = id
                                user.spouse!!.spouseInsuranceFrontViewFileName = fileName
                            }
                            fileName.contains(resources.getString(R.string.back_of_insurance), true) -> {
                                user.spouse!!.spouseInsuranceBackViewFileId = id
                                user.spouse!!.spouseInsuranceBackViewFileName = fileName
                            }
                        }
                    }
                    user.children.contains(currentPerson) -> {
                        val index = user.children.indexOf(currentPerson)
                        when {
                            currentItem.id == 6 -> {
                                user.children[index].birthCertificateFileId = id
                                user.children[index].birthCertificateFileName = fileName
                            }
                            currentItem.id == 3 -> {
                                user.children[index].ssnCardFileId = id
                            }
                        }
                    }
                }
            },
            {
                val regString = gson.toJson(user)
                val regEdit = preferences.edit()
                regEdit.putString(Constants.enroll, regString)
                regEdit.apply()

                var index = docList.indexOf(currentItem)
                if (index != -1) {
                    currentItem.uploaded = true

                    val updatedItems = docList.toMutableList().apply { this[index] = currentItem }

                    docList.clear()
                    docList.addAll(updatedItems)

                    _listCreated.value = Event(docList)
                }
            }
        )
    }
    fun uploadPicture(pic: File, currentDoc: ListDoc) {
        val file: File = pic
//
//        val filePart: MultipartBody.Part = MultipartBody.Part.createFormData(
//            "file",
//
//          pic.name,
//            RequestBody.create("image/*".toMediaTypeOrNull(), pic)
//        )

        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    enrollRepository.uploadDoc(
                        filePart,
                        HelperUtils.fileName(
                            currentDoc.docName!!.FileName,
                            currentPerson.firstName.toString(),
                            currentPerson.lastName.toString()
                        )
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().loading()
                        }
                        .subscribe({
                            saveUser()

                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().success(
                                it
                            )
                        }, {
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                                it.localizedMessage!!
                            )
                        })
                )
            }
        )
    }

    fun createDocList() {
        job = Coroutines.ioToMainThread(
            {
                user.subscriber?.let {
//                    resources.getString(R.string.front_driver_liscence)
//                    resources.getString(R.string.back_of_drivers_licence)
//                    resources.getString(R.string.social_security_card)
//                    resources.getString(R.string.proof_of_previous_insurance)
//                    resources.getString(R.string.social_security_card)
//                    resources.getString(R.string.marriage_certificate)
//                    resources.getString(R.string.birth_certificate)

                    docList.add(
                        ListDoc(
                            user.subscriber!!.licenseFrontFileId!!.isNotBlank(),
                            DocumentType.FrontDriverLicence,
                            it,
                            1
                        )
                    )
                    docList.add(
                        ListDoc(
                            user.subscriber!!.licenseBackFileId!!.isNotBlank(),
                            DocumentType.BackDriverLicence,
                            it,
                            2
                        )
                    )
                    docList.add(
                        ListDoc(
                            user.subscriber!!.ssnCardFileId!!.isNotBlank(),
                            DocumentType.Ssn,
                            it,
                            3
                        )
                    )
                    if (preferences.getString(Constants.ContinueCoverage, "").equals("true", true)) {
                        docList.add(
                            ListDoc(
                                user.subscriber!!.continueCoverageFileId!!.isNotBlank(),
                                DocumentType.ContinueCoverage,
                                it,
                                4
                            )
                        )
                    }
                }

                if (preferences.getString(Constants.maritalStatus, "").equals(
                        "married",
                        ignoreCase = true
                    )
                ) {
                    if (user.spouse?.firstName?.isBlank() == false) {
                        user.spouse?.let {
                            docList.add(
                                ListDoc(
                                    user.spouse!!.marriageCertificateFileId!!.isNotBlank(),
                                    DocumentType.MarriageCertificate,
                                    it,
                                    5
                                )
                            )

                            docList.add(
                                ListDoc(
                                    user.spouse!!.licenseFrontFileId!!.isNotEmpty(),
                                    DocumentType.FrontDriverLicence,
                                    it,
                                    1
                                )
                            )
                            docList.add(
                                ListDoc(
                                    user.spouse!!.licenseBackFileId!!.isNotEmpty(),
                                    DocumentType.BackDriverLicence,
                                    it,
                                    2
                                )
                            )
                            docList.add(
                                ListDoc(
                                    user.spouse!!.ssnCardFileId!!.isNotEmpty(),
                                    DocumentType.Ssn,
                                    it,
                                    3
                                )
                            )

                            if (StringUtils.endsWithIgnoreCase(
                                    it.healthInsuranceAvailable,
                                    "true"
                                )
                            ) {
//                                docList.add(ListDoc(false, "Front of Insurance Card", it))
//                                docList.add(ListDoc(false, "Back of Insurance Card", it))
                            }
                        }
                    }
                }

                user.children.isNotEmpty().let {
                    for (child in user.children) {
                        docList.add(
                            ListDoc(
                                child.birthCertificateFileId!!.isNotEmpty(),
                                DocumentType.BirtCertificate,
                                child,
                                6,
                                child.ssnOptional
                            )
                        )
                        docList.add(
                            ListDoc(
                                child.ssnCardFileId!!.isNotEmpty(),
                                DocumentType.Ssn,
                                child,
                                3,
                                child.ssnOptional
                            )
                        )
                    }
                }
            },
            {
                _listCreated.value = Event(docList)
            }
        )
    }

    fun checkIfAllComplete(): Boolean {
        return docList.all {
            if (!it.uploaded && it.optional) {
                true
            } else {
                it.uploaded
            }
        }
    }

    fun onPrevClicked() {
        _navigateToBack.value = Event(Unit)
    }
    fun onNextClicked() {
        _navigateToNext.value = Event(Unit)
    }

    fun saveUser() {
        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.enroll, regString)
        regEdit.apply()
    }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
