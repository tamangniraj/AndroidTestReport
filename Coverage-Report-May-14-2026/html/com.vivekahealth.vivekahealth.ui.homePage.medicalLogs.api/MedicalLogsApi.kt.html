package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.api

import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.*
import io.reactivex.Single
import retrofit2.http.*

interface MedicalLogsApi {

    @GET("member/v2/{memberId}/getMedicalLogs")
    fun getAllLogsList(
        @Path("memberId") memberId: String,
        @Query("type") medicalLogType: String,
        @Query(
            "sharedMemberIds"
        ) sharedMemberIds: String,
        @Query("sortAction") sortAction: String,
        @Query("sortBy") sortBy: String
    ): Single<MedicalLogsAll>

    @GET("member/v2/{memberId}/getMedicalLogs")
    fun getClaimsLogsList(
        @Path("memberId") memberId: String,
        @Query("type") medicalLogType: String,
        @Query(
            "sharedMemberIds"
        ) sharedMemberIds: String,
        @Query("sortAction") sortAction: String,
        @Query("sortBy") sortBy: String,
        @Query(
            "claimType"
        ) claimType: String,
        @Query("claimStatus") claimStatus: String,
        @Query("pageSize") pageSize: Int = 30,
        @Query("pageNo") pageNo: Int = 1
    ): Single<MedicalLogsAll>

    @GET("member/{memberId}/getClaimDetail/{claimNumber}")
    fun getClaimsDetails(
        @Path("memberId") memberId: String,
        @Path("claimNumber") claimNumber: String
    ): Single<ClaimsDetail>

    @GET("member/v2/{memberId}/diagnosisHistoryDetails/{icdCode}")
    fun getDiagnosisDetails(
        @Path("memberId") memberId: String,
        @Path("icdCode") claimNumber: String
    ): Single<DiagnosisDetailList>

    @GET("member/v2/{memberId}/getMedicalLogs")
    fun getDiagnosisLogsList(
        @Path("memberId") memberId: String,
        @Query("type") medicalLogType: String,
        @Query(
            "sharedMemberIds"
        ) sharedMemberIds: String,
        @Query("sortAction") sortAction: String,
        @Query("sortBy") sortBy: String,
        @Query("diagnosisFreqRange") diagnosisFreqRange: String,
        @Query("pageSize") pageSize: Int = 30,
        @Query("pageNo") pageNo: Int = 1
    ): Single<MedicalLogsAll>

    @GET("member/v2/{memberId}/getMedicalLogs")
    fun getPresLogsList(
        @Path("memberId") memberId: String,
        @Query("type") medicalLogType: String,
        @Query("sharedMemberIds") sharedMemberIds: String,
        @Query("sortAction") sortAction: String,
        @Query("sortBy") sortBy: String,
        @Query("drugName") drugName: String,
        @Query("pharmacyType") pharmacyType: String,
        @Query("pageSize") pageSize: Int = 30,
        @Query("pageNo") pageNo: Int = 1
    ): Single<MedicalLogsAll>

    @GET("member/v2/{memberId}/getMedicalLogs")
    fun getImmunizationLogsList(
        @Path("memberId") memberId: String,
        @Query("type") medicalLogType: String,
        @Query("sharedMemberIds") sharedMemberIds: String,
        @Query("sortAction") sortAction: String,
        @Query("sortBy") sortBy: String,
        @Query("vaccineName") vaccineType: String,
        @Query("pageSize") pageSize: Int = 30,
        @Query("pageNo") pageNo: Int = 1
    ): Single<MedicalLogsAll>

    @GET("member/{selfId}/getSharedWithMe")
    fun getDependentList(
        @Path("selfId") selfId: String,
        @Query("sharingTypes") sharingTypes: String
    ): Single<SharedMember>

    // Updated API endpoint with correct pagination parameters
    @GET("member/v2/{memberId}/getMedicalLogs")
    fun getAllLogsList(
        @Path("memberId") memberId: String,
        @Query("type") medicalLogType: String,
        @Query("sharedMemberIds") sharedMemberIds: String,
        @Query("sortAction") sortAction: String,
        @Query("sortBy") sortBy: String? = null,
        @Query("pageSize") pageSize: Int = 30,
        @Query("pageNo") pageNo: Int = 1
    ): Single<MedicalLogsAll>
}
