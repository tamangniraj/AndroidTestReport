    val showVerifyIdentity: LiveData<Event<Boolean?>>

    val showResetPasswordPrompt: LiveData<Event<Unit>>

    private val _showFab = MutableLiveData<Event<Boolean>>()
    val showFab: LiveData<Event<Boolean>>
        get() = _showFab

    private val _checkPushPermission = MutableLiveData<Event<Unit>>()
    val checkPushPermission: LiveData<Event<Unit>>
        get() = _checkPushPermission

    private val _checkBiometricPermission = MutableLiveData<Event<Unit>>()
    val checkBiometricPermission: LiveData<Event<Unit>>
        get() = _checkBiometricPermission

    private val _notificationCount = MutableLiveData<Event<Int>>()
    val notificationCount: LiveData<Event<Int>> get() = _notificationCount

    private val _showRequestInCustomerSupport = MutableLiveData<Event<Unit>>()
    val showRequestInCustomerSupport: LiveData<Event<Unit>>
        get() = _showRequestInCustomerSupport

    private val _showPendingSSN = MutableLiveData<Event<Unit>>()
    val showPendingSSN: LiveData<Event<Unit>>
        get() = _showPendingSSN

    var documentType: DocumentType = DocumentType.DL_FRONT
    var sessionId: String? = null

    private val _setLanguage = MutableLiveData<Event<String>>()
    val setLanguage: LiveData<Event<String>>
        get() = _setLanguage

    var savedUserNameEncoded: String = preferences.getString(
        Constants.encodedEmail,
        ""
    ).toString()

    var savedPasswordEncoded: String = preferences.getString(
        Constants.encodedPassword,
        ""
    ).toString()

    fun loadMemberInfo() {
        viewModelScope.launch {
            getMemberInfo()
        }
    }

    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }

    var email: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.email, null)
        }

    fun saveLanguageInPreference(memberId: String, language: String) {
        Constants.CURRENT_LANGUAGE = language
        val editor = preferences.edit()
        editor.putString(
            "$memberId${Constants.LanguagePref}",
            language
        )
        editor.apply()
    }

    private val _navFlag = MutableLiveData<Event<Unit>>()
    val navFlag: LiveData<Event<Unit>>
        get() = _navFlag

    var flagFindCare: Boolean = false

    fun saveLanguageToServer(language: String) {
        compositeDisposable.add(
            guestHomeRepository.setLanguagePreference(language)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }.subscribe({
                    loading = false
                    _setLanguage.value = Event(language)
                }, {
                    loading = false
                })
        )
    }

    fun getMemberInfo() {
        compositeDisposable.add(
            guestHomeRepository.getGuestInfo()
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }.subscribe({
                    it?.let {
                        if (HelperUtils.decrypt(savedUserNameEncoded).isEmpty()) {
                            val encodedEmail = HelperUtils.encrypt(it.email)
                            preferences.edit().putString(Constants.encodedEmail, encodedEmail).apply()
                            savedUserNameEncoded =
                                preferences.getString(Constants.encodedEmail, "").toString()
                        }
                        val locale = DateUtils.getCurrentLocale(resources)

                        if (it.languagePreference != null) {
                            if (locale.language != it.languagePreference) {
                                singleClick {
                                    _setLanguage.value = Event(it.languagePreference ?: "en")
                                }
                            }
                            saveLanguageInPreference(it.id, it.languagePreference ?: "en")
                        }

                        guestHomeRepository.insertGuestInfo(it)
                        _memberInfo.value = Event(it)
                        email = it.email
                        preferences.edit().putString(Constants.MEMBER_ID, it.id).apply()
                        // request push notification permission
                        // request notification -> request biometric -> check magic login -> checkIfVerifyIdentityNeedToShow
                        checkForPushPermission()
                    }
                    _showFab.value = Event(true)
                    loading = false
                }, {
                    if ("Unable to resolve host" in it.message.toString()) {
                        val decryptedEmail = HelperUtils.decrypt(savedUserNameEncoded)
                        email = decryptedEmail
                    }
                    _showFab.value = Event(true)
                    loading = false
                })
        )
    }

    fun checkIfVerifyIdentityNeedToShow(guestUserInfo: GuestUserInfo) {
        guestUserInfo.id.let {
            // First priority: Check for pending SSN on first login only
            val shouldShowSSN = checkPendingSSNOnFirstLogin(guestUserInfo)

            // If SSN flow is triggered, don't show verify identity bottom sheet
            if (shouldShowSSN) {
                return
            }

            var requestInCustomerSupport: Boolean = false
            memberInfo.value?.peekContent()?.let { guestUserInfo ->
                requestInCustomerSupport = guestUserInfo.requestInCustomerSupport
            }
            var showVerifyIdentity = preferences.getBoolean(
                Constants.VERIFY_IDENTITY_POPUP_SHOWN,
                false
            )

            if (!showVerifyIdentity && !requestInCustomerSupport) {
                // show bottom-sheet only if request is not in customer support
                showVerifyIdentityBottomSheet()
            }
            if (requestInCustomerSupport) {
                // if request in customer support just show FAB
                showVerifyIdentity = true
            }
            _showFab.value = Event(showVerifyIdentity)
        }
    }

    private fun checkPendingSSNOnFirstLogin(guestUserInfo: GuestUserInfo): Boolean {
        // Check if SSN flow has already been shown this session
        if (Constants.hasShownSSNFlowThisSession) {
            return false
        }

        if (guestUserInfo.pendingIdentityVerification) {
            // Mark as shown for this session
            sessionId = guestUserInfo.identificationSessionId
            Constants.hasShownSSNFlowThisSession = true
            _showPendingSSN.value = Event(Unit)
            return true
        }
        return false
    }

    public fun showVerifyIdentityBottomSheet() {
        singleClick {
            _showVerifyIdentity.value = Event(true)
        }
    }

    fun setVerifyIdentityShown() {
        // change this to guest id
        preferences.edit().putBoolean(
            Constants.VERIFY_IDENTITY_POPUP_SHOWN,
            true
        ).apply()
    }

    private val _clientDetails = MutableLiveData<Event<GuestClientDetails>>()
    val clientDetails: LiveData<Event<GuestClientDetails>>
        get() = _clientDetails

    fun loadContent() {
        compositeDisposable.add(
            guestHomeRepository.getGuestClientDetails().subscribeOn(
                Schedulers.io()
            )
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({
                    loading = false
                    _clientDetails.value = Event(it)
                    if (it.clientId.isNullOrBlank()) { /*THIS IS INTENTIONAL*/
                        /*For guest whose client we have not identified clientId is null*/
                        /*Since clientId is primary key for DAO we check for null and assign 0*/
                        it.clientId = "0"
                    }
                    flagFindCare = it.showHtmlBasedFindADoc
                    _navFlag.value = Event(Unit)

                    guestHomeRepository.insertGuestClientDetails(it)
                }, {
                    loading = false
                })
        )
    }

    fun clearMagicLogin() {
        preferences.edit().putString(Constants.MAGIC_TEMP_TOKEN, "").apply()
        preferences.edit().putBoolean(Constants.IS_MAGIC_LOGIN, false).apply()
    }

    private fun checkForPushPermission() {
        _checkPushPermission.value = Event(Unit)
    }

    fun checkForBiometricPermission() {
        proceedWithGuestLogin()
    }

    fun proceedWithGuestLogin() {
        if (preferences.getBoolean(Constants.IS_MAGIC_LOGIN, false)) {
            _showResetPasswordPrompt.value = Event(Unit)
        } else {
            memberInfo.value?.peekContent()?.let {
                checkIfVerifyIdentityNeedToShow(it)
            }
        }
    }

    fun updateFcmToken() {
        FirebaseMessaging.getInstance().token.addOnCompleteListener(
            OnCompleteListener { task ->
                if (!task.isSuccessful) {
                    return@OnCompleteListener
                }
                val token = task.result
                preferences.getString("FCM_TOKEN", "")

                memberInfo.value?.peekContent()?.id?.let { guestId ->
                    compositeDisposable.add(
                        guestHomeRepository.updateGuestFcmToken(guestId, token, true)
                            .observeOn(AndroidSchedulers.mainThread())
                            .subscribeOn(Schedulers.io())
                            .subscribe({ onSuccess ->
                            }, { onError ->
                            })
                    )
                }
            }
        )
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun saveBiometricPreference() {
        preferences.getString(Constants.USER_EMAIL, null)?.let { email ->
            preferences.edit().putBoolean(email + Constants.BIO_PERMISSION_REQUESTED, true).apply()
            preferences.edit().putString(Constants.emailLogin, email).apply()
        }

//            preferences.edit().putString(Constants.emailLogin, email).apply()
//        }
    }

    fun getUnreadNotificationCount() {
        compositeDisposable.add(
            guestHomeRepository.getGuestUnreadNotificationCount()
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .subscribe({
                    _notificationCount.value = Event(it.count)
                }, {
                    Timber.e(it.toString())
                })
        )
    }

    fun showPendingSsn() {
        _showPendingSSN.value = Event(Unit)
    }

    fun showRequestInCustomerSupport() {
        _showRequestInCustomerSupport.value = Event(Unit)
    }

    private var debounceJob: Job? = null
    private var isDebouncing = false
    private val debounceInterval = 500L // Set your debounce interval in milliseconds

    fun singleClick(action: () -> Unit) {
        if (isDebouncing) return
        action()
        isDebouncing = true
        debounceJob?.cancel()
        debounceJob = viewModelScope.launch {
            delay(debounceInterval)
            isDebouncing = false
        }
    }

    fun clearShowBottomSheet() {
        _showVerifyIdentity.value = Event(false)
    }
}
