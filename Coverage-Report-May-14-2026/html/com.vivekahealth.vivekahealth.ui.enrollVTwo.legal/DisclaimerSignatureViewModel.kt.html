package com.vivekahealth.vivekahealth.ui.enrollVTwo.legal

import android.content.SharedPreferences
import android.graphics.Bitmap
import android.view.View
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.enroll.DocumentUploadSuccessEnroll
import com.vivekahealth.vivekahealth.data.remote.enroll.OpenEnrollmentData
import com.vivekahealth.vivekahealth.data.remote.login.GenericErrorResponse
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.enrollVTwo.UploadedDocument

import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.launch
import okhttp3.MediaType.Companion.toMediaTypeOrNull
import okhttp3.MultipartBody
import okhttp3.RequestBody.Companion.asRequestBody
import java.io.File
import javax.inject.Inject

class DisclaimerSignatureViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    val homeRepository: HomeRepository,
    val guestHomeRepository: GuestHomeRepository
) : ViewModel(), Observable {

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack

    private val _navigateToNext = MutableLiveData<Event<Unit>>()
    val navigateToNext: LiveData<Event<Unit>>
        get() = _navigateToNext

    var gson = Gson()
    private lateinit var job: Job
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    val showError: LiveData<APIResponse<DataResponse>>
        get() = _showError
    private val _showError = MutableLiveData<APIResponse<DataResponse>>()

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun onPrevClicked() {
        singleClick {
            _navigateToBack.value = Event(Unit)
        }
    }

    fun onNextClicked() {
        singleClick {
            _navigateToNext.value = Event(Unit) // Process the first click immediately
        }
    }

    private var debounceJob: Job? = null
    private var isDebouncing = false
    private val debounceInterval = 500L // Set your debounce interval in milliseconds

    fun singleClick(action: () -> Unit) {
        if (isDebouncing) return
        action()
        isDebouncing = true
        debounceJob?.cancel()
        debounceJob = viewModelScope.launch {
            delay(debounceInterval)
            isDebouncing = false
        }
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
    var checkBox: Boolean = false
    var signed: Boolean = false

    private val _clearClicked = MutableLiveData<Event<Unit>>()
    val clearClicked: LiveData<Event<Unit>>
        get() = _clearClicked

    private val _check = MutableLiveData<Event<Boolean>>()
    val check: LiveData<Event<Boolean>>
        get() = _check

    var checkBoxErrorVisibility = View.GONE
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyChange(this, BR.checkBoxErrorVisibility)
        }

    fun ssnRadio() {
        checkBox = !checkBox

        _check.value = Event(checkBox)
        checkBoxErrorVisibility = if (signed) {
            if (checkBox) View.GONE else View.VISIBLE
        } else {
            View.GONE
        }
    }

    lateinit var signatureBitmap: Bitmap

    var currentDate: String = DateUtils.getCurrentLocalDateMonthName().toString()

    var currentNameFirst: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.currentNameFirst, null)
        }
    var currentNameLast: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.currentNameLast, null)
        }

    private val _navigateToUpload = MutableLiveData<APIResponse<DocumentUploadSuccessEnroll>>()
    val navigateToUpload: LiveData<APIResponse<DocumentUploadSuccessEnroll>>
        get() = _navigateToUpload

    fun uploadSign(pic: File) {
        val file: File = File(pic?.path)

        val filePart: MultipartBody.Part = MultipartBody.Part.createFormData(
            "file",
            file.name,
            file.asRequestBody("image/*".toMediaTypeOrNull())
        )

        viewModelScope.launch {
            compositeDisposable.add(
                homeRepository.uploadDoc(
                    filePart,
                    HelperUtils.fileName(Constants.Signature, currentNameFirst, currentNameLast),
                    "52"
                )
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().loading()
                    }
                    .subscribe({ response ->
                        _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().success(
                            response
                        )
                    }, {
                        _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                            it.localizedMessage!!
                        )
                    })
            )
        }
    }

    fun uploadSignOpen(pic: File, openEnrollmentData: OpenEnrollmentData) {
        val file: File = File(pic?.path)

        val filePart: MultipartBody.Part = MultipartBody.Part.createFormData(
            "file",
            file.name,
            file.asRequestBody("image/*".toMediaTypeOrNull())
        )

        viewModelScope.launch {
            compositeDisposable.add(
                guestHomeRepository.uploadOpenFile(
                    filePart,
                    52.toString()
                )
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().loading()
                    }
                    .subscribe({
                        openEnrollmentData.subscriber?.signatureFileId = it.id
                        val list = openEnrollmentData.subscriber?.uploadedDocuments as ArrayList<UploadedDocument>
                        list.add(UploadedDocument(52, it.id?.toInt() ?: 0))
                        openEnrollmentData.subscriber?.uploadedDocuments = list

                        callOpenEnrollment(openEnrollmentData)
                    }, {
                        _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                            it.localizedMessage!!
                        )
                    })
            )
        }
    }

    fun callOpenEnrollment(enrollment: OpenEnrollmentData) {
        viewModelScope.launch {
            compositeDisposable.add(
                guestHomeRepository.openEnrollMember(
                    enrollment
                )
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().loading()
                    }
                    .subscribe({
                        val body = it.body()
                        if (it.isSuccessful && body != null) {
                            if (it.body()?.code == 200) {
                                _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().success(
                                    body
                                )
                            } else {
                                _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                                    it.body()?.message ?: Constants.APOLOGY_MESSAGE
                                )
                            }
                        } else {
                            val errorMessage = try {
                                val errorResponse: GenericErrorResponse = gson.fromJson(
                                    it.errorBody()?.string(),
                                    GenericErrorResponse::class.java
                                )
                                errorResponse.getErrorMessage()
                            } catch (e: Exception) {
                                "Enrollment submission failed. Please try again."
                            }
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                                errorMessage
                            )
                        }
                    }, {
                        _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                            it.localizedMessage ?: "Enrollment submission failed. Please try again."
                        )
                    })
            )
        }
    }

    fun clearClicked() {
        _clearClicked.value = Event(Unit)
    }
}
