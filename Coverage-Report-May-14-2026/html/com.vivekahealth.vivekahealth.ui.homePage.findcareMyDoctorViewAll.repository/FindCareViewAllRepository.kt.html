package com.vivekahealth.vivekahealth.ui.homePage.findcareMyDoctorViewAll.repository

import com.vivekahealth.vivekahealth.data.local.home.MemberInfoDao
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.data.remote.home.findCare.doctorModelsNew.DoctorResponseNew
import com.vivekahealth.vivekahealth.ui.homePage.findcareMyDoctorViewAll.service.FindCareViewAllService
import io.reactivex.Single
import javax.inject.Inject
import javax.inject.Singleton
import kotlinx.coroutines.flow.Flow

/**
 * Repository that handles User objects.
 */
@Singleton
class FindCareViewAllRepository @Inject constructor(
    private val findCareService: FindCareViewAllService,
    private val dao: MemberInfoDao
) {

    fun getDoctorList(
        memberID: String,
        latitude: String,
        longitude: String,
        provider_type: String,
        radius: String
    ): Single<DoctorResponseNew> = findCareService.getFindCareViewAllApi().getDoctorList(
        memberID,
        latitude,
        longitude,
        provider_type,
        radius
    )

    fun getMemberFromDb(): Flow<List<MemberInfo>> {
        return dao.getAll()
    }

    fun getMemberSid(memberid: String): Flow<MemberInfo> {
        return dao.getMemberInfoLiveData(memberid)
    }
}
