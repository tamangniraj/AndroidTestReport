package com.vivekahealth.vivekahealth.data.remote.home.medicalLogs

import com.vivekahealth.vivekahealth.data.remote.home.*

data class MedicalLogsAll(
    val claimsList: List<ClaimsQuickView> ? = null,
    val diagnosisList: List<DiagnosisHistoryQuickView> ? = null,
    val labInfoList: List<LabResultQuickView> ? = null,
    val prescriptionList: List<PrescriptionQuickView>? = null,
    val immunizationList: List<ImmunizationQuickView>? = null,
    val availPrescriptionFilters: ArrayList<AvailPrescriptionFiltersItem> ? = null,
    val availImmunizationFilters: ArrayList<AvailImmunizationFiltersItem> ? = null,
    val paging: PagingInfo ? = null,
    val medicalLogMetaData: MedicalLogsMetaData ? = null
)

data class PagingInfo(
    val pageSize: Int,
    val currentPage: Int,
    val totalPages: Int,
    val totalHits: Int
) {
    val hasNextPage: Boolean
        get() = currentPage < totalPages
}

data class MedicalLogsMetaData(
    val maximumFrequency: Int? = null,
    val minimumFrequency: Int? = null,
    val earliestDate: String ? = null,
    val type: String ? = null,
    val seriesCompleted: Int ? = null,
    val totalRecords: Int ? = null
)
