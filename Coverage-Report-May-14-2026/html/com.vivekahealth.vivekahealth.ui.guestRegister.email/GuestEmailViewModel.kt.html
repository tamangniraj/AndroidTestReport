package com.vivekahealth.vivekahealth.ui.guestRegister.email

class GuestEmailViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val emailRepository: EmailRepository,
    private val loginRepository: LoginRepository,
    private val codeRepository: PhoneCodeRepository,
    private val recaptchaClient: RecaptchaClient
) : ViewModel(), Observable {

    var gson = Gson()
    var user = RegisterUser()
    private lateinit var job: Job
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _logInSuccess = MutableLiveData<APILoginResponse<Response<AuthToken>>>()
    val logInSuccess: LiveData<APILoginResponse<Response<AuthToken>>>
        get() = _logInSuccess
    private val _logInFail = MutableLiveData<Event<String>>()
    val logInFail: LiveData<Event<String>>
        get() = _logInFail
    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack
    private val _navigateToPassword = MutableLiveData<APIResponseState>()
    val navigateToPassword: LiveData<APIResponseState>
        get() = _navigateToPassword

    private val _displaySuggestionError = MutableLiveData<String?>()
    val displaySuggestionError: LiveData<String?>
        get() = _displaySuggestionError

    private val _suggestion = MutableLiveData<String?>()
    val suggestion: LiveData<String?>
        get() = _suggestion

    private val _emailChanged = MutableLiveData<Event<String>>()

    val emailChanged: LiveData<Event<String>>
        get() = _emailChanged

    val showError: LiveData<APIResponse<DataResponse>>
        get() = _showError
    private val _showError = MutableLiveData<APIResponse<DataResponse>>()

    private val _navigateSuccess = MutableLiveData<Event<Unit>>()
    val navigateSuccess: LiveData<Event<Unit>>
        get() = _navigateSuccess

    var email: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.email, null)
        }

    fun onPrevClicked() {
        _navigateToBack.value = Event(Unit)
    }

    private val _alternativeApp = MutableLiveData<Event<ErrorResponse>>()
    val alternativeApp: LiveData<Event<ErrorResponse>>
        get() = _alternativeApp

    var registrationId: String = ""

    var ssnOrMember: Boolean = true
    var firstName: String = ""
    var lastName: String = ""
    var dob: String = ""
    var password: String = ""
    var passwordConfirm: String = ""
    var gender: String = ""
    var mobile: String = ""
    var usernameType: String = ""
    var countryCode: String = ""
    var electronicDocuments: String = ""
    var overrideOldRegistration: String = ""
    var resendVerification: String = ""
    var tempUUID: String = ""

    @get:Bindable
    val emailTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
            _emailChanged.value = Event(s.toString())
        }
    }

    fun loginClicked(password: String) {
        compositeDisposable.add(

            loginRepository.getLogin(
                userName = email,
                password = password,
                user.tempUUID.toString(),
                false
            ).subscribeOn(
                Schedulers.io()
            )
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _logInSuccess.value = APILoginResponse<Response<AuthToken>>().loading()
                }
                .subscribe({
                    _logInSuccess.value = APILoginResponse<Response<AuthToken>>().success(it)
                }, {
                    _logInFail.value = Event(it.message.toString())
                })
        )
    }

    var lastSuggestEmail: String ? = null
    var lastSuggestion: String ? = null
    var continueAsItIs = false

    fun onNextClicked() {
        _displaySuggestionError.value = null
        lastSuggestion = null
        _suggestion.value = null
        lastSuggestEmail = null

        if (continueAsItIs) {
            continuePassword()
            return
        }

        compositeDisposable.add(
            Single.defer {
                // bypass email verification if not changed the email in step1
                val isEmailChanged = initialEmail != email
                emailRepository.validateEmail(email, isEmailChanged, registrationId)
            }.subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    // You can handle any subscription logic here, like showing a loading spinner
                    _navigateToPassword.value = APIResponseState.Loading
                }.subscribe({
                        verifyResponse ->

                    if (verifyResponse.errorBody() != null) {
                        try {
                            val errorResponse: GenericErrorResponse = gson.fromJson(
                                verifyResponse.errorBody()?.string(),
                                GenericErrorResponse::class.java
                            )

                            if (errorResponse.errorType == "EMAIL_ALREADY_SENT") {
                                _navigateSuccess.value = Event(Unit)
                            } else if (errorResponse.errorType == "ALTERNATE_APP_AVAILABLE") {
                                _alternativeApp.value = Event(
                                    ErrorResponse(
                                        android = errorResponse.otherDetails?.android.toString(),
                                        alternativeAppAvailable = errorResponse.otherDetails?.alternativeAppAvailable,
                                        appName = errorResponse.otherDetails?.appName ?: "",
                                        error_description = errorResponse.getErrorMessage()
                                    )
                                )
                            } else {
                                var errDes = errorResponse.getErrorMessage()

                                if (errDes == "REGISTERED_AND_VERIFIED") {
                                    errDes =
                                        "The user associated with this email address is already registered."
                                }
                                _navigateToPassword.value = APIResponseState.Error(
                                    errDes
                                )
                            }
                        } catch (exception: Exception) {
                            _navigateToPassword.value = APIResponseState.Error(
                                Constants.APOLOGY_MESSAGE
                            )
                        }
                    } else if (verifyResponse.body() != null) {
                        // send the result
                        verifyResponse.body()?.let {
                            // if response has suggestion display suggestion
                            if (it.suggestion != null && it.suggestion.isNotEmpty()) {
                                _suggestion.value = it.suggestion
                                lastSuggestion = it.suggestion
                            } else {
                                // if valid and ready to move ahead then move ahead
                                if (it.isValid && !it.accountExists) {
                                    continuePassword()
                                    lastSuggestEmail = null
                                } else {
                                    // else display the error message
                                    lastSuggestEmail = email
                                    _displaySuggestionError.value = it.errorMessage ?: "Unexpected error"
                                }
                            }
                        }
                    }
                }, {
                        error ->

                    val errorMessage = when {
                        error.message?.contains("Unable to resolve host") == true ||
                            error.message?.contains("Failed to connect to") == true ||
                            error?.message?.contains(
                            "Software caused connection abort"
                        ) == true ||
                            error?.message.equals(
                                "timeout"
                            )
                        ->
                            APOLOGY_CHECK_CONNECTION
                        else -> error.message ?: "Registration failed. Please try again."
                    }

                    _navigateToPassword.value = APIResponseState.Error(
                        errorMessage
                    )
                })
        )
    }

    fun saveTokens(data: AuthToken) {
        preferences.edit().putString(Constants.ACCESS_TOKEN, data.access_token).apply()
        preferences.edit().putString(Constants.TOKEN_TYPE, data.token_type).apply()
        preferences.edit().putString(Constants.EXPIRES_IN, data.expires_in).apply()
        preferences.edit().putString(Constants.SCOPE, data.scope).apply()
    }

    var initialEmail = ""

    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.register, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.register, ""),
                        RegisterUser::class.java
                    )
                }
            },
            {
                initialEmail = user.email.toString()
                _emailChanged.value = Event(user.email.toString())
            }
        )
    }

    fun continuePassword() {
        _navigateToPassword.value = APIResponseState.SuccessGeneric(
            DataResponse(
                code = "200",
                errorCode = "",
                reason = "NOT_REGISTERED",
                otherDetails = OtherDetails(sessionId = registrationId)
            )
        )
    }

    fun saveEmail() {
        user.email = email
        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.register, regString)
        regEdit.apply()
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
