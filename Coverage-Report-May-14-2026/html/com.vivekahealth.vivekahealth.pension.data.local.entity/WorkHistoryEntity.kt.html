package com.vivekahealth.vivekahealth.pension.data.local.entity

import androidx.room.Entity
import androidx.room.PrimaryKey
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.pension.data.model.WorkHistorySummary

@Entity(tableName = "work_history")
data class WorkHistoryEntity(
    @PrimaryKey
    val memberId: String,

    val dataJson: String,

    val fetchedAt: Long = System.currentTimeMillis()
) {
    fun toWorkHistorySummary(): WorkHistorySummary? {
        return try {
            Gson().fromJson(dataJson, WorkHistorySummary::class.java)
        } catch (e: Exception) {
            null
        }
    }

    companion object {
        fun fromWorkHistorySummary(
            memberId: String,
            data: WorkHistorySummary
        ): WorkHistoryEntity {
            val json = Gson().toJson(data)
            return WorkHistoryEntity(
                memberId = memberId,
                dataJson = json,
                fetchedAt = System.currentTimeMillis()
            )
        }
    }
}
