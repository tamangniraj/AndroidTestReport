package com.vivekahealth.vivekahealth.pension.data.local.entity

import androidx.room.Entity
import androidx.room.PrimaryKey
import com.vivekahealth.vivekahealth.pension.data.model.SummaryImpactData

@Entity(tableName = "summary_impact")
data class SummaryImpactEntity(
    @PrimaryKey
    val memberId: String,
    val totalCreditsEarned: Double,
    val creditsKept: Double,
    val creditsLost: Double,
    val lastUpdated: String,
    val isEligibleForPensionBenefits: Boolean ? = null,
    val showPensionDetails: Boolean? = null
) {
    fun toSummaryImpactData(): SummaryImpactData {
        return SummaryImpactData(
            totalCreditsEarned = totalCreditsEarned,
            creditsKept = creditsKept,
            creditsLost = creditsLost,
            date = lastUpdated,
            isEligibleForPensionBenefits = isEligibleForPensionBenefits,
            showPensionDetails = showPensionDetails
        )
    }

    companion object {
        fun fromSummaryImpactData(
            memberId: String,
            data: SummaryImpactData
        ): SummaryImpactEntity {
            return SummaryImpactEntity(
                memberId = memberId,
                totalCreditsEarned = data.totalCreditsEarned,
                creditsKept = data.creditsKept,
                creditsLost = data.creditsLost,
                lastUpdated = data.date,
                isEligibleForPensionBenefits = data.isEligibleForPensionBenefits,
                showPensionDetails = data.showPensionDetails
            )
        }
    }
}
