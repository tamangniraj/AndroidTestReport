package com.vivekahealth.vivekahealth.ui.homePage.documents.repository

import com.vivekahealth.vivekahealth.data.remote.enroll.DocumentUploadSuccessEnroll
import com.vivekahealth.vivekahealth.data.remote.home.*
import com.vivekahealth.vivekahealth.ui.homePage.documents.service.DocumentService
import com.vivekahealth.vivekahealth.utils.thread.AppExecutors
import io.reactivex.Single
import java.util.*
import javax.inject.Inject
import javax.inject.Singleton
import okhttp3.MultipartBody
import okhttp3.ResponseBody

/**
 * Repository that handles User objects.
 */
@Singleton
class DocumentRepository @Inject constructor(
    private val appExecutors: AppExecutors,
    private val documentService: DocumentService
) {

    fun getType(clientId: String): Single<DocumentTypeList> = documentService.getDocumentsApi().getDocType(
        clientId
    )

    fun getDocumentListSearched(
        memberId: String,
        sharedMemberIds: String,
        documentsType: String,
        documentCategoryId: String,
        viewed: String,
        createdDateSort: String,
        search: String
    ): Single<DocumentList> = documentService.getDocumentsApi().getDocumentListSearched(
        memberId,
        sharedMemberIds,
        documentsType,
        documentCategoryId,
        viewed,
        createdDateSort,
        search
    )
    fun getDocumentList(
        memberId: String,
        referencedMemberIds: String,
        documentsType: String,
        documentCategoryId: String,
        viewed: String,
        createdDateSort: String,
        search: String
    ): Single<DocumentList> = documentService.getDocumentsApi().getDocumentList(
        memberId,
        referencedMemberIds,
        documentsType,
        documentCategoryId,
        viewed,
        createdDateSort,
        search
    )

    fun uploadDoc(
        memberID: String,
        filePart: MultipartBody.Part,
        documentType: Int,
        documentCategory: Int,
        documentName: String,
        referencedMemberId: Int,
        longitude: Double?,
        latitude: Double?,
        requestId: String
    ): Single<DocumentUploadSuccessEnroll> = documentService.getDocumentsApi().getDoc(memberID, filePart, documentType, documentCategory, documentName, referencedMemberId, longitude, latitude, requestId)

    fun getPendingDocument(memberId: String): Single<PendingDocument> = documentService.getDocumentsApi().getPendingDocument(
        memberId
    )

    fun getDocumentOld(memberId: String, sharedMemberIds: String, documentFilterType: Int): Single<DocumentList> = documentService.getDocumentsApi().getDocumentOld(
        memberId,
        sharedMemberIds,
        documentFilterType
    )

    fun logDocumentView(
        memberID: String,
        documentCategoryId: String,
        documentsType: Int,
        referencedMemberId: String
    ): Single<ResponseBody> = documentService.getDocumentsApi().setDocumentViewedLog(
        memberID,
        documentCategoryId,
        documentsType,
        referencedMemberId
    )

    fun getPDF(url: String): Single<ResponseBody> = documentService.getDocumentsApi().getPDF(url)
}
