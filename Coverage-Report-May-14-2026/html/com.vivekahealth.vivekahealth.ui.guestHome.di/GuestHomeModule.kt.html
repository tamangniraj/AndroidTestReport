package com.vivekahealth.vivekahealth.ui.guestHome.di

import com.vivekahealth.vivekahealth.ui.guestHome.GuestBiometricRequestDialog
import com.vivekahealth.vivekahealth.ui.guestHome.GuestHomeFragment
import com.vivekahealth.vivekahealth.ui.guestHome.VerifyIdentityBottomSheet
import com.vivekahealth.vivekahealth.ui.guestHome.VerifyIdentityDialog
import dagger.Module
import dagger.android.ContributesAndroidInjector

@Module
abstract class GuestHomeModule {
    @ContributesAndroidInjector
    abstract fun contributeGuestHomeFragment(): GuestHomeFragment

    @ContributesAndroidInjector
    abstract fun contributeVerifyIdentityBottomSheet(): VerifyIdentityBottomSheet

    @ContributesAndroidInjector
    abstract fun contributeVerifyIdentityDialog(): VerifyIdentityDialog

    @ContributesAndroidInjector
    abstract fun contributeGuestBiometricRequestDialog(): GuestBiometricRequestDialog
}
