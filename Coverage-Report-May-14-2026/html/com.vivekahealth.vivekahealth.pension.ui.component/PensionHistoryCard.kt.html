package com.vivekahealth.vivekahealth.pension.ui.component

import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.res.painterResource
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import com.vivekahealth.vivekahealth.pension.data.model.WorkStatus
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.pension.data.model.YearGroup
import com.vivekahealth.vivekahealth.pension.ui.tagged

@Composable
fun PensionHistoryCard(
    data: YearGroup,
    modifier: Modifier = Modifier
) {
    Card(

        modifier = modifier
            .fillMaxWidth()
            .tagged("Pension History Card ${data.period}"),
        colors = CardDefaults.cardColors(
            containerColor = Color.White
        ),
        elevation = CardDefaults.cardElevation(defaultElevation = 0.dp)
    ) {
        Column(
            modifier = Modifier.padding(16.dp)
        ) {
            // Header with icon and period
            Row(
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier.fillMaxWidth()
            ) {
                Icon(
                    painter = when (WorkStatus.fromGroupType(data.groupType)) {
                        WorkStatus.VESTED -> painterResource(R.drawable.ic_success_pension)
                        WorkStatus.BREAK -> painterResource(R.drawable.ic_warning_pension)
                        WorkStatus.FORFEITED -> painterResource(R.drawable.ic_failed_pension)
                        else -> {
                            painterResource(R.drawable.ic_success_pension)
                        }
                    },
                    contentDescription = null,
                    tint = when (WorkStatus.fromGroupType(data.groupType)) {
                        WorkStatus.VESTED -> Color(0xFF00D671)
                        WorkStatus.BREAK -> Color.Black
                        WorkStatus.FORFEITED -> Color(0xFFEB4848)
                        else -> {
                            Color(0xFF00D671)
                        }
                    },
                    modifier = Modifier.size(20.dp)
                )

                Spacer(modifier = Modifier.width(8.dp))

                Text(
                    text = data.period + (
                        " - Break Years".takeIf {
                            WorkStatus.fromGroupType(
                                data.groupType
                            ) == WorkStatus.BREAK
                        } ?: ""
                        ),
                    style = MaterialTheme.typography.titleMedium,
                    fontWeight = FontWeight.Medium,
                    color = when (WorkStatus.fromGroupType(data.groupType)) {
                        WorkStatus.VESTED -> Color(0xFF00D671)
                        WorkStatus.BREAK -> Color.Black
                        WorkStatus.FORFEITED -> Color(0xFFF44336)
                        else -> Color(0xFF00D671)
                    }
                )
            }

            Spacer(modifier = Modifier.height(16.dp))

            // Credits and Vesting Stats
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Column(
                    modifier = Modifier
                        .weight(1f)
                        .tagged("Credits Earned Column ${data.period}")
                ) {
                    Text(
                        text = stringResource(R.string.credits_earned),
                        style = MaterialTheme.typography.bodySmall,
                        color = Color(0xFF757575),
                        fontSize = 16.sp,
                        modifier = Modifier.tagged("Credits Earned Label")
                    )
                    Spacer(modifier = Modifier.height(8.dp))
                    Text(
                        text = formatCreditValue(data.totalCredits),
                        style = MaterialTheme.typography.headlineSmall,
                        fontWeight = FontWeight.Bold,
                        color = Color(0xFF212121),
                        fontSize = 16.sp,
                        modifier = Modifier.tagged("Credits Earned Value")
                    )
                }

                Column(
                    modifier = Modifier
                        .weight(1f)
                        .tagged("Vesting Chips Column ${data.period}"),
                    horizontalAlignment = Alignment.Start
                ) {
                    Text(
                        text = stringResource(R.string.vesting_chips),
                        style = MaterialTheme.typography.bodySmall,
                        color = Color(0xFF757575),
                        fontSize = 16.sp,
                        modifier = Modifier.tagged("Vesting Chips Label")

                    )
                    Spacer(modifier = Modifier.height(8.dp))

                    Text(
                        text = formatCreditValue(data.totalVesting),
                        style = MaterialTheme.typography.headlineSmall,
                        fontWeight = FontWeight.Bold,
                        color = Color(0xFF212121),
                        fontSize = 16.sp,
                        modifier = Modifier.tagged("Vesting Chips Value")

                    )
                }
            }

            val notes = data.note

            // Bottom section (description or additional info)
            notes?.let {
                Spacer(modifier = Modifier.height(12.dp))

                Card(
                    colors = CardDefaults.cardColors(
                        containerColor = when (WorkStatus.fromGroupType(data.groupType)) {
                            WorkStatus.VESTED -> Color(0xFFE0FFF0)
                            WorkStatus.BREAK -> Color(0xFFF8F8F8)
                            WorkStatus.FORFEITED -> Color(0x20EB4848)
                            else -> Color(0xFFE0FFF0)
                        }
                    ),
                    modifier = Modifier.fillMaxWidth()
                ) {
                    Column(
                        modifier = Modifier.padding(12.dp)
                    ) {
                        Text(
                            text = notes.title ?: "",
                            style = MaterialTheme.typography.bodyMedium,
                            fontWeight = FontWeight.Medium,
                            color = Color.Black
                        )

                        if (notes.text?.isNotEmpty() == true) {
                            Spacer(modifier = Modifier.height(4.dp))
                            Text(
                                text = notes.text,
                                style = MaterialTheme.typography.bodySmall,
                                color = Color(0xFF757575)
                            )
                        }
                    }
                }
            }
        }
    }
}
