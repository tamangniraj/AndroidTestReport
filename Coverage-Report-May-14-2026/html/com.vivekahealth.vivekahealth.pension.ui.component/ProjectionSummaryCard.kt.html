package com.vivekahealth.vivekahealth.pension.ui.component

import android.annotation.SuppressLint
import androidx.compose.foundation.background
import androidx.compose.foundation.border
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.outlined.Info
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.res.colorResource
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.tooling.preview.Preview
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.pension.ui.tagged
import com.vivekahealth.vivekahealth.pension.ui.viewmodel.PensionViewModel

@SuppressLint("DefaultLocale")
@Composable
fun ProjectionSummaryCard(
    yearsToRetirement: Int?,
    additionalCreditsExpected: Double?,
    totalProjectedCredits: Double?,
    infoMessage: String? = null,
    errorMessage: String? = null,
    isCalculateEnabled: Boolean = true,
    isLoading: Boolean = false,
    onCalculateClick: () -> Unit,
    viewModel: PensionViewModel? = null
) {
    Column(
        modifier = Modifier
            .fillMaxWidth()
            .background(Color.White)
            .padding(16.dp)
            .tagged("Projection Summary Card")
    ) {
        Column(
            Modifier
                .background(
                    colorResource(R.color.primaryUltraLight),
                    shape = RoundedCornerShape(8.dp)
                )
                .border(
                    1.dp,
                    colorResource(R.color.primaryLight),
                    shape = RoundedCornerShape(8.dp)
                )
                .padding(12.dp)
        ) {
// Header
            Text(
                text = stringResource(R.string.projection_summary),
                style = MaterialTheme.typography.titleMedium,
                fontWeight = FontWeight.SemiBold,
                color = Color.Black,
                modifier = Modifier.padding(bottom = 16.dp)
            )

            var formattedYear = ""

            yearsToRetirement?.let {
                val years = it.coerceAtLeast(0)
                formattedYear = stringResource(R.string.years, years)
            }
            // Summary Items
            SummaryRow(
                label = stringResource(R.string.years_to_retirement),
                value = if (isLoading) "" else formattedYear,
                viewType = "Retirement"
            )

            Spacer(modifier = Modifier.height(8.dp))

            var formattedAdditionCredit = ""
            additionalCreditsExpected?.let {
                formattedAdditionCredit = formatCreditValue(it)
            }

            SummaryRow(
                label = stringResource(R.string.additional_credits_expected),
                value = if (isLoading) "" else formattedAdditionCredit,
                viewType = "Additional"
            )

            Spacer(modifier = Modifier.height(8.dp))

            var formattedTotalProjected = ""
            totalProjectedCredits?.let {
                formattedTotalProjected = formatCreditValue(it)
            }

            SummaryRow(
                label = stringResource(R.string.total_projected_credits),
                value = if (isLoading) "" else formattedTotalProjected,
                viewType = "Total"
            )

            Spacer(
                modifier = Modifier.height(16.dp)
            )

            // Info Message (if provided)
            infoMessage?.let { message ->
                Spacer(modifier = Modifier.height(16.dp))

                InfoMessageRow(
                    message = if (isLoading) "" else message,
                    isError = false
                )
                Spacer(modifier = Modifier.height(12.dp))
            }
        }

        // Error Message (if provided)
        errorMessage?.let { message ->
            if (message.isNotEmpty()) {
                Spacer(modifier = Modifier.height(4.dp))

                InfoMessageRow(
                    message = if (isLoading) "" else message,
                    isError = !isLoading
                )
            }
        }
    }
}

@Composable
private fun SummaryRow(
    label: String,
    value: String,
    viewType: String,
    modifier: Modifier = Modifier
) {
    Row(
        modifier = modifier
            .fillMaxWidth()
            .tagged("Summary Row $label"),
        horizontalArrangement = Arrangement.SpaceBetween,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(
            text = label,
            style = MaterialTheme.typography.bodyMedium,
            color = Color(0xFF606060),
            modifier = Modifier.weight(1f),
            fontSize = 16.sp
        )
        Text(
            text = value,
            style = MaterialTheme.typography.bodyMedium,
            fontWeight = FontWeight.Medium,
            color = Color.Black,
            modifier = Modifier.tagged("Summary Row Value $viewType")
        )
    }
}

@Composable
private fun InfoMessageRow(
    message: String,
    isError: Boolean,
    modifier: Modifier = Modifier
) {
    Row(
        modifier = modifier.fillMaxWidth(),
        verticalAlignment = Alignment.CenterVertically
    ) {
        if (isError) {
            Icon(
                imageVector = Icons.Outlined.Info,
                contentDescription = null,
                tint = Color(0xFFE53E3E),
                modifier = Modifier
                    .size(16.dp)
            )

            Spacer(modifier = Modifier.width(8.dp))
        }

        Text(
            text = message,
            style = MaterialTheme.typography.bodySmall,
            color = if (isError) Color(0xFFE53E3E) else colorResource(R.color.colorPrimary),
            lineHeight = 16.sp
        )
    }
}
