@Composable
fun WhyCreditsAreLostExpandable(
    modifier: Modifier = Modifier,
    scrollState: ScrollState? = null,
    rules: PlanRules? = null,
    viewModel: PensionViewModel? = null
) {
    var isExpanded by remember { mutableStateOf(false) }
    var hasLoggedInitialState by remember { mutableStateOf(false) }

    // Log analytics when expansion state changes
    LaunchedEffect(isExpanded) {
        if (viewModel != null) {
            // Skip logging on initial composition
            if (!hasLoggedInitialState) {
                hasLoggedInitialState = true
            } else {
                if (isExpanded) {
                    viewModel.logEvent(FirebaseTags.PENSION_CREDITS_LOST_EXPAND)
                } else {
                    viewModel.logEvent(FirebaseTags.PENSION_CREDITS_LOST_COLLAPSE)
                }
            }
        }
    }

    // Animation for the arrow rotation
    val arrowRotation by animateFloatAsState(
        targetValue = if (isExpanded) 0f else 360f,
        animationSpec = tween(300),
        label = "arrow_rotation"
    )

    // Auto-scroll to bottom when expanded
    LaunchedEffect(isExpanded) {
        if (isExpanded && scrollState != null) {
            kotlinx.coroutines.delay(350) // Wait for expansion animation to complete
            scrollState.animateScrollTo(scrollState.maxValue)
        }
    }

    Card(
        modifier = modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .tagged("Why Credits Are Lost Expandable"),
        colors = CardDefaults.cardColors(
            containerColor = Color.White
        ),
        elevation = CardDefaults.cardElevation(
            defaultElevation = 2.dp
        )
    ) {
        Column {
            // Header row
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .clickable { isExpanded = !isExpanded }
                    .padding(16.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Row(
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Icon(
                        painter = painterResource(R.drawable.ic_warning_pension),
                        contentDescription = "Info",
                        tint = colorResource(R.color.primaryLight),
                        modifier = Modifier.size(20.dp)
                    )

                    Spacer(modifier = Modifier.width(8.dp))

                    Text(
                        text = stringResource(R.string.why_credits_are_lost),
                        color = colorResource(R.color.primaryLight),
                        fontSize = 16.sp,
                        fontWeight = FontWeight.Medium
                    )
                }

                Icon(
                    imageVector = Icons.Default.KeyboardArrowDown,
                    contentDescription = "Expand",
                    tint = Color(0xFF666666),
                    modifier = Modifier
                        .size(24.dp)
                        .rotate(arrowRotation)
                )
            }

            // Expandable content
            AnimatedVisibility(
                visible = isExpanded,
                enter = expandVertically(
                    animationSpec = tween(300)
                ) + fadeIn(animationSpec = tween(300)),
                exit = shrinkVertically(
                    animationSpec = tween(300)
                ) + fadeOut(animationSpec = tween(300))
            ) {
                Column(
                    modifier = Modifier
                        .fillMaxWidth()
                        .background(Color(0xFFFFFFFF))
                        .padding(16.dp)
                ) {
                    rules?.ruleDescriptions?.forEachIndexed {
                            index, rule ->
                        val parts = parseRule(rule)
                        CreditLostItem(
                            title = parts.first,
                            description = parts.second,
                            isLast = index == rules.ruleDescriptions.size
                        )
                    }
                }
            }
        }
    }
}

@Composable
private fun CreditLostItem(
    title: String,
    description: String,
    isLast: Boolean = false
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(bottom = if (isLast) 0.dp else 8.dp),
        verticalAlignment = Alignment.Top
    ) {
        // Bullet point
        Text(
            text = "•",
            fontSize = 14.sp,
            lineHeight = 20.sp
        )

        Spacer(modifier = Modifier.width(8.dp))

        // Content
        Text(
            text = buildAnnotatedString {
                if (title.isNotEmpty()) {
                    withStyle(
                        style = SpanStyle(
                            fontWeight = FontWeight.SemiBold,
                            color = Color(0xFF333333)
                        )
                    ) {
                        append(title)
                    }
                    append(" ")
                }
                withStyle(
                    style = SpanStyle(
                        color = Color(0xFF666666)
                    )
                ) {
                    append(description)
                }
            },
            fontSize = 14.sp,
            lineHeight = 20.sp
        )
    }
}

fun parseRule(rule: String): Pair<String, String> {
    val delimiterIndex = when {
        rule.contains("=") -> rule.indexOf("=")
        rule.contains(":") -> rule.indexOf(":")
        else -> -1
    }
    return if (delimiterIndex != -1) {
        rule.substring(0, delimiterIndex).trim() to rule.substring(delimiterIndex)
    } else {
        "" to rule.trim()
    }
}
