/**
 * Formats credit values to display whole numbers without decimal points,
 * or up to 2 decimal places for non-whole numbers.
 * Examples: 0.0 → "0", 5.0 → "5", 5.5 → "5.5", 5.567 → "5.57"
 */
fun formatCreditValue(value: Double): String {
    val df = DecimalFormat("#.##")
    return df.format(value).toString()
}

fun formatAmountValue(value: Double): String {
    return "$" + String.format(Locale.getDefault(), "%#.2f", value)
}

@Composable
fun getTealTheme(
    gradientStartColor: Color = colorResource(R.color.gradient_home_end),
    gradientEndColor: Color = colorResource(R.color.gradient_home_start)
): SummaryTheme {
    return SummaryTheme(
        backgroundGradient = Brush.verticalGradient(
            colors = listOf(gradientStartColor, gradientEndColor)
        ),
        primaryTextColor = Color.White,
        secondaryTextColor = Color.White.copy(alpha = 0.9f),
        lostCreditsColor = Color.White,
        titleTextColor = Color.White,
        dividerColor = Color.White.copy(alpha = 0.5f)
    )
}

@Composable
fun getLightTheme(): SummaryTheme {
    return SummaryTheme(
        backgroundGradient = Brush.verticalGradient(
            colors = listOf(
                Color(0xFFFFFFFF), // Light gray
                Color(0xFFFFFFFF) // Slightly darker gray
            )
        ),
        primaryTextColor = Color(0xFF0088c3),
        secondaryTextColor = Color(0xFF757575),
        lostCreditsColor = Color.Red,
        titleTextColor = Color.Black,
        dividerColor = Color(0xFFDFDFDF)
    )
}

@Composable
fun SummaryImpactCard(
    modifier: Modifier = Modifier,
    data: SummaryImpactData,
    isOffline: Boolean,
    theme: SummaryTheme = getTealTheme(),
    onClick: (() -> Unit)? = null
) {
    Card(
        modifier = modifier
            .fillMaxWidth()
            .tagged("Summary Impact Card"),
        shape = RoundedCornerShape(12.dp),
        elevation = CardDefaults.cardElevation(defaultElevation = 0.dp),
        onClick = onClick ?: {}
    ) {
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .background(theme.backgroundGradient)
                .padding(20.dp)
        ) {
            Column {
                // Title
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = stringResource(R.string.total_pension_credit),
                        color = theme.titleTextColor,
                        fontSize = 16.sp,
                        fontWeight = FontWeight.Medium
                    )

                    if (isOffline) {
                        Text(
                            text = "As of: ${data.date}",
                            style = MaterialTheme.typography.bodySmall,
                            color = theme.secondaryTextColor,
                            fontSize = 12.sp
                        )
                    }
                }

                // Credits row
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween
                ) {
                    // Total Credits Earned
                    Column(
                        horizontalAlignment = Alignment.Start,
                        modifier = Modifier
                            .weight(1f)
                            .padding(top = 8.dp)
                            .tagged("Total Credits Earned Column")
                    ) {
                        BasicText(
                            text = formatCreditValue(data.totalCreditsEarned),
                            style = MaterialTheme.typography.headlineMedium.copy(
                                color = theme.primaryTextColor,
                                fontWeight = FontWeight.Bold
                            ),
                            autoSize = TextAutoSize.StepBased(
                                minFontSize = 24.sp,
                                maxFontSize = 32.sp,
                                stepSize = 1.sp
                                // Adjust step for smoother/faster scaling
                            ),
                            maxLines = 1,
                            modifier = Modifier.tagged("Total Credits Earned Value")

                        )
                    }

                    // Credits Kept
                    Column(
                        horizontalAlignment = Alignment.End,
                        modifier = Modifier
                            .weight(1f)

                    ) {
                        Icon(
                            painter = painterResource(R.drawable.verified_user),
                            contentDescription = "verified",
                            modifier = Modifier,
                            tint = Color.White
                        )
                    }
                }
            }
        }
    }
}
