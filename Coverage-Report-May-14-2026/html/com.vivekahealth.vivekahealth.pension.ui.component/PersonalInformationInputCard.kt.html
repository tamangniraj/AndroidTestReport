package com.vivekahealth.vivekahealth.pension.ui.component

import androidx.compose.foundation.background
import androidx.compose.foundation.border
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.selection.selectable
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.foundation.text.KeyboardActions
import androidx.compose.foundation.text.KeyboardOptions
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.filled.Close
import androidx.compose.material.icons.filled.KeyboardArrowDown
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalFocusManager
import androidx.compose.ui.platform.LocalSoftwareKeyboardController
import androidx.compose.ui.res.colorResource
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.text.input.ImeAction
import androidx.compose.ui.text.input.KeyboardType
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import com.vivekahealth.vivekahealth.FirebaseTags
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.pension.data.model.RetirementOption
import com.vivekahealth.vivekahealth.pension.ui.tagged
import com.vivekahealth.vivekahealth.pension.ui.viewmodel.PensionViewModel
import androidx.compose.foundation.layout.navigationBars
import androidx.compose.foundation.layout.windowInsetsPadding

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun PersonalInformationInputCard(
    selectedRetirementAge: RetirementOption?,
    retirementOption: List<RetirementOption>,
    expectedHours: String?,
    currentHours: String?,
    editableExpectedHour: Boolean? = true,
    min: Int?,
    max: Int?,
    onRetirementAgeChange: (RetirementOption) -> Unit,
    onExpectedHoursChange: (String) -> Unit,
    viewModel: PensionViewModel? = null
) {
    var showBottomSheet by remember { mutableStateOf(false) }

    val focusManger = LocalFocusManager.current
    val keyboardController = LocalSoftwareKeyboardController.current
    // Log analytics when bottom sheet opens
    LaunchedEffect(showBottomSheet) {
        if (showBottomSheet && viewModel != null) {
            viewModel.logEvent(FirebaseTags.PENSION_RETIREMENT_AGE_OPEN)
        }
    }

    // Log analytics when expected hours input changes (with debounce)
    LaunchedEffect(expectedHours) {
        if (expectedHours?.isNotEmpty() == true && viewModel != null) {
            kotlinx.coroutines.delay(1000) // Debounce for 1 second
            viewModel.logEvent(FirebaseTags.PENSION_EXPECTED_HOURS_INPUT)
        }
    }

    val bottomSheetState = rememberModalBottomSheetState(
        skipPartiallyExpanded = true
    )

    Box(
        modifier = Modifier
            .fillMaxWidth()
            .padding(top = 12.dp)
            .background(color = Color(0xFFFFFFFF))
            .tagged("Personal Information Input Card")

    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp)
        ) {
            // Title
            Text(
                text = stringResource(R.string.personal_information),
                fontSize = 18.sp,
                fontWeight = FontWeight.SemiBold,
                color = Color.Black,
                modifier = Modifier.padding(bottom = 24.dp)
            )

            // Retirement Age Field
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(bottom = 24.dp)
            ) {
                Row(
                    modifier = Modifier.padding(bottom = 8.dp)
                ) {
                    Text(
                        text = stringResource(R.string.retirement_option),
                        fontSize = 14.sp,
                        fontWeight = FontWeight.Medium,
                        color = Color.Black
                    )
                    Text(
                        text = " *",
                        fontSize = 14.sp,
                        color = Color.Red
                    )
                }

                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(56.dp)
                        .border(
                            width = 1.dp,
                            color = Color.Gray,
                            shape = RoundedCornerShape(4.dp)
                        )
                        .clickable { showBottomSheet = true }
                        .padding(horizontal = 16.dp)
                        .tagged("Retirement Age Selector"),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = selectedRetirementAge?.label ?: "",
                        fontSize = 16.sp,
                        color = Color.Black,
                        modifier = Modifier.tagged("Retirement Age Value")
                    )

                    Icon(
                        imageVector = Icons.Default.KeyboardArrowDown,
                        contentDescription = "Select retirement age",
                        tint = Color.Gray
                    )
                }
            }

            // Expected Hours Field
            Column(
                modifier = Modifier.fillMaxWidth()
            ) {
                Row(
                    modifier = Modifier.padding(bottom = 8.dp)
                ) {
                    Text(
                        text = stringResource(R.string.expected_hours_per_year),
                        fontSize = 14.sp,
                        fontWeight = FontWeight.Medium,
                        color = Color.Black
                    )
                    Text(
                        text = " *",
                        fontSize = 14.sp,
                        color = Color.Red
                    )
                }

                val hasError = expectedHours?.toDoubleOrNull()?.let { hours ->
                    (min == null || hours >= min) && (max == null || hours <= max)
                } != true

                var isFocused by remember { mutableStateOf(false) }

                OutlinedTextField(
                    value = expectedHours ?: "",
                    onValueChange = { input ->
                        // Allow empty, digits, and a single decimal point
                        onExpectedHoursChange(input)
                    },
                    enabled = editableExpectedHour ?: true,
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(0.dp)
                        .border(
                            width = 1.dp,
                            color = if (isFocused) {
                                colorResource(R.color.primaryLight)
                            } else if (hasError) {
                                Color.Red
                            } else {
                                MaterialTheme.colorScheme.outline
                            },
                            shape = RoundedCornerShape(4.dp)
                        )
                        .tagged("Expected Hours Input"),

                    placeholder = {
                    },
                    keyboardOptions = KeyboardOptions(
                        keyboardType = KeyboardType.Decimal,
                        imeAction = ImeAction.Done
                    ),
                    keyboardActions = KeyboardActions(onDone = {
                        focusManger.clearFocus()
                        keyboardController?.hide()
                    }),
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedTextColor = Color.Black,
                        unfocusedTextColor = Color.Black,
                        disabledTextColor = Color.Gray,
                        focusedBorderColor = if (hasError) {
                            Color.Red
                        } else {
                            colorResource(
                                R.color.primaryLight
                            )
                        },
                        unfocusedBorderColor = if (hasError) Color.Red else Color.Gray,
                        errorBorderColor = Color.Red
                    ),
                    shape = RoundedCornerShape(4.dp),
                    isError = hasError
                )

                // Error message
                if (hasError) {
                    val msg = if (max != null) {
                        stringResource(R.string.error_valid_number_range, min ?: 0, max)
                    } else {
                        stringResource(R.string.error_valid_number_min, min ?: 0)
                    }

                    Text(
                        text = msg,
                        fontSize = 12.sp,
                        color = Color.Red,
                        modifier = Modifier.padding(top = 4.dp, start = 8.dp)
                    )
                }

                Text(
                    text = stringResource(
                        R.string.current_year_s_hrs,
                        currentHours ?: ""
                    ),
                    fontSize = 12.sp,
                    color = Color.Gray,
                    modifier = Modifier.padding(top = 4.dp)
                )
            }
        }
    }

    // Bottom Sheet Dialog
    if (showBottomSheet) {
        ModalBottomSheet(
            onDismissRequest = { showBottomSheet = false },
            sheetState = bottomSheetState,
            containerColor = Color.White,
            shape = RoundedCornerShape(topStart = 16.dp, topEnd = 16.dp),
            modifier = Modifier.padding(top = 0.dp),
            dragHandle = null
        ) {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .windowInsetsPadding(WindowInsets.navigationBars)
            ) {
                // Header
                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(horizontal = 24.dp),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = stringResource(R.string.retirement_option),
                        fontSize = 18.sp,
                        fontWeight = FontWeight.SemiBold,
                        color = Color.Black

                    )

                    IconButton(
                        onClick = { showBottomSheet = false }
                    ) {
                        Icon(
                            imageVector = Icons.Default.Close,
                            contentDescription = "Close",
                            tint = Color.Gray,
                            modifier = Modifier.padding(8.dp)
                        )
                    }
                }
                HorizontalDivider(color = Color(0xFFf8f8f8), thickness = 2.dp)
                Spacer(Modifier.height(6.dp))
                // Retirement Options
                retirementOption.forEach { option ->
                    Row(
                        modifier = Modifier
                            .fillMaxWidth()
                            .selectable(
                                selected = selectedRetirementAge?.label == option.label,
                                onClick = {
                                    viewModel?.logEvent(FirebaseTags.PENSION_RETIREMENT_AGE_SELECT)
                                    onRetirementAgeChange(option)
                                    showBottomSheet = false
                                }
                            )
                            .padding(horizontal = 12.dp),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        RadioButton(
                            selected = selectedRetirementAge?.label == option.label,
                            onClick = {
                                viewModel?.logEvent(FirebaseTags.PENSION_RETIREMENT_AGE_SELECT)
                                onRetirementAgeChange(option)
                                showBottomSheet = false
                            },
                            colors = RadioButtonDefaults.colors(
                                selectedColor = colorResource(R.color.primaryLight),
                                disabledSelectedColor = colorResource(R.color.light_grey)
                            )
                        )

                        Spacer(modifier = Modifier.width(12.dp))

                        Text(
                            text = option.label ?: "",
                            fontSize = 16.sp,
                            color = Color.Black
                        )
                    }
                }

                Spacer(modifier = Modifier.height(16.dp))
            }
        }
    }
}
