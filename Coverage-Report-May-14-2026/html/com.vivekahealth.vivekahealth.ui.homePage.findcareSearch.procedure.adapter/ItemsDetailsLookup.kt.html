package com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.procedure.adapter

import android.view.MotionEvent
import androidx.recyclerview.selection.ItemDetailsLookup
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityprofile.adapter.ProceduresAdapter

class ItemsDetailsLookup(private val recyclerView: RecyclerView) : ItemDetailsLookup<Long>() {

    // 2
    override fun getItemDetails(event: MotionEvent): ItemDetails<Long>? {
        // 3
        val view = recyclerView.findChildViewUnder(event.x, event.y)
        if (view != null) {
            // 4
            return (recyclerView.getChildViewHolder(view) as ProceduresAdapter.ViewHeaderViewHolder).getItem()
        }
        return null
    }
}
