package com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.procedure.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.databinding.ProcedureDesignBinding
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityModel.Procedure

class ProcedureAdapter(
    private var onClick:
        (Procedure) -> Unit
) :
    ListAdapter<Procedure, RecyclerView.ViewHolder>(ValueDiffCallback()) {

    var selectedPosition: Int = 0

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return ProcedureViewHolder(
            ProcedureDesignBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)
        (holder as ProcedureViewHolder).binding.item = item

        (holder as ProcedureViewHolder).binding.setClickListener {
            onClick(item)
        }
    }

    class ProcedureViewHolder(
        val binding: ProcedureDesignBinding,
        private val onClick: (Procedure) -> Unit
    ) : RecyclerView.ViewHolder(binding.root)
}

private class ValueDiffCallback : DiffUtil.ItemCallback<Procedure>() {
    override fun areItemsTheSame(oldItem: Procedure, newItem: Procedure): Boolean {
        return oldItem.code == newItem.code
    }

    override fun areContentsTheSame(oldItem: Procedure, newItem: Procedure): Boolean {
        return oldItem.code == newItem.code && // Check the primary key
            oldItem.price == newItem.price && // Compare price
            oldItem.priceLoading == newItem.priceLoading && // Compare loading state
            oldItem.minPrice == newItem.minPrice && // Compare min price
            oldItem.maxPrice == newItem.maxPrice && // Compare max price
            oldItem.occurrence == newItem.occurrence // Compare occurrence
    }
}
