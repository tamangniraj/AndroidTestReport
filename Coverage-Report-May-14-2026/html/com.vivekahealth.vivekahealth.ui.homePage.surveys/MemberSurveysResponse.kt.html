package com.vivekahealth.vivekahealth.ui.homePage.surveys

import java.io.Serializable
import java.text.DateFormat
import java.text.SimpleDateFormat
import java.util.Locale

data class MemberSurveysResponse(val responseData: List<SurveyData>?)

data class SurveyData(
    val title: String,
    val description: String,
    val endDate: String,
    val submissionDate: String,
    val status: SurveyStatus,
    val fullScreenSplash: Boolean,
    val mobileDisplaySection: String,
    val links: List<Link>
) : Serializable {
    /**
     * Get the [endDate] of the survey as a user-friendly string.
     */
    fun getDisplayDate(): String {
        val date = SimpleDateFormat("yyyy-MM-dd", Locale("en")).parse(endDate)
        val mediumDate = DateFormat.getDateInstance(DateFormat.MEDIUM).format(date)
        return "$mediumDate"
    }

    fun getSubmittedDate(): String {
        val date = SimpleDateFormat("yyyy-MM-dd", Locale("en")).parse(submissionDate)
        val mediumDate = DateFormat.getDateInstance(DateFormat.MEDIUM).format(date)
        return "$mediumDate"
    }
}

data class Link(val rel: String, val href: String)

enum class SurveyStatus {
    NEW,
    COMPLETED,
    EXPIRED
}
