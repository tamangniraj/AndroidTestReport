package com.vivekahealth.vivekahealth.ui.homePage.surveys

import android.content.SharedPreferences
import androidx.core.content.edit
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.login.GenericErrorResponse
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import javax.inject.Inject

class SurveysViewModel @Inject constructor(
    private val homeRepository: HomeRepository,
    private val preferences: SharedPreferences
) : ViewModel(), Observable {

    private val _dataReceived = MutableLiveData<Event<MemberSurveysResponse>>()
    val dataReceived: LiveData<Event<MemberSurveysResponse>>
        get() = _dataReceived

    private val _newSurveyAvailable = MutableLiveData<Event<SurveyStatusResponse>>()
    val newSurveyAvailable: LiveData<Event<SurveyStatusResponse>>
        get() = _newSurveyAvailable

    private val _onCloseClicked = MutableLiveData<Event<Unit>>()
    val onCloseClicked: LiveData<Event<Unit>>
        get() = _onCloseClicked

    private val _goToSurveysClicked = MutableLiveData<Event<Unit>>()
    val goToSurveysClicked: LiveData<Event<Unit>>
        get() = _goToSurveysClicked

    private val _surveyOptOutComplete = MutableLiveData<Event<Unit>>()
    val surveyOptOutComplete: LiveData<Event<Unit>>
        get() = _surveyOptOutComplete
    private val _surveyOptOutError = MutableLiveData<Event<String>>()
    val surveyOptOutError: LiveData<Event<String>>
        get() = _surveyOptOutError

    var isFetchingSplash: Boolean = false

    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    var isCallingOptOut: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.callingOptOut, null)
        }
    var hasUserOptedOut: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.hasUserOptedOut, null)
        }
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    fun fetchSurveys() {
        if (!Constants.IS_OFFLINE) {
            compositeDisposable.add(
                homeRepository.getSurveys()
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .subscribe({
                        if (it.isSuccessful && it.body() != null) {
                            _dataReceived.value = Event(it.body()!!)
                        }
                    }, {})
            )
        }
    }

    fun fetchSplashScreens() {
        if (!Constants.IS_OFFLINE) {
            compositeDisposable.add(
                homeRepository.getSplashScreens()
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        isFetchingSplash = true
                    }
                    .doFinally {
                        isFetchingSplash = false
                    }
                    .subscribe({
                        if (it.isSuccessful && it.body() != null) {
                            _dataReceived.value = Event(it.body()!!)
                        }
                    }, {})

            )
        }
    }

    fun getSurveyStatus() {
        if (!Constants.IS_OFFLINE) {
            compositeDisposable.add(
                homeRepository.getSurveyStatus()
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .subscribe({
                        if (it.isSuccessful) {
                            it.body()?.let {
                                // save in preference for opt out
                                preferences.edit()
                                    .putBoolean(
                                        preferences.getString(Constants.MEMBER_ID, "") + Constants.OPT_OUT_SURVEY,
                                        it.hasOptedOut
                                    )
                                    .apply()
                                _newSurveyAvailable.value = Event(it)
                            }
                        } else {
                        }
                    }, {
                    })
            )
        }
    }

    fun optOutSurvey(optOut: Boolean) {
        isCallingOptOut = true
        compositeDisposable.add(
            homeRepository.optOutSurvey(
                preferences.getString(Constants.MEMBER_ID, "") ?: "",
                optOut
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .subscribe({
                    isCallingOptOut = false
                    if (it.isSuccessful) {
                        hasUserOptedOut = optOut
                        preferences.edit(commit = true) {
                            putBoolean(
                                preferences.getString(
                                    Constants.MEMBER_ID,
                                    ""
                                ) + Constants.OPT_OUT_SURVEY,
                                hasUserOptedOut
                            )
                        }
                        _surveyOptOutComplete.value = Event(Unit)
                    } else {
                        try {
                            val errorResponse: GenericErrorResponse = Gson().fromJson(
                                it.errorBody()?.string(),
                                GenericErrorResponse::class.java
                            )
                            _surveyOptOutError.value = Event(errorResponse.getErrorMessage())
                        } catch (e: Exception) {
                            e.printStackTrace()
                        }
                    }
                }, {
                    isCallingOptOut = false
                    _surveyOptOutError.value = Event(it.message ?: "")
                })
        )
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
