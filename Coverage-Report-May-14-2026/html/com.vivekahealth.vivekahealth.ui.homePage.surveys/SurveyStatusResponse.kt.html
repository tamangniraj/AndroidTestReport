package com.vivekahealth.vivekahealth.ui.homePage.surveys

import android.os.Parcelable
import com.google.gson.annotations.SerializedName
import kotlinx.parcelize.Parcelize

data class SurveyStatusResponse(
    @SerializedName("memberId")
    val memberId: String,
    @SerializedName("hasOptedOut")
    val hasOptedOut: Boolean,
    @SerializedName("newSurveyAvailable")
    val newSurveyAvailable: Boolean,
    @SerializedName("notificationContent")
    val notificationContent: NotificationContent?,
    @SerializedName("url")
    val url: String ? = null
)

@Parcelize
data class NotificationContent(
    @SerializedName("title")
    val title: String = "",
    @SerializedName("description")
    val description: String = ""
) : Parcelable
