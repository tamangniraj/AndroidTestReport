package com.vivekahealth.vivekahealth.ui.homePage.findCare.selectSpecialty

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.local.VivekaDatabase
import com.vivekahealth.vivekahealth.data.remote.home.findCare.FindCareFilterModel
import com.vivekahealth.vivekahealth.data.remote.home.findCare.specialityValue.RecentAll
import com.vivekahealth.vivekahealth.data.remote.home.findCare.specialityValue.SpecialityAll
import com.vivekahealth.vivekahealth.data.remote.home.findCare.specialityValue.ValueSelected
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.thread.ThreadUtils
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import javax.inject.Inject

class SelectSpecialityViewModel@Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository,
    private val vivekaDatabase: VivekaDatabase
) : ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    var valueList: ArrayList<ValueSelected> = arrayListOf()
    private val _updateList = MutableLiveData<Event<List<ValueSelected>>>()
    val updateList: LiveData<Event<List<ValueSelected>>>
        get() = _updateList

    private val _updateListRecent = MutableLiveData<Event<List<ValueSelected>>>()
    val updateListRecent: LiveData<Event<List<ValueSelected>>>
        get() = _updateListRecent
    private val _updateListSpecialities = MutableLiveData<Event<SpecialityAll>>()
    val updateListSpecialities: LiveData<Event<SpecialityAll>>
        get() = _updateListSpecialities
    private val _close = MutableLiveData<Event<Unit>>()
    val close: LiveData<Event<Unit>>
        get() = _close
    var allSpecialist: SpecialityAll? = null
    var recentSpecialist: RecentAll? = null

    var recentSelected: ArrayList<ValueSelected> = arrayListOf()
    private val _done = MutableLiveData<Event<Unit>>()
    val done: LiveData<Event<Unit>>
        get() = _done
    var filterOption: FindCareFilterModel = FindCareFilterModel()
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.filterOption, null)
        }

    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun apply() {
//        filterOption.selectedSubSpeciality.clear()
//        filterOption.selectedSubSpeciality.addAll(valueList.filter { valueSelected ->  valueSelected.selected})
        _done.value = Event(Unit)
    }

    fun close() {
        _close.value = Event(Unit)
    }

    init {
        if (recentSpecialist == null) {
            recent()
        }
        if (allSpecialist == null) {
            allSpecialities()
        }
    }

    fun reCalculate() {
        allSpecialist?.let {
            _updateListSpecialities.value = Event(it)
        }
    }
    fun recent() {
        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    homeRepository.getRecentSpecialities()
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                        }
                        .subscribe({
                            recentSpecialist = it
                            valueList.clear()
                            valueList.addAll(it.values)
                            loading = false
                            var listItem = filterOption.selectedSubSpeciality.map { it -> it.speciality }
                            valueList.forEach { item ->
                                item.selected = listItem.contains(item.speciality)
                            }
                            _updateListRecent.value = Event(valueList)
                        }, {
                            loading = false
                        })
                )
            }
        )
    }

    fun allSpecialities() {
        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    homeRepository.getAllSpecialities()
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                        }
                        .subscribe({
                            allSpecialist = it

                            loading = false
                            _updateListSpecialities.value = Event(it)
                        }, {
                            loading = false
                        })
                )
            }
        )
    }

    fun remove(valueSelected: ValueSelected, position: Int) {
        var item = valueList.find { item -> valueSelected.speciality.equals(item.speciality, true) }
        var index = valueList.indexOf(item)
        if (index != -1) {
            item!!.selected = false
            val updatedItems = valueList.toMutableList().apply { this[index] = item }
            valueList.clear()
            valueList.addAll(updatedItems)
            _updateListRecent.value = Event(valueList)
        }
        var item2 = filterOption.selectedSubSpeciality.find { item2 ->
            valueSelected.speciality.equals(
                item2.speciality,
                true
            )
        }
        var index2 = filterOption.selectedSubSpeciality.indexOf(item2)
        if (index2 != -1) {
            filterOption.selectedSubSpeciality.removeAt(index2)
            _updateList.value = Event(filterOption.selectedSubSpeciality)
        }
    }
    fun listUpdate(mainList: List<ValueSelected>, changed: Int, isChecked: Boolean) {
        var index = changed

        var value = mainList[index]
        var originalIndex = valueList.indexOf(value)
        value.selected = isChecked

        val updatedItems = valueList.toMutableList().apply { this[originalIndex] = value }
        val updatedItemsView = mainList.toMutableList().apply { this[index] = value }

        valueList.clear()
        valueList.addAll(updatedItems)
        var match = filterOption.selectedSubSpeciality.find { pre ->
            pre.speciality.equals(
                value.speciality,
                true
            )
        }
        if (match == null && isChecked) {
            filterOption.selectedSubSpeciality.add(value)
        }
        if (!isChecked) {
            remove(value, changed)
        }

        _updateList.value = Event(filterOption.selectedSubSpeciality)
        _updateListRecent.value = Event(updatedItems)
    }

    fun updateSub() {
        var listItem = filterOption.selectedSubSpeciality.map { it -> it.speciality }
        valueList.forEach { item ->
            item.selected = listItem.contains(item.speciality)
        }
        _updateList.value = Event(filterOption.selectedSubSpeciality)

        _updateListRecent.value = Event(valueList)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
