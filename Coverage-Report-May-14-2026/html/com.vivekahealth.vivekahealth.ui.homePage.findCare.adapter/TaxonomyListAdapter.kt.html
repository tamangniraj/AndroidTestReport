package com.vivekahealth.vivekahealth.ui.homePage.findCare.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.findCare.Value
import com.vivekahealth.vivekahealth.databinding.TaxonomyCellBinding
import java.util.*

class TaxonomyListAdapter(private var onClick: (Value) -> Unit) : ListAdapter<Value, RecyclerView.ViewHolder>(
    TaxonomyListDiffCallback()
) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return TaxonomyViewHolder(
            TaxonomyCellBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }

    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as TaxonomyViewHolder).binding.setClickListener {
            onClick(item)
        }

//        val color = holder.binding.imageView10.context?.getColor(
//            com.vivekahealth.vivekahealth.R.color.iconColor
//        )
//        color?.let {
//            val colorFilter = PorterDuffColorFilter(color, PorterDuff.Mode.MULTIPLY)
//            holder.binding.imageView10.setColorFilter(colorFilter)
//        }

        (holder as TaxonomyViewHolder).binding.item = item
    }

    class TaxonomyViewHolder(
        val binding: TaxonomyCellBinding,
        private val onClick: (Value) -> Unit
    ) : RecyclerView.ViewHolder(binding.root)
}

private class TaxonomyListDiffCallback : DiffUtil.ItemCallback<Value>() {
    override fun areItemsTheSame(oldItem: Value, newItem: Value): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: Value, newItem: Value): Boolean {
        return oldItem == newItem
    }
}
