package com.vivekahealth.vivekahealth.ui.homePage.findCare.adapter
import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.findCare.doctorModelsNew.Value
import com.vivekahealth.vivekahealth.databinding.MydoctorcardBinding

class FindCareAdapter(
    private val onClick: (Value) -> Unit,
    private val onPhoneClick: (String) -> Unit
) : RecyclerView.Adapter<FindCareAdapter.ViewHolder>() {

    private val doctorList = mutableListOf<Value>()

    inner class ViewHolder(private val binding: MydoctorcardBinding) :
        RecyclerView.ViewHolder(binding.root) {

        fun bind(item: Value) {
            binding.value = item

            val street2 = if (!item.address2.isNullOrEmpty()) "${item.address2}, " else ""
            val address =
                "${item.address1 ?: ""}, $street2${item.city ?: ""}, ${item.state ?: ""}, ${item.postalCode ?: ""}"

            binding.address = address

            binding.cardView.setOnClickListener { onClick(item) }
            binding.call.setOnClickListener { item.telephone?.let { onPhoneClick(it) } }
        }
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): ViewHolder {
        val binding = MydoctorcardBinding.inflate(
            LayoutInflater.from(parent.context),
            parent,
            false
        )
        return ViewHolder(binding)
    }

    override fun onBindViewHolder(holder: ViewHolder, position: Int) {
        if (position in 0 until doctorList.size) {
            holder.bind(doctorList[position])
        }
    }

    override fun getItemCount(): Int = doctorList.size

    /**
     * Updates the list efficiently using DiffUtil.
     */
    fun submitList(newList: List<Value>) {
        val diffCallback = DoctorDiffCallback(doctorList, newList)
        val diffResult = DiffUtil.calculateDiff(diffCallback)

        doctorList.clear()
        doctorList.addAll(newList)

        diffResult.dispatchUpdatesTo(this)
    }
}

/**
 * DiffUtil Callback for efficient list updates.
 */
class DoctorDiffCallback(
    private val oldList: List<Value>,
    private val newList: List<Value>
) : DiffUtil.Callback() {

    override fun getOldListSize(): Int = oldList.size

    override fun getNewListSize(): Int = newList.size
    override fun areItemsTheSame(oldItemPosition: Int, newItemPosition: Int): Boolean {
        return oldList[oldItemPosition] == newList[newItemPosition]
    }

    override fun areContentsTheSame(oldItemPosition: Int, newItemPosition: Int): Boolean {
        return oldList[oldItemPosition].npi == newList[newItemPosition].npi
    }
}
