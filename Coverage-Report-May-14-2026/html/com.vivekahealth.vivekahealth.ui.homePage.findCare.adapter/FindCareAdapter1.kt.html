package com.vivekahealth.vivekahealth.ui.homePage.findCare.adapter

class FindCareAdapter1(
    private var onClick:
        (Value) -> Unit,
    var listMain: List<DoctorResponseNew> = ArrayList(),
    var listLength: ArrayList<Any>,
    var click: List<Value> = ArrayList(),
    var dial_click: listnerclick
) :
    RecyclerView.Adapter<RecyclerView.ViewHolder>() {

    var selectedPosition: Int = 0

    private inner class ViewHeaderViewHolder(
        val binding: MydoctorcardBinding
    ) : RecyclerView.ViewHolder(binding.root) {
        fun bind(position: Int) {
        }
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return ViewHeaderViewHolder(
            MydoctorcardBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            )
        )
    }

    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        // val member = getItem(position)

        (holder as ViewHeaderViewHolder).binding.cardView.setOnClickListener {
            onClick(click[position])
        }

        (holder as ViewHeaderViewHolder).binding.doctorName.setText(
            listMain[position].values[position].displayName
        )
        (holder as ViewHeaderViewHolder).binding.mobileNo.setText(
            listMain[position].values[position].telephone
        )
        (holder as ViewHeaderViewHolder).binding.type.setText(
            listMain[position].values[position].primaryTaxonomy
        )

        if (listMain[position].values[position].providerNetwork.equals("IN_NETWORK_PROVIDER")) {
            (holder as ViewHeaderViewHolder).binding.networkinOrOut.setText("In Network")
            (holder as ViewHeaderViewHolder).binding.networkinOrOut.setTextColor(
                Color.parseColor("#00d671")
            )
        } else {
            (holder as ViewHeaderViewHolder).binding.networkinOrOut.setText("Out of Network")
            (holder as ViewHeaderViewHolder).binding.networkinOrOut.setTextColor(
                Color.parseColor("#DC0C2B")
            )
        }

        try {
            val df = DecimalFormat("#.##")
            df.roundingMode = RoundingMode.FLOOR

            (holder as ViewHeaderViewHolder).binding.miles.setText(
                df.format(listMain[position].values[position].distance).toString() + " miles"
            )
            (holder as ViewHeaderViewHolder).binding.rating.setText(
                listMain[position].values[position].nsa?.composite!!.symbolQuantity
            )
        } catch (e: Exception) {
        }
        (holder as ViewHeaderViewHolder).binding.call.setOnClickListener {
            dial_click.clickdata(listMain[position].values[position].telephone!!)
        }
    }
    fun roundOffDecimal(number: Double): Double? {
        val df = DecimalFormat("#.##")
        df.roundingMode = RoundingMode.FLOOR
        return df.format(number).toDouble()
    }
    class MemberViewHolder(
        val binding: MydoctorcardBinding,
        private val onClick: (
            Value
        ) -> Unit,
        private var selectedPosition: Int
    ) : RecyclerView.ViewHolder(binding.root)

    override fun getItemCount(): Int {
        return listLength.size
    }

    interface listnerclick {
        fun clickdata(phone_no: String)
    }
}
