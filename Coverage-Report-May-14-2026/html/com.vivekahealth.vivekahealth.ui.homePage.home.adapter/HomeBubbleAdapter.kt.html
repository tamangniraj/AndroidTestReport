package com.vivekahealth.vivekahealth.ui.homePage.home.adapter

import android.content.Context
import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.bumptech.glide.Glide
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.databinding.AddDependentBubbleBinding
import com.vivekahealth.vivekahealth.databinding.SingleRowBubbleBinding
import com.vivekahealth.vivekahealth.utils.HelperUtils

private const val MEMBER_ITEM = 0
private const val ADD_ITEM = 1

class HomeBubbleAdapter(
    private var onClick: (MemberInfo) -> Unit,
    private var onAddDependent: (MemberInfo) -> Unit,
    private var switchUser: (MemberInfo) -> Unit
) : ListAdapter<MemberInfo, RecyclerView.ViewHolder>(
    MemberDiffCallback()
) {

    var currentTimeMills: String ? = null

    var selectedPosition: Int = 0
    lateinit var context: Context

    var isSsnRequired: Boolean = false

    override fun getItemViewType(position: Int): Int {
        return if (getItem(position).isDummy) {
            ADD_ITEM
        } else {
            MEMBER_ITEM
        }
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        context = parent.context

        return if (viewType == MEMBER_ITEM) {
            MemberViewHolder(
                SingleRowBubbleBinding.inflate(
                    LayoutInflater.from(context),
                    parent,
                    false
                ),
                onClick,
                onAddDependent,
                selectedPosition
            )
        } else {
            AddDependentViewHolder(
                AddDependentBubbleBinding.inflate(
                    LayoutInflater.from(context),
                    parent,
                    false
                )
            )
        }
    }

    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val member = getItem(position)
        if (holder is MemberViewHolder) {
            holder.binding.member = member

            holder.binding.setClickListener {
                val previousPosition = selectedPosition

                // Only update if selection actually changed
                if (previousPosition != position) {
                    selectedPosition = position

                    // Use payload-based updates for better performance
                    notifyItemChanged(previousPosition, "selection_changed")
                    notifyItemChanged(selectedPosition, "selection_changed")

                    // Clear image asynchronously to avoid blocking UI
                    holder.binding.imageBubble.post {
                        Glide.with(context).clear(holder.binding.imageBubble)
                    }
                }

                onClick(member)
            }

            if (member.selfieMandatorySwitch) {
                holder.binding.showEnroll =
                    (
                    member.birthDate.isNullOrBlank() || member.gender.isNullOrBlank() || (
                        isSsnRequired && HelperUtils.checkIfInfant(
                            member.ssn,
                            member.birthDate
                        )
                        ) || !member.hasAvatar.equals(
                        "true",
                        true
                    )
                    ) && position != itemCount - 1 && !member.memberRelationship.toString()
                    .equals(
                        "insured",
                        true
                    )
                if (!(
                    member.birthDate.isNullOrBlank() || member.gender.isNullOrBlank() || (
                        isSsnRequired && HelperUtils.checkIfInfant(
                                member.ssn,
                                member.birthDate
                            )
                        ) || !member.hasAvatar.equals(
                            "true",
                            true
                        )
                    ) && position != itemCount - 1
                ) {
                    holder.binding.showSwitcher =
                        member.selfId.equals(member.loggedInId) && member.selfId.isNotEmpty()
                }
            } else {
                holder.binding.showEnroll =
                    (
                    member.birthDate.isNullOrBlank() || member.gender.isNullOrBlank() || (
                        isSsnRequired && HelperUtils.checkIfInfant(
                            member.ssn,
                            member.birthDate
                        )
                        )
                    ) && position != itemCount - 1 && !member.memberRelationship.toString()
                    .equals(
                        "insured",
                        true
                    )
                if (!(
                    member.birthDate.isNullOrBlank() || member.gender.isNullOrBlank() || (
                        isSsnRequired && HelperUtils.checkIfInfant(
                                member.ssn,
                                member.birthDate
                            )
                        )
                    ) && position != itemCount - 1
                ) {
                    holder.binding.showSwitcher =
                        member.selfId.equals(member.loggedInId) && member.selfId.isNotEmpty()
                }
            }

            holder.binding.showSwitcher =
                member.selfId.equals(member.loggedInId) && member.selfId.isNotEmpty()

            holder.binding.switcher.setOnClickListener {
                switchUser(member)
            }

            holder.binding.clicked = member.isSelected

            holder.binding.currentTime = currentTimeMills.toString()

            if (member.isDummy) {
                holder.binding.addGone = !member.isInsured
                holder.binding.isAdd = true
            } else {
                holder.binding.hasImage = member.hasAvatar.toBooleanStrict()

                holder.binding.isAdd = false
            }
        } else if (holder is AddDependentViewHolder) {
            holder.binding.root.setOnClickListener {
                onAddDependent(member)
            }
        }
    }

    class MemberViewHolder(
        val binding: SingleRowBubbleBinding,
        private val onClick: (MemberInfo) -> Unit,
        private val onAddDependent: (MemberInfo) -> Unit,
        private var selectedPosition: Int
    ) : RecyclerView.ViewHolder(binding.root)

    class AddDependentViewHolder(
        val binding: AddDependentBubbleBinding
    ) : RecyclerView.ViewHolder(binding.root)
}

private class MemberDiffCallback : DiffUtil.ItemCallback<MemberInfo>() {

    override fun areItemsTheSame(oldItem: MemberInfo, newItem: MemberInfo): Boolean {
        return oldItem.selfId == newItem.selfId
    }

    override fun areContentsTheSame(oldItem: MemberInfo, newItem: MemberInfo): Boolean {
        return oldItem == newItem
    }
}
