package com.vivekahealth.vivekahealth.ui.homePage.home.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.InsuranceCardQuickViewDTO
import com.vivekahealth.vivekahealth.databinding.BenefitCardItemViewBinding
import com.vivekahealth.vivekahealth.utils.Constants

class HomeBenefitCardAdapter(private var onClick: (InsuranceCardQuickViewDTO, String) -> Unit) : ListAdapter<InsuranceCardQuickViewDTO, RecyclerView.ViewHolder>(
    InsuranceCardQuickViewDTODiffCallback()
) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return BenefitCardViewHolder(
            BenefitCardItemViewBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            )
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as BenefitCardViewHolder).binding.viewModel = item
        holder.binding.asOfGone = Constants.IS_OFFLINE
//        (holder as BenefitCardViewHolder).bind(item)
        (holder as BenefitCardViewHolder).binding.setClickListener {
            onClick(item, item.cardType)
        }
    }

    class BenefitCardViewHolder(
        val binding: BenefitCardItemViewBinding
    ) : RecyclerView.ViewHolder(binding.root) {

//        init {
//
//
//
//
//        }

        fun bind(item: InsuranceCardQuickViewDTO) {
            binding.apply {
                viewModel = item
                executePendingBindings()
            }
        }
    }
}

private class InsuranceCardQuickViewDTODiffCallback : DiffUtil.ItemCallback<InsuranceCardQuickViewDTO>() {

    override fun areItemsTheSame(
        oldItem: InsuranceCardQuickViewDTO,
        newItem: InsuranceCardQuickViewDTO
    ): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(
        oldItem: InsuranceCardQuickViewDTO,
        newItem: InsuranceCardQuickViewDTO
    ): Boolean {
        return oldItem == newItem
    }
}
