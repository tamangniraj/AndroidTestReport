package com.vivekahealth.vivekahealth.ui.homePage.home.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.*
import com.vivekahealth.vivekahealth.databinding.*

sealed class ItemType(val value: Int) {
    object Diagnosis : ItemType(0)
    object Prescription : ItemType(1)
    object Lab : ItemType(2)
    object Claims : ItemType(3)
}
class HomeRecentActivitiesAdapter(
    private var onClick: (ItemType, String?) -> Unit
) :
    RecyclerView.Adapter<RecyclerView.ViewHolder>() {

    private var listMain: ArrayList<List<Any>> = arrayListOf()
    private var listLength: ArrayList<Int> = arrayListOf()
    private var viewTypeList: ArrayList<ItemType> = arrayListOf()
    private var flatList: ArrayList<Any> = arrayListOf()
    private var labSize: Int = 0
    private var prescriptionSize: Int = 0

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        val inflater = LayoutInflater.from(parent.context)
        val binding = when (viewType) {
            ItemType.Diagnosis.value -> RecentDiagnosisCardItemViewBinding.inflate(
                inflater,
                parent,
                false
            )
            ItemType.Prescription.value -> RecentPrescriptionCardItemViewBinding.inflate(
                inflater,
                parent,
                false
            )
            ItemType.Lab.value -> RecentLabResultCardItemViewBinding.inflate(
                inflater,
                parent,
                false
            )
            ItemType.Claims.value -> RecentClaimsCardItemViewBinding.inflate(
                inflater,
                parent,
                false
            )
            else -> throw IllegalArgumentException("Unknown viewType: $viewType")
        }

        return when (viewType) {
            ItemType.Diagnosis.value -> ViewDiagnosisViewHolder(
                binding as RecentDiagnosisCardItemViewBinding
            )
            ItemType.Prescription.value -> ViewPrescriptionViewHolder(
                binding as RecentPrescriptionCardItemViewBinding
            )
            ItemType.Lab.value -> ViewLabViewHolder(binding as RecentLabResultCardItemViewBinding)
            ItemType.Claims.value -> ViewClaimsViewHolder(
                binding as RecentClaimsCardItemViewBinding
            )
            else -> throw IllegalArgumentException("Unknown viewType: $viewType")
        }
    }

    override fun getItemCount(): Int {
        return listLength.size
    }

    fun updateData(
        list: ArrayList<List<Any>>,
        listLength1: ArrayList<Int>,
        flat: ArrayList<Any>,
        lab: Int,
        prescription: Int
    ) {
        listMain = list
        listLength = listLength1
        flatList = flat
        labSize = lab
        prescriptionSize = prescription
        notifyDataSetChanged()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = flatList[position]
        when (item) {
            is DiagnosisHistoryQuickView -> {
                (holder as ViewDiagnosisViewHolder).binding.viewModel = item
                holder.binding.setClickListener {
                    onClick(ItemType.Diagnosis, item.icd10Code.code)
                }
            }
            is PrescriptionQuickView -> {
                (holder as ViewPrescriptionViewHolder).binding.viewModel = item
                (holder as ViewPrescriptionViewHolder).binding.totalPres =
                    (prescriptionSize - 1).toString()
                (holder as ViewPrescriptionViewHolder).binding.setClickListener {
                    onClick(ItemType.Prescription, item.drugCode)
                }
            }
            is LabResultQuickView -> {
                (holder as ViewLabViewHolder).binding.viewModel = item
                (holder as ViewLabViewHolder).binding.totalTest = (labSize - 1).toString()
                (holder as ViewLabViewHolder).binding.setClickListener {
                    onClick(ItemType.Lab, item.resultCode)
                }
            }
            is ClaimsQuickView -> {
                (holder as ViewClaimsViewHolder).binding.viewModel = item
                (holder as ViewClaimsViewHolder).binding.setClickListener {
                    onClick(ItemType.Claims, item.claimNumber)
                }
            }
        }
    }

    override fun getItemViewType(position: Int): Int {
        return when (flatList[position]) {
            is DiagnosisHistoryQuickView -> ItemType.Diagnosis.value
            is PrescriptionQuickView -> ItemType.Prescription.value
            is LabResultQuickView -> ItemType.Lab.value
            is ClaimsQuickView -> ItemType.Claims.value
            else -> ItemType.Claims.value
        }
    }

    private inner class ViewClaimsViewHolder(
        val binding: RecentClaimsCardItemViewBinding
    ) : RecyclerView.ViewHolder(binding.root) {
        // Handle ViewClaimsViewHolder
    }

    private inner class ViewDiagnosisViewHolder(
        val binding: RecentDiagnosisCardItemViewBinding
    ) : RecyclerView.ViewHolder(binding.root) {
        // Handle ViewDiagnosisViewHolder
    }

    private inner class ViewLabViewHolder(
        val binding: RecentLabResultCardItemViewBinding
    ) : RecyclerView.ViewHolder(binding.root) {
        // Handle ViewLabViewHolder
    }

    private inner class ViewPrescriptionViewHolder(
        val binding: RecentPrescriptionCardItemViewBinding
    ) : RecyclerView.ViewHolder(binding.root) {
        // Handle ViewPrescriptionViewHolder
    }
}
