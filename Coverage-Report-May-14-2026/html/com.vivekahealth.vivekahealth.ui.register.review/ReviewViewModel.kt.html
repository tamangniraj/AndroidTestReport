    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.register, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.register, ""),
                        RegisterUser::class.java
                    )
                }
            },
            {
                user.usernameType = "1"
                user.electronic_documents = "false"
                user.overrideOldRegistration = "false"

                firstName = user.firstName.toString()
                lastName = user.lastName.toString()
                email = user.email.toString()

                if (user.gender.toString().equals("M", ignoreCase = true)) {
                    genderFull = "Male"
                    _maleClicked.value = Event(Unit)
                } else {
                    genderFull = "Female"
                    _femaleClicked.value = Event(Unit)
                }

                dob = ActivityUtils.convertDateToFormat(
                    DateUtils.stringServerDateToDate(user.dob.toString()),
                    "MMddyyyy"
                )

                dob = DateUtils.Convert(user.dob)
                ssn = user.ssn.toString()
                _ssnChangedPrev.value = Event(user.ssn.toString())
                if (ssn.isNotEmpty()) {
                    showSsn = true
                }
                user.hipaaNumber?.let {
                    memberId = it
                }
                _idChangedPrev.value = Event(user.hipaaNumber.toString())

                if (memberId.isNotEmpty() && ssn.isEmpty()) {
                    showMemberId = true
                }
                _datePrev.value = Event(user.dob.toString())

//                .substring(0, 2) + "-" + user.ssn?.substring(3, 4) +"-" + user.ssn?.substring(5, 8)
            }
        )
    }
