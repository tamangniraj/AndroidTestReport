package com.vivekahealth.vivekahealth.ui.homePage.home.profilePictureUpload

import android.content.SharedPreferences
import android.net.Uri
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import androidx.lifecycle.asLiveData
import com.vivekahealth.vivekahealth.data.remote.enroll.DocumentUploadSuccessEnroll
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.ui.enroll.fragments.uploadDocuments.uploadUI.UploadDocCallback
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.remote.APIResponse
import com.vivekahealth.vivekahealth.utils.thread.ThreadUtils
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import java.io.File
import javax.inject.Inject
import okhttp3.MultipartBody

class ProfilePictureUploadViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository
) : ViewModel(), Observable {
    internal lateinit var uploadDocCallback: UploadDocCallback
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    var memberInfoDb = homeRepository.getMemberSid(preferences.getString("memberID", "").toString()).asLiveData()
    var memberInfo: MemberInfo = MemberInfo()
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.memberInfo, null)
        }

    private val _navigateToChangeProfile = MutableLiveData<Event<Unit>>()
    val navigateToChangeProfile: LiveData<Event<Unit>>
        get() = _navigateToChangeProfile
    private val _navigateToSkip = MutableLiveData<Event<Unit>>()
    val navigateToSkip: LiveData<Event<Unit>>
        get() = _navigateToSkip
    var uriImage: Uri? = null
    lateinit var picFile: File
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }

    lateinit var filePart: MultipartBody.Part
    private val _navigateToUpload = MutableLiveData<APIResponse<DocumentUploadSuccessEnroll>>()
    val navigateToUpload: LiveData<APIResponse<DocumentUploadSuccessEnroll>>
        get() = _navigateToUpload
    var name: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.name, null)
        }

    var hideSkip: Boolean = true
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.hideSkip, null)
        }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun changeProfilePic() {
        _navigateToChangeProfile.value = Event(Unit)
    }

    fun loadData(data: MemberInfo) {
        memberInfo = data.copy()
//        name = memberInfo.displayName.toString()
//        hideSkip = memberInfo.selfieMandatorySwitch
    }
    fun skip() {
        preferences.edit().putBoolean(memberInfo.selfId + Constants.skippedProfileUpload, true).apply()
        _navigateToSkip.value = Event(Unit)
    }

    fun uploadPicture(pic: File) {
        picFile = pic

        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    homeRepository.uploadProfile(memberInfo.selfId, filePart)
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                        }
                        .subscribe({
                            loading = false
                            preferences.edit().remove("uri${memberInfo.selfId}").apply()

                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().success(
                                it
                            )
                        }, {
                            loading = false

                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                                it.localizedMessage!!
                            )
                        })
                )
            }
        )
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
