package com.vivekahealth.vivekahealth.ui.appUpdate

import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.utils.Event
import javax.inject.Inject

class AppUpdateViewModel @Inject constructor() : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    private val _goToPlayStore = MutableLiveData<Event<Unit>>()
    val goToPlayStore: LiveData<Event<Unit>>
        get() = _goToPlayStore

    private val _dismissDialog = MutableLiveData<Event<Unit>>()
    val dismissDialog: LiveData<Event<Unit>>
        get() = _dismissDialog

    var appUpdateAvailable: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.appUpdateAvailable, null)
        }

    var appVersion: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.appVersion, null)
        }
    fun onNoThankYouClick() {
        _dismissDialog.value = Event(Unit)
    }

    fun onUpdateAppClick() {
        _goToPlayStore.value = Event(Unit)
    }

    fun onGreatClick() {
        _dismissDialog.value = Event(Unit)
    }
}
