package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.claims.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.DiagnosisCodesDTO
import com.vivekahealth.vivekahealth.databinding.DiagnosisListItemBinding

class DiagnosisListAdapter(private var onClick: (DiagnosisCodesDTO) -> Unit) : ListAdapter<DiagnosisCodesDTO, RecyclerView.ViewHolder>(
    DiagnosisDiffCallback()
) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return MemberViewHolder(
            DiagnosisListItemBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)
        (holder as MemberViewHolder).binding.viewModel = item
    }
    class MemberViewHolder(
        val binding: DiagnosisListItemBinding,
        private val onClick: (DiagnosisCodesDTO) -> Unit
    ) : RecyclerView.ViewHolder(binding.root)
}

private class DiagnosisDiffCallback : DiffUtil.ItemCallback<DiagnosisCodesDTO>() {

    override fun areItemsTheSame(oldItem: DiagnosisCodesDTO, newItem: DiagnosisCodesDTO): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: DiagnosisCodesDTO, newItem: DiagnosisCodesDTO): Boolean {
        return oldItem.code == newItem.code
    }
}
