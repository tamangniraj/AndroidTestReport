    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    var uriImage: Uri? = null
    lateinit var filePart: MultipartBody.Part
    lateinit var picFile: File

    private val _navigateMemberInfo = MutableLiveData<Event<MemberInfo>>()
    val navigateMemberInfo: LiveData<Event<MemberInfo>>
        get() = _navigateMemberInfo
    private val _navigateToChangeProfile = MutableLiveData<Event<MemberInfo>>()
    val navigateToChangeProfile: LiveData<Event<MemberInfo>>
        get() = _navigateToChangeProfile

    private val _navigateToChange = MutableLiveData<Event<MemberInfo>>()
    val navigateToChange: LiveData<Event<MemberInfo>>
        get() = _navigateToChange

    private val _navigateToUpload = MutableLiveData<APIResponse<DocumentUploadSuccessEnroll>>()
    val navigateToUpload: LiveData<APIResponse<DocumentUploadSuccessEnroll>>
        get() = _navigateToUpload

    private val _navigateToRemove = MutableLiveData<APIResponse<DataResponse>>()
    val navigateToRemove: LiveData<APIResponse<DataResponse>>
        get() = _navigateToRemove
    var memberInfoDb = homeRepository.getMemberSid(preferences.getString("memberID", "").toString()).asLiveData()

    private val _navigateToRemoveDependent = MutableLiveData<APIResponse<DataResponse>>()
    val navigateToRemoveDependent: LiveData<APIResponse<DataResponse>>
        get() = _navigateToRemoveDependent
    var data: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.data, null)
        }
    var reason: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.reason, null)
        }
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    var memberInfo: MemberInfo = MemberInfo()
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.memberInfo, null)
        }

    var currentMemberInfo: MemberInfo = MemberInfo()
    var dependentList: ArrayList<MemberInfo> = arrayListOf()

    fun changePicture(memberInfo: MemberInfo) {
        currentMemberInfo = memberInfo
        _navigateToChangeProfile.value = Event(memberInfo)
    }

    fun changedData(status: String) {
        dependentList
        currentMemberInfo.hasAvatar = status

        for (member in memberInfo.dependentsArrayList!!) {
            if (member.selfId == currentMemberInfo.selfId) {
                member.hasAvatar = status
            }
        }

        homeRepository.updateMember(memberInfo)

        _navigateToChange.value = Event(memberInfo)
    }

    fun initiateHome() {
        compositeDisposable.add(
            homeRepository.getMember()
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({
                    loading = false

                    memberInfo = it
                    db()
                }, {
                    loading = false
                })
        )
    }

    fun db() {
        if (!Constants.IS_OFFLINE) {
            homeRepository.insertMember(memberInfo)

            getMem()
        }
    }
    fun getMem() {
        viewModelScope.launch {
            memberInfoDb = homeRepository.getMemberSid(memberInfo.selfId).asLiveData()
        }
    }
    fun createDependentList() {
        dependentList.clear()

        memberInfo.dependentsArrayList?.let { it1 ->

            for (member in it1) {
                if (!member.memberAdditionInReview) {
                    if (StringUtils.equalsIgnoreCase(member.memberRelationship, "Spouse")) {
                        dependentList.add(0, member)
                    } else {
                        if (!member.hideProfileFromHome) {
                            dependentList.add(member)
                        }
                    }
                }
            }

            // reverse the order
            dependentList.sortByDescending { it ->
                DateUtils.stringServerDateToDate(
                    it.birthDate.toString()
                )
            }

            var reversed = dependentList.reversed()

            var grouped = reversed.groupBy { it.memberRelationship }
            grouped.forEach {
                it.value.sortedByDescending { item ->
                    DateUtils.stringServerDateToDate(
                        item.birthDate.toString()
                    )
                }
                it.value.reversed()
            }

            dependentList.clear()

            grouped.forEach {
                if (it.key.equals("Spouse", true)) {
                    dependentList.add(0, it.value[0])
                } else if (it.key.equals("Child", true)) {
                    val dateGroup = it.value.groupBy { date ->
                        DateUtils.stringServerDateToDate(
                            date.birthDate.toString()
                        )
                    }

                    dateGroup.forEach { item ->
                        val sorted = item.value.sortedBy { item1 -> item1.firstName }
                        dependentList.addAll(sorted)
                    }
                }
            }
            val tempOthers: ArrayList<MemberInfo> = arrayListOf()
            grouped.forEach {
                if (!it.key.equals("Insured", true) && !it.key.equals(
                        "Spouse",
                        true
                    ) && !it.key.equals("Child", true)
                ) {
                    tempOthers.addAll(it.value)
                }
            }

            val dateGroup = tempOthers.groupBy { date ->
                DateUtils.stringServerDateToDate(
                    date.birthDate.toString()
                )
            }
            dateGroup.forEach { item ->
                val sorted = item.value.sortedBy { item1 -> item1.firstName }
                dependentList.addAll(sorted)
            }
        }

        data = dependentList.isNullOrEmpty()
    }

    fun uploadPicture(pic: File) {
        picFile = pic
//
//        val filePart: MultipartBody.Part = MultipartBody.Part.createFormData(
//            "file",
//
//          pic.name,
//            RequestBody.create("image/*".toMediaTypeOrNull(), pic)
//        )

        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    homeRepository.uploadProfile(currentMemberInfo.selfId, filePart)
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().loading()
                        }
                        .subscribe({
                            loading = false
                            preferences.edit().remove("uri${currentMemberInfo.selfId}").apply()
                            currentCacheTime = System.currentTimeMillis().toString()
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().success(
                                it
                            )
                        }, {
                            loading = false

                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                                it.localizedMessage!!
                            )
                        })
                )
            }
        )
    }

    fun removePicture() {
//
//

        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    homeRepository.removeAvatar(currentMemberInfo.selfId)
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                            _navigateToRemove.value = APIResponse<DataResponse>().loading()
                        }
                        .subscribe({
                            loading = false

                            _navigateToRemove.value = APIResponse<DataResponse>().success(it)
                        }, {
                            loading = false

                            _navigateToRemove.value = APIResponse<DataResponse>().error(
                                it.localizedMessage!!
                            )
                        })
                )
            }
        )
    }
    fun removeDependent() {
//
//

        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    homeRepository.removeDependent(
                        memberInfo.selfId,
                        AddDependent(
                            currentMemberInfo.firstName!!,
                            currentMemberInfo.lastName!!,
                            currentMemberInfo.birthDate!!,
                            currentMemberInfo.gender!!,
                            currentMemberInfo.ssn!!,
                            "",
                            "",
                            currentMemberInfo.selfId
                        )
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                            _navigateToRemoveDependent.value = APIResponse<DataResponse>().loading()
                        }
                        .subscribe({
                            loading = false
                            initiateHome()
                            _navigateToRemoveDependent.value = APIResponse<DataResponse>().success(
                                it
                            )
                        }, {
                            loading = false

                            _navigateToRemoveDependent.value = APIResponse<DataResponse>().error(
                                it.localizedMessage!!
                            )
                        })
                )
            }
        )
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
