package com.vivekahealth.vivekahealth.data.local.guest

import com.vivekahealth.vivekahealth.data.remote.guest.AdditionalRequirementResponse
import com.vivekahealth.vivekahealth.data.remote.guest.DocumentUploadResponse
import com.vivekahealth.vivekahealth.utils.remote.RetrofitException
import okhttp3.internal.http2.StreamResetException
import java.io.IOException
import java.net.SocketTimeoutException

enum class LoadingState {
    UPLOAD,
    VERIFY,
    IDENTIFY,
    PROCESS,
    RESET
}

sealed class UploadState {

    object Uploaded : UploadState()
    object CustomerSupport : UploadState()
    object AddZip : UploadState() // Multi-match request zip
    object AddAddress : UploadState() // Multi-match request address
    object NoMatchAddZip : UploadState() // No match request zip
    object NoMatchAddAddress : UploadState() // No match request address

    data class MemberAlreadyRegistered(val email: String) : UploadState()
    data class ResetSuccess(val email: String, val msg: String) : UploadState()
    data class ResetError(val msg: String) : UploadState()

    data class Loading(val state: LoadingState) : UploadState()
    data class Success(
        val qrResult: Any? = null,
        val imageResult: DocumentUploadResponse? = null,
        val additionalRequirement: AdditionalRequirementResponse? = null
    ) : UploadState() // Replace `String` with your data type

    data class SuccessScan(val data: Any? = null) : UploadState()
    data class FailedScan(val msg: String) : UploadState()

    data class Error(val message: String) : UploadState() {
        constructor(exception: Throwable) : this(

            when (exception) {
                is SocketTimeoutException, is StreamResetException -> "Request timed out. Please try again."
                is IOException -> "No internet connection."
                is RetrofitException -> handleRetrofitException(exception)
                else -> exception.localizedMessage ?: "An unexpected error occurred."
            }
        )

        companion object {
            private fun handleRetrofitException(exception: RetrofitException): String {
                val ex = exception.exception

                if ((ex.message?.contains("SocketException") == true) &&
                    (ex.message?.contains("StreamResetException") == true)
                ) {
                    return "Upload failed due to a network interruption. Please check your connection and try again."
                }

                return when (val ex = exception.exception) {
                    is SocketTimeoutException, is StreamResetException -> "Request timed out. Please try again."
                    is IOException -> "No internet connection."
                    else -> ex.localizedMessage ?: "An unexpected error occurred."
                }
            }
        }
    }

    object Idle : UploadState() // Optional, for the initial state
}
