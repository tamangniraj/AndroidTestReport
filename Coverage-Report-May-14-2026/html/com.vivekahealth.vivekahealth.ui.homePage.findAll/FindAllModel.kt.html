package com.vivekahealth.vivekahealth.ui.homePage.findAll

class FindAllModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val findCareRepository: FindAllRepository,
    private val homeRepository: HomeRepository,
    private val findCareViewAllRepository: FindCareViewAllRepository
) : ViewModel(), Observable {

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    var eligibleValueList: ArrayList<DoctorResponseNew> = ArrayList()
    var listForDoctorAdapter: List<DoctorResponseNew> = listOf()
    var currentLocation: Location? = null

    var providerCallData: ProviderCallData = ProviderCallData()
    var filterDataNew: FindCareFilterModel = FindCareFilterModel()
    var filterOption: FindCareFilterModel = FindCareFilterModel()
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, com.vivekahealth.vivekahealth.BR.filterOption, null)
        }
    var showFilterView: Boolean = true
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, com.vivekahealth.vivekahealth.BR.showFilterView, null)
        }

    var listSpecialities: SpecialityAll? = null
    var page = 1
    var key: String = ""
    var lati_curr: String = ""
    var longi_current: String = ""
    var locat: String = ""
    var selected_location: String? = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.selected_location, null)
        }

    var resultReceiver: ResultReceiver? = null
    var selected_specialities: String = ""
    var within: String? = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.within, null)
        }

    var memberInfoList = findCareRepository.getMemberFromDb().asLiveData()
    var memberInfoDb = findCareRepository.getMemberSid(
        preferences.getString("memberID", "").toString()
    ).asLiveData()
    final var accesstoken = preferences.getString(Constants.ACCESS_TOKEN, "").toString()

    var latitude: String = ""
    var longitude: String = ""

    var gson = Gson()
    var valueString = ""
    private val _DoctorListGenerated = MutableLiveData<Event<List<Value>>>()
    val doctorListGenerated: LiveData<Event<List<Value>>>
        get() = _DoctorListGenerated

    private val _showFilter = MutableLiveData<Event<Unit>>()
    val showFilter: LiveData<Event<Unit>>
        get() = _showFilter
    private val _logsListGenerated = MutableLiveData<Event<ArrayList<Value>>>()
    val logsListGenerated: LiveData<Event<ArrayList<Value>>>
        get() = _logsListGenerated
    var tempListFlat: ArrayList<Any> = arrayListOf()
    var locationType: String = ""
    var providerType: String = ""
    var lastDistance: String = ""
    var valueList: ArrayList<Value> = arrayListOf()

    var filterData: FilterData = FilterData()
    var pageSize: String = "20"
    var speciality: String = ""
    var searchText: String = ""
    var searchType: String = ""
    var taxonomyGroup: String = ""
    var codeType: String = ""
    var codeValue: String = ""
    var memberID = preferences.getString("memberID", "").toString()
    var isLastPage = false
    var isScrolling = false

    var dataAvailableWithOutFilter: Boolean = false
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    final var accessToken = preferences.getString(Constants.ACCESS_TOKEN, "").toString()
    var data: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.data, null)
        }
    var reason: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.reason, null)
        }

    fun data(value: String) {
        val filter = gson.fromJson(value, FindCareFilterModel::class.java)
//        filterOption.showSpeciality =
        filterOption = filter
        if (!filterOption.specific) {
            allSpecialities()
        }
    }

    fun retainList(list: ArrayList<Value>) {
        val listTemp: ArrayList<Value> = arrayListOf()
        listTemp.addAll(list)
        valueList.clear()
        valueList.addAll(listTemp)
        data = valueList.isEmpty()
        _logsListGenerated.value = Event(valueList)
    }

    fun getProviderList() {
        if (key.equals("doctor", true)) {
            showFilterView = false
            filterOption.speciality = "My Doctor"
            filterOption = filterOption
            getDoctorList()
        } else {
            showFilterView = true
            provider()
        }
    }
    fun removeSpeciality() {
        filterOption.selectedSubSpeciality.clear()
        filterOption.specific = false
        filterOption.taxonomyGroup = ""
        filterOption.taxonomyCodes.clear()
    }

    fun provider() {
        val list = filterOption.selectedSubSpeciality.map { it.speciality }
        val listTaxonomy = filterOption.taxonomyCodes

        var tempSpeciality = list.joinToString(separator = "~") { item -> item }
        val tempTaxonomyCode = listTaxonomy.joinToString(separator = ",") { st -> st }

        if (tempSpeciality.isBlank() && filterOption.speciality?.isNotBlank() == true) {
            tempSpeciality = filterOption.speciality.toString()
        }
        compositeDisposable.add(
            findCareRepository.getProviderList(
                filterOption.current?.latitude.toString(), filterOption.current?.longitude.toString(), page, pageSize, tempSpeciality,
                "", memberID, filterOption.searchType.toString(), filterOption.taxonomyGroup.toString(), filterOption.locationType.toString(), filterOption.providerType.toString(), filterOption.radius.toString(),
                filterOption.codeType.toString(), filterOption.codeValue.toString(), tempTaxonomyCode
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->
                    if (page == 1) {
                        valueList.clear()
                    }

                    loading = false
                    dataAvailableWithOutFilter = it.dataAvailableWithOutFilter

                    page++
                    valueString = gson.toJson(it.values)
                    isLastPage = it.paging?.totalPages == page
                    valueList.addAll(it.values)
                    listForDoctorAdapter = eligibleValueList.toList()
                    data = valueList.isEmpty()

                    _logsListGenerated.value = Event(valueList)
                }, {
                    loading = false
                    data = valueList.isEmpty()
                })
        )
    }

    fun allSpecialities() {
        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    homeRepository.getAllSpecialities()
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                        }
                        .subscribe({
                            listSpecialities = it

                            if (filterOption.taxonomyGroup.toString().isNotBlank()) {
                                var list = it.values.filter { item ->
                                    item.specialityGroup.equals(
                                        filterOption.taxonomyGroup,
                                        true
                                    )
                                }
                                var selected: ArrayList<ValueSelected> = arrayListOf()

                                list.forEach { item ->
                                    item.specialities.map { value ->
                                        selected.add(
                                            ValueSelected(
                                                speciality = value,
                                                taxonomyGroup = item.specialityGroup,
                                                selected = true
                                            )
                                        )
                                    }
                                }
                                filterOption.selectedSubSpeciality.clear()

                                filterOption.selectedSubSpeciality = selected
                                val listComa = filterOption.selectedSubSpeciality.map { item -> item.speciality }
                                filterData.speciality = listComa.joinToString(separator = ",") { value -> value }
                            }
                        }, {
                            loading = false
                        })
                )
            }
        )
    }

    fun getDoctorList() {
        compositeDisposable.add(
            findCareViewAllRepository.getDoctorList(
                preferences.getString("memberID", "").toString(),
                filterOption.current?.latitude.toString(),
                filterOption.current?.longitude.toString(),
                "",
                ""
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->
                    loading = false
                    if (page == 1) {
                        valueList.clear()
                    }
                    page++
                    valueString = gson.toJson(it.values)

                    it.paging?.totalPages?.let {
                        isLastPage = it == page
                    } ?: run {
                        isLastPage = true
                    }

                    valueList.addAll(it.values)

                    listForDoctorAdapter = eligibleValueList.toList()
                    data = valueList.isEmpty()
                    _logsListGenerated.value = Event(valueList)
                }, {
                    loading = false
                    data = valueList.isEmpty()
                })
        )
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun openFilter() {
        _showFilter.value = Event(Unit)
    }

    fun saveLocation(latitude: String, longitude: String) {
        preferences.edit().putString(Constants.LATITUDE, latitude).apply()
        preferences.edit().putString(Constants.LONGITUDE, longitude).apply()
    }
    fun saveLocationString(location: String) {
        preferences.edit().putString(Constants.LOCATIONSTRING, location).apply()
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
