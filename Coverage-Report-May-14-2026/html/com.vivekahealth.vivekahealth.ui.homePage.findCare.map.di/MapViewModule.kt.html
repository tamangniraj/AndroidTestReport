package com.vivekahealth.vivekahealth.ui.homePage.findCare.map.di
import android.content.SharedPreferences
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.ui.homePage.findCare.map.MapViewFragment
import com.vivekahealth.vivekahealth.ui.homePage.findCare.service.FindCareService
import dagger.Module
import dagger.Provides
import dagger.android.ContributesAndroidInjector
import okhttp3.OkHttpClient

@Module
abstract class MapViewModule {

    @ContributesAndroidInjector
    abstract fun contributesMapViewFragment(): MapViewFragment

    companion object {
        @JvmStatic
        @Provides
        fun provideMapViewFragmentService(
            gson: Gson,
            builder: OkHttpClient.Builder,
            sharedPreferences: SharedPreferences
        ): FindCareService = FindCareService(
            gson = gson,
            builder = builder,
            preferences = sharedPreferences
        )
    }
}
