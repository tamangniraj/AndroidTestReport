package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.immunization.immunizationFilter

import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.AvailImmunizationFiltersItem
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.disposables.CompositeDisposable
import javax.inject.Inject

class ImmunizationFilterViewModel @Inject constructor() : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _navigateToImmunization = MutableLiveData<Event<Unit>>()
    val navigateToImmunization: LiveData<Event<Unit>>
        get() = _navigateToImmunization

    private val _navigateToImmunizationFilter = MutableLiveData<Event<Unit>>()
    val navigateToImmunizationFilter: LiveData<Event<Unit>>
        get() = _navigateToImmunizationFilter

    private val _navigateToClear = MutableLiveData<Event<Unit>>()
    val navigateToClear: LiveData<Event<Unit>>
        get() = _navigateToClear

    var typeList: ArrayList<AvailImmunizationFiltersItem> = arrayListOf()
    var typeListID: ArrayList<AvailImmunizationFiltersItem> = arrayListOf()
    var typeListIDFinal: ArrayList<AvailImmunizationFiltersItem> = arrayListOf()

    var count: Int = 0
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.count, null)
        }

    var showTypeHeader: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showTypeHeader, null)
        }

    var showType: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showType, null)
        }

    var groupedType: List<AvailImmunizationFiltersItem>? = null

    fun type() {
        showType = !showType
    }

    fun apply() {
        typeListIDFinal = typeListID
        _navigateToImmunizationFilter.value = Event(Unit)
    }

    fun close() {
        _navigateToImmunization.value = Event(Unit)
    }

    fun clear() {
        typeListID.clear()
        count = 0

        if (!groupedType.isNullOrEmpty()) {
            groupedType!!.forEach { it.selected = false }
        }
        _navigateToClear.value = Event(Unit)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
