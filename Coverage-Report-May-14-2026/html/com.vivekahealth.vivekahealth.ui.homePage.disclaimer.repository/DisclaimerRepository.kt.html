package com.vivekahealth.vivekahealth.ui.homePage.disclaimer.repository

import com.vivekahealth.vivekahealth.data.remote.guest.GuestClientDetails
import com.vivekahealth.vivekahealth.data.remote.home.ClientDetailResponse
import com.vivekahealth.vivekahealth.ui.homePage.disclaimer.service.DisclaimerService
import com.vivekahealth.vivekahealth.utils.thread.AppExecutors
import io.reactivex.Single
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class DisclaimerRepository @Inject constructor(
    private val appExecutors: AppExecutors,
    private val disclaimerService: DisclaimerService
) {

    fun getDisclaimer(): Single<ClientDetailResponse> = disclaimerService.getDisclaimerApi().getClientDetail()

    fun getDisclaimerGuest(): Single<GuestClientDetails> = disclaimerService.getDisclaimerApi().getGuestClientDetails()
}
