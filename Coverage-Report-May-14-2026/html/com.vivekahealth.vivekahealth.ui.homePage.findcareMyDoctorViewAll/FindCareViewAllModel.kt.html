class FindCareViewAllModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val findCareRepository: FindCareViewAllRepository
) : ViewModel(), Observable {

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    var ValueList: ArrayList<DoctorResponseNew> = ArrayList()
    var eligibleValueList: ArrayList<DoctorResponseNew> = ArrayList()
    var listForDoctorAdapter: List<DoctorResponseNew> = listOf()

    private val _DoctorListGenerated = MutableLiveData<Event<List<Value>>>()
    val doctorListGenerated: LiveData<Event<List<Value>>>
        get() = _DoctorListGenerated

    private val _logsListGenerated = MutableLiveData<Event<DoctorResponseNew>>()
    val logsListGenerated: LiveData<Event<DoctorResponseNew>>
        get() = _logsListGenerated
    var tempListFlat: ArrayList<Any> = arrayListOf()
    final var memberID = preferences.getString("memberID", "").toString()
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }

    var memberInfo: MemberInfo = MemberInfo()
    var memberInfoList = findCareRepository.getMemberFromDb().asLiveData()
    var memberInfoDb = findCareRepository.getMemberSid(
        preferences.getString("memberID", "").toString()
    ).asLiveData()
    final var accesstoken = preferences.getString(Constants.ACCESS_TOKEN, "").toString()

    var data: Boolean = false
    fun getDoctorList(lat: String, long: String, provider: String, radious: String) {
        var radiu: String
        if (radious.equals("No limits", ignoreCase = true)) {
            radiu = ""
        } else {
            radiu = radious
        }
        loading = true
        compositeDisposable.add(
            findCareRepository.getDoctorList(
                memberID,
                lat,
                long,
                provider,
                radiu
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->

                    // data =  it.doctorList.isNotEmpty()
//                    Value.isSelected = true
                    //   ValueList = it.values

                    eligibleValueList = ValueList
                    listForDoctorAdapter = eligibleValueList.toList()
                    _logsListGenerated.value = Event(it)

                    /*   eligibleValueList.clear()
                       eligibleValueList =  ValueList

                       listForDoctorAdapter = eligibleValueList.toList()



                       loading = false

                       _DoctorListGenerated.value = Event(listForDoctorAdapter)


   */
                }, {
                    loading = false
                })
        )
    }

    fun saveLocation(latitude: String, longitude: String) {
        preferences.edit().putString(Constants.LATITUDE, latitude).apply()
        preferences.edit().putString(Constants.LONGITUDE, longitude).apply()
    }
    fun saveLocationString(location: String) {
        preferences.edit().putString(Constants.LOCATIONSTRING, location).apply()
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
    }
}
