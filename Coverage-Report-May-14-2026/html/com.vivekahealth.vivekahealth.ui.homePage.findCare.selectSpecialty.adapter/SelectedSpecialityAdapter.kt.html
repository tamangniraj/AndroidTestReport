package com.vivekahealth.vivekahealth.ui.homePage.findCare.selectSpecialty.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.findCare.specialityValue.ValueSelected
import com.vivekahealth.vivekahealth.databinding.SelectedListItemBinding

class SelectedSpecialityAdapter(
    private var onClick:
        (ValueSelected, Int) -> Unit
) :
    ListAdapter<ValueSelected, RecyclerView.ViewHolder>(ValueItemDiffCallback()) {

    var selectedPosition: Int = 0

    private inner class ViewHeaderViewHolder(
        val binding: SelectedListItemBinding,
        private val onClick: (ValueSelected, Int) -> Unit
    ) : RecyclerView.ViewHolder(binding.root) {
        fun bind(position: Int) {
        }
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return ViewHeaderViewHolder(
            SelectedListItemBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)
        (holder as ViewHeaderViewHolder).binding.member = item

        (holder as ViewHeaderViewHolder).binding.setClickListener {
            onClick(item, position)
        }
    }

    override fun getItemId(position: Int): Long {
        return position.toLong()
    }
}

private class ValueItemDiffCallback : DiffUtil.ItemCallback<ValueSelected>() {
    override fun areItemsTheSame(oldItem: ValueSelected, newItem: ValueSelected): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(
        oldItem: ValueSelected,
        newItem: ValueSelected
    ): Boolean {
        return oldItem.speciality == newItem.speciality
    }
}
