package com.vivekahealth.vivekahealth.ui.homePage.accountsDetails.changePassword

class ChangePasswordViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository,
    private val vivekaDatabase: VivekaDatabase,
    private val guestHomeRepository: GuestHomeRepository
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _navigateToName = MutableLiveData<Event<Unit>>()
    val navigateToName: LiveData<Event<Unit>>
        get() = _navigateToName

    var errorMessage: String = "Passwords do not match"
    val errorOrNot: Boolean = true

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack
    private val _passwordChanged = MutableLiveData<Event<String>>()

    val passwordChanged: LiveData<Event<String>>
        get() = _passwordChanged

    private val _passwordOldChanged = MutableLiveData<Event<String>>()

    val passwordOldChanged: LiveData<Event<String>>
        get() = _passwordOldChanged

    private val _confirmPasswordChanged = MutableLiveData<Event<String>>()

    val confirmPasswordChanged: LiveData<Event<String>>
        get() = _confirmPasswordChanged

    var password: String = ""

    var confirmPassword: String = ""
    var oldPassword: String = ""

    //        @Bindable get
//        set(value) {
//            field = value
//            callbacks.notifyCallbacks(this, BR.password, null)
//        }
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    var isMagicLogin: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.magicLogin, null)
        }
    var userName: String? = null
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.userName, null)
        }

    var ssnOrMember: Boolean = true
    var firstName: String = ""
    var lastName: String = ""
    var email: String = ""
    var dob: String = ""
    var passwordConfirm: String = ""
    var gender: String = ""
    var mobile: String = ""
    var usernameType: String = ""
    var countryCode: String = ""
    var electronicDocuments: String = ""
    var overrideOldRegistration: String = ""
    var resendVerification: String = ""
    var tempUUID: String = ""

    private val _navigateToSuccess = MutableLiveData<APIResponse<DataResponse>>()
    val navigateToSuccess: LiveData<APIResponse<DataResponse>>
        get() = _navigateToSuccess

    @get:Bindable
    val passwordTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
            if (s.toString().isNotBlank()) {
                _passwordChanged.value = Event(s.toString())
            }
        }
    }

    @get:Bindable
    val passwordOldTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
            if (s.toString().isNotEmpty()) {
                _passwordOldChanged.value = Event(s.toString())
            }
        }
    }

    @get:Bindable
    val confirmPasswordTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
//            confirmPassword = s.toString()
            if (s.toString().isNotBlank()) {
                _confirmPasswordChanged.value = Event(s.toString())
            }
        }
    }

    fun passwordCheck() {
    }

    fun passLen(password: String): Boolean {
        return StringUtils.isNotEmpty(password) && password.length >= 8
    }

    fun passNumber(password: String): Boolean {
        val letter: Pattern = Pattern.compile(".*\\d.*")
        val hasLetter: Matcher = letter.matcher(password)

        return hasLetter.find()
    }

    fun passCaps(password: String): Boolean {
        val letterLow: Pattern = Pattern.compile(".*[A-Z]")
        val hasLetterLow: Matcher = letterLow.matcher(password)
        val letterUp: Pattern = Pattern.compile(".*[a-z]")
        val hasLetterUp: Matcher = letterUp.matcher(password)
        return hasLetterLow.find() && hasLetterUp.find()
//        return password.contains(".*[A-Z]") && password.contains(".*[a-z])")
    }

    fun passSpecialChar(password: String): Boolean {
        val letter: Pattern = Pattern.compile("[!@#$%&*()_+=|<>?{}\\[\\]~-]")
        val hasLetter: Matcher = letter.matcher(password)
        return hasLetter.find()

//        return password.contains("[!@#$%&*()_+=|<>?{}\\[\\]~-]")
    }

    fun passAndConfirm(password: String, confirm: String): Boolean {
        return if (StringUtils.isNotEmpty(password) && StringUtils.isNotEmpty(confirm)) {
            StringUtils.equals(password, confirm)
        } else {
            false
        }
//        return password.contains("[!@#$%&*()_+=|<>?{}\\[\\]~-]")
    }

    fun onPrevClicked() {
        _navigateToBack.value = Event(Unit)
    }

    fun onNextClicked() {
        val isGuestLogin = preferences.getBoolean(Constants.IS_GUEST, false)

        compositeDisposable.add(
            if (!isGuestLogin) { // FULL REGISTER USER
                if (isMagicLogin) {
                    homeRepository.changePassword(
                        ChangePasswordV2DTO(tempUUID, userName ?: "", password, confirmPassword)
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            _navigateToSuccess.value = APIResponse<DataResponse>().loading()

                            loading = true
                        }
                        .subscribe({
                            _navigateToSuccess.value = APIResponse<DataResponse>().success(it)

                            loading = false
                        }, {
                            _navigateToSuccess.value = APIResponse<DataResponse>().error(
                                it.localizedMessage
                            )

                            loading = false
                        })
                } else {
                    homeRepository.updatePassword(
                        preferences.getString("memberID", "").toString(),
                        UpdatePassword(oldPassword, password, confirmPassword)
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            _navigateToSuccess.value = APIResponse<DataResponse>().loading()

                            loading = true
                        }
                        .subscribe({
                            _navigateToSuccess.value = APIResponse<DataResponse>().success(it)

                            loading = false
                        }, {
                            _navigateToSuccess.value = APIResponse<DataResponse>().error(
                                it.localizedMessage
                            )

                            loading = false
                        })
                }
            } else { // GUEST USER
                if (isMagicLogin) {
                    guestHomeRepository.changeGuestPassword(
                        ChangePasswordV3DTO(
                            userName ?: "",
                            null,
                            password,
                            confirmPassword,
                            tempUUID
                        )
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            _navigateToSuccess.value = APIResponse<DataResponse>().loading()

                            loading = true
                        }
                        .subscribe({
                            _navigateToSuccess.value = APIResponse<DataResponse>().success(it)

                            loading = false
                        }, {
                            _navigateToSuccess.value = APIResponse<DataResponse>().error(
                                it.localizedMessage
                            )

                            loading = false
                        })
                } else {
                    guestHomeRepository.updateGuestPassword(
                        ChangePasswordV3DTO(null, oldPassword, password, confirmPassword, null)
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            _navigateToSuccess.value = APIResponse<DataResponse>().loading()

                            loading = true
                        }
                        .subscribe({
                            _navigateToSuccess.value = APIResponse<DataResponse>().success(it)

                            loading = false
                        }, {
                            _navigateToSuccess.value = APIResponse<DataResponse>().error(
                                it.localizedMessage
                            )

                            loading = false
                        })
                }
            }
        )
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
