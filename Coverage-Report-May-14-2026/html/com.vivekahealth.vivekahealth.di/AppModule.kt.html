    @Provides
    @Singleton
    fun okHttpClient(): OkHttpClient {
        val logging = HttpLoggingInterceptor()
        if (BuildConfig.DEBUG) {
            logging.level = HttpLoggingInterceptor.Level.BODY
        } else {
            logging.level = HttpLoggingInterceptor.Level.NONE
        }

        return OkHttpClient.Builder()
            .connectTimeout(45, TimeUnit.SECONDS)
            .readTimeout(45, TimeUnit.SECONDS)
            .writeTimeout(45, TimeUnit.SECONDS)
            .addInterceptor(logging)
            .build()
    }

    @Provides
    internal fun provideOkHttpClientBuilder(application: Application): OkHttpClient.Builder {

        val dispatcher = Dispatcher().apply {
            maxRequests = 64 // Max parallel requests (default: 64)
            maxRequestsPerHost = 10 // Max parallel requests per host (default: 5)
        }

// Configure Connection Pool
        val connectionPool = ConnectionPool(
            maxIdleConnections = 10, // Keep 10 idle connections
            keepAliveDuration = 5, // Keep alive for 5 minutes
            TimeUnit.MINUTES
        )

        val builder = OkHttpClient.Builder()
            .dispatcher(dispatcher)
            .connectionPool(connectionPool)
            .connectTimeout(45, TimeUnit.SECONDS)
            .readTimeout(45, TimeUnit.SECONDS)
            .writeTimeout(45, TimeUnit.SECONDS)
            .retryOnConnectionFailure(true)
            .followRedirects(false)
            .followSslRedirects(false)

        val loggingInterceptor = HttpLoggingInterceptor()
        if (BuildConfig.DEBUG) {
            loggingInterceptor.level = HttpLoggingInterceptor.Level.BODY
        } else {
            loggingInterceptor.level = HttpLoggingInterceptor.Level.BODY
        }

//        builder.addInterceptor(loggingInterceptor)
        builder.addNetworkInterceptor(ResponseInterceptor(application as VivekaHealthApplication))
//        builder.addInterceptor(ServerDownInterceptor(application))
        return builder
    }

    internal class ResponseInterceptor constructor(val appContext: VivekaHealthApplication) :
        Interceptor {

        // Helper function to set server down status for both activity types
        private fun setServerDownStatus(status: Boolean) {
            when (val activity = appContext.currentActivity) {
                is BaseActivity -> activity._serverDown.postValue(status)
                is BaseComposeActivity -> activity._serverDown.postValue(status)
                else -> Timber.w("Current activity is neither BaseActivity nor BaseComposeActivity")
            }
        }

        override fun intercept(chain: Interceptor.Chain): Response {
            val request: Request = chain.request()
            try {
                val response: Response = chain.proceed(request)
                val currentActivity = appContext.currentActivity

                when (response.code) {
                    401 -> {
                        when (currentActivity) {
                            is BaseActivity -> {
                                val body = response.peekBody(1000000)
                                val s = bodyToString(body)
                                if (!s.contains("An Authentication object was not found in the SecurityContext")) {
                                    SessionLogoutHelper.showSessionLogoutDialog(currentActivity)
                                }
                            }
                            is BaseComposeActivity -> {
                                val body = response.peekBody(1000000)
                                val s = bodyToString(body)
                                if (!s.contains("An Authentication object was not found in the SecurityContext")) {
                                    SessionLogoutHelper.showSessionLogoutDialog(currentActivity as AppCompatActivity)
                                }
                            }
                        }
                        return response
                    }
                    500 -> {
                        when (currentActivity) {
                            is BaseActivity -> {
                                if (currentActivity !is GuestRegisterActivity && currentActivity !is GuestVerifyActivity) {
                                    val body = response.peekBody(1000000)
                                    val s = NetworkMessage.getErrorMessage(body)
                                    SessionLogoutHelper.handleServer(currentActivity, s)
                                }
                            }
                            is BaseComposeActivity -> {
                                val body = response.peekBody(1000000)
                                val s = NetworkMessage.getErrorMessage(body)
                                SessionLogoutHelper.handleServer(currentActivity as AppCompatActivity, s)
                            }
                        }
                        return response
                    }
                    408 -> {
                        when (currentActivity) {
                            is BaseActivity -> {
                                if (currentActivity !is GuestRegisterActivity) {
                                    SessionLogoutHelper.showTimeoutError(currentActivity)
                                }
                            }
                            is BaseComposeActivity -> {
                                SessionLogoutHelper.showTimeoutError(currentActivity as AppCompatActivity)
                            }
                        }
                        return response
                    }
                    502, 503 -> {
                        Timber.e("502 or 503 NETWORK CHECK")
                        setServerDownStatus(true)
                        return response
                    }
                    200 -> {
                        Timber.d(" e ${request.url}")
                        setServerDownStatus(false)
                    }
                }

                return response
            } catch (e: Exception) {
                throw e
            }

        }

        companion object {

            fun bodyToString(response: ResponseBody): String {
                return try {
                    val bufferedSource = response.byteStream().source().buffer()
                    bufferedSource.readUtf8()
                } catch (e: IOException) {
                    "did not work"
                }
            }
        }
    }

    internal class ServerDownInterceptor(private var appContext: VivekaHealthApplication) :
        Interceptor {
        private var isShowingServerDialog: Boolean = false
        private var errorMsg: String = appContext.resources.getString(R.string.apologies_msg)
        override fun intercept(chain: Interceptor.Chain): Response {
            val request: Request = chain.request()
            return try {
                chain.proceed(request)
            } catch (e: IOException) {
                val context = appContext.currentActivity as Context
                var shouldShowForException = false
                val isOffline = Constants.IS_OFFLINE && !HelperUtils.isNetworkConnected(context)
                (e as? StreamResetException)?.let {
                    shouldShowForException = !e.errorCode.name.equals("cancel", true)
                } ?: run {
                    (e as? UnknownHostException)?.let {
                        shouldShowForException = !isOffline
                        errorMsg = if (!HelperUtils.isNetworkConnected(context)) {
                            context.resources.getString(R.string.no_internet_msg)
                        } else {
                            context.resources.getString(R.string.apologies_msg)
                        }
                    } ?: run {
                        shouldShowForException = !e.message.equals(
                            "canceled",
                            true
                        ) && !e.message.equals("socket closed", true)
                        errorMsg = if (!HelperUtils.isNetworkConnected(context)) {
                            context.resources.getString(R.string.no_internet_msg)
                        } else {
                            context.resources.getString(R.string.apologies_msg)
                        }
                    }
                }
                if (!isShowingServerDialog
                    && shouldShowForException
                    && !isOffline
                    && !request.url.pathSegments.contains("unreadNotificationCount")
                    && !request.url.pathSegments.contains("pendingDocuments")
                    && !request.url.pathSegments.contains("logPageView")
                ) {
                    val isChatAPI = request.url.pathSegments.contains("chat")
                    var isChatShown = false
                    ((appContext.currentActivity as AppCompatActivity).supportFragmentManager.primaryNavigationFragment)?.let {
                        isChatShown =
                            (it as NavHostFragment).navController.currentDestination?.label == "customerCare"
                    }
                    if (isChatAPI && !isChatShown) {
                        throw (e)
                    }
                    isShowingServerDialog = true
                    Handler(Looper.getMainLooper()).postDelayed({
                        val dialog = MaterialDialog.Builder(context)
                            .tag("network")
                            .content(errorMsg)
                            .positiveText(R.string.ok_string)
                            .onPositive { dialog, _ ->
                                isShowingServerDialog = false
                                dialog.dismiss()
                            }
                            .dismissListener {
                                isShowingServerDialog = false
                            }
                            .autoDismiss(false)
                            .cancelable(false)
                            .build()
                        if (!(context as AppCompatActivity).isDestroyed) {
                            isShowingServerDialog = true
                            dialog.show()
                        } else {
                            isShowingServerDialog = false
                        }
                    }, 1000)
                }
                throw (e)
            }
        }
    }
