package com.vivekahealth.vivekahealth.ui.homePage.informationSharing.repository

import com.vivekahealth.vivekahealth.data.remote.home.sharing.SharingDataResponse
import com.vivekahealth.vivekahealth.data.remote.home.sharing.SharingDataWithDetailsResponse
import com.vivekahealth.vivekahealth.data.remote.home.sharing.SharingDetailData
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.homePage.informationSharing.SharingService
import com.vivekahealth.vivekahealth.utils.thread.AppExecutors
import io.reactivex.Single
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class SharingRepository @Inject constructor(
    private val appExecutors: AppExecutors,
    private val sharingService: SharingService
) {

    fun getMembersWithSharingDetails(selfId: String): Single<SharingDataWithDetailsResponse> = sharingService.getSharingApi().getFamilyResourceSharingDetails(
        selfId
    )

    fun getMembers(selfId: String): Single<SharingDataResponse> = sharingService.getSharingApi().getFamilyMembers(
        selfId
    )

    fun postShareDetails(selfId: String, detail: SharingDetailData): Single<DataResponse> = sharingService.getSharingApi().postShareDetail(
        selfId,
        detail
    )
}
