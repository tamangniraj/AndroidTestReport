    fun getProviderList(
        latitude: String,
        longitude: String,
        pageNo: String,
        pageSize: String,
        speciality: String,
        searchText: String,
        memberId: String,
        searchType: String,
        taxonomyGroup: String,
        locationType: String,
        providerType: String,
        lastDistance: String,
        codeType: String,
        codeValue: String
    ): LiveData<DoctorResponseNew> {
        mutableSignup = MutableLiveData()
        getProviderLists(
            latitude, longitude, pageNo, pageSize, speciality, searchText, memberId, searchType, taxonomyGroup,
            locationType, providerType, lastDistance, codeType, codeValue
        )
        return mutableSignup!!
    }

    fun getDoctorLis(search: String, memberID: String, latitude: String, longitude: String): LiveData<Specialties> {
        doctorList = MutableLiveData()
        getDoctorList(search, memberID, latitude, longitude)
        return doctorList!!
    }

    private fun getProviderLists(
        latitude: String,
        longitude: String,
        pageNo: String,
        pageSize: String,
        speciality: String,
        searchText: String,
        memberId: String,
        searchType: String,
        taxonomyGroup: String,
        locationType: String,
        providerType: String,
        lastDistance: String,
        codeType: String,
        codeValue: String
    ) {
        val call: Call<DoctorResponseNew> = RetrofitClient.getApi()!!.getAllProviderList(
            latitude, longitude,
            pageNo, pageSize,
            speciality, searchText, memberId, searchType, taxonomyGroup,
            locationType, providerType, "10", codeType, codeValue
        )
        call.enqueue(object : Callback<DoctorResponseNew> {
            override fun onResponse(
                call: Call<DoctorResponseNew>,
                response: Response<DoctorResponseNew>
            ) {
                try {
                    mutableSignup?.value = response.body()
                } catch (e: Exception) {
                }
            }

            override fun onFailure(call: Call<DoctorResponseNew>, t: Throwable) {
                getNavigator()?.onError("Email or Password Invalid")
                //  Log.d(TAG, "onFailure: ${t.localizedMessage}")
            }
        })
    }

    private fun getDoctorList(search: String, memberID: String, latitude: String, longitude: String) {
        val call: Call<Specialties> = RetrofitClient.getApi()!!.getFindCareSearchList(
            search,
            memberID,
            latitude,
            longitude
        )
        call.enqueue(object : Callback<Specialties> {
            override fun onResponse(
                call: Call<Specialties>,
                response: Response<Specialties>
            ) {
                try {
                    doctorList?.value = response.body()
                } catch (e: Exception) {
                }
            }

            override fun onFailure(call: Call<Specialties>, t: Throwable) {
                getNavigator()?.onError("Email or Password Invalid")
                //  Log.d(TAG, "onFailure: ${t.localizedMessage}")
            }
        })
    }
