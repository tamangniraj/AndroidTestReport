package com.vivekahealth.vivekahealth.ui.enroll.fragments.personalDetail

import android.content.SharedPreferences
import android.text.Editable
import android.text.TextWatcher
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.enroll.EnrollMember
import com.vivekahealth.vivekahealth.data.remote.register.RegisterUser
import com.vivekahealth.vivekahealth.ui.enroll.NextClickListener
import com.vivekahealth.vivekahealth.ui.enroll.repository.EnrollRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Coroutines
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.disposables.CompositeDisposable
import javax.inject.Inject
import kotlinx.coroutines.Job
import org.apache.commons.lang3.StringUtils

class MemberDetailViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val enrollRepository: EnrollRepository
) : ViewModel(), Observable {

    val gson = Gson()
    var user = EnrollMember()
    private lateinit var job: Job

    var listAv = arrayListOf<String>()

    var listState = arrayListOf<String>()
    internal lateinit var nextClickListener: NextClickListener

    var selectedInsurance: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.selectedInsurance, null)
        }
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack
    private val _navigateToNext = MutableLiveData<Event<Unit>>()
    val navigateToNext: LiveData<Event<Unit>>
        get() = _navigateToNext

    private val _nameChanged = MutableLiveData<Event<String>>()

    val nameChanged: LiveData<Event<String>>
        get() = _nameChanged

    private val _cityChanged = MutableLiveData<Event<String>>()

    val cityChanged: LiveData<Event<String>>
        get() = _cityChanged

    private val _setState = MutableLiveData<Event<Int>>()
    val setState: LiveData<Event<Int>>
        get() = _setState

    private val _yesClicked = MutableLiveData<Event<Unit>>()
    val yesClicked: LiveData<Event<Unit>>
        get() = _yesClicked
    private val _noClicked = MutableLiveData<Event<Unit>>()
    val noClicked: LiveData<Event<Unit>>
        get() = _noClicked

    private val _zipChanged = MutableLiveData<Event<String>>()

    var continueCoverage: String = ""

    var clicked: Boolean = false

    var zipExtension: String = ""
    var zipFirst: String = ""
    val zipChanged: LiveData<Event<String>>
        get() = _zipChanged

    var addressStreet: String = ""
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.addressStreet, null)
        }
    var zip: String = ""
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.zip, null)
        }

    var state: String = ""
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.state, null)
        }

    var apartment: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.apartment, null)
        }

    var city: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.city, null)
        }

    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.enroll, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.enroll, ""),
                        EnrollMember::class.java
                    )
                }
            },
            {
                addressStreet = user.subscriber?.street1.toString()
                city = user.subscriber?.city.toString()
                val stateStr = user.subscriber?.state.toString()
                apartment = user.subscriber?.street2.toString()
                zipFirst = user.subscriber?.zip.toString()

                zipExtension = user.subscriber?.zipExtension.toString()
                var zipOld = zipFirst + zipExtension
                zip = zipOld

                if (stateStr.isNotBlank()) {
                    listAv.clear()
                    for (list in listState) {
                        listAv.add(list.substring(0, list.indexOf("-")))
                    }

                    val index = listAv.indexOf(stateStr)

                    _setState.value = Event(index)
                }

                continueCoverage = preferences.getString(Constants.ContinueCoverage, "").toString()
                if (continueCoverage.equals("true", true)) {
                    yesClicked()
                } else if (continueCoverage.equals("false", true)) {
                    noClicked()
                }
            }
        )
    }

    fun onNextClicked() {
        _navigateToNext.value = Event(Unit)
    }

    fun addSecondStep() {
        val regEdit = preferences.edit()
        regEdit.putString(Constants.ContinueCoverage, continueCoverage)
        regEdit.apply()
    }
    fun addFirstStep(id: String, firstName: String, lastName: String) {
        var reg = gson.fromJson(
            preferences.getString(Constants.register, ""),
            RegisterUser::class.java
        )

        if (reg == null && id.isNotBlank()) {
            reg = RegisterUser(firstName = firstName, lastName = lastName)
            user.subscriber?.memberId = id
        }
        user.subscriber?.street1 = addressStreet.trim()
        user.subscriber?.city = city.trim()
        user.subscriber?.state = state
        user.subscriber?.street2 = apartment.trim()
        user.subscriber?.zip = zipFirst
        user.subscriber?.zipExtension = zipExtension
        user.subscriber?.firstName = reg.firstName?.trim()
        user.subscriber?.lastName = reg.lastName?.trim()
        user.subscriber?.ssn = reg.ssn
        user.subscriber?.gender = reg.gender
        user.subscriber?.dateOfBirth = reg.dob
        user.subscriber?.phoneNumber = reg.mobile

        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.enroll, regString)
        regEdit.apply()
    }

    @get:Bindable
    val addressTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _nameChanged.value = Event(s.toString())
        }
    }

    @get:Bindable
    val cityTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _cityChanged.value = Event(s.toString())
        }
    }

    fun yesClicked() {
        selectedInsurance = "Yes"

        clicked = true
        _yesClicked.value = Event(Unit)
    }
    fun noClicked() {
        clicked = true

        selectedInsurance = "No"
        _noClicked.value = Event(Unit)
    }

    @get:Bindable
    val zipTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _zipChanged.value = Event(s.toString())
        }
    }

    fun onPrevClicked() {
        _navigateToBack.value = Event(Unit)
    }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
