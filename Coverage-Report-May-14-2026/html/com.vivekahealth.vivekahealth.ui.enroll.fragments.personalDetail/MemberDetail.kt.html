package com.vivekahealth.vivekahealth.ui.enroll.fragments.personalDetail

import android.content.Intent
import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.widget.AdapterView
import android.widget.ArrayAdapter
import androidx.lifecycle.ViewModelProvider
import androidx.navigation.fragment.findNavController
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.databinding.FragmentEnrollmentPersonalDetailBinding
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseFragment
import com.vivekahealth.vivekahealth.ui.enroll.EnrollActivity
import com.vivekahealth.vivekahealth.ui.enroll.NextClickListener
import com.vivekahealth.vivekahealth.ui.login.LoginActivity
import com.vivekahealth.vivekahealth.utils.EventObserver
import javax.inject.Inject

class MemberDetail : BaseFragment(), Injectable, NextClickListener {

    companion object {
        fun newInstance() = MemberDetail()
    }

    private lateinit var viewModel: MemberDetailViewModel
    private lateinit var binding: FragmentEnrollmentPersonalDetailBinding

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = FragmentEnrollmentPersonalDetailBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onResume() {
        if (viewModel.city.isNotBlank()) {
            binding.show1 = false
            binding.show2 = true
        } else {
            binding.show1 = true
            binding.show2 = false
        }

//        reInitSpinner()
        super.onResume()
    }
    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)

        viewModel = ViewModelProvider(this, viewModelFactory).get(MemberDetailViewModel::class.java).apply {
            nextClickListener = this@MemberDetail
        }
        binding.viewModel = viewModel
        (activity as EnrollActivity).changeState(1)
        viewModel.loadData()
        if ((activity as EnrollActivity).enrollId().isNotBlank()) {
            viewModel.user.subscriber?.firstName = (activity as EnrollActivity).enrollFName()
            viewModel.user.subscriber?.lastName = (activity as EnrollActivity).enrollLName()
        }

        binding.show1 = true
        binding.show2 = false

        var State = resources.getStringArray(R.array.us_states)

        viewModel.listState.addAll(State)

        // Create an ArrayAdapter using the string array and a default spinner
        val staticAdapter = ArrayAdapter
            .createFromResource(
                requireContext(),
                R.array.us_states,
                android.R.layout.simple_spinner_dropdown_item
            )

        // Specify the layout to use when the list of choices appears
        staticAdapter
            .setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item)

        // Apply the adapter to the spinner
        binding.stateSpinner.adapter = staticAdapter

//        binding.stateSpinner.onItemSelectedListener = object : AdapterView.OnItemSelectedListener {
//            override fun onItemSelected(parent: AdapterView<*>?, view: View, position: Int, id: Long) {
//                val st  = parent!!.getItemAtPosition(position).toString()
//             viewModel.state =  st.substring(0, st.indexOf("-"));
//
//            }
//
//            override fun onNothingSelected(parent: AdapterView<*>?) {
//
//
//            }
//        }

        viewModel.setState.observe(
            viewLifecycleOwner,
            EventObserver {
                binding.stateSpinner.setSelection(it)
            }
        )

        viewModel.navigateToBack.observe(
            viewLifecycleOwner,
            EventObserver {
                if (binding.show2) {
                    binding.show1 = true
                    binding.show2 = false
                    binding.nextEnable = true
                } else {
                    val intent = Intent(requireContext(), LoginActivity::class.java)
                    startActivity(intent)
                    requireActivity().finish()
                }
            }
        )
        viewModel.navigateToNext.observe(
            viewLifecycleOwner,
            EventObserver {
                if (binding.show1) {
                    val st = viewModel.listState[binding.stateSpinner.selectedItemPosition]
                    viewModel.state = st.substring(0, st.indexOf("-"))
                    binding.show1 = false
                    binding.show2 = true
                    binding.nextEnable = viewModel.continueCoverage.isNotBlank()
                    viewModel.addFirstStep(
                        (activity as EnrollActivity).enrollId(),
                        (activity as EnrollActivity).enrollFName(),
                        (activity as EnrollActivity).enrollLName()
                    )
                } else {
                    viewModel.addSecondStep()
                    findNavController().navigate(
                        R.id.action_memberDetailFragment_to_martialStatusFragment
                    )
                }

//            (this.requireActivity() as RegisterActivity).navigationFragmentManager.addFragmentToBackStack(StepGenderFragment.newInstance(viewModel.mobile,viewModel.countryCode,viewModel.tempUUID,viewModel.email,viewModel.password,viewModel.passwordConfirm,viewModel.firstName,viewModel.lastName))
            }
        )
        viewModel.nameChanged.observe(
            viewLifecycleOwner,
            EventObserver { name ->

                binding.nextEnable = viewModel.addressStreet.isNotBlank() && viewModel.city.isNotBlank() && viewModel.zip.isNotBlank() && viewModel.zip.length >= 5
            }
        )
        viewModel.cityChanged.observe(
            viewLifecycleOwner,
            EventObserver { name ->

                binding.nextEnable = viewModel.addressStreet.isNotBlank() && viewModel.city.isNotBlank() && viewModel.zip.isNotBlank() && viewModel.zip.length >= 5
            }
        )

        viewModel.zipChanged.observe(
            viewLifecycleOwner,
            EventObserver { zip ->

                val zips: List<String> = zip.split("-")

                if (zips[0].length == 5) {
                    viewModel.zipFirst = zips[0]
                }
                if (zips.size > 1) {
                    if (zips[1].length == 4) {
                        viewModel.zipExtension = zips[1]
                    }
                }

                if (viewModel.zip.length >= 5) {
                    binding.nextEnable =
                        viewModel.addressStreet.isNotBlank() && viewModel.city.isNotBlank() && viewModel.zip.isNotBlank()
                } else {
                    binding.nextEnable = false
                }
            }
        )

        viewModel.yesClicked.observe(
            viewLifecycleOwner,
            EventObserver {
                viewModel.continueCoverage = "true"

                binding.yesBtn.background = requireActivity().getDrawable(
                    R.drawable.gender_button_negative_clear
                )

                binding.noBtn.background = requireActivity().getDrawable(
                    R.drawable.gender_button_positive_clear
                )
                binding.nextEnable = binding.show2 && viewModel.clicked
            }
        )
        viewModel.noClicked.observe(
            viewLifecycleOwner,
            EventObserver {
                viewModel.continueCoverage = "false"

                binding.noBtn.background = requireActivity().getDrawable(
                    R.drawable.gender_button_negative_clear
                )

                binding.yesBtn.background = requireActivity().getDrawable(
                    R.drawable.gender_button_positive_clear
                )

                binding.nextEnable = binding.show2 && viewModel.clicked
            }
        )
    }

    fun reInitSpinner() {
        binding.stateSpinner.onItemSelectedListener = object : AdapterView.OnItemSelectedListener {
            override fun onItemSelected(
                parent: AdapterView<*>,
                view: View,
                position: Int,
                id: Long
            ) {
                val st = parent.getItemAtPosition(position).toString()
                viewModel.state = st.substring(0, st.indexOf("-"))
            }

            override fun onNothingSelected(parent: AdapterView<*>?) {
            }
        }
    }
}
