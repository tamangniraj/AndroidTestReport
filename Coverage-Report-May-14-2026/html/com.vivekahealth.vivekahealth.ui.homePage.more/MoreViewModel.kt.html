    private fun showTutorialIfConditionMatched() {
        val isGuestLogin = preferences.getBoolean(Constants.IS_GUEST, false)
        val isNotShownForSession = !preferences.getBoolean(
            Constants.SHOWN_MORE_TUT_FOR_SESSION,
            false
        ) && !tutorialShown
        if (!isGuestLogin && isNotShownForSession
        ) {
            _showTutorial.value = Event(Unit)
            tutorialShown = true
            preferences.edit { putBoolean(Constants.SHOWN_MORE_TUT_FOR_SESSION, true) }
        }
    }

    fun dontShowTutorial() {
        preferences.edit { putBoolean(memberInfo.selfId + "moreTutorialShown", true) }
    }

    fun resetTutorial() {
        _tutorialsAdded.value = false
    }

    fun shouldShowTutorial(): Boolean {
        return !preferences.getBoolean(memberInfo.selfId + "moreTutorialShown", false)
    }

    private fun showMenu(member: ClientDetailResponse) {
        hasUnion = member.hasUnionCard
        hasSaveMoney = member.showSaveMoney
        hasGuardianNurse = member.showGuardianNurse

        hasRewards = member.showRewards
        hasTelemedicine = member.showTelemedicine
        setupClientCustomMenu(member.activeCustomMenus, member.activeCustomSubMenus)
        showTutorialIfConditionMatched()
    }

    fun setBiometric() {
        if (biometricEnabled) {
            biometricEnabled = false
            preferences.edit()
//                .putBoolean(preferences.getString("memberID", "") + Constants.biometricPref, false)
                .putBoolean(
                    preferences.getString(Constants.USER_EMAIL, "") + Constants.BIO_PERMISSION_REQUESTED,
                    false
                )
                .apply()
        } else {
            biometricEnabled = true
            preferences.edit()
//                .putBoolean(preferences.getString("memberID", "") + Constants.biometricPref, true)
                .putBoolean(
                    preferences.getString(Constants.USER_EMAIL, "") + Constants.BIO_PERMISSION_REQUESTED,
                    true
                )
                .apply()
        }
    }
