package com.vivekahealth.vivekahealth.ui.guestRegister.repository

import com.vivekahealth.vivekahealth.data.remote.guest.GuestRegisterUser
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.register.step8.service.RegisterService
import io.reactivex.Single
import retrofit2.Response
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class GuestRegisterRepository @Inject constructor(private val registerService: RegisterService) {
    fun registerGuest(guestRegisterUser: GuestRegisterUser): Single<Response<DataResponse>> {
        return registerService.getRegisterApi().guestRegister(guestRegisterUser)
    }

    fun resendActivationCode(email: String): Single<Response<DataResponse>> {
        return registerService.getRegisterApi().resendAccountActivationEmail(email)
    }
}
