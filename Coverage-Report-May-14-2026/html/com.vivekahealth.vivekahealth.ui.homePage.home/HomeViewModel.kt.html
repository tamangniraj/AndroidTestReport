    fun fetchForPrimaryDevice(memberInfo: MemberInfo?) {
        memberInfo?.let {
            if (memberInfo.selfieMandatorySwitch && !memberInfo.hasAvatar.equals(
                    "true",
                    true
                ) || (
                    !memberInfo.selfieMandatorySwitch && !memberInfo.hasAvatar.equals(
                            "true",
                            true
                        ) && !preferences.getBoolean(
                            memberInfo.selfId + Constants.skippedProfileUpload,
                            false
                        )
                    )
            ) {
                _uploadProfile.value = Event(memberInfo)
                _directlySetDeviceAsPrimary.value = Event(Unit)
            } else {
                // check if device prompt has already been request in this session
                if (!preferences.getBoolean(
                        Constants.REQUEST_DEVICE_PROMPT,
                        false
                    )
                ) {
                    // set device prompt flag for this session (flag is reset when login screen is shown)
                    preferences.edit().putBoolean(
                        Constants.REQUEST_DEVICE_PROMPT,
                        true
                    ).apply()

                    _checkForPrimaryDevice.value = Event(Unit)
                }
            }
        }
        pendingSelfiePrimaryDevice = false
    }

    fun updateUserLanguage(memberInfo: MemberInfo?) {
        memberInfo?.let {
            val locale = DateUtils.getCurrentLocale(resources)

            if (locale.language != "en" && it.languagePreference != locale.language) {
                compositeDisposable.add(
                    homeRepository.updateUserDetails(
                        preferences.getString("memberID", "").toString(),
                        UpdateDetails(languagePreference = locale.language)
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                        }.subscribe({
                            loading = false
                            _setLanguage.value = Event(locale.language)
                        }, {
                            loading = false
                        })
                )
            } else {
                val editor = preferences.edit()
                val memberId = preferences.getString("memberID", "")
                editor.putString(
                    "$memberId${Constants.LanguagePref}",
                    it.languagePreference ?: "en"
                )
                editor.apply()
                it.languagePreference ?.let { langPref ->
                    if (langPref != locale.language) {
                        _setLanguage.value = Event(langPref)
                        Constants.LOCAL_LANG_MODIFIED = langPref
                    }
                }
            }
        }
    }

    fun setConstants(it: ClientDetailResponse, memberInfo: MemberInfo) {
        Constants.CLIENT_DISCLAIMER = it.benefitDisclaimerTextV2 ?: ""
        Constants.BENEFIT_ROLLING_PERIOD = 12
        Constants.SHOW_BANKED_HOURS = it.hasHoursbank == true
        Constants.hasHoursBank = it.hasHoursbank ?: false
        Constants.showCardsOnlyIfEligible = it.cardsOnlyIfEligibleSwitch
        if (it.mobileNavigationSectionVisibility != null) {
            it.mobileNavigationSectionVisibility.forEach { value ->

                if (value.type.equals("MEDICAL_LOGS")) {
                    flagMedicalLog = value.value == true
                    // if flag medical log is true, then  hide medical log should be false
                    Constants.HIDE_MEDICAL_LOG = !flagMedicalLog
                    _navFlag.value = Event(Unit)
                } else if (value.type.equals("RECENT_ACTIVITY")) {
                    flagRecentActivity = value.value == true
                }
            }
        } else {
            flagRecentActivity = !memberInfo.registrationType.equals("3")
        }

        preferences.edit().putString("last_logged_client_id", memberInfo.clientId.toString())
            .apply()

        Constants.LoggedIn = true
        Constants.FIRST_NAME = memberInfo.firstName.toString().replaceFirstChar(Char::uppercase)
        Constants.TRANSPERENCY = memberInfo.registrationType.equals("3")

        if ((memberInfo.isEligible || !Constants.showCardsOnlyIfEligible) && !Constants.IS_OFFLINE && memberInfo.selfId.isNotBlank()) {
            preFetchInsurance(memberInfo.selfId)
        }
    }

    var recentCardDisposable: Disposable? = null
    var recentBenefitDisposable: Disposable? = null

    private fun getRecentCards(memberId: String, apiDataShow: Boolean, currentMember: MemberInfo) {
        recentCardDisposable?.let {
            it.dispose()
            compositeDisposable.remove(it)
        }
        viewModelScope.launch {
            recentCardDisposable = homeRepository.getRecentActivities(
                currentMember.selfId,
                memberId
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                }
                .subscribe({
                    loading = false

                    it?.let {
                        val recentGone =
                            (it.claimsQuickViewList?.isEmpty() ?: true) &&
                                (it.diagnosisHistoryQuickViewList?.isEmpty() ?: true) &&
                                (it.labResultQuickViewList?.isEmpty() ?: true) &&
                                (it.prescriptionQuickViewList?.isEmpty() ?: true)

                        it.selfId = memberId
                        if (recentGone) {
                            homeRepository.deleteRecent(memberId)
                        } else {
                            homeRepository.insertRecent(it)
                        }
                        if (apiDataShow && !recentGone) {
                            _recentCardsData.value = Event(it)
                        }
                        if (memberId == currentMember.selfId) {
                            recentActivityFetched = true
                            showTutorialIfConditionMatched()
                        }
                    }
                }, {
                    loading = false
                })
            compositeDisposable.add(recentCardDisposable!!)
        }
    }

    private fun getBenefitCards(memberId: String, apiDataShow: Boolean, currentMember: MemberInfo) {
        recentBenefitDisposable?.let {
            it.dispose()
            compositeDisposable.remove(it)
        }
        viewModelScope.launch {
            recentBenefitDisposable = homeRepository.getBenefitsCards(memberId)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({
                    loading = false

                    cardGone = it.insuranceCardQuickViewDTOList.isNullOrEmpty()
                    it.selfId = memberId
                    it.insuranceCardQuickViewDTOList?.forEach { data ->
                        data.date = DateUtils.getSystemDateTimeAsString().toString()
                    }
                    if (cardGone) {
                        homeRepository.deleteBenefit(memberId)
                    } else {
                        homeRepository.insertBenefit(it)
                    }

                    if (apiDataShow && !cardGone) {
                        _insuranceCardsData.value = Event(it)
                    }
                    if (memberId == currentMember.selfId) {
                        benefitCardsFetched = true
                        showTutorialIfConditionMatched()
                    }
                }, {
                    loading = false
                    Timber.d(it)
                    if (it is RetrofitException) {
                        _serverError.value = Event(
                            NetworkMessage.getErrorMessage(
                                it.response?.errorBody()
                            )
                        )
                    }
                })
            compositeDisposable.add(recentBenefitDisposable!!)
        }
    }

    fun getOfflineDataBenefit(selfId: String, currentMember: MemberInfo) {
        val data = homeRepository.getBenefit(selfId)
        cardGone = data?.insuranceCardQuickViewDTOList.isNullOrEmpty()

        if (data != null) {
            _insuranceCardsData.value = Event(data)
            if (selfId == currentMember.selfId) {
                benefitCardsFetched = true
                showTutorialIfConditionMatched()
            }
        }

        if (!Constants.IS_OFFLINE) {
            getBenefitCards(selfId, cardGone, currentMember)
        }
    }

    fun getOfflineDataRecent(selfId: String, currentMember: MemberInfo) {
        val data = homeRepository.getRecent(selfId)
        recentGone =
            data?.claimsQuickViewList.isNullOrEmpty() and data?.diagnosisHistoryQuickViewList.isNullOrEmpty() and data?.labResultQuickViewList.isNullOrEmpty() and data?.prescriptionQuickViewList.isNullOrEmpty()

        if (!recentGone) {
            _recentCardsData.value = Event(data)
            if (selfId == currentMember.selfId) {
                recentActivityFetched = true
                showTutorialIfConditionMatched()
            }
        }

        if (!Constants.IS_OFFLINE) {
            if (selfId.toLong() > 0) {
                getRecentCards(selfId, recentGone, currentMember)
            }
        }
    }

    fun setSelectedMember(memberInfo: MemberInfo, loginMember: MemberInfo, context: Context) {
        firstName = memberInfo.firstName.toString()
        lastName = memberInfo.lastName.toString()
        eligible = memberInfo.isEligible
        if (!memberInfo.employeeRole.toString().equals("null", true)) {
            memberRole = memberInfo.employeeRole.toString()
        }

        inReviewSelfie = !memberInfo.hasAvatar.equals("true", true)

        memberType = when {
            memberInfo.memberRelationship.toString().equals("insured", true) -> {
                context.getString(R.string.primary_benificiary)
            }

            else -> {
                memberInfo.relationDesc.toString()
            }
        }
        status = memberInfo.employeeStatus.toString()

        getOfflineDataRecent(memberInfo.selfId, loginMember)
        getOfflineDataBenefit(memberInfo.selfId, loginMember)
    }
