package com.vivekahealth.vivekahealth.ui.homePage.home

import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView

class SmoothScrollListener(
    private val targetPosition: Int,
    private val onScrollComplete: () -> Unit
) :
    RecyclerView.OnScrollListener() {

    override fun onScrolled(recyclerView: RecyclerView, dx: Int, dy: Int) {
        super.onScrolled(recyclerView, dx, dy)
        val layoutManager = recyclerView.layoutManager
        if (layoutManager != null && layoutManager is LinearLayoutManager) {
            val firstVisibleItemPosition = layoutManager.findFirstVisibleItemPosition()
            val lastVisibleItemPosition = layoutManager.findLastVisibleItemPosition()

            if (targetPosition in firstVisibleItemPosition..lastVisibleItemPosition) {
                onScrollComplete()
                recyclerView.removeOnScrollListener(this)
            }
        }
    }
}
