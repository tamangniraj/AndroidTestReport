package com.vivekahealth.vivekahealth.ui.step

import android.os.Bundle
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.R
import timber.log.Timber

class StepCounterViewModel : ViewModel() {

    private val _stepIds = mutableListOf<Pair<Int, Boolean>>(
        Pair(R.id.guestPhoneFragment, true),
        Pair(R.id.guestEmailFragment, true),
        Pair(R.id.guestPasswordFragment, true),
        Pair(R.id.documentScanFragment, true), // DL_FRONT
        Pair(R.id.documentScanFragment, true), // DL_BACK
        Pair(R.id.documentScanFragment, false), // SSN - disabled by default
        Pair(R.id.selfieCaptureFragment, true),
        Pair(R.id.zipUpdateFragment, false), // ZIP - disabled by default
        Pair(R.id.zipUpdateFragment, false) // Full address - disabled by default
    )

    fun enableSSN(enable: Boolean) {
        _stepIds[5] = Pair(R.id.documentScanFragment, enable) // SSN step
    }

    fun enableZip(enable: Boolean) {
        _stepIds[7] = Pair(R.id.zipUpdateFragment, enable) // ZIP step
    }

    fun enableFullAddress(enable: Boolean) {
        _stepIds[8] = Pair(R.id.zipUpdateFragment, enable) // Full address step
    }

    private fun getEnabledSteps(): List<Int> {
        return _stepIds.filter { it.second }.map { it.first }
    }

    fun getStepsTextAndPercent(destinationId: Int, arguments: Bundle? = null): Triple<Int, Int, Double>? {
        val enabledSteps = getEnabledSteps()
        val enabledStepIndices = _stepIds.mapIndexedNotNull { index, pair ->
            if (pair.second) index else null
        }

        if (destinationId == R.id.documentScanFragment && arguments?.getSerializable("documentType")?.toString() == "SSN_ONLY") {
            return Triple(1, 1, 100.0)
        }

        val position = when (destinationId) {
            R.id.documentScanFragment -> {
                val documentType = arguments?.getSerializable("documentType")?.toString()
                Timber.d("DocumentType: $documentType")
                when (documentType) {
                    "DL_FRONT" -> if (enabledStepIndices.contains(3)) enabledStepIndices.indexOf(3) else -1
                    "DL_BACK" -> if (enabledStepIndices.contains(4)) enabledStepIndices.indexOf(4) else -1
                    else -> if (enabledStepIndices.contains(5)) enabledStepIndices.indexOf(5) else -1
                }
            }

            R.id.zipUpdateFragment -> {
                val isFull = arguments?.getBoolean("isFull")
                if (isFull == true) {
                    if (enabledStepIndices.contains(8)) enabledStepIndices.indexOf(8) else -1
                } else {
                    if (enabledStepIndices.contains(7)) enabledStepIndices.indexOf(7) else -1
                }
            }

            else -> {
                val stepIndex = _stepIds.indexOfFirst { it.first == destinationId }
                if (stepIndex != -1 && enabledStepIndices.contains(stepIndex)) {
                    enabledStepIndices.indexOf(stepIndex)
                } else {
                    -1
                }
            }
        }

        return if (position != -1) {
            val currentStep = position + 1
            val totalStep = enabledSteps.size
            val percent = (currentStep.toDouble() / totalStep.toDouble()) * 100
            val roundedPercent = kotlin.math.round(percent * 100) / 100.0
            Triple(currentStep, totalStep, roundedPercent)
        } else {
            null
        }
    }
}
