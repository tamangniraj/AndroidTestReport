    private fun getBarDetails(
        barInfo: Value,
        networkType: NetworkType,
        expenseType: ExpenseType,
        expenseBarType: ExpenseBarType
    ): BarDetails {
        var consumed = 0.0
        var total = 0.0
        var consumedRatio = 0.0
        var remainingRatio = 0.0
        var message = ""
        var hideBar = false
        when (networkType) {
            NetworkType.IN_NETWORK -> {
                when (expenseType) {
                    ExpenseType.DEDUCTIBLE -> {
                        if (expenseBarType == ExpenseBarType.INDIVIDUAL) {
                            consumed = barInfo.individualExpenses?.deductibleAmount ?: 0.0
                            total = barInfo.individualExpenses?.deductibleMaxAmount ?: 0.0
                            consumedRatio = consumed / total
                            remainingRatio = 1 - consumedRatio
                            if (total == 0.0) {
                                message = resources.getString(R.string.no_deductible_benefit)
                                hideBar = true
                            } else if (consumed == total) {
                                message = resources.getString(R.string.deductible_met)
                            }
                        } else {
                            consumed = barInfo.familyExpenses?.familyDeductibleAmount ?: 0.0
                            total = barInfo.familyExpenses?.familyDeductibleMaxAmount ?: 0.0
                            consumedRatio = consumed / total
                            remainingRatio = 1 - consumedRatio
                            if (total == 0.0) {
                                hideBar = true
                                message = resources.getString(R.string.no_deductible_benefit)
                            } else if (consumed == total) {
                                message = resources.getString(R.string.deductible_met)
                            }
                        }
                    }

                    ExpenseType.OOP -> {
                        if (expenseBarType == ExpenseBarType.INDIVIDUAL) {
                            consumed = barInfo.individualExpenses?.outOfPocketAmount ?: 0.0
                            total = barInfo.individualExpenses?.outOfPocketMaxAmount ?: 0.0
                            consumedRatio = consumed / total
                            remainingRatio = 1 - consumedRatio
                            if (total == 0.0) {
                                message = resources.getString(R.string.no_oop)
                                hideBar = true
                            } else if (consumed == total) {
                                message = resources.getString(R.string.oop_met)
                            }
                        } else {
                            consumed = barInfo.familyExpenses?.familyOutOfPocketAmount ?: 0.0
                            total = barInfo.familyExpenses?.familyOutOfPocketMaxAmount ?: 0.0
                            consumedRatio = consumed / total
                            remainingRatio = 1 - consumedRatio
                            if (total == 0.0) {
                                hideBar = true
                                message = resources.getString(R.string.no_oop)
                            } else if (consumed == total) {
                                message = resources.getString(R.string.oop_met)
                            }
                        }
                    }
                }
            }

            NetworkType.OUT_NETWORK -> {
                when (expenseType) {
                    ExpenseType.DEDUCTIBLE -> {
                        if (expenseBarType == ExpenseBarType.INDIVIDUAL) {
                            consumed = barInfo.individualExpenses?.oonDeductibleAmount ?: 0.0
                            total = barInfo.individualExpenses?.oonDeductibleMaxAmount ?: 0.0
                            consumedRatio = consumed / total
                            remainingRatio = 1 - consumedRatio
                            if (total == 0.0) {
                                hideBar = true
                                message = resources.getString(R.string.no_deductible_benefit)
                            } else if (consumed == total) {

                            }
                        } else {

                            total = barInfo.familyExpenses?.familyOonDeductibleMaxAmount ?: 0.0
                            consumedRatio = consumed / total
                            remainingRatio = 1 - consumedRatio
                            if (total == 0.0) {
                                hideBar = true
                                message = resources.getString(R.string.no_deductible_benefit)
                            } else if (consumed == total) {

                    }

                    ExpenseType.OOP -> {
                        if (expenseBarType == ExpenseBarType.INDIVIDUAL) {
                            consumed = barInfo.individualExpenses?.oonOutOfPocketAmount ?: 0.0
                            total = barInfo.individualExpenses?.oonOutOfPocketMaxAmount ?: 0.0

    private fun loadChart(barInfo: Value?, holder: YearViewHolder, expenseBarType: ExpenseBarType) {
        if (barInfo != null) {
            holder.binding.hideYourDeductibleSection =
                !(
                    barInfo.displayConfig?.showDeductibleBar ?: true ||
                        barInfo.displayConfig?.showOOPBar ?: true ||
                        barInfo.displayConfig?.showCopay ?: true ||
                        barInfo.displayConfig?.showCoinsurance ?: true ||
                        barInfo.displayConfig?.showOONBalance ?: true ||
                        barInfo.displayConfig?.showMemberResponsible ?: true ||
                        barInfo.displayConfig?.showPlanResponsible ?: true
                    )

            holder.binding.hideOutOfNetwork = if (expenseBarType == ExpenseBarType.INDIVIDUAL) {
                (barInfo.individualExpenses?.oonDeductibleMaxAmount == null || barInfo.individualExpenses.oonDeductibleMaxAmount == 0.0) &&
                    (barInfo.individualExpenses?.oonOutOfPocketMaxAmount == null || barInfo.individualExpenses.oonOutOfPocketMaxAmount == 0.0)
            } else {
                (barInfo.familyExpenses?.familyOonOutOfPocketMaxAmount == null || barInfo.familyExpenses.familyOonOutOfPocketMaxAmount == 0.0) &&
                    (barInfo.familyExpenses?.familyOonDeductibleMaxAmount == null || barInfo.familyExpenses.familyOonDeductibleMaxAmount == 0.0)
            }

            holder.binding.barDetails = getBarDetails(
                barInfo,
                NetworkType.IN_NETWORK,
                ExpenseType.DEDUCTIBLE,
                expenseBarType
            )
            holder.binding.barDetails2 = getBarDetails(
                barInfo,
                NetworkType.IN_NETWORK,
                ExpenseType.OOP,
                expenseBarType
            )
            holder.binding.barDetails3 = getBarDetails(
                barInfo,
                NetworkType.OUT_NETWORK,
                ExpenseType.DEDUCTIBLE,
                expenseBarType
            )
            holder.binding.barDetails4 = getBarDetails(
                barInfo,
                NetworkType.OUT_NETWORK,
                ExpenseType.OOP,
                expenseBarType
            )
        } else {
            holder.binding.hideYourDeductibleSection = true
            holder.binding.hideOutOfNetwork = true
        }
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)
        (holder as YearViewHolder).binding.benefit = item
        holder.binding.expandPlan = item.expanded
        var barInfo = barModelIndividual.values!!.find { it.id!! == (item.benefitId!!) }

        loadChart(barInfo, holder, ExpenseBarType.INDIVIDUAL)

        holder.binding.isIndividual = true
        holder.binding.setOnTabChanged {
            if (it.id == R.id.individual) {
                holder.binding.isIndividual = true
                loadChart(barInfo, holder, ExpenseBarType.INDIVIDUAL)
            } else {
                holder.binding.isIndividual = false
                val familyBarInfo = barModel.values!!.find { it.id!! == (item.benefitId!!) }
                loadChart(familyBarInfo, holder, ExpenseBarType.FAMILY)
            }
        }

        holder.binding.planSummaryGone = barInfo?.familyExpenses == null
        holder.binding.setClickListener {
            if (item.expanded) {
                item.expanded = false
                holder.binding.expandPlan = false
            } else {
                item.expanded = true
                holder.binding.expandPlan = true
            }
            loadBar(item, holder)
            loadDetails(item, holder)
        }

        holder.binding.setClickListenerPast {
            if (item.expandedPast) {
                item.expandedPast = false
                holder.binding.expandPast = false
            } else {
                item.expandedPast = true
                holder.binding.expandPast = true
            }
        }

        holder.binding.setClickListenerDependents {
            if (item.expanded) {
                item.expanded = false
                holder.binding.dependentsExpanded = false
            } else {
                item.expanded = true
                holder.binding.dependentsExpanded = true
            }
        }

        if (barInfo?.familyExpenses != null) {
            val toggle = barInfo.displayConfig
            holder.binding.viewDeductibleMet = (toggle?.showDeductibleBar ?: false && barInfo.familyExpenses?.familyDeductibleMaxAmount != 0.0)
            holder.binding.showDeductibleMet =
                (barInfo.familyExpenses?.familyDeductibleMaxAmount == barInfo.familyExpenses?.familyDeductibleAmount)
        }
        holder.binding.currentYear = currentYear
        val adapter = BenefitMemberListAdapter()
        val exDependentAdapter = BenefitMemberListAdapter()

        holder.binding.benefitList.adapter = adapter

        val dividerItemDecoration = DividerItemDecoration(context = context)
        holder.binding.benefitList.addItemDecoration(dividerItemDecoration)
        holder.binding.exDependentList.adapter = exDependentAdapter
        // Only show non-primary and non-ex-dependent members
        val filtered = item.memberList?.filter {
            !it.exDependent && it.relationshipDesc != "Insured"
        }
        // Show only ex-dependents
        val filteredExOnly = item.memberList?.filter {
            it.exDependent
        }
        holder.binding.dependentsCountString = resources.getString(
            R.string.dependents_count,
            filtered?.size ?: 0
        )
        holder.binding.exDependentsCountString = resources.getString(
            R.string.ex_dependents_count,
            filteredExOnly?.size ?: 0
        )
        adapter.submitList(filtered)
        exDependentAdapter.submitList(filteredExOnly)
        holder.binding.noDependents = filtered.isNullOrEmpty()
        holder.binding.noPastDependent = filteredExOnly.isNullOrEmpty()
    }

            barInfo.familyExpenses?.familyDeductibleMaxAmount?.let { value ->
