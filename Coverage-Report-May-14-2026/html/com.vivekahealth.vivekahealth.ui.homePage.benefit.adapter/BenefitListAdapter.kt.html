class BenefitListAdapter(
    private var reason: String,

) : ListAdapter<BenefitsDTO, RecyclerView.ViewHolder>(

) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return BenefitViewHolder(
            SingleCellBenefitPerMonthBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            )
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }
    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)
        var barInfo = barModel.values!!.find { it.name!!.contains(item.benefitName!!) }

        (holder as BenefitViewHolder).binding.benefit = item

        holder.binding.benefitNameText = item.benefitName?.let {
            it
        }
        holder.binding.reasonGone = (reason.isNullOrBlank() || reason.equals("null", true)) && item.reasons.isNullOrBlank()
//        holder.binding.reasonGone = eligible

        if (barInfo?.familyExpenses != null) {
            holder.binding.outGone = false
            holder.binding.inGone = false
            var maxWidth = HelperUtils.getWidth(holder.binding.nameView.context)
            var totalViewWidth = maxWidth
            var ratio: Double = 0.0
            ratio =
                (barInfo.familyExpenses!!.familyDeductibleMaxAmount?.toDouble() ?: 0.0) / (barInfo.familyExpenses!!.familyOutOfPocketMaxAmount?.toDouble() ?: 0.0)
            var deductibleMainWidth = totalViewWidth * (ratio)
            var temp = totalViewWidth * (1 - ratio)
            var coinsuranceMainWidth = temp * 0.8
            var deductibleWidthRatio: Double = 0.0
            deductibleWidthRatio = (barInfo.familyExpenses!!.familyDeductibleAmount ?: 0.0) / (barInfo.familyExpenses!!.familyDeductibleMaxAmount ?: 0.0)

            var deductibleWidth = deductibleWidthRatio * deductibleMainWidth
            var oopWidthRatio: Double = 0.0
            oopWidthRatio = (barInfo.familyExpenses!!.familyOutOfPocketAmount ?: 0.0) / (barInfo.familyExpenses!!.familyOutOfPocketMaxAmount ?: 0.0)
            var oopWidth = 0.0
            if ((barInfo.familyExpenses!!.familyDeductibleAmount ?: 0.0) >= (barInfo.familyExpenses!!.familyDeductibleMaxAmount ?: 0.0)) {
                oopWidth = oopWidthRatio * coinsuranceMainWidth
            }

            var ratioOon: Double = 0.0

            ratioOon = (barInfo.familyExpenses!!.familyOonDeductibleMaxAmount?.toDouble() ?: 0.0) / (barInfo.familyExpenses!!.familyOonOutOfPocketMaxAmount?.toDouble() ?: 0.0)
            var deductibleMainWidthOon = totalViewWidth * (ratioOon)
            var tempOon = totalViewWidth * (1 - ratioOon)
            var coinsuranceMainWidthOon = tempOon * 0.8
            var deductibleWidthRatioOon: Double = 0.0
            deductibleWidthRatioOon = (barInfo.familyExpenses!!.familyOonDeductibleAmount ?: 0.0) / (barInfo.familyExpenses!!.familyOonDeductibleMaxAmount ?: 0.0)

            var deductibleWidthOon = deductibleWidthRatioOon * deductibleMainWidthOon
            var oopWidthRatioOon: Double = 0.0
            oopWidthRatioOon = (barInfo.familyExpenses!!.familyOonOutOfPocketAmount ?: 0.0) / (barInfo.familyExpenses!!.familyOonOutOfPocketMaxAmount ?: 0.0)
            var oopWidthOon = 0.0
            if ((barInfo.familyExpenses!!.familyOonDeductibleAmount ?: 0.0) >= (barInfo.familyExpenses!!.familyOonDeductibleMaxAmount ?: 0.0)) {
                oopWidthOon = oopWidthRatioOon * coinsuranceMainWidthOon
            }

            if ((barInfo.familyExpenses!!.familyDeductibleMaxAmount ?: 0.0) >= (barInfo.familyExpenses!!.familyOutOfPocketMaxAmount ?: 0.0)) {
                deductibleMainWidth = 0.0
                holder.binding.barIn.deductibleSlide.visibility = View.GONE
                holder.binding.barIn.coinsuranceView.visibility = View.INVISIBLE

//                val containerParamsDed = holder.binding.barIn.deductibleSlide.layoutParams as ConstraintLayout.LayoutParams
//                containerParamsDed.endToEnd = holder.binding.barIn.coinsurance.id
//                holder.binding.barIn.deductibleView.layoutParams =
//                    ViewGroup.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT - 8, 20)

                val containerParams = holder.binding.barIn.coinsurance.layoutParams as ConstraintLayout.LayoutParams
                containerParams.startToStart = ConstraintLayout.LayoutParams.UNSET
                holder.binding.barIn.coinsurance.layoutParams = containerParams

                val containerParamsView = holder.binding.barIn.deductibleView.layoutParams as ConstraintLayout.LayoutParams
                containerParamsView.endToEnd = ConstraintLayout.LayoutParams.PARENT_ID
                holder.binding.barIn.deductibleView.layoutParams = containerParamsView

                coinsuranceMainWidth = 0.0
                val containerParamsEnd = holder.binding.barIn.linearLayout2.layoutParams as ConstraintLayout.LayoutParams
                containerParamsEnd.marginEnd = -2
                containerParamsEnd.startToEnd = ConstraintLayout.LayoutParams.UNSET
                containerParamsEnd.endToEnd = ConstraintLayout.LayoutParams.UNSET
                containerParamsEnd.topToBottom = holder.binding.barIn.coinsuranceView.id
                containerParamsEnd.endToEnd = holder.binding.barIn.deductibleView.id
                holder.binding.barIn.linearLayout2.layoutParams = containerParamsEnd

                val containerParamsText = holder.binding.barIn.textView23.layoutParams as ConstraintLayout.LayoutParams
                containerParamsText.topToBottom = holder.binding.barIn.linearLayout2.id
                containerParamsText.endToStart = holder.binding.barIn.linearLayout2.id
                holder.binding.barIn.textView23.layoutParams = containerParamsText
            } else {
                val containerParamsEnd = holder.binding.barIn.linearLayout2.layoutParams as ConstraintLayout.LayoutParams
                containerParamsEnd.endToEnd = ConstraintLayout.LayoutParams.PARENT_ID
                containerParamsEnd.startToEnd = holder.binding.barIn.deductibleSlide.id
                holder.binding.barIn.linearLayout2.layoutParams = containerParamsEnd
            }

            if ((barInfo.familyExpenses!!.familyOonDeductibleMaxAmount ?: 0.0) >= (barInfo.familyExpenses!!.familyOonOutOfPocketMaxAmount ?: 0.0)) {
//                deductibleMainWidthOon = (holder.binding.barOut.main.layoutParams.width - 8).toDouble()
                deductibleMainWidthOon = 0.0
                holder.binding.barOut.deductibleSlide.visibility = View.GONE
                holder.binding.barOut.coinsuranceView.visibility = View.INVISIBLE

//                val containerParamsDed = holder.binding.barIn.deductibleSlide.layoutParams as ConstraintLayout.LayoutParams
//                containerParamsDed.endToEnd = holder.binding.barIn.coinsurance.id
//                holder.binding.barIn.deductibleView.layoutParams =
//                    ViewGroup.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT - 8, 20)

                val containerParams = holder.binding.barOut.coinsurance.layoutParams as ConstraintLayout.LayoutParams
                containerParams.startToStart = ConstraintLayout.LayoutParams.UNSET
                holder.binding.barOut.coinsurance.layoutParams = containerParams

                val containerParamsView = holder.binding.barOut.deductibleView.layoutParams as ConstraintLayout.LayoutParams
                containerParamsView.endToEnd = ConstraintLayout.LayoutParams.PARENT_ID
                holder.binding.barOut.deductibleView.layoutParams = containerParamsView

                coinsuranceMainWidth = 0.0
                val containerParamsEnd = holder.binding.barOut.linearLayout2.layoutParams as ConstraintLayout.LayoutParams
                containerParamsEnd.marginEnd = -2
                containerParamsEnd.startToEnd = ConstraintLayout.LayoutParams.UNSET
                containerParamsEnd.endToEnd = ConstraintLayout.LayoutParams.UNSET
                containerParamsEnd.topToBottom = holder.binding.barOut.coinsuranceView.id
                containerParamsEnd.endToEnd = holder.binding.barOut.deductibleView.id
                holder.binding.barOut.linearLayout2.layoutParams = containerParamsEnd

                val containerParamsText = holder.binding.barOut.textView23.layoutParams as ConstraintLayout.LayoutParams
                containerParamsText.topToBottom = holder.binding.barOut.linearLayout2.id
                containerParamsText.endToStart = holder.binding.barOut.linearLayout2.id
                holder.binding.barOut.textView23.layoutParams = containerParamsText
            } else {
                val containerParamsEnd = holder.binding.barOut.linearLayout2.layoutParams as ConstraintLayout.LayoutParams
                containerParamsEnd.endToEnd = ConstraintLayout.LayoutParams.PARENT_ID
                containerParamsEnd.startToEnd = holder.binding.barOut.deductibleSlide.id
                holder.binding.barOut.linearLayout2.layoutParams = containerParamsEnd
            }
            holder.binding.inBar = InNetworkBarDetails(
                barInfo.familyExpenses!!.coinsurance,
                barInfo.familyExpenses!!.familyDeductibleMaxAmount?.toFloat() ?: 0.0f,
                barInfo.familyExpenses!!.familyOutOfPocketMaxAmount?.toFloat() ?: 0.0f,
                deductibleMainWidth.toInt(),
                deductibleWidth.toInt(),
                coinsuranceMainWidth.toInt(),
                oopWidth.toInt(),
                coinsuranceMainWidth.toInt(),
                0f,
                100f, holder.itemView.context.getString(R.string.family)
            )
            if ((barInfo.familyExpenses!!.familyDeductibleMaxAmount ?: 0.0) >= (barInfo.familyExpenses!!.familyOutOfPocketMaxAmount ?: 0.0)) {
                holder.binding.barIn.coinsuranceView.visibility = View.INVISIBLE
            } else {
                val containerParamsText = holder.binding.barIn.textView23.layoutParams as ConstraintLayout.LayoutParams
                containerParamsText.topToBottom = ConstraintLayout.LayoutParams.UNSET
                containerParamsText.topToBottom = holder.binding.barIn.deductibleSlide.id
                containerParamsText.endToStart = ConstraintLayout.LayoutParams.UNSET
                containerParamsText.startToStart = holder.binding.barIn.deductibleView.id
                holder.binding.barIn.textView23.layoutParams = containerParamsText
            }
            holder.binding.outBar = OutNetworkBarDetails(
                barInfo.familyExpenses!!.oonCoinsurance,
                barInfo.familyExpenses!!.familyOonDeductibleMaxAmount?.toFloat() ?: 0.0f,
                barInfo.familyExpenses!!.familyOonOutOfPocketMaxAmount?.toFloat() ?: 0.0f,
                deductibleMainWidthOon.toInt(),
                deductibleWidthOon.toInt(),
                coinsuranceMainWidthOon.toInt(),
                oopWidthOon.toInt(),
                coinsuranceMainWidthOon.toInt(),
                0f,
                100f, holder.itemView.context.getString(R.string.family)
            )

            if ((barInfo.familyExpenses!!.familyOonDeductibleMaxAmount ?: 0.0) >= (barInfo.familyExpenses!!.familyOonOutOfPocketMaxAmount ?: 0.0)) {
                holder.binding.barOut.coinsuranceView.visibility = View.INVISIBLE
            } else {
                val containerParamsOut = holder.binding.barOut.textView23.layoutParams as ConstraintLayout.LayoutParams
                containerParamsOut.topToBottom = holder.binding.barOut.deductibleSlide.id
                containerParamsOut.endToStart = ConstraintLayout.LayoutParams.UNSET
                containerParamsOut.startToStart = holder.binding.barOut.deductibleView.id
                holder.binding.barOut.textView23.layoutParams = containerParamsOut
            }
            holder.binding.inGone = barInfo.familyExpenses!!.familyOutOfPocketMaxAmount == null || barInfo.familyExpenses!!.familyOutOfPocketMaxAmount == 0.0
            holder.binding.outGone = barInfo.familyExpenses!!.familyOonOutOfPocketMaxAmount == null || barInfo.familyExpenses!!.familyOonOutOfPocketMaxAmount == 0.0
        } else {
            holder.binding.outGone = true
            holder.binding.inGone = true
        }
        holder.binding.setClickListener {
            if (eligible && holder.binding.reasonGone == true) {
                onClickBenefit(item)
            }
        }

        if (!item.benefitLogo.isNullOrEmpty()) {
            val imageBit = Base64.decode(item.benefitLogo.toByteArray(), Base64.DEFAULT)

            holder.binding.logo.setImageBitmap(
                BitmapFactory.decodeByteArray(
                    imageBit,
                    0,
                    imageBit.size
                )
            )
        }

        var adapterWork = WorkHourListBenefit()

        holder.binding.workHourList.adapter = adapterWork
        adapterWork.submitList(item.employerHourDTOList)

        holder.binding.reasonBtn.setOnClickListener {
            onClickReason(item.reasons)
        }

        var adapterBenefit = BenefitDependentListAdapter(onClick = { MemberBenefitDto ->
            Timber.d(holder.itemView.context.getString(R.string.add_dependent_benefit))

            onClick(MemberBenefitDto)
        })

        holder.binding.memberProfilePicture.adapter = adapterBenefit

        var bubbleMemberList: ArrayList<MemberBenefitDto> = arrayListOf()
        item.memberBenefitDtoList.let { listItem ->

            var grouped = listItem.groupBy { value -> value.relation }
            grouped.forEach {
                it.value.sortedByDescending { item ->
                    DateUtils.stringServerDateToDate(
                        item.dateOfBirth.toString()
                    )
                }
                it.value.reversed()
            }

            bubbleMemberList.clear()

            grouped.forEach {
                if (it.key.equals("Insured", true)) {
                    bubbleMemberList.add(0, it.value[0])
                } else if (it.key.equals("Spouse", true)) {
                    if (grouped.containsKey("Insured")) {
                        bubbleMemberList.add(1, it.value[0])
                    } else {
                        bubbleMemberList.add(0, it.value[0])
                    }
                } else if (it.key.equals("Child", true)) {
                    val dateGroup = it.value.groupBy { date ->
                        DateUtils.stringServerDateToDate(
                            date.dateOfBirth.toString()
                        )
                    }

                    dateGroup.forEach { item ->
                        val sorted = item.value.sortedBy { item1 -> item1.firstName }
                        bubbleMemberList.addAll(sorted)
                    }
                }
            }
            val tempOthers: ArrayList<MemberBenefitDto> = arrayListOf()
            grouped.forEach {
                if (!it.key.equals("Insured", true) && !it.key.equals(
                        "Spouse",
                        true
                    ) && !it.key.equals("Child", true)
                ) {
                    tempOthers.addAll(it.value)
                }
            }

            val dateGroup = tempOthers.groupBy { date ->
                DateUtils.stringServerDateToDate(
                    date.dateOfBirth.toString()
                )
            }
            dateGroup.forEach { item ->
                val sorted = item.value.sortedBy { item1 -> item1.firstName }
                bubbleMemberList.addAll(sorted)
            }
        }

        adapterBenefit.submitList(bubbleMemberList)

//        holder.binding.clicked = position == itemCount-1
    }

    class BenefitViewHolder(

private class BenefitDiffCallback : DiffUtil.ItemCallback<BenefitsDTO>() {

    override fun areItemsTheSame(oldItem: BenefitsDTO, newItem: BenefitsDTO): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: BenefitsDTO, newItem: BenefitsDTO): Boolean {
        return oldItem == newItem
    }
}
