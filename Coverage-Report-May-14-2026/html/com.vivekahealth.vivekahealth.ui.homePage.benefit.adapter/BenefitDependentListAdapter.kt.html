package com.vivekahealth.vivekahealth.ui.homePage.benefit.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.data.remote.home.*
import com.vivekahealth.vivekahealth.databinding.*
import timber.log.Timber

class BenefitDependentListAdapter(private var onClick: (MemberBenefitDto) -> Unit) : ListAdapter<MemberBenefitDto, RecyclerView.ViewHolder>(
    MemberDepDiffCallback()
) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return MemberBenefitViewHolder(
            SingleRowBubbleBenefitBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            )
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as MemberBenefitViewHolder).binding.member = item
        holder.binding.setClickListener {
            Timber.d(holder.itemView.context.getString(R.string.add_dependent_benefit))

            onClick(item)
        }

//        holder.binding.clicked = position == itemCount-1
    }

    class MemberBenefitViewHolder(
        val binding: SingleRowBubbleBenefitBinding
    ) : RecyclerView.ViewHolder(binding.root) {

        //        var toggle = binding.clicked
        init {

//            binding.clicked = adapterPosition == count
        }
    }
}

private class MemberDepDiffCallback : DiffUtil.ItemCallback<MemberBenefitDto>() {

    override fun areItemsTheSame(oldItem: MemberBenefitDto, newItem: MemberBenefitDto): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: MemberBenefitDto, newItem: MemberBenefitDto): Boolean {
        return oldItem == newItem
    }
}
