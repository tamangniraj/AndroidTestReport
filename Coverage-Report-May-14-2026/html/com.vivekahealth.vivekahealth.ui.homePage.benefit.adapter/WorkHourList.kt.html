package com.vivekahealth.vivekahealth.ui.homePage.benefit.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.*
import com.vivekahealth.vivekahealth.databinding.SingleCellEmployerListBinding
import com.vivekahealth.vivekahealth.utils.DateUtils
import java.text.DecimalFormat

class WorkHourList() : ListAdapter<EmployerDTO, RecyclerView.ViewHolder>(EmployerDiffCallback()) {

//    var benefitList = benefitsDTOLists
    var maxWidth: Int = 0

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return EmployerViewHolder(
            SingleCellEmployerListBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            )
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as EmployerViewHolder).binding.name = item.employerName
        val df = DecimalFormat("#.##")

        holder.binding.hour = df.format(item.hoursContributed).toString()
        holder.binding.year = DateUtils.getMonthYearFromServerDateSlashedToMMMMonth(
            item.month.toString() + "/" + item.year.toString()
        )

//            item.yearMonthString

        holder.itemView.post {
            val layoutParamsCell: ViewGroup.LayoutParams = holder.binding.cell.layoutParams

            val cellWidth = layoutParamsCell.width

            if (maxWidth < cellWidth) {
                maxWidth = cellWidth
                val layoutParams: ViewGroup.LayoutParams = holder.binding.cell.layoutParams
                layoutParams.width = maxWidth

                notifyDataSetChanged()
            }
        }

//        holder.binding.clicked = position == itemCount-1
    }

    class EmployerViewHolder(
        val binding: SingleCellEmployerListBinding
    ) : RecyclerView.ViewHolder(binding.root) {

        //        var toggle = binding.clicked
        init {

//            binding.clicked = adapterPosition == count
        }
    }
}

private class EmployerDiffCallback : DiffUtil.ItemCallback<EmployerDTO>() {

    override fun areItemsTheSame(oldItem: EmployerDTO, newItem: EmployerDTO): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: EmployerDTO, newItem: EmployerDTO): Boolean {
        return oldItem == newItem
    }
}
