package com.vivekahealth.vivekahealth.ui.homePage.benefit.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.workBenefit.benefitPerMonth.ResponseDataMemberBenefitPerMonth
import com.vivekahealth.vivekahealth.databinding.DotLayoutBinding

class EligibleBenefitScrollAllAdapter() : ListAdapter<ResponseDataMemberBenefitPerMonth, RecyclerView.ViewHolder>(
    EligibleBenefitScrollAllDiffCallback()
) {
    var currentState: Int? = null

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return EligibleBenefitViewHolder(
            DotLayoutBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            )
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }
    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)
        (holder as EligibleBenefitViewHolder).binding.selected = item.benefitId == currentState
//        holder.binding.currentState = currentState
    }

    fun currentState(state: Int) {
        val prev = currentList.indexOf(
            currentList.find { item ->
                item.benefitId == currentState
            }
        )
        currentState = state
        val next = currentList.indexOf(
            currentList.find { item ->
                item.benefitId == state
            }
        )
        notifyItemChanged(prev)
        notifyItemChanged(next)
    }
    class EligibleBenefitViewHolder(
        val binding: DotLayoutBinding
    ) : RecyclerView.ViewHolder(binding.root)
}

private class EligibleBenefitScrollAllDiffCallback : DiffUtil.ItemCallback<ResponseDataMemberBenefitPerMonth>() {

    override fun areItemsTheSame(
        oldItem: ResponseDataMemberBenefitPerMonth,
        newItem: ResponseDataMemberBenefitPerMonth
    ): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(
        oldItem: ResponseDataMemberBenefitPerMonth,
        newItem: ResponseDataMemberBenefitPerMonth
    ): Boolean {
        return oldItem == newItem
    }
}
