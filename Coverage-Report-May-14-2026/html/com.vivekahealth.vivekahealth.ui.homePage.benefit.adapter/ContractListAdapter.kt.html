package com.vivekahealth.vivekahealth.ui.homePage.benefit.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.hoursSummary.ContractBasicInfoDTO
import com.vivekahealth.vivekahealth.databinding.CellEmployerListBinding
import com.vivekahealth.vivekahealth.utils.Constants

class ContractListAdapter(private var onClick: (Int) -> Unit) : ListAdapter<ContractBasicInfoDTO, RecyclerView.ViewHolder>(
    ContractDiffCallback()
) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return YearViewHolder(
            CellEmployerListBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)
        (holder as YearViewHolder).binding.contract = item.contractName
        holder.binding.bankHours = item.bankedHour
        holder.binding.hasHoursBank = Constants.hasHoursBank
    }

    class YearViewHolder(
        val binding: CellEmployerListBinding,
        private val onClick: (Int) -> Unit
    ) : RecyclerView.ViewHolder(binding.root) {

        //        var toggle = binding.clicked
        init {

//            binding.clicked = adapterPosition == count
        }
    }
}

private class ContractDiffCallback : DiffUtil.ItemCallback<ContractBasicInfoDTO>() {
    override fun areItemsTheSame(oldItem: ContractBasicInfoDTO, newItem: ContractBasicInfoDTO): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: ContractBasicInfoDTO, newItem: ContractBasicInfoDTO): Boolean {
        return oldItem == newItem
    }
}
