package com.vivekahealth.vivekahealth.ui.homePage.benefit.adapter

import android.content.res.Resources
import android.graphics.Rect
import android.util.TypedValue
import android.view.View
import androidx.recyclerview.widget.RecyclerView

class CustomItemDecoration(
    private val firstItemMargin: Float,
    private val lastItemMargin: Float,
    private val otherItemMargin: Float,
    private val resources: Resources
) : RecyclerView.ItemDecoration() {

    override fun getItemOffsets(
        outRect: Rect,
        view: View,
        parent: RecyclerView,
        state: RecyclerView.State
    ) {
        val position = parent.getChildAdapterPosition(view)
        val itemCount = parent.adapter?.itemCount ?: 0

        val displayMetrics = resources.displayMetrics

        if (position == 0) {
            outRect.left = TypedValue.applyDimension(
                TypedValue.COMPLEX_UNIT_DIP,
                firstItemMargin,
                displayMetrics
            ).toInt()
            outRect.right = TypedValue.applyDimension(
                TypedValue.COMPLEX_UNIT_DIP,
                otherItemMargin,
                displayMetrics
            ).toInt()
        } else if (position == itemCount - 1) {
            outRect.left = TypedValue.applyDimension(
                TypedValue.COMPLEX_UNIT_DIP,
                otherItemMargin,
                displayMetrics
            ).toInt()
            outRect.right = TypedValue.applyDimension(
                TypedValue.COMPLEX_UNIT_DIP,
                lastItemMargin,
                displayMetrics
            ).toInt()
        } else {
            outRect.left = TypedValue.applyDimension(
                TypedValue.COMPLEX_UNIT_DIP,
                otherItemMargin,
                displayMetrics
            ).toInt()
            outRect.right = TypedValue.applyDimension(
                TypedValue.COMPLEX_UNIT_DIP,
                otherItemMargin,
                displayMetrics
            ).toInt()
        }
    }
}
