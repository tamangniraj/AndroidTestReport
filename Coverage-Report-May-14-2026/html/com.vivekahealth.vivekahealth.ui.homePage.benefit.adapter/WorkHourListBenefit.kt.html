        val item = getItem(position)

        (holder as EmployerViewHolder).binding.name = item.employerName

            holder.binding.bankedHours = df.format(item.bankedHoursRemaining).toString()
            holder.binding.hoursBanked = true
        } else {
            holder.binding.hour = df.format(item.hoursContributed).toString()
            holder.binding.hoursBanked = false

private class EmployerBenefitDiffCallback : DiffUtil.ItemCallback<EmployerHourDTO>() {

    override fun areItemsTheSame(oldItem: EmployerHourDTO, newItem: EmployerHourDTO): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: EmployerHourDTO, newItem: EmployerHourDTO): Boolean {
        return oldItem == newItem
