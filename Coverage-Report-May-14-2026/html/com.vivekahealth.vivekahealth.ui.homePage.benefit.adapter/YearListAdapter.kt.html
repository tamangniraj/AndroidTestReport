package com.vivekahealth.vivekahealth.ui.homePage.benefit.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.databinding.SingleCellYearBinding
import java.util.*

class YearListAdapter(private var onClick: (Int) -> Unit) : ListAdapter<Int, RecyclerView.ViewHolder>(
    MemberDiffCallback()
) {

    var selectedPosition: Int = 0
    var lastSelected: Int = -1
    var count: Int = 0
    var currentYearPos: Int = 0

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return YearViewHolder(
            SingleCellYearBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick,
            selectedPosition,
            count
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        count = itemCount - 1
        (holder as YearViewHolder).binding.setClickListener {
            notifyItemChanged(selectedPosition)

            if (lastSelected == -1) {
                notifyItemChanged(itemCount - 1)

                lastSelected = position
            } else {
                notifyItemChanged(lastSelected)
                lastSelected = position
            }

            selectedPosition = position

            onClick(item)
            notifyItemChanged(position)
        }

        if (lastSelected == -1) {
            if (Calendar.getInstance().get(Calendar.YEAR) == item) {
                currentYearPos = position
                selectedPosition = position
            }

            holder.binding.clicked = Calendar.getInstance().get(Calendar.YEAR) == item

            if (!currentList.contains(Calendar.getInstance().get(Calendar.YEAR))) {
                if (getItem(itemCount - 1) != Calendar.getInstance().get(Calendar.YEAR)) {
                    holder.binding.clicked = position == itemCount - 1
                }
            }
//            else {
//                holder.binding.clicked = position == itemCount - 1
//            }
        } else {
            holder.binding.clicked = position == selectedPosition
        }

        holder.binding.work = item

//        holder.binding.clicked = position == itemCount-1
    }

    fun selectYear(year: Int) {
        var selected = currentList.indexOf(year)
        notifyItemChanged(selectedPosition)

        if (lastSelected == -1) {
            notifyItemChanged(itemCount - 1)

            lastSelected = selected
        } else {
            notifyItemChanged(lastSelected)
            lastSelected = selected
        }

        selectedPosition = selected

        onClick(currentList[selected])
        notifyItemChanged(selected)
    }

    class YearViewHolder(
        val binding: SingleCellYearBinding,
        private val onClick: (Int) -> Unit,
        private var selectedPosition: Int,
        private var count: Int
    ) : RecyclerView.ViewHolder(binding.root) {

        //        var toggle = binding.clicked
        init {

//            binding.clicked = adapterPosition == count
        }
    }
}

private class MemberDiffCallback : DiffUtil.ItemCallback<Int>() {

    override fun areItemsTheSame(oldItem: Int, newItem: Int): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: Int, newItem: Int): Boolean {
        return oldItem == newItem
    }
}
