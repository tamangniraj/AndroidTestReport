package com.vivekahealth.vivekahealth.ui.homePage.benefit.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.workBenefit.workLog.BenefitsLogo
import com.vivekahealth.vivekahealth.data.remote.home.workBenefit.workLog.WorkLog
import com.vivekahealth.vivekahealth.databinding.YearTilesCellBinding
import com.vivekahealth.vivekahealth.utils.DateUtils
import com.vivekahealth.vivekahealth.utils.MonthCardUtils

class YearTilesAdapter(
    private var onClick: (Int, Int, Boolean) -> Unit,
    private var benefitsLogos: List<BenefitsLogo>?,
    private var currentMonth: Int,
    private var currentYear: Int
) : ListAdapter<WorkLog, RecyclerView.ViewHolder>(
    YearTilesDiffCallback()
) {

    var currentYearSelected = currentYear
    var selectedPosition: Int = 0
    var lastSelected: Int = currentMonth - 1
    var count: Int = 0
    var currentYearPos: Int = 0

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return YearTileViewHolder(
            YearTilesCellBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)
        (holder as YearTileViewHolder).binding.setTileClick {
            if (item.isSelected) {
                // Do nothing if already selected
                return@setTileClick
            }
            notifyItemChanged(selectedPosition)

            if (lastSelected == -1) {
                notifyItemChanged(itemCount - 1)

                lastSelected = position
            } else {
                val lastItem = currentList.find { it.isSelected == true }
                lastItem?.isSelected = false
                notifyItemChanged(currentList.indexOf(lastItem))
                lastSelected = position
                item.isSelected = true
            }

            selectedPosition = position

            onClick(item.month.toInt(), item.year.toInt(), item.addedItem)
            notifyItemChanged(position)
        }
        holder.binding.month = item.month
        holder.binding.hours = item.hoursWorked
        holder.binding.premium = item.memberContribution
        holder.binding.eligible = item.eligibilityStatus == "ELIGIBLE"
        holder.binding.isFutureAddedMonth = isFutureMonthCard(item.year, item.month) && item.addedItem
        if (!item.employerDTOList.isNullOrEmpty()) {
            holder.binding.employerName = item.employerDTOList.first().shopName ?: ""
        }
        item.employerDTOList?.let {
            if (item.employerDTOList.isNotEmpty()) {
                holder.binding.employerCount = (item.employerDTOList.count() - 1).toString()
            } else {
                holder.binding.employerCount = "0"
            }
        }
        if ((item.month.isNotBlank()) && item.year.isNotBlank()) {
            holder.binding.clicked = item.month.toInt() == currentMonth && item.year.toInt() == currentYearSelected
        } else {
            holder.binding.clicked = item.month.toInt() == currentMonth && DateUtils.getCurrentYear() == currentYearSelected
        }
        holder.binding.clicked = item.isSelected
        holder.binding.selected = item.isSelected
        holder.binding.reason = item.inEligibilityReasons
        val noMemberContribution = item.memberContribution == null || item.memberContribution == 0.0
        val noHoursWorked = item.hoursWorked == null || item.hoursWorked == 0.0
        holder.binding.hidePremium = noMemberContribution || !noHoursWorked

        val eligible = item.eligibleBenefits?.map {
            val benefitLogo = benefitsLogos?.first { benefitLogo -> benefitLogo.type == it }
            benefitLogo
        }

        val adapter = EligibleBenefitListAdapter(eligible?.size ?: 0)
        holder.binding.benefitList.adapter = adapter
        adapter.submitList(eligible?.take(5))
    }

    private fun isFutureMonthCard(year: String, month: String): Boolean {
        return MonthCardUtils.isFutureMonthCard(
            year,
            month,
            DateUtils.getCurrentYear(),
            DateUtils.getCurrentMonthInt()
        )
    }

    fun currentYear() {
        val selected = currentList.indexOf(
            currentList.find { item -> item.year.toInt() == currentYearSelected && item.month.toInt() == currentMonth }
        )
        notifyItemChanged(selected)
    }
    class YearTileViewHolder(
        val binding: YearTilesCellBinding,
        private val onClick: (Int, Int, Boolean) -> Unit
    ) : RecyclerView.ViewHolder(binding.root) {

        //        var toggle = binding.clicked
        init {

//            binding.clicked = adapterPosition == count
        }
    }
}

private class YearTilesDiffCallback : DiffUtil.ItemCallback<WorkLog>() {
    override fun areItemsTheSame(oldItem: WorkLog, newItem: WorkLog): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: WorkLog, newItem: WorkLog): Boolean {
        return oldItem.month == newItem.month && oldItem.year == newItem.year
    }
}
