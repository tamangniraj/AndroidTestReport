package com.vivekahealth.vivekahealth.ui.homePage.home.di

import android.content.Context
import android.content.SharedPreferences
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.ui.enrollVTwo.continueEnrollment.ContinuePreviousEnrollmentV2Fragment
import com.vivekahealth.vivekahealth.ui.enrollVTwo.start.StartEnrollmentDialog
import com.vivekahealth.vivekahealth.ui.homePage.home.HomeFragment
import com.vivekahealth.vivekahealth.ui.homePage.service.HomeService
import com.vivekahealth.vivekahealth.ui.homePage.surveys.NewSurveyAvailableFragment
import com.vivekahealth.vivekahealth.ui.homePage.surveys.SplashScreenFragment
import dagger.Module
import dagger.Provides
import dagger.android.ContributesAndroidInjector
import okhttp3.OkHttpClient

@Module
abstract class HomeModule {

    @ContributesAndroidInjector
    abstract fun contributesHomeFragment(): HomeFragment

    @ContributesAndroidInjector
    abstract fun contributeStartEnrollment(): StartEnrollmentDialog

    @ContributesAndroidInjector
    abstract fun contributeContinuePreviousEnrollmentV2(): ContinuePreviousEnrollmentV2Fragment

    @ContributesAndroidInjector
    abstract fun contributesSplashScreenFragment(): SplashScreenFragment

    @ContributesAndroidInjector
    abstract fun contributesNewSurveyAvailable(): NewSurveyAvailableFragment

    companion object {
        @JvmStatic
        @Provides
        fun provideHomeService(
            gson: Gson,
            builder: OkHttpClient.Builder,
            sharedPreferences: SharedPreferences,
            context: Context
        ): HomeService = HomeService(
            context = context,
            gson = gson,
            builder = builder,
            preferences = sharedPreferences
        )
    }
}
