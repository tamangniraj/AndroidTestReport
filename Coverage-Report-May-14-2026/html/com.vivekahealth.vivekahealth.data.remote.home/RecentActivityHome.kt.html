package com.vivekahealth.vivekahealth.data.remote.home

import androidx.room.Entity
import androidx.room.PrimaryKey
import androidx.room.TypeConverters
import com.vivekahealth.vivekahealth.data.local.typeConverter.VivekaTypeConverters

@Entity(tableName = "recent_activity_home")
@TypeConverters(VivekaTypeConverters::class)
data class RecentActivityHome(
    @PrimaryKey
    var selfId: String,
    @TypeConverters(VivekaTypeConverters::class)
    val claimsQuickViewList: List<ClaimsQuickView> = emptyList(),
    @TypeConverters(VivekaTypeConverters::class)
    val diagnosisHistoryQuickViewList: List<DiagnosisHistoryQuickView> = emptyList(),
    @TypeConverters(VivekaTypeConverters::class)
    val labResultQuickViewList: List<LabResultQuickView> = emptyList(),
    @TypeConverters(VivekaTypeConverters::class)
    val prescriptionQuickViewList: List<PrescriptionQuickView> = emptyList()
)
