package com.vivekahealth.vivekahealth.data.remote.home

import com.vivekahealth.vivekahealth.utils.HelperUtils
import java.util.Date

data class PrescriptionQuickView(
    val dateOfService: String,
    val dose: Double,
    val drugCode: String,
    val drugName: String,
    val measurementCode: String ? = null,
    val quantity: Double,
    var viewType: Int = 1,
    var sortingDate: Date? = null
) {

    fun doseConverted(): String = HelperUtils.floor(dose)
    fun quantityConverted(): String = HelperUtils.floor(quantity)
}
