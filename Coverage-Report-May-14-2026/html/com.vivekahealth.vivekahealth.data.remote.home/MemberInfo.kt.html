object MemberDiff : DiffUtil.ItemCallback<MemberInfo>() {
    override fun areItemsTheSame(oldItem: MemberInfo, newItem: MemberInfo) =
        oldItem.selfId == newItem.selfId

    override fun areContentsTheSame(oldItem: MemberInfo, newItem: MemberInfo) = oldItem == newItem
}

object MemberDiffShare : DiffUtil.ItemCallback<MemberInfoShare>() {
    override fun areItemsTheSame(oldItem: MemberInfoShare, newItem: MemberInfoShare) =
        oldItem.selfId == newItem.selfId

    override fun areContentsTheSame(oldItem: MemberInfoShare, newItem: MemberInfoShare) = oldItem == newItem
}
