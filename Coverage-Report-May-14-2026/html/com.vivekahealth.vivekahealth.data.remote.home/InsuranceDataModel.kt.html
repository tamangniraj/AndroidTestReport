package com.vivekahealth.vivekahealth.data.remote.home

import androidx.room.Embedded
import androidx.room.Entity
import androidx.room.PrimaryKey
import androidx.room.Relation

data class InsuranceDataModel(
    var selfId: String = "",
    var values: List<ValueInsurance>,
    var updatedDate: String? = null
)

@Entity(tableName = "insurance_data_new")
data class InsuranceDataNewModel(
    @PrimaryKey
    var selfId: Long,
    var updatedDate: String? = null
)

@Entity(tableName = "card_data", primaryKeys = ["selfId", "benefitType"])
data class CardsInsurance(
    var selfId: Long,
    var front: String? = null,
    val back: String? = null,
    val benefitType: String,
    val indexOrder: Int
)

data class InsuranceWithCards(
    @Embedded val data: InsuranceDataNewModel,
    @Relation(
        parentColumn = "selfId",
        entityColumn = "selfId"
    )
    val cards: List<CardsInsurance>
)
