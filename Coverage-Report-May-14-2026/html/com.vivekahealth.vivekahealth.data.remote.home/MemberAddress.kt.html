package com.vivekahealth.vivekahealth.data.remote.home

import com.google.gson.annotations.Expose
import com.google.gson.annotations.SerializedName

data class MemberAddress(
    @SerializedName("street1")
    @Expose
    var street1: String? = "",

    @SerializedName("street2")
    @Expose
    var street2: String? = "",

    @SerializedName("city")
    @Expose
    var city: String? = "",

    @SerializedName("state")
    @Expose
    var state: String? = "",

    @SerializedName("zip")
    @Expose
    var zip: String? = "",

    @SerializedName("zipExtension")
    @Expose
    var zipExtension: String? = "",

    @SerializedName("latitude")
    @Expose
    var latitude: Double? = null,

    @SerializedName("longitude")
    @Expose
    var longitude: Double? = null,

    @SerializedName("country")
    @Expose
    var country: String? = "",

    @SerializedName("addressStatus")
    @Expose
    var addressStatus: String? = "",

    @SerializedName("current")
    @Expose
    var current: Boolean = false,

    @SerializedName("type")
    @Expose
    var type: String? = "",

    @SerializedName("checkEnrollment")
    @Expose
    var checkEnrollment: Boolean = false,

    @SerializedName("postalCode")
    @Expose
    var postalCode: String? = ""
) {
    /**
     * Returns the full address. Skip parts if they are `null` or blank.
     */
    fun fullAddress(): String = buildString {
        if (!street1.isNullOrBlank()) {
            append("$street1, ")
        }
        if (!street2.isNullOrBlank()) {
            append("$street2, ")
        }
        if (!city.isNullOrBlank()) {
            append("$city, ")
        }
        if (!state.isNullOrBlank()) {
            append("$state, ")
        }
        if (!postalCode.isNullOrBlank()) {
            append("$postalCode")
        }
    }
}
