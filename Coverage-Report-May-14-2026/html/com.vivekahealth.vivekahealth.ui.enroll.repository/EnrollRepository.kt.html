package com.vivekahealth.vivekahealth.ui.enroll.repository

import com.vivekahealth.vivekahealth.data.remote.enroll.DocumentUploadSuccessEnroll
import com.vivekahealth.vivekahealth.data.remote.enroll.EnrollMember
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.enroll.service.EnrollService
import io.reactivex.Single
import javax.inject.Inject
import javax.inject.Singleton
import okhttp3.MultipartBody

/**
 * Repository that handles User objects.
 */
@Singleton
class EnrollRepository @Inject constructor(
    private val enrollService: EnrollService
) {

    fun enrollMember(member: EnrollMember, memberId: String): Single<DataResponse> = enrollService.getEnrollApi().getEnroll(
        member,
        memberId
    )
    fun uploadDoc(filePart: MultipartBody.Part, filename: String): Single<DocumentUploadSuccessEnroll> = enrollService.getEnrollApi().getDoc(
        filePart,
        filename
    )
}
