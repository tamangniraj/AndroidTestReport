package com.vivekahealth.vivekahealth.ui.enrollVTwo.di

import com.vivekahealth.vivekahealth.di.ActivityScope
import com.vivekahealth.vivekahealth.ui.enroll.fragments.spouse.di.SpouseModule
import com.vivekahealth.vivekahealth.ui.enroll.fragments.uploadDocuments.uploadUI.di.DocumentUploadModule
import com.vivekahealth.vivekahealth.ui.enrollVTwo.EnrollmentV2Activity
import com.vivekahealth.vivekahealth.ui.enrollVTwo.dependentStatus.di.DependentStatusModule
import com.vivekahealth.vivekahealth.ui.homePage.accountsDetails.pickerBottomSheet.di.PickerBottomSheetModule
import dagger.Module
import dagger.android.ContributesAndroidInjector

@Module
abstract class EnrollV2Module {

    @ActivityScope
    @ContributesAndroidInjector(
        modules = [
            DependentStatusModule::class,
            SpouseModule::class,
            PickerBottomSheetModule::class,
            DocumentUploadModule::class
        ]
    )

    abstract fun contributesEnrollNewActivity(): EnrollmentV2Activity
}
