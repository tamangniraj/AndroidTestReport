package com.vivekahealth.vivekahealth.ui.login.service

import android.content.SharedPreferences
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.BuildConfig
import com.vivekahealth.vivekahealth.ui.login.api.LoginApi
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.remote.RxErrorHandlingCallAdapterFactory
import java.io.IOException
import java.util.Locale
import java.util.concurrent.TimeUnit
import javax.inject.Inject
import okhttp3.*
import okhttp3.logging.HttpLoggingInterceptor
import retrofit2.Retrofit
import retrofit2.converter.gson.GsonConverterFactory

class LoginService @Inject constructor(
    private val gson: Gson,
    private val builder: OkHttpClient.Builder,
    private val preferences: SharedPreferences,
    private val baseUrl: String = BuildConfig.SERVER_URL
) {

    private var sharedPreferences: SharedPreferences = preferences
    private var loginApi: LoginApi? = null

    private fun getOkHttpClient(): OkHttpClient {
        val logging = HttpLoggingInterceptor()
        if (BuildConfig.DEBUG) {
            logging.level = HttpLoggingInterceptor.Level.BODY
        } else {
            logging.level = HttpLoggingInterceptor.Level.NONE
        }
        builder.readTimeout(45, TimeUnit.SECONDS)
            .writeTimeout(45, TimeUnit.SECONDS)
            .connectTimeout(45, TimeUnit.SECONDS)
            .followRedirects(false)
            .followSslRedirects(false)
            .retryOnConnectionFailure(false).addInterceptor(logging)
        builder.authenticator { route, response ->

            if (response.code == 401) {
                null
            } else {
                response.request.newBuilder()
                    .header(
                        "Accept-Language",
                        Constants.LOCAL_LANG_MODIFIED ?: Locale.getDefault().language.toString()
                    )
                    .header("Device-Id", Constants.deviceUUID)
                    .header("App-Version", Constants.versionNumber)
                    .header("Device-Info", Constants.deviceID)
                    .header("Client-ID", Constants.clientID)
                    .header("X-API-Version", Constants.apiVersion)
                    .build()
            }
        }.addInterceptor(SupportInterceptor(sharedPreferences))

        return builder
            .build()
    }

    private fun createLogInApiRetrofit(): Retrofit {
        return Retrofit.Builder()
            .baseUrl(baseUrl)
            .addConverterFactory(GsonConverterFactory.create())
            .addCallAdapterFactory(RxErrorHandlingCallAdapterFactory.create())
            .client(getOkHttpClient())
            .build()
    }

    fun getLoginApi(): LoginApi {
        if (loginApi == null) {
            loginApi = createLogInApiRetrofit().create(LoginApi::class.java)
        }
        return loginApi!!
    }

    class SupportInterceptor @Inject constructor(
        private val sharedPreferences: SharedPreferences
    ) : Interceptor {

        /**
         * Interceptor class for setting of the headers for every request
         */
        @Throws(IOException::class)

        override fun intercept(chain: Interceptor.Chain): Response {
//

            var request = chain.request().newBuilder()
            request.addHeader("Accept-Language", Locale.getDefault().language.toString())
            request.addHeader("Device-Id", Constants.deviceUUID)
            request.addHeader("App-Version", Constants.versionNumber)
            request.addHeader("Device-Info", Constants.deviceID)
            request.addHeader("Client-ID", Constants.clientID)
            request.header("X-API-Version", Constants.apiVersion)

            return chain.proceed(request.build())
        }
    }
}
