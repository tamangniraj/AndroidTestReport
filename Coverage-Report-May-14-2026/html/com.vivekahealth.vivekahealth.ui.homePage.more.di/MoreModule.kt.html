package com.vivekahealth.vivekahealth.ui.homePage.more.di
import android.content.SharedPreferences
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.ui.homePage.more.MoreFragment
import com.vivekahealth.vivekahealth.ui.homePage.surveys.SurveyOptOutConfirmDialogFragment
import com.vivekahealth.vivekahealth.ui.homePage.surveys.SurveysFragment
import com.vivekahealth.vivekahealth.ui.register.step1.service.PhoneService
import dagger.Module
import dagger.Provides
import dagger.android.ContributesAndroidInjector
import okhttp3.OkHttpClient

@Module
abstract class MoreModule {

    @ContributesAndroidInjector
    abstract fun contributesMoreFragment(): MoreFragment

    @ContributesAndroidInjector
    abstract fun contributesSurveysFragment(): SurveysFragment

    @ContributesAndroidInjector
    abstract fun contributesSurveyOptOutConfirmDialogFragment(): SurveyOptOutConfirmDialogFragment

    companion object {
        @JvmStatic
        @Provides
        fun provideMoreFragmentService(
            gson: Gson,
            builder: OkHttpClient.Builder,
            sharedPreferences: SharedPreferences
        ): PhoneService = PhoneService(
            gson = gson,
            builder = builder,
            preferences = sharedPreferences
        )
    }
}
