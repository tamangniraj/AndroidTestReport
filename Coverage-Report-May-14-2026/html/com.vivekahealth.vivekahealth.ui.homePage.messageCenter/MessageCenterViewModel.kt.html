package com.vivekahealth.vivekahealth.ui.homePage.messageCenter

import android.content.SharedPreferences

import com.vivekahealth.vivekahealth.data.local.home.more.MenuType

enum class NotificationType(val value: Int) {
    DOCUMENT(3),
    DOCUMENT_ALT(5),
    CUSTOMER_CARE(4),
    BENEFIT(8);

    companion object {
        fun fromValue(value: Int?) = entries.find { it.value == value }
    }
}

sealed class NotificationNavEvent {
    object ToBenefit : NotificationNavEvent()
    object ToCustomerCare : NotificationNavEvent()
    object ToDocuments : NotificationNavEvent()
    data class ToDynamicMenu(val menuTypeId: String, val menuName: String) : NotificationNavEvent()
    data class ShowDialog(val title: String, val message: String) : NotificationNavEvent()
    data class ShowGuestDialog(val title: String, val message: String) : NotificationNavEvent()
}

class MessageCenterViewModel @javax.inject.Inject constructor(
    private val preferences: SharedPreferences,
    private val medicalLogsRepository: MedicalLogsRepository,
    private val homeRepository: HomeRepository,
    private val analyticsRepository: AnalyticsRepository,
    private val guestHomeRepository: GuestHomeRepository
) : ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    var isLastPage = false
    var isScrolling = false
    var sourcePage = ""
    var notificationList: ArrayList<MessageCenterData> = arrayListOf()
    var data: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.data, null)
        }
    var reason: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.reason, null)
        }
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    var page: Int = 1
    private val _notificationListGenerated = MutableLiveData<Event<ArrayList<MessageCenterData>>>()
    val notificationListGenerated: LiveData<Event<ArrayList<MessageCenterData>>>
        get() = _notificationListGenerated

    private val _navigationEvent = MutableLiveData<Event<NotificationNavEvent>>()
    val navigationEvent: LiveData<Event<NotificationNavEvent>> get() = _navigationEvent

    fun onNotificationClicked(item: MessageCenterData) {
        if (preferences.getBoolean(Constants.IS_GUEST, false)) {
            _navigationEvent.value = Event(
                NotificationNavEvent.ShowGuestDialog(item.title.orEmpty(), item.message.orEmpty())
            )
            return
        }
        when (NotificationType.fromValue(item.notificationType)) {
            NotificationType.DOCUMENT, NotificationType.DOCUMENT_ALT -> {
                val faqMenu = activeCustomSubMenus?.find { it.menuTypeId == MenuType.DOCUMENT_ST_LOIS.id }
                _navigationEvent.value = if (faqMenu != null) {
                    Event(
                        NotificationNavEvent.ToDynamicMenu(
                            faqMenu.menuTypeId?.toString().orEmpty(),
                            faqMenu.menuName.orEmpty()
                        )
                    )
                } else {
                    Event(NotificationNavEvent.ToDocuments)
                }
            }
            NotificationType.CUSTOMER_CARE -> _navigationEvent.value = Event(
                NotificationNavEvent.ToCustomerCare
            )
            NotificationType.BENEFIT -> _navigationEvent.value = Event(
                NotificationNavEvent.ToBenefit
            )
            null -> _navigationEvent.value = Event(
                NotificationNavEvent.ShowDialog(item.title.orEmpty(), item.message.orEmpty())
            )
        }
    }
    var clientInfoDb =
        homeRepository.getClientDetail(preferences.getString("memberID", "").toString()).asLiveData()
    var activeCustomSubMenus: List<ClientMenuConfig>? = null
    fun getNotificationList() {
        val isGuestLogin = preferences.getBoolean(Constants.IS_GUEST, false)
        var messageCenterList: Single<MessageCenterList>
        if (!isGuestLogin) {
            messageCenterList = homeRepository.getNotifications(
                preferences.getString(Constants.MEMBER_ID, "").toString(),
                page
            )
        } else {
            messageCenterList = guestHomeRepository.getGuestNotifications(page, 10)
        }
        compositeDisposable.add(
            messageCenterList
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({
                    Timber.d(it.toString())
                    loading = false

                    page++
                    notificationList.addAll(it.values ?: arrayListOf())
                    data = notificationList.isEmpty()
                    isLastPage = it.paging?.totalPages == page

                    _notificationListGenerated.value = Event(notificationList)
                }, {
                    Timber.d(it.toString())
                    loading = false
                })
        )
    }
    fun setupClientCustomMenu(
        menu: List<ClientMenuConfig>?
    ) {
        activeCustomSubMenus = menu
    }

    fun getLogView() {
        val isGuestLogin = preferences.getBoolean(Constants.IS_GUEST, false)
        if (isGuestLogin) { return }
        compositeDisposable.add(
            analyticsRepository.getLogPageView(
                preferences.getString(
                    "memberID",
                    ""
                ).toString(),
                PageType.MESSAGE_CENTER.pageName,
                sourcePage
            )
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread()).doOnSubscribe {
                }.subscribe({
                }, {
                })
        )
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }
}
