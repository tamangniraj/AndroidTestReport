package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.immunization

class ImmunizationViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val medicalLogsRepository: MedicalLogsRepository,
    private val homeRepository: HomeRepository
) : ViewModel(), Observable {

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack

    private val _navigateToNext = MutableLiveData<Event<Unit>>()
    val navigateToNext: LiveData<Event<Unit>>
        get() = _navigateToNext

    private val _navigateToFilter = MutableLiveData<Event<Unit>>()
    val navigateToFilter: LiveData<Event<Unit>>
        get() = _navigateToFilter

    var gson = Gson()
    private lateinit var job: Job
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    val showError: LiveData<APIResponse<DataResponse>>
        get() = _showError
    private val _showError = MutableLiveData<APIResponse<DataResponse>>()

    var typeList: ArrayList<AvailImmunizationFiltersItem> = arrayListOf()
    var filterList: ArrayList<AvailImmunizationFiltersItem>? = null

    // Sorting and filtering
    var sortby = DEFAULT_SORT_BY
    var sortAction = DEFAULT_SORT_ACTION

    var range: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.range, null)
        }

    // Member and data management
    final var memberID = preferences.getString("memberID", "").toString()
    var currentMemberId: String = memberID
    var sharedIds = currentMemberId
    var selectedMember = ""

    var lastMemberId = ""

    // Data state
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }

    var hasData: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.hasData, null)
        }

    var hasResponse: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.hasResponse, null)
        }

    var furthestDate: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.furthestDate, null)
        }

    var totalRecord: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.totalRecord, null)
        }

    var seriesCompleted: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.seriesCompleted, null)
        }

    var reason: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.reason, null)
        }

    var displayNew: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.displayNew, null)
        }

    // Filter ranges
    var maxValue: Float? = null
    var minValue: Float? = null
    var minRange: Float? = null
    var maxRange: Float? = null

    // Pagination
    var currentPage = 1
    private val pageSize = PAGE_SIZE
    var hasMorePages = false
    var isLoadedFirstPage = false

    // Type for API call - temporarily using "diagnosis" until immunization API is ready
    var type: String = "vaccination"

    // LiveData for processed immunization list
    private val _processedLogs = MutableLiveData<List<Any>>()
    val processedLogs: LiveData<List<Any>> = _processedLogs

    // UI State
    private val _uiState = MutableLiveData<MedicalLogsUiState>()
    val uiState: LiveData<MedicalLogsUiState> = _uiState

    var shouldAppendAllData: Boolean = false
    val dataSaver = MedicalLogAllDataSaver()

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun onPrevClicked() {
        singleClick {
            _navigateToBack.value = Event(Unit)
        }
    }

    fun onNextClicked() {
        singleClick {
            _navigateToNext.value = Event(Unit) // Process the first click immediately
        }
    }

    private val _navigateToSort = MutableLiveData<Event<Unit>>()
    val navigateToSort: LiveData<Event<Unit>>
        get() = _navigateToSort

    private var debounceJob: Job? = null
    private var isDebouncing = false
    private val debounceInterval = 500L // Set your debounce interval in milliseconds

    fun singleClick(action: () -> Unit) {
        if (isDebouncing) return
        action()
        isDebouncing = true
        debounceJob?.cancel()
        debounceJob = viewModelScope.launch {
            delay(debounceInterval)
            isDebouncing = false
        }
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }

    fun sort() {
        _navigateToSort.value = Event(Unit)
    }
    fun filter() {
        _navigateToFilter.value = Event(Unit)
    }

    var title: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.title, null)
        }

    // Data loading methods
    fun onMemberIdChanged(newMemberId: String) {
        if (lastMemberId == newMemberId) {
            return
        }

        if (newMemberId.isNotEmpty()) {
            lastMemberId = newMemberId
        }

        sharedIds = newMemberId
        minValue = null
        maxValue = null
        minRange = null
        maxRange = null

        typeList.clear()
        filterList?.forEach { it.selected = false }
        presType = ""
        sortAction = DEFAULT_SORT_ACTION
        sortby = DEFAULT_SORT_BY
        range = ""
        loadDataWithDebounce(sharedIds, isRefresh = true)
    }

    private val loadDebouncer = ParameterizedDebouncer<LoadParams>()

    fun loadDataWithDebounce(sharedId: String, isRefresh: Boolean = false) {
        loadDebouncer.debounce(LoadParams(sharedId, isRefresh)) { params ->
            loadMedicalLogs(params.sharedIds, params.isRefresh)
        }
    }
    var presType: String = ""

    private fun loadMedicalLogs(selected: String, isRefresh: Boolean = false) {
        selectedMember = if (selected.isNullOrBlank() || selected.equals("null", true)) {
            currentMemberId
        } else {
            selected
        }

        // do not call api for the member id whose id is negative
        if (selectedMember.isNotEmpty() && selectedMember.toLong() < 0) {
            return
        }

        if (isRefresh) {
            currentPage = 1
            minValue = null
            maxValue = null
            dataSaver.clearSavedData()
            _processedLogs.value = emptyList()
            compositeDisposable.clear()
            hasData = false
            hasResponse = false
        }

        presType = typeList.mapNotNull { it.id }.joinToString(separator = "~") {
            it.trim()
        }

        compositeDisposable.clear()

        compositeDisposable.add(
            // Temporarily using diagnosis API until immunization API is ready
            medicalLogsRepository.getImmunizationMedicalLogs(
                currentMemberId,
                type,
                selectedMember,
                sortAction,
                sortby,
                presType,
                PAGE_SIZE,
                currentPage
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _uiState.value = MedicalLogsUiState.LoadingMore
                }
                .subscribe({
                    isLoadedFirstPage = true

                    if (filterList == null) {
                        filterList = it.availImmunizationFilters
                    }

                    if (currentPage == 1) {
                        filterList = it.availImmunizationFilters
                        hasData = it.immunizationList?.isNotEmpty() ?: false
                        hasResponse = true

                        it.medicalLogMetaData?.let { meta ->
                            maxValue = meta.maximumFrequency?.toFloat() ?: 0f
                            minValue = meta.minimumFrequency?.toFloat() ?: 10f

                            meta.earliestDate?.let {
                                furthestDate = DateUtils.dateToNamePres(
                                    DateUtils.stringServerDateToDate(it)
                                )
                            }

                            meta.seriesCompleted?.let {
                                seriesCompleted = it.toString()
                            }

                            meta.totalRecords?.let {
                                totalRecord = it.toString()
                            }
                        }
                    }

                    handleSuccessResponse(it)
                }, {
                    Timber.d(it.toString())
                    _uiState.value = MedicalLogsUiState.Error(
                        it.message ?: "Unknown error occurred"
                    )
                })
        )
    }

    fun loadNextPage() {
        if (_uiState.value !is MedicalLogsUiState.LoadingMore && _uiState.value !is MedicalLogsUiState.Loading && hasMorePages) {
            currentPage++
            loadMedicalLogs(selectedMember)
        }
    }

    fun hasMorePages(): Boolean {
        return hasMorePages
    }

    private fun handleSuccessResponse(response: MedicalLogsAll) {
        // Update pagination info
        response.paging?.let { paging ->
            hasMorePages = paging.hasNextPage
            currentPage = paging.currentPage
        }

        val isEmpty = isDataEmpty(response)

        if (isEmpty) {
            _uiState.value = MedicalLogsUiState.Empty
        } else {
            val processedData = dataSaver.saveData(response, shouldAppendAllData)
            _processedLogs.value = MedicalLogsSortingUtils.sortImmunizationsOnly(
                processedData
            )
            _uiState.value = MedicalLogsUiState.Success(response)
        }
    }

    private fun isDataEmpty(data: MedicalLogsAll): Boolean {
        return data.immunizationList.isNullOrEmpty()
    }

    /**
     * Check if the "new" indicator should be displayed
     * This checks SharedPreferences to see if the user has clicked the immunization tab before
     */
    fun checkNewAvailable() {
        val hasClickedImmunizationTab = preferences.getBoolean(
            PREF_KEY_IMMUNIZATION_TAB_CLICKED,
            false
        )
        displayNew = !hasClickedImmunizationTab
    }

    /**
     * Hide the "new" indicator
     * Called when the user clicks on the immunization tab
     * Saves the state to SharedPreferences so it won't show again
     */
    fun hideNew() {
        displayNew = false
        preferences.edit()
            .putBoolean(PREF_KEY_IMMUNIZATION_TAB_CLICKED, true)
            .apply()
    }

    companion object {
        private const val PREF_KEY_IMMUNIZATION_TAB_CLICKED = "immunization_tab_clicked"
    }
}
