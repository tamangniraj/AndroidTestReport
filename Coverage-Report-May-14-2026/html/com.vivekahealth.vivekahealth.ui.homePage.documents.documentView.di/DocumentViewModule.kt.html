package com.vivekahealth.vivekahealth.ui.homePage.documents.documentView.di

import android.content.SharedPreferences
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.ui.homePage.documents.documentView.DocumentViewFragment
import com.vivekahealth.vivekahealth.ui.homePage.documents.service.DocumentService
import dagger.Module
import dagger.Provides
import dagger.android.ContributesAndroidInjector
import okhttp3.OkHttpClient

@Module
abstract class DocumentViewModule {

    @ContributesAndroidInjector
    abstract fun contributesDocumentViewFragment(): DocumentViewFragment

    companion object {
        @JvmStatic
        @Provides
        fun provideDocumentViewService(
            gson: Gson,
            builder: OkHttpClient.Builder,
            sharedPreferences: SharedPreferences
        ): DocumentService = DocumentService(
            gson = gson,
            builder = builder,
            preferences = sharedPreferences
        )
    }
}
