package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.di
import android.content.SharedPreferences
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.MedicalLogsFragment
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.service.MedicalLogsService
import dagger.Module
import dagger.Provides
import dagger.android.ContributesAndroidInjector
import okhttp3.OkHttpClient

@Module
abstract class MedicalLogsModule {

    @ContributesAndroidInjector
    abstract fun contributesMedicalLogsFragment(): MedicalLogsFragment

    companion object {
        @JvmStatic
        @Provides
        fun provideMedicalLogsFragmentService(
            gson: Gson,
            builder: OkHttpClient.Builder,
            sharedPreferences: SharedPreferences
        ): MedicalLogsService = MedicalLogsService(
            gson = gson,
            builder = builder,
            preferences = sharedPreferences
        )
    }
}
