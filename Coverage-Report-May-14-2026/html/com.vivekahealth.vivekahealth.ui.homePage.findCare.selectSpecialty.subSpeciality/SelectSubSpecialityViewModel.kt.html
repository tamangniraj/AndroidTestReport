package com.vivekahealth.vivekahealth.ui.homePage.findCare.selectSpecialty.subSpeciality

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.local.VivekaDatabase
import com.vivekahealth.vivekahealth.data.remote.home.findCare.FindCareFilterModel
import com.vivekahealth.vivekahealth.data.remote.home.findCare.specialityValue.Value
import com.vivekahealth.vivekahealth.data.remote.home.findCare.specialityValue.ValueSelected
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.disposables.CompositeDisposable
import javax.inject.Inject

class SelectSubSpecialityViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository,
    private val vivekaDatabase: VivekaDatabase
) : ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    var valueList: ArrayList<ValueSelected> = arrayListOf()
    private val _updateList = MutableLiveData<Event<List<ValueSelected>>>()
    val updateList: LiveData<Event<List<ValueSelected>>>
        get() = _updateList

    private val _close = MutableLiveData<Event<Unit>>()
    val close: LiveData<Event<Unit>>
        get() = _close
    var listMain: Value? = null
    private val _done = MutableLiveData<Event<Unit>>()
    val done: LiveData<Event<Unit>>
        get() = _done
    var filterOption: FindCareFilterModel = FindCareFilterModel()
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.filterOption, null)
        }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun apply() {
        filterOption.selectedSubSpeciality.clear()
        var selected = valueList.filter { valueSelected -> valueSelected.selected }
        filterOption.selectedSubSpeciality.addAll(selected)
        var listTemp = filterOption.selectedSubSpeciality.toSet().toList()
        filterOption.selectedSubSpeciality.clear()
        filterOption.selectedSubSpeciality.addAll(listTemp)
//        filterOption.selectedSubSpeciality.removeIf { item -> !(item.taxonomyGroup.equals(listMain?.specialityGroup,true)) }

        _done.value = Event(Unit)
    }

    fun close() {
        _close.value = Event(Unit)
    }

    fun listUpdate(mainList: List<ValueSelected>, changed: Int, isChecked: Boolean) {
        var index = changed

        var value = mainList[index]
        var originalIndex = valueList.indexOf(value)
        value.selected = isChecked

        val updatedItems = valueList.toMutableList().apply { this[originalIndex] = value }
        val updatedItemsView = mainList.toMutableList().apply { this[index] = value }

        valueList.clear()
        valueList.addAll(updatedItems)

        _updateList.value = Event(updatedItemsView)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
