package com.vivekahealth.vivekahealth.data.remote.login

import android.os.Parcelable
import com.google.gson.annotations.SerializedName
import kotlinx.parcelize.Parcelize

@Parcelize
data class ErrorResponse(

    @SerializedName("error")
    public var error: String? = null,

    @SerializedName("error_description")
    public var error_description: String? = null,

    @SerializedName("errorType")
    public val error_type: String? = null,

    @SerializedName("registeredEmail")
    public var registeredEmail: String? = null,
    var enrolledDate: String? = null,
    var firstName: String? = null,
    var lastName: String? = null,
    var memberId: String? = null,

    @SerializedName("alternate_app_available")
    public var alternativeAppAvailable: String? = null,

    @SerializedName("android")
    var android: String? = null,

    @SerializedName("app_name")
    var appName: String? = null

) : Parcelable

data class GenericErrorResponse(
    val code: String? = null,
    val reason: String? = null,
    val action: String? = null,
    val errorCode: String? = null,
    val error: String? = null,
    val otherDetails: OtherDetails? = null,
    val errorType: String? = null,
    val message: String? = null,
    private val errorDetails: String? = null,
    @SerializedName("error_description")
    private val errorDescription: String? = null
) {
    fun getErrorMessage(): String {
        return errorDetails ?: errorDescription ?: message ?: "Something went wrong. Please try again later."
    }
}

data class OtherDetails(
    val remainingTime: String? = null,

    @SerializedName("alternate_app_available")
    public var alternativeAppAvailable: String? = null,

    @SerializedName("android")
    var android: String? = null,

    @SerializedName("app_name")
    var appName: String? = null
)
