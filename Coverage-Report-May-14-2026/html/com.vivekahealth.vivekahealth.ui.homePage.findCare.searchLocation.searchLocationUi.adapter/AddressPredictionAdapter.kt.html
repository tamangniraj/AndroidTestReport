package com.vivekahealth.vivekahealth.ui.homePage.findCare.searchLocation.searchLocationUi.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.google.android.libraries.places.api.model.AutocompletePrediction
import com.vivekahealth.vivekahealth.databinding.AddressPredictionBinding

class AddressPredictionAdapter(
    private var onClick:
        (AutocompletePrediction) -> Unit
) :
    ListAdapter<AutocompletePrediction, RecyclerView.ViewHolder>(AddressDiffCallback()) {

    var selectedPosition: Int = 0

    private inner class ViewHeaderViewHolder(
        val binding: AddressPredictionBinding,
        private val onClick: (AutocompletePrediction) -> Unit
    ) : RecyclerView.ViewHolder(binding.root) {
        fun bind(position: Int) {
        }
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return ViewHeaderViewHolder(
            AddressPredictionBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)
        (holder as ViewHeaderViewHolder).binding.value = item.getFullText(null).toString()

        (holder as ViewHeaderViewHolder).binding.setClickListener {
            onClick(item)
        }
    }

    override fun getItemId(position: Int): Long {
        return position.toLong()
    }
}

private class AddressDiffCallback : DiffUtil.ItemCallback<AutocompletePrediction>() {
    override fun areItemsTheSame(
        oldItem: AutocompletePrediction,
        newItem: AutocompletePrediction
    ): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(
        oldItem: AutocompletePrediction,
        newItem: AutocompletePrediction
    ): Boolean {
        return oldItem.placeId == newItem.placeId
    }
}
