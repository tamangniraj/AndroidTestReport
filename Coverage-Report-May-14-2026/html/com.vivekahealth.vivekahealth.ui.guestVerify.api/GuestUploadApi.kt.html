package com.vivekahealth.vivekahealth.ui.guestVerify.api

import com.vivekahealth.vivekahealth.data.remote.guest.AdditionalRequirementResponse
import com.vivekahealth.vivekahealth.data.remote.guest.DocumentUploadResponse
import com.vivekahealth.vivekahealth.data.remote.guest.IdentifyErrorResponse
import com.vivekahealth.vivekahealth.data.remote.guest.UploadDetailsSendParams
import com.vivekahealth.vivekahealth.data.remote.guest.UploadResponse
import io.reactivex.Single
import okhttp3.MultipartBody
import retrofit2.Response
import retrofit2.http.Body
import retrofit2.http.GET
import retrofit2.http.Multipart
import retrofit2.http.POST
import retrofit2.http.Part
import retrofit2.http.Path
import retrofit2.http.Query

interface GuestUploadApi {

    @Multipart
    @POST("guest/v2/identifications/{identificationId}/media")
    fun uploadImage(
        @Path("identificationId") sessionId: String,
        @Part filePart: MultipartBody.Part,
        @Query("fileType") documentType: Int
    ): Single<DocumentUploadResponse>

    // //guest/v2/identifications/e773b2cc-58b3-44d2-bc44-fe48d95d79f8/scanBarcode
    @Multipart
    @POST("guest/v2/identifications/{identificationId}/scanBarcode")
    fun scanBarcode(
        @Path("identificationId") sessionId: String,
        @Part filePart: MultipartBody.Part,
        @Query("fileType") documentType: Int
    ): Single<Response<DocumentUploadResponse>>

    @POST("guest/v2/identifications/{identificationId}/uploadDetails")
    fun uploadDetails(
        @Path("identificationId") sessionId: String,
        @Body uploadDetails: UploadDetailsSendParams
    ): Single<Response<UploadResponse>>

    @POST("guest/v2/identifications/{identificationId}/verify")
    fun verify(
        @Path("identificationId") sessionId: String
    ): Single<Response<UploadResponse>>

    @POST("guest/v2/identifications/{identificationId}/identify")
    fun identify(
        @Path("identificationId") sessionId: String,
        @Query("overrideOldRegistration") overrideOldRegistration: Boolean? = null
    ): Single<Response<IdentifyErrorResponse>>

    @GET("guest/v2/identifications/{identificationId}/additionalRequirements")
    fun additionalRequirements(@Path("identificationId") sessionId: String): Single<Response<AdditionalRequirementResponse>>

    @GET("guest/v2/identifications/{identificationId}/processDocument")
    fun processDocument(@Path("identificationId") sessionId: String): Single<Response<UploadResponse>>

    @GET("guest/v2/identifications/{identificationId}/validateAlreadyRegisteredEmail")
    fun validateRegisteredEmail(
        @Path("identificationId") sessionId: String,
        @Query("email") email: String
    ): Single<Response<IdentifyErrorResponse>>

    @POST("guest/v2/identifications/{identificationId}/logFailedAttempt")
    fun logFailedAttempt(
        @Path("identificationId") sessionId: String,
        @Query("verificationType") verificationType: Int
    ): Single<Response<IdentifyErrorResponse>>
}
