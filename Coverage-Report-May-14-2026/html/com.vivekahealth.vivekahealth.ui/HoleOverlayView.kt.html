package com.vivekahealth.vivekahealth.ui

import android.content.Context
import android.graphics.Canvas
import android.graphics.Color
import android.graphics.Paint
import android.graphics.PorterDuff
import android.graphics.PorterDuffXfermode
import android.graphics.RectF
import android.util.AttributeSet
import android.view.View

class HoleOverlayView @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
    defStyle: Int = 0
) : View(context, attrs, defStyle) {

    private val holePaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        xfermode = PorterDuffXfermode(PorterDuff.Mode.CLEAR)
    }

    private val overlayPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = Color.parseColor("#B3000000") // semi-transparent black
    }

    // Size of the hole
    var holeWidth = 450
    var holeHeight = 800

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)

        // Set layer type to make CLEAR mode work
        setLayerType(LAYER_TYPE_HARDWARE, null)

        val parentWidth = width.toFloat()
        val parentHeight = height.toFloat()

        // Hole size: 60% of width, height from 9:16 aspect ratio
        val holeWidth = parentWidth * 0.7f
        val holeHeight = holeWidth * 6f / 5f

        val centerX = parentWidth / 2f
        val centerY = parentHeight / 2f

        val left = centerX - holeWidth / 2f
        val top = centerY - holeHeight / 2f
        val right = centerX + holeWidth / 2f
        val bottom = centerY + holeHeight / 2f

        val holeRect = RectF(left, top, right, bottom)

        // Draw dim background
        canvas.drawRect(0f, 0f, parentWidth, parentHeight, overlayPaint)

        // Clear hole in the center
        canvas.drawOval(holeRect, holePaint)
    }
}
