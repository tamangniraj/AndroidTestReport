package com.vivekahealth.vivekahealth.ui

import android.content.Context
import android.util.AttributeSet
import android.view.LayoutInflater
import android.widget.ImageView
import android.widget.LinearLayout
import android.widget.TextView
import com.vivekahealth.vivekahealth.R

class SecondaryButton @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
    defStyleAttr: Int = 0
) : LinearLayout(context, attrs, defStyleAttr) {

    private val iconView: ImageView
    private val textView: TextView

    init {
        LayoutInflater.from(context).inflate(R.layout.custom_secondary_button, this, true)

        iconView = findViewById(R.id.iv_button_icon)
        textView = findViewById(R.id.tv_button_text)

        // Handle custom attributes
        attrs?.let {
            val typedArray = context.obtainStyledAttributes(it, R.styleable.SecondaryButton)

            val text = typedArray.getString(R.styleable.SecondaryButton_buttonText)
            val iconRes = typedArray.getResourceId(
                R.styleable.SecondaryButton_buttonIcon,
                R.drawable.ic_sms
            )
            val enabled = typedArray.getBoolean(R.styleable.SecondaryButton_buttonEnabled, true)

            setText(text ?: "")
            setIcon(iconRes)

            setEnabled(enabled)

            typedArray.recycle()
        }
    }

    fun setText(text: String) {
        textView.text = text
    }

    fun setIcon(iconRes: Int) {
        iconView.setImageResource(iconRes)
    }

    override fun setEnabled(enabled: Boolean) {
        super.setEnabled(enabled)
        iconView.isEnabled = enabled
        textView.isEnabled = enabled

        // Control clickable and focusable based on enabled state
        isClickable = enabled
        isFocusable = enabled

        alpha = if (enabled) 1.0f else 0.6f
    }

    override fun performClick(): Boolean {
        // Only perform click if button is enabled
        return if (isEnabled) {
            super.performClick()
        } else {
            false
        }
    }
}
