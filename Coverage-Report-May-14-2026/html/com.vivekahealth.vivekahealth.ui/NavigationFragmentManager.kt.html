package com.vivekahealth.vivekahealth.ui

import androidx.annotation.IdRes
import androidx.fragment.app.Fragment
import androidx.fragment.app.FragmentManager
import androidx.fragment.app.FragmentTransaction
import com.vivekahealth.vivekahealth.R
import javax.inject.Inject

class NavigationFragmentManager @Inject internal constructor() {

    var fragmentManager: FragmentManager? = null

    @IdRes
    var containerId: Int = 0

    enum class TransitionAnimation {
        SLIDE_FROM_RIGHT,
        SLIDE_FROM_BOTTOM
    }

    @JvmOverloads
    fun addAsBaseFragment(
        fragment: Fragment,
        @IdRes containerViewId: Int = containerId,
        tag: String = getDefaultTagForFragment(fragment)
    ) {
        fragmentManager?.popBackStackImmediate(null, FragmentManager.POP_BACK_STACK_INCLUSIVE)
        val transaction = fragmentManager?.beginTransaction()
        transaction?.replace(containerViewId, fragment, tag)
        transaction?.commit()
        fragmentManager?.executePendingTransactions()
    }

    @JvmOverloads
    fun addFragmentToBackStack(
        fragment: Fragment,
        @IdRes containerViewId: Int = containerId,
        transitionAnimation: TransitionAnimation? = null,
        tag: String = getDefaultTagForFragment(fragment),
        backStackName: String = tag
    ) {
        val transaction = fragmentManager?.beginTransaction()
        setTransactionAnimation(transaction, transitionAnimation)
        transaction?.addToBackStack(backStackName)

        transaction?.replace(containerViewId, fragment, tag)

        transaction?.commit()
//        fragmentManager?.executePendingTransactions()
    }

    @JvmOverloads
    fun addFragmentToBackStackNow(
        fragment: Fragment,
        @IdRes containerViewId: Int = containerId,
        transitionAnimation: TransitionAnimation? = null,
        tag: String = getDefaultTagForFragment(fragment),
        backStackName: String = tag
    ) {
        val transaction = fragmentManager?.beginTransaction()
        setTransactionAnimation(transaction, transitionAnimation)
        transaction?.replace(containerViewId, fragment, tag)
        transaction?.addToBackStack(backStackName)
        transaction?.commit()
        fragmentManager?.executePendingTransactions()
    }

    fun getTopFragment(): Fragment? {
        fragmentManager?.run {
            return when (backStackEntryCount) {
                0 -> null
                else -> findFragmentByTag(getBackStackEntryAt(backStackEntryCount - 1).name)
            }
        }

        return null
    }

    fun popBackStackImmediate(backStackName: String, inclusive: Boolean = true) {
        val flags = if (inclusive) FragmentManager.POP_BACK_STACK_INCLUSIVE else 0
        fragmentManager?.popBackStackImmediate(backStackName, flags)
    }

    fun popBackStackImmediate() {
//        fragmentManager?.executePendingTransactions()

        fragmentManager?.popBackStackImmediate()
    }

    private fun setTransactionAnimation(
        transaction: FragmentTransaction?,
        transitionAnimation: TransitionAnimation?
    ) =
        when (transitionAnimation) {
            TransitionAnimation.SLIDE_FROM_RIGHT ->
                transaction?.setCustomAnimations(
                    R.anim.slide_in_right,
                    R.anim.fade_out,
                    R.anim.fade_in,
                    R.anim.slide_out_right
                )
            TransitionAnimation.SLIDE_FROM_BOTTOM ->
                transaction?.setCustomAnimations(
                    R.anim.slide_in_bottom,
                    R.anim.fade_out,
                    R.anim.fade_in,
                    R.anim.slide_out_bottom
                )
            null -> transaction
        }

    private fun getDefaultTagForFragment(fragment: Fragment): String = fragment.javaClass.simpleName
}
