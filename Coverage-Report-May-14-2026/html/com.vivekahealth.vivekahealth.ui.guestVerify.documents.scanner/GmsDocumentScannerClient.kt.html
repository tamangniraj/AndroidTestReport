package com.vivekahealth.vivekahealth.ui.guestVerify.documents.scanner

import android.app.Activity
import android.net.Uri
import androidx.activity.result.ActivityResultLauncher
import androidx.activity.result.IntentSenderRequest
import androidx.activity.result.contract.ActivityResultContracts
import androidx.fragment.app.Fragment
import com.google.mlkit.vision.documentscanner.GmsDocumentScannerOptions
import com.google.mlkit.vision.documentscanner.GmsDocumentScannerOptions.RESULT_FORMAT_JPEG
import com.google.mlkit.vision.documentscanner.GmsDocumentScannerOptions.SCANNER_MODE_BASE_WITH_FILTER
import com.google.mlkit.vision.documentscanner.GmsDocumentScanning
import com.google.mlkit.vision.documentscanner.GmsDocumentScanningResult
import com.vivekahealth.vivekahealth.ui.guestVerify.documents.DocumentType
import javax.inject.Inject

class GmsDocumentScannerClient @Inject constructor() : DocumentScannerClient {

    private lateinit var scannerLauncher: ActivityResultLauncher<IntentSenderRequest>
    private var onPageCaptured: ((Uri, DocumentType) -> Unit)? = null
    private var currentDocumentType: DocumentType? = null

    override fun register(fragment: Fragment, onPageCaptured: (Uri, DocumentType) -> Unit) {
        this.onPageCaptured = onPageCaptured
        scannerLauncher = fragment.registerForActivityResult(
            ActivityResultContracts.StartIntentSenderForResult()
        ) { result ->
            if (result.resultCode == Activity.RESULT_OK) {
                val docType = currentDocumentType ?: return@registerForActivityResult
                GmsDocumentScanningResult.fromActivityResultIntent(result.data)
                    ?.pages?.getOrNull(0)?.imageUri
                    ?.let { uri -> onPageCaptured(uri, docType) }
            }
        }
    }

    override fun startScan(
        activity: Activity,
        documentType: DocumentType,
        onFailure: (Exception) -> Unit
    ) {
        currentDocumentType = documentType
        val options = GmsDocumentScannerOptions.Builder()
            .setScannerMode(SCANNER_MODE_BASE_WITH_FILTER)
            .setResultFormats(RESULT_FORMAT_JPEG)
            .setPageLimit(1)
            .setGalleryImportAllowed(false)
            .build()
        GmsDocumentScanning.getClient(options)
            .getStartScanIntent(activity)
            .addOnSuccessListener { intentSender ->
                scannerLauncher.launch(IntentSenderRequest.Builder(intentSender).build())
            }
            .addOnFailureListener { e -> onFailure(e) }
    }
}
