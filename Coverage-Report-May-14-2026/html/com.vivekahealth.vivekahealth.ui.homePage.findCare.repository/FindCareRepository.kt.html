package com.vivekahealth.vivekahealth.ui.homePage.findCare.repository

import com.vivekahealth.vivekahealth.data.local.home.MemberInfoDao
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.data.remote.home.findCare.TaxonomyGroupList
import com.vivekahealth.vivekahealth.data.remote.home.findCare.doctorModelsNew.DoctorResponseNew
import com.vivekahealth.vivekahealth.ui.homePage.findCare.service.FindCareService
import io.reactivex.Single
import javax.inject.Inject
import javax.inject.Singleton
import kotlinx.coroutines.flow.Flow

/**
 * Repository that handles User objects.
 */
@Singleton
class FindCareRepository @Inject constructor(
    private val findCareService: FindCareService,
    private val dao: MemberInfoDao
) {

    fun getDoctorList(
        memberID: String,
        latitude: String,
        longitude: String,
        provider_type: String,
        radius: String
    ): Single<DoctorResponseNew> = findCareService.getFindCareApi().getAllDoctorList(
        memberID,
        latitude,
        longitude,
        provider_type,
        radius
    )
    fun getTaxonomyList(): Single<TaxonomyGroupList> = findCareService.getFindCareApi().getTaxonomyGroup()

    fun getMemberFromDb(): Flow<List<MemberInfo>> {
        return dao.getAll()
    }

    fun getMemberSid(memberid: String): Flow<MemberInfo> {
        return dao.getMemberInfoLiveData(memberid)
    }
}
