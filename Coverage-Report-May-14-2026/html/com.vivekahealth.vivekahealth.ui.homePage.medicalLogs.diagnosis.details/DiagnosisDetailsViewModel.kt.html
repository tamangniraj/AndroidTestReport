package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.diagnosis.details

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.home.DiagnosisHistoryQuickView
import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.ClaimsDetailItem
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.repository.MedicalLogsRepository
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import javax.inject.Inject
import timber.log.Timber

class DiagnosisDetailsViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val medicalLogsRepository: MedicalLogsRepository,
    private val homeRepository: HomeRepository
) : ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _logsListGenerated = MutableLiveData<Event<List<DiagnosisHistoryQuickView>>>()
    val logsListGenerated: LiveData<Event<List<DiagnosisHistoryQuickView>>>
        get() = _logsListGenerated

    private val _viewEob = MutableLiveData<Event<Unit>>()
    val viewEob: LiveData<Event<Unit>>
        get() = _viewEob

    var claimsDetail: ClaimsDetailItem? = null
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.claimsDetail, null)
        }
    var memberID: String = ""
    var claimsNumber: String = ""
    var diagnosis: DiagnosisHistoryQuickView? = null

    var earliestDos: String = ""
    var frequency: Int = 0
    val code: String = ""
    var mostRecentDos: String = ""
    var viewType: Int = 0
    var diagnosisDate: String = ""

    var serviceDate: String = ""
    var description: String = ""
    var actionDate: String = ""
    var status: String = ""
    var gson = Gson()
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }

    fun diagnosisStringToObject(item: String) {
        if (item.isNotEmpty()) {
            diagnosis = gson.fromJson(item, DiagnosisHistoryQuickView::class.java)
        }
    }
    fun getList() {
        compositeDisposable.add(

            medicalLogsRepository.getDiagnosisDetails(memberID, diagnosis!!.icd10Code.codeString())
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({
                    Timber.d(it.toString())

                    loading = false

                    _logsListGenerated.value = Event(it.values)
                }, {
                    Timber.d(it.toString())

                    loading = false
                })
        )
    }

    fun viewEob() {
        _viewEob.value = Event(Unit)
    }
    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }
}
