class AddressChangeViewModel @javax.inject.Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository
) : ViewModel(), Observable {
    var user = EnrollMember()
    private lateinit var job: Job

    var listAv = arrayListOf<String>()

    var listState = arrayListOf<String>()

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    var memberInfoDb = homeRepository.getMemberSid(preferences.getString("memberID", "").toString()).asLiveData()

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack
    private val _navigateToNext = MutableLiveData<Event<Unit>>()
    val navigateToNext: LiveData<Event<Unit>>
        get() = _navigateToNext

    private val _navigateToSuccess = MutableLiveData<Event<DataResponse>>()
    val navigateToSuccess: LiveData<Event<DataResponse>>
        get() = _navigateToSuccess

    private val _apartmentChanged = MutableLiveData<Event<String>>()
    val apartmentChanged: LiveData<Event<String>>
        get() = _apartmentChanged
    private val _addressChanged = MutableLiveData<Event<String>>()
    val addressChanged: LiveData<Event<String>>
        get() = _addressChanged

    private val _stateChanged = MutableLiveData<Event<String>>()
    val stateChanged: LiveData<Event<String>>
        get() = _stateChanged

    private val _cityChanged = MutableLiveData<Event<String>>()

    val cityChanged: LiveData<Event<String>>
        get() = _cityChanged

    private val _setState = MutableLiveData<Event<Int>>()
    val setState: LiveData<Event<Int>>
        get() = _setState

    private val _yesClicked = MutableLiveData<Event<Unit>>()
    val yesClicked: LiveData<Event<Unit>>
        get() = _yesClicked
    private val _noClicked = MutableLiveData<Event<Unit>>()
    val noClicked: LiveData<Event<Unit>>
        get() = _noClicked

    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }

    private val _zipChanged = MutableLiveData<Event<String>>()

    var memberInfo: MemberInfo = MemberInfo()
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.memberInfo, null)
        }

    var continueCoverage: String = ""

    var clicked: Boolean = false

    var zipExtension: String = ""
    var zipFirst: String = ""
    val zipChanged: LiveData<Event<String>>
        get() = _zipChanged

    var addressStreet: String = ""
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.addressStreet, null)
        }

    var enableButton: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.enableButton, null)
        }

    var zip: String = ""
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.zip, null)
        }

    var state: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.state, null)
        }

    var apartment: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.apartment, null)
        }

    var city: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.city, null)
        }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    @get:Bindable
    val zipTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _zipChanged.value = Event(s.toString())
        }
    }

    @get:Bindable
    val addressTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _addressChanged.value = Event(s.toString())
        }
    }

    @get:Bindable
    val apartmentTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _apartmentChanged.value = Event(s.toString())
        }
    }

    @get:Bindable
    val homeTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
//            _Changed.value =Event( s.toString())
        }
    }

    @get:Bindable
    val cityTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _cityChanged.value = Event(s.toString())
        }
    }

    fun updateAddress() {
        compositeDisposable.add(
            homeRepository.updateAddress(
                preferences.getString("memberID", "").toString(),
                MemberAddress(addressStreet, apartment, city, state, zipFirst, zipExtension, null, null, null, null, false, null, false, null)
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({
                    loading = false
                    _navigateToSuccess.value = Event(it)
                }, {
                    loading = false
                })
        )
    }

    fun isSame(): Boolean {
        if (zipExtension.isEmpty()) {
            return (
                memberInfo.homeAddress?.street1.equals(
                    addressStreet,
                    true
                ) && memberInfo.homeAddress?.street2.equals(
                    apartment,
                    true
                ) && memberInfo.homeAddress?.city.equals(
                    city,
                    true
                ) && memberInfo.homeAddress?.state.equals(
                    state,
                    true
                ) && memberInfo.homeAddress?.zip.equals(zipFirst, true)
                )
        } else {
            return (
                memberInfo.homeAddress?.street1.equals(
                    addressStreet,
                    true
                ) && memberInfo.homeAddress?.street2.equals(
                    apartment,
                    true
                ) && memberInfo.homeAddress?.city.equals(
                    city,
                    true
                ) && memberInfo.homeAddress?.state.equals(
                    state,
                    true
                ) && memberInfo.homeAddress?.zip.equals(zipFirst, true) && memberInfo.homeAddress?.zipExtension.equals(
                    zipExtension,
                    true
                )
                )
        }
    }

    fun onChangeClicked() {
        updateAddress()
    }
    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }

    fun checkEnableButton() {
        enableButton = AddressChangedValidator.isAddressReady(
            memberInfo = memberInfo,
            apartmentChanged = apartmentChanged.value?.peekContent(),
            addressChanged = addressChanged.value?.peekContent(),
            cityChanged = cityChanged.value?.peekContent(),
            state = stateChanged.value?.peekContent(),
            zipChanged = zipChanged.value?.peekContent()
        )
    }

    fun setSelectedState(changedState: String) {
        _stateChanged.value = Event(changedState)
        state = changedState
    }

    fun updateZip(zip: String?, zipExtension: String?) {
        zip?.let {
            var temp: String = it

            zipExtension?.let {
                temp = "$temp-$it"
            }
            _zipChanged.value = Event(temp)
        }
    }
}
