package com.vivekahealth.vivekahealth.ui.guestRegister.password

import android.content.SharedPreferences
import androidx.core.content.edit
import android.os.Bundle
import android.text.Editable
import android.text.TextWatcher
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.guest.GuestRegisterUser
import com.vivekahealth.vivekahealth.data.remote.login.AuthToken
import com.vivekahealth.vivekahealth.data.remote.login.GenericErrorResponse
import com.vivekahealth.vivekahealth.ui.guestHome.repository.GuestHomeRepository
import com.vivekahealth.vivekahealth.ui.guestRegister.repository.GuestRegisterRepository
import com.vivekahealth.vivekahealth.ui.login.repository.LoginRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Constants.APOLOGY_CHECK_CONNECTION
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import java.util.regex.Pattern
import javax.inject.Inject

class GuestPasswordViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val guestRegisterRepository: GuestRegisterRepository,
    var loginRepository: LoginRepository,
    val guestHomeRepository: GuestHomeRepository
) :
    ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _passwordChanged = MutableLiveData<Event<String>>()
    val passwordChanged: LiveData<Event<String>>
        get() = _passwordChanged

    private val _confirmPasswordChanged = MutableLiveData<Event<String>>()
    val confirmPasswordChanged: LiveData<Event<String>>
        get() = _confirmPasswordChanged

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack

    private val _loading = MutableLiveData<Event<Boolean>>()
    val loading: LiveData<Event<Boolean>>
        get() = _loading

    private val _registerError = MutableLiveData<Event<String>>()
    val registerError: LiveData<Event<String>>
        get() = _registerError

    private val _startSession = MutableLiveData<Event<String>>()
    val startSession: LiveData<Event<String>>
        get() = _startSession

    val passwordError: Boolean = false

    var gson = Gson()

    var phone: String? = null
    var cCode: String? = null
    var registrationId: String? = null
    var email: String? = null
    var password: String? = null
    private var confirmPassword: String? = null
    var guestRegisterUser: GuestRegisterUser? = null
        private set
    private var lastSuccessfulRegistration: GuestRegisterUser? = null

    @get:Bindable
    val passwordTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
            _passwordChanged.value = Event(s.toString())
            if (s.toString().isNotEmpty()) {
                password = s.toString()
                updateGuestRegisterUser()
            }
        }
    }

    @get:Bindable
    val confirmPasswordTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
            if (s.toString().isNotEmpty()) {
                confirmPassword = s.toString()
                _confirmPasswordChanged.value = Event(confirmPassword!!)
                updateGuestRegisterUser()
            }
        }
    }

    fun hasUpperAndLower(pass: String): Boolean {
        val upperLowerPattern = Pattern.compile("(?=.*[a-z])(?=.*[A-Z])")
        return upperLowerPattern.matcher(pass).find()
    }

    fun hasAtLeastOneNumber(pass: String): Boolean {
        val atLeastOneNumber = Pattern.compile("(?=.*\\d)")
        return atLeastOneNumber.matcher(pass).find()
    }

    fun hasOneSpecialCharacter(pass: String): Boolean {
        val atLeastOneSpecialChar = Pattern.compile("(?=.*\\W)")
        return atLeastOneSpecialChar.matcher(pass).find()
    }

    fun isAtLeastEightCharLong(pass: String): Boolean {
        val atLeastEightCharLong = Pattern.compile(".{8,}")
        return atLeastEightCharLong.matcher(pass).find()
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun onPrevClicked() {
        _navigateToBack.value = Event(Unit)
    }

    fun onNextClicked() {
    }

    fun handleArguments(arguments: Bundle?) {
        phone = arguments?.getString("phone")
        cCode = arguments?.getString("cCode")
        registrationId = arguments?.getString(("registrationId"))
        email = arguments?.getString("email")
        updateGuestRegisterUser()
    }

    private fun updateGuestRegisterUser() {
        guestRegisterUser =
            GuestRegisterUser(registrationId, email, phone, cCode, password, confirmPassword)
    }

    private fun isSameAsLastSuccessfulRegistration(user: GuestRegisterUser): Boolean {
        return lastSuccessfulRegistration?.let {
            it.email == user.email &&
                it.password == user.password &&
                it.phone == user.phone &&
                it.countryCode == user.countryCode &&
                it.registrationId == user.registrationId
        } ?: false
    }

    fun registerGuestUser() {
        guestRegisterUser?.let { user ->
            val skipRegistration = isSameAsLastSuccessfulRegistration(user)

            val observable = if (skipRegistration) {
                loginRepository.getCompleteRegistrationToken(
                    email.toString(),
                    password.toString()
                )
            } else {
                guestRegisterRepository
                    .registerGuest(user)
                    .flatMap { registerResponse ->
                        if (registerResponse.isSuccessful) {
                            lastSuccessfulRegistration = user.copy()
                            loginRepository.getCompleteRegistrationToken(
                                email.toString(),
                                password.toString()
                            )
                        } else {
                            val errorResponse: GenericErrorResponse = gson.fromJson(
                                registerResponse.errorBody()?.string(),
                                GenericErrorResponse::class.java
                            )
                            throw Exception(
                                "Registration failed: ${errorResponse.getErrorMessage()}"
                            )
                        }
                    }
            }

            val disposable = observable
                .flatMap { tokenResponse ->
                    if (tokenResponse.isSuccessful) {
                        saveTokens(tokenResponse.body()!!)
                        preferences.edit().putString(Constants.USER_EMAIL, email).commit()
                        guestHomeRepository.startGuestRegistration()
                    } else {
                        val errorBody = tokenResponse.errorBody()?.string()
                        val errorMessage = try {
                            val errorResponse: GenericErrorResponse = gson.fromJson(
                                errorBody,
                                GenericErrorResponse::class.java
                            )
                            errorResponse.getErrorMessage()
                        } catch (e: Exception) {
                            "Authentication failed"
                        }
                        throw Exception("Token retrieval failed: $errorMessage")
                    }
                }
                .map { startRegistrationResponse ->
                    if (startRegistrationResponse.identificationId.isNotEmpty()) {
                        startRegistrationResponse
                    } else {
                        throw Exception(
                            "Verification session failed: Unable to start verification process"
                        )
                    }
                }
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _loading.value = Event(true)
                }
                .subscribe({ startRegistrationResponse ->
                    _loading.value = Event(false)
                    _startSession.value = Event(startRegistrationResponse.identificationId)
                }, { error ->
                    _loading.value = Event(false)
                    val errorMessage = when {
                        error.message?.contains("Unable to resolve host") == true ||
                            error.message?.contains("Failed to connect to") == true ||
                            error?.message?.contains(
                            "Software caused connection abort"
                        ) == true ||
                            error?.message.equals(
                                "timeout"
                            )
                        ->
                            APOLOGY_CHECK_CONNECTION

                        else -> error.message ?: "Registration failed. Please try again."
                    }
                    _registerError.value = Event(errorMessage)
                })
            compositeDisposable.add(disposable)
        }
    }

    fun registerOpenEnrollmentUser() {
        guestRegisterUser?.let { user ->
            val skipRegistration = isSameAsLastSuccessfulRegistration(user)

            val disposable = if (skipRegistration) {
                loginRepository.getCompleteRegistrationToken(
                    email.toString(),
                    password.toString()
                )
            } else {
                guestRegisterRepository
                    .registerGuest(user)
                    .flatMap { registerResponse ->
                        if (registerResponse.isSuccessful) {
                            lastSuccessfulRegistration = user.copy()
                            loginRepository.getOpenRegistrationToken(
                                email.toString(),
                                password.toString()
                            )
                        } else {
                            val errorResponse: GenericErrorResponse = gson.fromJson(
                                registerResponse.errorBody()?.string(),
                                GenericErrorResponse::class.java
                            )
                            throw Exception(
                                "Registration failed: ${errorResponse.getErrorMessage()}"
                            )
                        }
                    }
            }
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _loading.value = Event(true)
                }
                .subscribe({ tokenResponse ->
                    _loading.value = Event(false)

                    if (tokenResponse.isSuccessful) {
                        saveTokens(tokenResponse.body()!!)
                        preferences.edit().putString(Constants.USER_EMAIL, email).commit()
                        _startSession.value = Event("")
                    } else {
                        val errorBody = tokenResponse.errorBody()?.string()
                        val errorMessage = try {
                            val errorResponse: GenericErrorResponse = gson.fromJson(
                                errorBody,
                                GenericErrorResponse::class.java
                            )
                            "Token retrieval failed: ${errorResponse.getErrorMessage()}"
                        } catch (e: Exception) {
                            "Token retrieval failed: Authentication failed"
                        }
                        _registerError.value = Event(errorMessage)
                    }
                }, { error ->
                    _loading.value = Event(false)
                    val errorMessage = when {
                        error.message?.contains("Unable to resolve host") == true ||
                            error.message?.contains("Failed to connect to") == true ||
                            error?.message?.contains(
                            "Software caused connection abort"
                        ) == true ||
                            error?.message.equals(
                                "timeout"
                            )
                        ->
                            APOLOGY_CHECK_CONNECTION

                        else -> error.message ?: "Registration failed. Please try again."
                    }
                    _registerError.value = Event(errorMessage)
                })
            compositeDisposable.add(disposable)
        }
    }

    private fun saveTokens(data: AuthToken) {
        preferences.edit {
            putString(Constants.ACCESS_TOKEN, data.access_token)
            putString(Constants.TOKEN_TYPE, data.token_type)
            putString(Constants.EXPIRES_IN, data.expires_in)
            putString(Constants.SCOPE, data.scope)
            putBoolean(Constants.IS_GUEST, data.isGuest)
        }
    }
}
