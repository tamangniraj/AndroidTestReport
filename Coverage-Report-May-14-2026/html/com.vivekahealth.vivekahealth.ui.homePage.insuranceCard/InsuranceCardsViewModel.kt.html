    fun missingData(): List<String> {
        var list = arrayListOf<String>()
        currentMember.lastName.let {
            if (it.isNullOrBlank()) {
                list.add("Last Name, First Name")
            }
        } ?: run {
            list.add("Last Name, First Name")
        }

        currentMember.gender.let {
            if (it.isNullOrBlank() || it.equals("U", true)) {
                list.add("Gender")
            }
        } ?: run {
            list.add("Gender")
        }
        currentMember.birthDate.let {
            if (it.isNullOrBlank() || it.equals("U", true)) {
                list.add("Date of Birth")
            }
        } ?: run {
            list.add("Date of Birth")
        }

        currentMember.ssn.let {
            if (it.isNullOrBlank()) {
                list.add(resources.getString(R.string.social_security_number))
            }
        } ?: run {
            list.add(resources.getString(R.string.social_security_number))
        }

        return list
    }

    fun loadInsuranceCard(selfId: String, cardType: String) {
        Timber.d("loading: loadInsuraceCard")

        compositeDisposable.clear()

        compositeDisposable.add(
            benefitWorkRepository.getInsuranceCard(selfId, cardType)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _loadingState.value = true
                }
                .subscribe({
                    _loadingState.value = false
                    it.selfId = selfId
                    benefitWorkRepository.insertInsurance(it)
                    insurance = it
                }, {
                    if (it is HttpException) {
//                    if (it.code() != 500) {
                        _serverError.value = Event(
                            NetworkMessage.getErrorMessage(
                                it.response()?.errorBody()
                            )
                        )

//                    }
                    }
                    _loadingState.value = false
                })
        )
    }

    fun bubbleMemberMap() {
//        memberInfo.isSelected = currentMemberID.isBlank()
        bubbleMemberList.clear()
        bubbleMemberList.add(memberInfo)

        memberInfo.dependentsArrayList?.sortByDescending {
            DateUtils.stringServerDateToDate(
                it.birthDate.toString()
            )
        }

        memberInfo.dependentsArrayList?.reverse()

        memberInfo.dependentsArrayList?.let { it1 ->

            for (member in it1) {
                if (StringUtils.equalsIgnoreCase(member.memberRelationship, "Spouse")) {
                    bubbleMemberList.add(1, member)
                } else {
                    if (!member.hideProfileFromHome) {
                        bubbleMemberList.add(member)
                    }
                }
            }

            var grouped = bubbleMemberList.groupBy { it.memberRelationship }
            grouped.forEach {
                it.value.sortedByDescending {
                    DateUtils.stringServerDateToDate(
                        it.birthDate.toString()
                    )
                }
                it.value.reversed()
            }

            bubbleMemberList.clear()

            grouped.forEach {
                if (it.key.equals("Insured", true)) {
                    bubbleMemberList.add(0, it.value[0])
                } else if (it.key.equals("Spouse", true)) {
                    if (grouped.containsKey("Insured")) {
                        bubbleMemberList.add(1, it.value[0])
                    } else {
                        bubbleMemberList.add(0, it.value[0])
                    }
                } else if (it.key.equals("Child", true)) {
                    val dateGroup = it.value.groupBy { date ->
                        DateUtils.stringServerDateToDate(
                            date.birthDate.toString()
                        )
                    }

                    dateGroup.forEach { item ->
                        val sorted = item.value.sortedBy { item1 -> item1.firstName }
                        bubbleMemberList.addAll(sorted)
                    }
                }
            }
            val tempOthers: ArrayList<MemberInfo> = arrayListOf()
            grouped.forEach {
                if (!it.key.equals("Insured", true) && !it.key.equals(
                        "Spouse",
                        true
                    ) && !it.key.equals("Child", true)
                ) {
                    tempOthers.addAll(it.value)
                }
            }

            val dateGroup = tempOthers.groupBy { date ->
                DateUtils.stringServerDateToDate(
                    date.birthDate.toString()
                )
            }
            dateGroup.forEach { item ->
                val sorted = item.value.sortedBy { item1 -> item1.firstName }
                bubbleMemberList.addAll(sorted)
            }
        }

        eligibleBubbleMemberList.clear()
        for (member in bubbleMemberList) {
            if (member.selfId.equals(currentMemberID, true)) {
                member.isSelected = true
                currentMember = member
            } else {
                member.isSelected = false
            }
//            if (member.isEligible == "true" || !member.showCardsOnlyIfEligible) {
            eligibleBubbleMemberList.add(member)
//            }
        }

        listForHomeBubbleAdapter = eligibleBubbleMemberList.toList()

        currentMemberID = ""

        // only hide the loading if the request if not first time.
        // updatedDate is not null means, the we have data on db, so no need to show the loader
        if (updatedDate.isNotBlank()) {
            loading = false
        }

        _bubbleListGenerated.value = Event(listForHomeBubbleAdapter)
    }
