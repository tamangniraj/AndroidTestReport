package com.vivekahealth.vivekahealth.ui.homePage.addDependent.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.local.home.DocumentAddDependent
import com.vivekahealth.vivekahealth.databinding.CellDocumentBinding

class DocumentUploadAdapter(private var onClick: (DocumentAddDependent) -> Unit) : ListAdapter<DocumentAddDependent, RecyclerView.ViewHolder>(
    DocListDiffCallback()
) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return DocViewHolder(
            CellDocumentBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }

    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)

        (holder as DocViewHolder).binding.docCard.setOnClickListener {
            onClick(item)
            notifyItemChanged(position)
        }

        holder.binding.docName = holder.binding.docCard.context.getString(
            item.docName!!.TranslatedName
        )
        holder.binding.docUploaded = item.uploaded
        holder.binding.person = item
    }

    class DocViewHolder(
        val binding: CellDocumentBinding,
        private val onClick: (DocumentAddDependent) -> Unit
    ) : RecyclerView.ViewHolder(binding.root) {

        //        var toggle = binding.clicked
        init {
        }
    }
}
private class DocListDiffCallback : DiffUtil.ItemCallback<DocumentAddDependent>() {

    override fun areItemsTheSame(oldItem: DocumentAddDependent, newItem: DocumentAddDependent): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: DocumentAddDependent, newItem: DocumentAddDependent): Boolean {
        return oldItem.uploaded == newItem.uploaded
    }
}
