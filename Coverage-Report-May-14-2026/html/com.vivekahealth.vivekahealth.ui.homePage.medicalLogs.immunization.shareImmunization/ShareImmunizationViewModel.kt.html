package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.immunization.shareImmunization

class ShareImmunizationViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository,
    private val resources: Resources
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    var title: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.title, null)
        }

    private val _navigateToOk = MutableLiveData<Event<Unit>>()
    val navigateToOK: LiveData<Event<Unit>>
        get() = _navigateToOk

    var loadedFile: Boolean = false
    private val _navigateToPDFDownloaded = MutableLiveData<Event<ResponseBody>>()
    val navigateToPDFDownloaded: LiveData<Event<ResponseBody>>
        get() = _navigateToPDFDownloaded

    private val _navigateToNoCards = MutableLiveData<Event<Unit>>()
    val navigateToNoCards: LiveData<Event<Unit>>
        get() = _navigateToNoCards

    private val _navigateToShareNeeded = MutableLiveData<Event<String>>()
    val navigateToShareNeeded: LiveData<Event<String>>
        get() = _navigateToShareNeeded

    private var _emailResponse =
        MutableLiveData<Event<ShareCardResponse>>()
    val emailResponse: LiveData<Event<ShareCardResponse>>
        get() = _emailResponse

    private val _bubbleListGeneratedNew = MutableLiveData<Event<List<MemberInfo>>>()
    val bubbleListGeneratedNew: LiveData<Event<List<MemberInfo>>>
        get() = _bubbleListGeneratedNew

    var dependentMemberList: ArrayList<MemberInfo> = ArrayList()
    var eligibleDependentMemberList: ArrayList<MemberInfo> = ArrayList()
    var listForDependentAdapter: List<MemberInfo> = listOf()
    private val _dependentListGenerated = MutableLiveData<Event<List<MemberInfo>>>()
    val dependentListGenerated: LiveData<Event<List<MemberInfo>>>
        get() = _dependentListGenerated

    var memberInfoDb = homeRepository.getMemberSid(preferences.getString("memberID", "").toString()).filterNotNull().asLiveData()
    private var selectedMemberId: String = ""

    var showTitleEmail: Boolean = true
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showTitleEmail, null)
        }

    var myFile: File? = null

    var showEmail: Boolean = true
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showEmail, null)
        }

    var memberInfo: MemberInfo = MemberInfo()
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.memberInfo, null)
        }

    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }

    private val _emailChanged = MutableLiveData<Event<String>>()
    val emailChanged: LiveData<Event<String>>
        get() = _emailChanged

    var sendEmail: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.sendEmail, null)
        }

    @get:Bindable
    val emailTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {}
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {}
        override fun afterTextChanged(s: Editable) {
            _emailChanged.value = Event(s.toString())
        }
    }

    fun onCrossClicked() {
        _navigateToOk.value = Event(Unit)
    }

    fun loadData(memberInfo: MemberInfo) {
        dependentMemberList.clear()
        var member = memberInfo
        member.isSelected = true
        member.firstName += "  ${resources.getString(R.string.you)}"
        dependentMemberList.add(member)

        selectedMemberId = memberInfo.selfId

        memberInfo.dependentsArrayList?.sortByDescending {
            DateUtils.stringServerDateToDate(it.birthDate.toString())
        }

        memberInfo.dependentsArrayList?.reverse()

        memberInfo.dependentsArrayList?.let { it1 ->
            for (memberItem in it1) {
                if (memberItem.memberRelationship.equals("Insured", true)) {
                    dependentMemberList.add(0, memberItem)
                } else if (StringUtils.equalsIgnoreCase(memberItem.memberRelationship, "Spouse")) {
                    if (memberItem.memberRelationship.equals("Insured", true)) {
                        dependentMemberList.add(0, memberItem)
                    } else {
                        dependentMemberList.add(1, memberItem)
                    }
                } else {
                    dependentMemberList.add(memberItem)
                }
            }

            var grouped = dependentMemberList.groupBy { it.memberRelationship }
            grouped.forEach {
                it.value.sortedByDescending { item ->
                    DateUtils.stringServerDateToDate(item.birthDate.toString())
                }
                it.value.reversed()
            }

            dependentMemberList.clear()

            grouped.forEach {
                if (it.key.equals("Insured", true)) {
                    dependentMemberList.add(0, it.value[0])
                } else if (it.key.equals("Spouse", true)) {
                    if (grouped.containsKey("Insured")) {
                        dependentMemberList.add(1, it.value[0])
                    } else {
                        dependentMemberList.add(0, it.value[0])
                    }
                } else if (it.key.equals("Child", true)) {
                    val dateGroup = it.value.groupBy { date ->
                        DateUtils.stringServerDateToDate(date.birthDate.toString())
                    }

                    dateGroup.forEach { item ->
                        val sorted = item.value.sortedBy { item1 -> item1.firstName }
                        dependentMemberList.addAll(sorted)
                    }
                }
            }

            val tempOthers: ArrayList<MemberInfo> = arrayListOf()
            grouped.forEach {
                if (!it.key.equals("Insured", true) && !it.key.equals("Spouse", true) && !it.key.equals(
                        "Child",
                        true
                    )
                ) {
                    tempOthers.addAll(it.value)
                }
            }

            val dateGroup = tempOthers.groupBy { date ->
                DateUtils.stringServerDateToDate(date.birthDate.toString())
            }
            dateGroup.forEach { item ->
                val sorted = item.value.sortedBy { item1 -> item1.firstName }
                dependentMemberList.addAll(sorted)
            }
        }

        // Eligible member only
        eligibleDependentMemberList.clear()
        for (member1 in dependentMemberList) {
            eligibleDependentMemberList.add(member1)
        }

        var grouped = eligibleDependentMemberList.groupBy { it.memberRelationship }
        grouped.forEach {
            it.value.sortedByDescending { DateUtils.stringServerDateToDate(it.birthDate.toString()) }
            it.value.reversed()
        }

        eligibleDependentMemberList.clear()

        grouped.forEach {
            if (it.key.equals("Insured", true)) {
                eligibleDependentMemberList.add(0, it.value[0])
            } else if (it.key.equals("Spouse", true)) {
                if (grouped.containsKey("Insured")) {
                    eligibleDependentMemberList.add(1, it.value[0])
                } else {
                    eligibleDependentMemberList.add(0, it.value[0])
                }
            } else if (it.key.equals("Child", true)) {
                val dateGroup = it.value.groupBy { date ->
                    DateUtils.stringServerDateToDate(date.birthDate.toString())
                }

                dateGroup.forEach { item ->
                    val sorted = item.value.sortedBy { item1 -> item1.firstName }
                    eligibleDependentMemberList.addAll(sorted)
                }
            }
        }
        val tempOthers: ArrayList<MemberInfo> = arrayListOf()
        grouped.forEach {
            if (!it.key.equals("Insured", true) && !it.key.equals("Spouse", true) && !it.key.equals(
                    "Child",
                    true
                )
            ) {
                tempOthers.addAll(it.value)
            }
        }

        val dateGroup = tempOthers.groupBy { date ->
            DateUtils.stringServerDateToDate(date.birthDate.toString())
        }
        dateGroup.forEach { item ->
            val sorted = item.value.sortedBy { item1 -> item1.firstName }
            eligibleDependentMemberList.addAll(sorted)
        }

        listForDependentAdapter = eligibleDependentMemberList.toList()
        _dependentListGenerated.value = Event(listForDependentAdapter)
    }

    fun sendInformation() {
        if (showEmail) {
            compositeDisposable.add(
                homeRepository.shareVaccinationCards(
                    selectedMemberId,
                    sendEmail
                ).subscribeOn(
                    Schedulers.io()
                )
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        loading = true
                    }
                    .subscribe({ it ->

                        loading = false

                        _emailResponse.value = Event(it)
                    }, {
                        loading = false
                        errorMessage(it.message)
                    })
            )
        } else {
            compositeDisposable.add(
                homeRepository.downloadVaccinePDF(
                    selectedMemberId
                ).subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        loading = true
                    }
                    .subscribe({ it ->
                        loading = false
                        _navigateToPDFDownloaded.value = Event(it)
                    }, {
                        loading = false
//
                        errorMessage(it.message)
                    })
            )
        }
    }

    fun errorMessage(message: String?) {
        val errorMessage = when {
            message?.contains("Unable to resolve host") == true ||
                message?.contains("Failed to connect to") == true ||
                message?.contains(
                "Software caused connection abort"
            ) == true ||
                message.equals(
                    "timeout"
                )
            ->
                APOLOGY_CHECK_CONNECTION
            else -> message ?: "Registration failed. Please try again."
        }

        _navigateToShareNeeded.value = Event(errorMessage)
    }

    fun setSelectedMember(memberInfo: MemberInfo) {
        val index = eligibleDependentMemberList.indexOf(memberInfo)

        for (item in eligibleDependentMemberList) {
            if (eligibleDependentMemberList.indexOf(item) == index) {
                eligibleDependentMemberList[index].apply {
                    isSelected = true
                }
            } else {
                eligibleDependentMemberList[eligibleDependentMemberList.indexOf(item)].apply {
                    isSelected = false
                }
            }
        }

        listForDependentAdapter = eligibleDependentMemberList.toList()
        _bubbleListGeneratedNew.value = Event(listForDependentAdapter)

        selectedMemberId = memberInfo.selfId
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun clearEmail() {
        _emailChanged.value = Event("")
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.clear()
    }
}
