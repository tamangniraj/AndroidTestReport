package com.vivekahealth.vivekahealth.ui.guestHome.api

import com.vivekahealth.vivekahealth.data.remote.enroll.DocumentUploadSuccessEnroll
import com.vivekahealth.vivekahealth.data.local.enroll.RootDocumentResponse
import com.vivekahealth.vivekahealth.data.remote.enroll.OpenEnrollmentData
import com.vivekahealth.vivekahealth.data.remote.enroll.RootMissingDocumentResponse
import com.vivekahealth.vivekahealth.data.remote.guest.ProcessLicenseResponse
import com.vivekahealth.vivekahealth.data.remote.guest.GuestClientDetails
import com.vivekahealth.vivekahealth.data.remote.guest.GuestUserInfo
import com.vivekahealth.vivekahealth.data.remote.guest.LanguagePreference
import com.vivekahealth.vivekahealth.data.remote.guest.StartIdentificationsReceiveParams
import com.vivekahealth.vivekahealth.data.remote.home.findCare.HtmlFindCareResponse
import com.vivekahealth.vivekahealth.data.remote.home.messageCenter.MessageCenterList
import com.vivekahealth.vivekahealth.data.remote.home.messageCenter.NotificationCount
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.data.remote.resetPassword.ChangePasswordV3DTO
import io.reactivex.Single
import okhttp3.MultipartBody
import retrofit2.Response
import retrofit2.http.Body
import retrofit2.http.GET
import retrofit2.http.Multipart
import retrofit2.http.POST
import retrofit2.http.Part
import retrofit2.http.Query

interface GuestHomeApi {

    @GET("guest/v2/info")
    fun getGuestInfo(): Single<GuestUserInfo>

    @GET("guest/v2/clientDetails")
    fun getGuestClientDetails(): Single<GuestClientDetails>

    @GET("guest/v2/refreshFcmId")
    fun refreshGuestFcm(
        @Query("memberId") memberId: String,
        @Query("fcmId") fcmId: String,
        @Query("isApns") isAPN: Boolean,
        @Query("isPrimary") isPrimary: Boolean
    ): Single<Void>

    @GET("guest/v2/identifications/start")
    fun startRegistration(): Single<StartIdentificationsReceiveParams>

    @POST("guest/v2/changePassword")
    fun changePasswordGuest(@Body changePasswordV3DTO: ChangePasswordV3DTO): Single<DataResponse>

    @POST("guest/v2/updatePassword")
    fun updatePasswordGuest(@Body changePasswordV3DTO: ChangePasswordV3DTO): Single<DataResponse>

    @GET("guest/v2/logout")
    fun getLogoutGuest(): Single<DataResponse>

    @GET("guest/v2/messages")
    fun getGuestNotifications(
        @Query("pageNo") pageNo: Int,
        @Query("pageSize") pageSize: Int
    ): Single<MessageCenterList>

    @GET("guest/v2/unreadNotificationCount")
    fun getGuestUnreadNotificationCount(): Single<NotificationCount>

    @POST("guest/v2/languagePreference")
    fun setGuestLanguagePreference(@Body languagePreference: LanguagePreference): Single<DataResponse>

    @GET("guest/v2/findADocPage")
    fun getHtmlForFindDoc(): Single<HtmlFindCareResponse>

    @POST("guest/v2/selfEnroll")
    fun getOpenEnroll(
        @Body enrollMember: OpenEnrollmentData
    ): Single<Response<DocumentUploadSuccessEnroll>>

    @GET("guest/v2/missingDocuments")
    fun getMissingRequiredDocuments(): Single<RootMissingDocumentResponse>

    @GET("guest/v2/requiredDocuments")
    fun getRequiredDocuments(): Single<RootDocumentResponse>

    @Multipart
    @POST("guest/v2/processLicense")
    fun processLicense(
        @Part file: MultipartBody.Part
    ): Single<Response<ProcessLicenseResponse>>

    @Multipart
    @POST("guest/v2/selfEnroll/uploadFile")
    fun uploadOpenFile(
        @Part filePart: MultipartBody.Part,
        @Query("fileType") fileName: String,
        @Query("enrollmentId") enrollmentId: Long? = null
    ): Single<DocumentUploadSuccessEnroll>
}
