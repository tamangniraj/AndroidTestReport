package com.vivekahealth.vivekahealth.ui.homePage

import android.content.SharedPreferences
import android.os.Handler
import android.os.Looper
import androidx.lifecycle.LiveData
import androidx.lifecycle.MediatorLiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.local.BadgeCount
import com.vivekahealth.vivekahealth.ui.homePage.documents.repository.DocumentRepository
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import timber.log.Timber

class CountReceiverViewModel() : ViewModel() {

    var notificationAction: String? = null

    // <chat, notification, doc>
    val countLiveData = MediatorLiveData<Event<BadgeCount>>()

    private val _docCount = MutableLiveData<Event<Int>>()
    val docCount: LiveData<Event<Int>>
        get() = _docCount

    private val _chatCount = MutableLiveData<Event<Int>>()
    val chatCount: LiveData<Event<Int>>
        get() = _chatCount

    private val _notificationCount = MutableLiveData<Event<Int>>()
    val notificationCount: LiveData<Event<Int>>
        get() = _notificationCount

    init {
        // Triple<chat, notification, doc>
        countLiveData.addSource(chatCount) {
            debounceUpdate()
        }
        countLiveData.addSource(notificationCount) {
            debounceUpdate()
        }
        countLiveData.addSource(docCount) {
            debounceUpdate()
        }
    }

    private val handler = Handler(Looper.getMainLooper())
    private var debounceRunnable: Runnable? = null

    private fun debounceUpdate() {
        debounceRunnable?.let { handler.removeCallbacks(it) } // Remove previous callbacks

        debounceRunnable = Runnable {
            countLiveData.value = Event(
                BadgeCount(
                    chat = chatCount.value?.peekContent() ?: 0,
                    notification = notificationCount.value?.peekContent() ?: 0,
                    doc = docCount.value?.peekContent() ?: 0
                )
            )
        }

        handler.postDelayed(debounceRunnable!!, 300) // Adjust debounce time (e.g., 300ms)
    }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    fun updateCount(
        preferences: SharedPreferences,
        homeRepository: HomeRepository,
        documentRepository: DocumentRepository
    ) {
        val memberId = preferences.getString("memberID", "").toString()
        if (memberId.isEmpty()) {
            return
        }

        if (Constants.IS_OFFLINE) {
            return
        }

        // notification
        notificationCounter(memberId, homeRepository)
        // chat
        chatCounter(homeRepository)

        getDocumentPending(preferences, documentRepository)
    }

    private fun notificationCounter(memberId: String, homeRepository: HomeRepository) {
        compositeDisposable.add(
            homeRepository.getUnreadNotification(memberId)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                }
                .subscribe({
                    _notificationCount.value = Event(it.count)
                }, {
                    Timber.d(it.toString())
                })
        )
    }

    private fun chatCounter(homeRepository: HomeRepository) {
        compositeDisposable.add(
            homeRepository.getChat("new", null, null, null, null, false)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                }
                .subscribe({
                    Timber.d(it.toString())

                    try {
                        it?.let {
                            if (it.values != null && it.values.isNotEmpty()) {
                                _chatCount.value = Event(it.values.count())
                            } else {
                                _chatCount.value = Event(0)
                            }
                        } ?: run {
                            _chatCount.value = Event(0)
                        }
                    } catch (e: NullPointerException) {
                        _chatCount.value = Event(0)
                    }
                }, {
                    Timber.d(it.toString())
                })
        )
    }

    fun getDocumentPending(preferences: SharedPreferences, documentRepository: DocumentRepository) {
        compositeDisposable.add(
            documentRepository.getPendingDocument(preferences.getString("memberID", "").toString())
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                }
                .subscribe({
                    if (it.pendingRequestDTOList != null) {
                        _docCount.value = Event(it.pendingRequestDTOList.count())
                    } else {
                        _docCount.value = Event(0)
                    }
                }, {
                })
        )
    }

    fun updateThePendingDocCount(int: Int) {
        _docCount.value = Event(int)
    }

    override fun onCleared() {
        super.onCleared()
        handler.removeCallbacksAndMessages(null)
    }
}
