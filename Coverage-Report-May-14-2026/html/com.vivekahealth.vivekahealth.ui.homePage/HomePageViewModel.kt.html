package com.vivekahealth.vivekahealth.ui.homePage

class HomePageViewModel @Inject constructor(
    val preferences: SharedPreferences,
    private val homeRepository: HomeRepository,
    private val documentRepository: DocumentRepository,
    private val benefitWorkRepository: BenefitWorkRepository
) :
    ViewModel(),
    Observable {
    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    // Debouncer instances for avoiding duplicate requests
    private val insuranceDebouncer: ParameterizedDebouncer<String> = ParameterizedDebouncer()
    private val dataDebouncer: ParameterizedDebouncer<Pair<MemberInfo, String>> =
        ParameterizedDebouncer()

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    var notificationDataParser: NotificationDataParser ? = null
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    var uriImage: Uri? = null
    private val _navigateToUpload = MutableLiveData<APIResponse<DocumentUploadSuccessEnroll>>()
    val navigateToUpload: LiveData<APIResponse<DocumentUploadSuccessEnroll>>
        get() = _navigateToUpload
    var sendToReg: Boolean = false
    var regType: String = ""
    var currentId: String = ""
    var currentMember: MemberInfo = MemberInfo()
    var name = currentMember.firstName + " " + currentMember.lastName
    lateinit var filePart: MultipartBody.Part
    lateinit var fragmentManager: FragmentManager

    var memberInfoPicUpload: MemberInfo = MemberInfo()
    lateinit var currentFragment: Fragment

    var appRepositoryValue: AppRepository = AppRepository()
    var filterOption: FindCareFilterModel = FindCareFilterModel()
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.filterOption, null)
        }
    var useThirdPartyProviderSearch: Boolean = false
    var showHtmlBasedFindADoc: Boolean = false
    private val _currentMemberID = MutableLiveData<String>()
    val currentMemberID: LiveData<String>
        get() = _currentMemberID
    private val _notificationCountMore = MutableLiveData<Int>()
    val getNotificationCountMore: LiveData<Int>
        get() = _notificationCountMore
    var offlineState: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.offlineState, null)
        }

    val _isInternetConnected = MutableLiveData<Boolean>()
    val isInternetConnected: LiveData<Boolean>
        get() = _isInternetConnected

    lateinit var memberId: String

    fun init() {
        memberId = preferences.getString("memberID", "") ?: ""
        _notificationCountMore.postValue(0)
        // initial client data

        getClientData()
    }

    fun checkInternet(identifier: String, isConnected: Boolean) {
        if (identifier == HomePageViewModel::class.java.simpleName) {
            if (isConnected) {
                isInternetAvailable()
            } else {
                _isInternetConnected.postValue(true)
            }
        }
    }

    override fun onCleared() {
        super.onCleared()

        compositeDisposable.dispose()
    }

    private fun isInternetAvailable() {
        CheckInternetTask(this).execute()
    }

    fun name() {
        name = currentMember.firstName + " " + currentMember.lastName
    }

    fun uploadPicture(pic: File, memberInfo: MemberInfo) {
        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    homeRepository.uploadProfile(memberInfo.selfId, filePart)
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            loading = true
                            _navigateToUpload.value =
                                APIResponse<DocumentUploadSuccessEnroll>().loading()
                        }
                        .subscribe({
                            loading = false
                            preferences.edit().remove("uri${memberInfo.selfId}").apply()
                            memberInfoRefresh(it)
                        }, {
                            loading = false

                            _navigateToUpload.value =
                                APIResponse<DocumentUploadSuccessEnroll>().error(
                                    it.localizedMessage!!
                                )
                        })
                )
            }
        )
    }

    fun memberInfoRefresh(message: DocumentUploadSuccessEnroll) {
        compositeDisposable.add(
            homeRepository.getMember()
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnDispose {
                    Timber.d("here")
                }.doOnSubscribe {
                }.doOnError {
                }.subscribe({
                    if (!Constants.IS_OFFLINE) {
                        homeRepository.insertMember(it)
                    }
                    _navigateToUpload.value =
                        APIResponse<DocumentUploadSuccessEnroll>().success(message)
                }, {
                })
        )
    }

    fun removeSpeciality() {
        filterOption.speciality = ""
        filterOption.selectedSubSpeciality.clear()
        filterOption.taxonomyCodes.clear()
    }

    fun offlineValue(value: Boolean) {
        offlineState = value
    }

    fun postCountValue(count: Int) {
        _notificationCountMore.postValue(count)
    }

    fun postID(memberID: String) {
        _currentMemberID.value = memberID
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    fun getHomeRepository(): HomeRepository {
        return homeRepository
    }

    fun getDocumentRepository(): DocumentRepository {
        return documentRepository
    }

    private val _imageClient = MutableLiveData<Event<ByteArray>>()
    val imageClient: LiveData<Event<ByteArray>>
        get() = _imageClient

    var configDb: LiveData<ClientDetailResponse>? = null

    fun getClientData() {
        val isGuestUser = preferences.getBoolean(Constants.IS_GUEST, false)

        if (isGuestUser) {
            return
        }

        Timber.d("onlyOnce---client data")
        viewModelScope.launch {
            memberId = preferences.getString("memberID", "") ?: ""
            compositeDisposable.add(
                homeRepository.getClientLogo()
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        loading = true
                    }
                    .subscribe({
                        it.selfID = memberId ?: ""
                        preferences.edit()
                            .putString(
                                memberId + "_client_disclaimer",
                                it.benefitDisclaimerText ?: ""
                            )
                            .apply()
                        Constants.CLIENT_DISCLAIMER_BILL_PREVIEW =
                            it.advancedEOBDisclaimerText ?: ""
                        preferences.edit()
                            .putString(
                                memberId + "_client_disclaimer_bill_preview",
                                it.advancedEOBDisclaimerText ?: ""
                            )
                            .apply()
                        homeRepository.insertClientDetail(it)
                        loading = false
                    }, {
                        loading = false
                    })
            )
        }
    }

    fun getMemberInfo() {
        val isGuestUser = preferences.getBoolean(Constants.IS_GUEST, false)

        if (isGuestUser) {
            return
        }

        if (!Constants.IS_OFFLINE) {
            viewModelScope.launch {
                homeRepository.getMember()
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .subscribe({
                        if (!Constants.IS_OFFLINE) {
                            homeRepository.insertMember(it)
                        }
                    }, {
                    })
            }
        }
    }

    fun perFetchDataForDependent(memberInfo: MemberInfo) {
        viewModelScope.launch {
            // child
            val bubbleMemberList = memberInfo.dependentsArrayList

            // primary
            memberInfo.insured?.let {
                bubbleMemberList?.add(it)
            }

            bubbleMemberList?.forEachIndexed { index, item ->
                if ((!Constants.IS_OFFLINE) && item.selfId.isNotBlank() && item.selfId.toLong() > 0) {
                    // Use debouncer to avoid duplicate insurance requests with 1 second delay
                    insuranceDebouncer.debounce(item.selfId, 1000L) { memberId ->
                        preFetchInsurance(memberId)
                    }

                    // Use debouncer to avoid duplicate data requests with 1 second delay
                    dataDebouncer.debounce(
                        Pair(memberInfo, item.selfId),
                        1000L
                    ) { (memberInfo, memberId) ->
                        preFetchData(memberInfo, memberId)
                    }
                }
            }
        }
    }

    var prefetchRecentCardDisposable: Disposable? = null
    var prefetchRecentBenefitDisposable: Disposable? = null

    private fun preFetchData(memberInfo: MemberInfo, memberId: String) {
        var selectedMember = memberId

        val selfId = memberInfo.selfId
        prefetchRecentCardDisposable?.let {
            it.dispose()
            compositeDisposable.remove(it)
        }
        prefetchRecentBenefitDisposable?.let {
            it.dispose()
            compositeDisposable.remove(it)
        }

        prefetchRecentCardDisposable = homeRepository.getRecentActivities(selfId, selectedMember)
            .subscribeOn(Schedulers.io())
            .observeOn(AndroidSchedulers.mainThread())
            .doOnSubscribe {
            }
            .subscribe({
                it.selfId = memberId
                homeRepository.insertRecent(it)
            }, {
            })
        compositeDisposable.add(prefetchRecentCardDisposable!!)

        prefetchRecentBenefitDisposable = homeRepository.getBenefitsCards(memberId)
            .subscribeOn(Schedulers.io())
            .observeOn(AndroidSchedulers.mainThread())
            .doOnSubscribe {
            }
            .subscribe({
                val cardGone = it.insuranceCardQuickViewDTOList.isNullOrEmpty()
                it.selfId = memberId

                if (!cardGone) {
                    it.insuranceCardQuickViewDTOList?.forEach { data ->
                        data.date = DateUtils.getSystemDateTimeAsString().toString()
                    }
                    homeRepository.insertBenefit(it)
                }
            }, {
            })
        compositeDisposable.add(prefetchRecentBenefitDisposable!!)
    }

    private fun preFetchInsurance(memberId: String) {
        compositeDisposable.add(
            benefitWorkRepository.getInsuranceCard(memberId, "")
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                }
                .subscribe({
                    if (!it.values.isNullOrEmpty()) {
                        it.selfId = memberId
                        benefitWorkRepository.insertInsurance(it)
                    }
                }, {
                })
        )
    }
}
