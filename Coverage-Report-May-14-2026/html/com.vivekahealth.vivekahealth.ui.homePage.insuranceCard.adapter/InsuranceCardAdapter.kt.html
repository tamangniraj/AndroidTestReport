package com.vivekahealth.vivekahealth.ui.homePage.insuranceCard.adapter

import android.annotation.SuppressLint
import android.content.Context
import android.content.Intent
import android.net.Uri
import android.util.Base64
import android.view.LayoutInflater
import android.view.ViewGroup
import android.webkit.WebSettings
import android.webkit.WebView
import android.webkit.WebViewClient
import androidx.navigation.Navigation
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.data.remote.home.ValueInsurance
import com.vivekahealth.vivekahealth.databinding.SingleCardViewBinding
import com.vivekahealth.vivekahealth.ui.homePage.insuranceCard.InsuranceCardsFragmentDirections
import com.vivekahealth.vivekahealth.utils.OnSwipeTouchListener
import java.net.URLDecoder

class InsuranceCardAdapter :
    ListAdapter<ValueInsurance, InsuranceCardAdapter.InsuranceCardViewHolder>(
        InsuranceCardDiffCallback()
    ) {
    lateinit var context: Context
    var count: Boolean = true
    var selectedPosition: Int = 0
    lateinit var cardAdapter: InsuranceCardFlipAdapter
    private val itemHeights = mutableMapOf<Int, Int>()

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): InsuranceCardViewHolder {
        context = parent.context
        return InsuranceCardViewHolder(
            SingleCardViewBinding.inflate(
                LayoutInflater.from(context),
                parent,
                false
            )
        )
    }

    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    @SuppressLint("ClickableViewAccessibility")
    override fun onBindViewHolder(holder: InsuranceCardViewHolder, position: Int) {
        val item = getItem(position)

        item.back?.let {
            item.front?.let {
                val cards = mutableListOf(item.back!!, item.front!!)
                cardAdapter = InsuranceCardFlipAdapter(
                    cards
                )

                with(holder.binding) {
                    easyFlipView.setOnFlipListener { _, newCurrentSide ->

                        tvSwipeInfo.text = if (newCurrentSide.name == "BACK_SIDE") {
                            context.getString(R.string.swipe_to_see_front)
                        } else {
                            context.getString(
                                R.string.swipe_to_see_back
                            )
                        }
                    }

                    wvFront.setOnTouchListener(object : OnSwipeTouchListener(
                        holder.itemView.context
                    ) {
                            override fun onSwipeLeft() {
                                easyFlipView.flipTheView()
                            }
                            override fun onSwipeRight() {
                                easyFlipView.flipTheView()
                            }
                        })
                    wvBack.setOnTouchListener(object : OnSwipeTouchListener(
                        holder.itemView.context
                    ) {
                            override fun onSwipeLeft() {
                                easyFlipView.flipTheView()
                            }
                            override fun onSwipeRight() {
                                easyFlipView.flipTheView()
                            }
                        })

                    val card = item
                    val webSettingsF: WebSettings = wvFront.settings
                    webSettingsF.javaScriptEnabled = true

                    val webSettingsB: WebSettings = wvBack.settings
                    webSettingsB.javaScriptEnabled = true

                    wvFront.clearCache(true)
                    wvBack.clearCache(true)
                    wvFront.isClickable = true
                    wvBack.isClickable = true

                    wvBack.webViewClient = object : WebViewClient() {
                        override fun shouldOverrideUrlLoading(view: WebView, url: String?): Boolean {
                            val shouldHandleDeepLink = url?.contains("app.vivekahealth.com") ?: false
                            return if (shouldHandleDeepLink) {
                                handleDeeplink(view, url ?: "")
                                true
                            } else {
                                view.context.startActivity(
                                    Intent(Intent.ACTION_VIEW, Uri.parse(url))
                                )
                                true
                            }
                        }

                        override fun onLoadResource(view: WebView?, url: String?) {
                        }
                    }

                    wvFront.webViewClient = object : WebViewClient() {
                        override fun shouldOverrideUrlLoading(view: WebView, url: String?): Boolean {
                            val shouldHandleDeepLink = url?.contains("app.vivekahealth.com") ?: false
                            return if (shouldHandleDeepLink) {
                                handleDeeplink(view, url ?: "")
                                true
                            } else {
                                view.context.startActivity(
                                    Intent(Intent.ACTION_VIEW, Uri.parse(url))
                                )
                                true
                            }
                        }

                        override fun onLoadResource(view: WebView?, url: String?) {
                        }
                    }

                    val base64Front = Base64.encodeToString(
                        card.front!!.toByteArray(),
                        Base64.DEFAULT
                    )
                    val htmlDecoded = String(Base64.decode(base64Front, Base64.DEFAULT))
                    wvFront.loadDataWithBaseURL(
                        null,
                        htmlDecoded,
                        "text/html",
                        "utf-8",
                        null
                    )

                    val base64Back = Base64.encodeToString(
                        card.back!!.toByteArray(),
                        Base64.DEFAULT
                    )
                    val htmlDecodedBack = String(Base64.decode(base64Back, Base64.DEFAULT))
                    wvBack.loadDataWithBaseURL(
                        null,
                        htmlDecodedBack,
                        "text/html",
                        "utf-8",
                        null
                    )

                    flipButton.setOnClickListener {
                        easyFlipView.flipTheView()
                    }
                }
            }
        }
    }

    fun handleDeeplink(view: WebView, url: String) {
        val decodedUrl = URLDecoder.decode(url, "UTF-8")
        val dynamicLink = Uri.parse(decodedUrl)
        val path = dynamicLink.path
        val queryParams = dynamicLink.queryParameterNames
        if (path.contentEquals("/customMenu", true)) {
            var menuName = ""
            var menuTypeId = "1"
            for (q in queryParams) {
                if (q.equals("menuName", true)) {
                    menuName = dynamicLink.getQueryParameter(q) ?: ""
                }
                if (q.equals("menuType", true)) {
                    menuTypeId = (dynamicLink.getQueryParameter(q) ?: "1")
                }
            }
            val directToCustomMenu =
                InsuranceCardsFragmentDirections.actionInsuranceCardsFragmentToDynamicMenuFragment(
                    menuTypeId,
                    menuName
                )
            Navigation.findNavController(view).navigate(directToCustomMenu)
        }
    }

    class InsuranceCardViewHolder(
        val binding: SingleCardViewBinding
    ) : RecyclerView.ViewHolder(binding.root)
}

private class InsuranceCardDiffCallback : DiffUtil.ItemCallback<ValueInsurance>() {

    override fun areItemsTheSame(oldItem: ValueInsurance, newItem: ValueInsurance): Boolean {
        return oldItem.benefitType == newItem.benefitType
    }

    override fun areContentsTheSame(oldItem: ValueInsurance, newItem: ValueInsurance): Boolean {
        return oldItem == newItem
    }
}
