package com.vivekahealth.vivekahealth.ui.homePage.insuranceCard.adapter

import android.content.Context
import android.content.Intent
import android.net.Uri
import android.util.Base64
import android.util.LruCache
import android.view.LayoutInflater
import android.view.ViewGroup
import android.webkit.WebSettings
import android.webkit.WebView
import android.webkit.WebViewClient
import androidx.navigation.Navigation
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView.ViewHolder
import com.vivekahealth.vivekahealth.databinding.ItemCardBinding
import com.vivekahealth.vivekahealth.ui.homePage.insuranceCard.InsuranceCardsFragmentDirections
import com.vivekahealth.vivekahealth.utils.getManifestPlaceHolderValue
import java.net.URLDecoder

class InsuranceCardFlipAdapter(private val cardImages: List<String>) :
    ListAdapter<String, InsuranceCardFlipViewHolder>(InsuranceCardFlipDiffUtilCallBack()) {
    lateinit var context: Context
    private val htmlCache = LruCache<String, String>(10)
    val maxSize = Int.MAX_VALUE
    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): InsuranceCardFlipViewHolder {
        context = parent.context
        return InsuranceCardFlipViewHolder(
            ItemCardBinding.inflate(
                LayoutInflater.from(
                    context
                ),
                parent,
                false

            )
        )
    }

    override fun getItemCount() = maxSize

    override fun onBindViewHolder(holder: InsuranceCardFlipViewHolder, position: Int) {
        with(holder.itemCardBinding) {
            val cardImage = cardImages[position % 2]
            val cachedHtml = htmlCache[cardImage]
            val webSettings: WebSettings = wvCard.settings
            webSettings.javaScriptEnabled = true
            webSettings.textZoom = 100
            wvCard.webViewClient = object : WebViewClient() {
                override fun shouldOverrideUrlLoading(view: WebView, url: String?): Boolean {
                    val appHost = getManifestPlaceHolderValue(context, "APP_HOST")
                    val shouldHandleDeepLink = url?.contains(appHost ?: "app.vivekahealth.com") ?: false
                    return if (shouldHandleDeepLink) {
                        handleDeeplink(view, url ?: "")
                        true
                    } else {
                        view.context.startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(url)))
                        true
                    }
                }

                override fun onLoadResource(view: WebView?, url: String?) {
                }
            }
            wvCard.setInitialScale(1)
            webSettings.loadWithOverviewMode = true
            webSettings.useWideViewPort = true
            webSettings.javaScriptEnabled = true
            webSettings.builtInZoomControls = true
            webSettings.displayZoomControls = false

            if (cachedHtml != null) {
                wvCard.loadData(
                    cachedHtml,
                    "text/html; charset=UTF-8",
                    "base64"
                )
            } else {
                val base64Front =
                    Base64.encodeToString(cardImage.toByteArray(), Base64.DEFAULT)
                wvCard.loadData(
                    base64Front,
                    "text/html; charset=UTF-8",
                    "base64"
                )
                htmlCache.put(cardImage, base64Front)
            }
        }
    }

    fun handleDeeplink(view: WebView, url: String) {
        val decodedUrl = URLDecoder.decode(url, "UTF-8")
        val dynamicLink = Uri.parse(decodedUrl)
        val path = dynamicLink.path
        val queryParams = dynamicLink.queryParameterNames
        if (path.contentEquals("/customMenu", true)) {
            var menuName = ""
            var menuTypeId = "1"
            for (q in queryParams) {
                if (q.equals("menuName", true)) {
                    menuName = dynamicLink.getQueryParameter(q) ?: ""
                }
                if (q.equals("menuType", true)) {
                    menuTypeId = (dynamicLink.getQueryParameter(q) ?: "1")
                }
            }
            val directToCustomMenu =
                InsuranceCardsFragmentDirections.actionInsuranceCardsFragmentToDynamicMenuFragment(
                    menuTypeId,
                    menuName
                )
            Navigation.findNavController(view).navigate(directToCustomMenu)
        }
    }
}

class InsuranceCardFlipViewHolder(val itemCardBinding: ItemCardBinding) :
    ViewHolder(itemCardBinding.root)

class InsuranceCardFlipDiffUtilCallBack : DiffUtil.ItemCallback<String>() {
    override fun areItemsTheSame(oldItem: String, newItem: String) = oldItem == newItem

    override fun areContentsTheSame(oldItem: String, newItem: String) = oldItem == newItem
}
