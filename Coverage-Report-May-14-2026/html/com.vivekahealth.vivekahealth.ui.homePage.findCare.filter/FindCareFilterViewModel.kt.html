package com.vivekahealth.vivekahealth.ui.homePage.findCare.filter

import android.content.SharedPreferences
import android.location.Location
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.android.gms.maps.model.LatLng
import com.vivekahealth.vivekahealth.data.local.VivekaDatabase
import com.vivekahealth.vivekahealth.data.remote.home.findCare.FindCareFilterModel
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.disposables.CompositeDisposable
import javax.inject.Inject

class FindCareFilterViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository,
    private val vivekaDatabase: VivekaDatabase
) : ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    final var accessToken = preferences.getString(Constants.ACCESS_TOKEN, "").toString()
    var wentToSetting: Boolean = false
    var defaultLocationType: Int? = null
    var defaultAddressName: String? = null
    var defaultLatLng: LatLng? = null

    var count: Int = 0
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.count, null)
        }
    var filterOptionOriginal: FindCareFilterModel = FindCareFilterModel()
    var filterOption: FindCareFilterModel = FindCareFilterModel()
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.filterOption, null)
        }

    private val _navigateToPres = MutableLiveData<Event<Unit>>()
    val navigateToPres: LiveData<Event<Unit>>
        get() = _navigateToPres
    private val _navigateToDocFilter = MutableLiveData<Event<Unit>>()
    val navigateToDocFilter: LiveData<Event<Unit>>
        get() = _navigateToDocFilter

    private val _navigateToClear = MutableLiveData<Event<Unit>>()
    val navigateToClear: LiveData<Event<Unit>>
        get() = _navigateToClear

    fun clear() {
        _navigateToClear.value = Event(Unit)
        filterOption.locationType = defaultLocationType.toString()
        filterOption.radius = "10"
        filterOption.current = defaultLatLng
        filterOption.addressName = defaultAddressName
        filterOption.speciality = ""
        filterOption.selectedSubSpeciality = arrayListOf()
        filterOption.outNetwork = false
        filterOption.providerType = "1"
        removeSpeciality()
        filterOption = filterOption
    }

    fun removeSpeciality() {
        if (filterOption.showSpeciality) {
            preferences.edit().putString("key", "").apply()
        }
        filterOption.speciality = ""
        filterOption.selectedSubSpeciality.clear()
        filterOption.taxonomyGroup = ""
        filterOption.taxonomyCodes.clear()
    }

    fun assignValue(value: FindCareFilterModel) {
        filterOption = value
    }

    fun getSpeciality() {
        if (filterOption.showSpeciality) {
            val list = filterOption.selectedSubSpeciality.map { it.speciality }
            filterOption.speciality = list.joinToString(separator = ",") { it -> it }
            filterOption = filterOption
        }
    }

    fun locationCurrent(currentLocation: Location) {
        filterOption.current = LatLng(currentLocation.latitude, currentLocation.longitude)
        filterOption.near = LatLng(currentLocation.latitude, currentLocation.longitude)
        filterOption = filterOption
    }

    fun close() {
        filterOption = filterOptionOriginal
        _navigateToPres.value = Event(Unit)
    }

    fun outNetworkClick() {
        filterOption.providerType = "1"
        filterOption.outNetwork = false
        filterOption.outNetwork = filterOption.outNetwork
    }

    fun apply() {
        _navigateToDocFilter.value = Event(Unit)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
