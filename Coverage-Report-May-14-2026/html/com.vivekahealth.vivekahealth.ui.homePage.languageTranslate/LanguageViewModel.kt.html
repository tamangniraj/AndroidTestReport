package com.vivekahealth.vivekahealth.ui.homePage.languageTranslate

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.AppScope
import com.vivekahealth.vivekahealth.data.remote.home.LanguageList
import com.vivekahealth.vivekahealth.data.remote.home.UpdateDetails
import com.vivekahealth.vivekahealth.ui.guestHome.repository.GuestHomeRepository
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Constants
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.schedulers.Schedulers
import kotlinx.coroutines.launch
import timber.log.Timber
import javax.inject.Inject

class LanguageViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository,
    private val guestHomeRepository: GuestHomeRepository
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private val _languages = MutableLiveData<List<LanguageList>>()
    val languages: LiveData<List<LanguageList>> get() = _languages

    var selectedCode: String? = null
    private var languageMap: List<LanguageList> = emptyList()

    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }

    fun initLanguages(localizedLabels: List<Pair<String, String>>) {
        val memId = preferences.getString("memberID", "") ?: ""
        val savedCode = preferences.getString("$memId${Constants.LanguagePref}", "en") ?: "en"

        // Use saved preference OR Constants.LOCAL_LANG_MODIFIED (if non-blank)
        selectedCode = Constants.CURRENT_LANGUAGE.takeIf { !it.isNullOrBlank() } ?: savedCode

        languageMap = localizedLabels.map { (label, code) ->
            LanguageList(label, selectedCode.equals(code, ignoreCase = true), code)
        }

        _languages.value = languageMap
    }

    fun selectLanguage(selected: LanguageList) {
        selectedCode = selected.code
        Constants.LOCAL_LANG_MODIFIED = selected.code
        Constants.CURRENT_LANGUAGE = selected.code
        saveLanguage(selected)

        _languages.value = languageMap.map {
            it.copy(selected = it.code == selected.code)
        }

        sendToServer(selected)
    }

    private fun saveLanguage(language: LanguageList) {
        preferences.edit()
            .putString(
                preferences.getString("memberID", "") + Constants.LanguagePref,
                language.code
            )
            .apply()
    }

    private fun sendToServer(language: LanguageList) {
        if (!Constants.IS_OFFLINE) {
            loading = true
            AppScope.scope.launch {
                try {
                    val isGuest = preferences.getBoolean(Constants.IS_GUEST, false)
                    val memberId = preferences.getString("memberID", "") ?: ""
                    if (isGuest) {
                        guestHomeRepository.setLanguagePreference(language.code)
                            .subscribeOn(Schedulers.io())
                            .observeOn(AndroidSchedulers.mainThread())
                            .subscribe({
                                loading = false
                            }, {
                                loading = false
                            })
                    } else {
                        homeRepository.updateUserDetails(
                            memberId,
                            UpdateDetails(languagePreference = language.code)
                        ).subscribeOn(Schedulers.io())
                            .observeOn(AndroidSchedulers.mainThread())
                            .subscribe({
                                loading = false
                            }, {
                                loading = false
                            })
                    }
                } catch (e: Exception) {
                    loading = false
                    Timber.e("Language update failed: ${e.localizedMessage}")
                }
            }
        }
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }
}
