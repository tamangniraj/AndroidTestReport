package com.vivekahealth.vivekahealth.ui.resetPassword

import android.content.SharedPreferences
import android.text.Editable
import android.text.TextWatcher
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.register.RegisterUser
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.data.remote.resetPassword.ChangePasswordV3DTO
import com.vivekahealth.vivekahealth.ui.login.repository.LoginRepository
import com.vivekahealth.vivekahealth.ui.resetPassword.repository.ResetPasswordRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.remote.APIResponse
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import java.util.regex.Matcher
import java.util.regex.Pattern
import javax.inject.Inject
import org.apache.commons.lang3.StringUtils

class ResetPasswordViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val loginRepository: LoginRepository,
    private val resetPasswordRepository: ResetPasswordRepository
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    private val _endVideoLive = MutableLiveData<Event<Unit>>()

    private val _passwordChanged = MutableLiveData<Event<String>>()

    var errorMessage: String = "Passwords do not match"
    val errorOrNot: Boolean = true
    val passwordChanged: LiveData<Event<String>>
        get() = _passwordChanged

    var resetOrChange: Boolean = false
    private val _confirmPasswordChanged = MutableLiveData<Event<String>>()

    val confirmPasswordChanged: LiveData<Event<String>>
        get() = _confirmPasswordChanged

    private val _navigateToPassword = MutableLiveData<APIResponse<DataResponse>>()
    val navigateToPassword: LiveData<APIResponse<DataResponse>>
        get() = _navigateToPassword
    private val _navigateToChangePassword = MutableLiveData<APIResponse<DataResponse>>()
    val navigateToChangePassword: LiveData<APIResponse<DataResponse>>
        get() = _navigateToChangePassword

    var password: String = ""
        @Bindable get set(value) {
        field = value
        callbacks.notifyChange(this, BR.password)
    }

    var confirmPassword: String = ""
    var email: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.email, null)
        }

    val emailChanged: LiveData<Event<String>>
        get() = _emailChanged

    private val _emailChanged = MutableLiveData<Event<String>>()

    @get:Bindable
    val emailTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            _emailChanged.value = Event(s.toString())
        }
    }

    @get:Bindable
    val passwordTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
            if (s.toString().isNotEmpty()) {
                _passwordChanged.value = Event(s.toString())
            }
        }
    }

    @get:Bindable
    val confirmPasswordTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }
        override fun afterTextChanged(s: Editable) {
//            confirmPassword = s.toString()
            if (s.toString().isNotEmpty()) {
                _confirmPasswordChanged.value = Event(s.toString())
            }
        }
    }

    fun passwordCheck() {
    }

    fun passLen(password: String): Boolean {
        return StringUtils.isNotEmpty(password) && password.length >= 8
    }

    fun passNumber(password: String): Boolean {
        val letter: Pattern = Pattern.compile(".*\\d.*")
        val hasLetter: Matcher = letter.matcher(password)

        return hasLetter.find()
    }

    fun passCaps(password: String): Boolean {
        val letterLow: Pattern = Pattern.compile(".*[A-Z]")
        val hasLetterLow: Matcher = letterLow.matcher(password)
        val letterUp: Pattern = Pattern.compile(".*[a-z]")
        val hasLetterUp: Matcher = letterUp.matcher(password)
        return hasLetterLow.find() && hasLetterUp.find()
//        return password.contains(".*[A-Z]") && password.contains(".*[a-z])")
    }
    fun passSpecialChar(password: String): Boolean {
        val letter: Pattern = Pattern.compile("[!@#$%&*()_+=|<>?{}\\[\\]~-]")
        val hasLetter: Matcher = letter.matcher(password)
        return hasLetter.find()

//        return password.contains("[!@#$%&*()_+=|<>?{}\\[\\]~-]")
    }

    fun passAndConfirm(password: String, confirm: String): Boolean {
        return if (StringUtils.isNotEmpty(password) && StringUtils.isNotEmpty(confirm)) {
            StringUtils.equals(password, confirm)
        } else {
            false
        }
//        return password.contains("[!@#$%&*()_+=|<>?{}\\[\\]~-]")
    }

    fun callAPi() {
        if (!resetOrChange) {
            compositeDisposable.add(
                resetPasswordRepository.getForgotPassword(email)
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _navigateToPassword.value = APIResponse<DataResponse>().loading()
                    }
                    .subscribe({
                        _navigateToPassword.value = APIResponse<DataResponse>().success(it)
                    }, {
                        _navigateToPassword.value = APIResponse<DataResponse>().error(
                            it.localizedMessage
                        )
                    })
            )
        } else {
            compositeDisposable.add(
                resetPasswordRepository.changePassword(
                    ChangePasswordV3DTO(email, null, password, confirmPassword, null)
                )
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _navigateToChangePassword.value = APIResponse<DataResponse>().loading()
                    }
                    .subscribe({
                        _navigateToChangePassword.value = APIResponse<DataResponse>().success(it)
                    }, {
                        _navigateToChangePassword.value = APIResponse<DataResponse>().error(
                            it.localizedMessage
                        )
                    })
            )
        }
    }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun removeReg() {
        val regEdit = preferences.edit()
        regEdit.putString(Constants.Registration, "")
        regEdit.apply()
    }

    fun removeEnroll() {
        val regEdit = preferences.edit()
        regEdit.putString(Constants.enroll, "")
        regEdit.apply()
    }

    fun saveEmailAndOverrideOldReg() {
        val user = RegisterUser()
        user.email = email
        user.overrideOldRegistration = "true"
        val regString = Gson().toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.register, regString)
        regEdit.apply()
    }
    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }

    private val _navigateToRegister = MutableLiveData<Event<Boolean>>()
    val navigateToRegister: LiveData<Event<Boolean>>
        get() = _navigateToRegister

    fun goToRegister() {
        removeEnroll()
        removeReg()
        saveEmailAndOverrideOldReg()
        _navigateToRegister.value = Event(true)
    }
}
