package com.vivekahealth.vivekahealth.ui.guestRegister.phone

import android.content.SharedPreferences
import android.text.Editable

import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.lifecycle.LiveData

import androidx.lifecycle.ViewModel

import com.google.android.recaptcha.RecaptchaClient
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.guest.ResponseOtpReceiveParams
import com.vivekahealth.vivekahealth.data.remote.guest.Source
import com.vivekahealth.vivekahealth.data.remote.login.GenericErrorResponse

class GuestPhoneViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val phoneCodeRepository: PhoneCodeRepository,
    private val recaptchaClient: RecaptchaClient
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    var gson = Gson()
    var user = RegisterUser()
    private lateinit var job: Job

    private val _phoneNumberChanged = MutableLiveData<Event<String>>()

    val phoneNumberChanged: LiveData<Event<String>>
        get() = _phoneNumberChanged

    private val _phoneNumberPrev = MutableLiveData<Event<String>>()

    val phoneNumberPrev: LiveData<Event<String>>
        get() = _phoneNumberPrev
    val showError: LiveData<APIResponse<ResponseOtpReceiveParams>>
        get() = _showError
    private val _showError = MutableLiveData<APIResponse<ResponseOtpReceiveParams>>()
    private val _navigateToBack1 = MutableLiveData<Event<Unit>>()
    val navigateToBack1: LiveData<Event<Unit>>
        get() = _navigateToBack1

    private val _navigateToCode = MutableLiveData<APIResponse<ResponseOtpReceiveParams>>()
    val navigateToCode: LiveData<APIResponse<ResponseOtpReceiveParams>>
        get() = _navigateToCode

    var cCode: String = "+1"

    var phoneNumber: String = ""

    var source: Source ? = null

    private val _isCheckConsent = MutableLiveData<Boolean>(false)
    val isCheckConsent: LiveData<Boolean>
        get() = _isCheckConsent

    fun setCheckConsent(isChecked: Boolean) {
        _isCheckConsent.value = isChecked
    }

    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.Registration, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.Registration, ""),
                        RegisterUser::class.java
                    )
                }
            },
            {
                cCode = user.countryCode.toString()
                if (cCode.isEmpty()) {
                    cCode = "+1"
                }
                phoneNumber = user.mobile.toString()
                _phoneNumberPrev.value = Event(user.mobile.toString())
                email = user.email
                registrationId = user.registrationId.toString()
                _isCheckConsent.value = user.consentChecked ?: false
            }
        )
    }

    fun savePhone() {
        user.mobile = phoneNumber
        user.countryCode = cCode

        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.Registration, regString)
        regEdit.apply()
    }
    var modifiedText: String = ""

    @get:Bindable
    val phoneTextWatcher = object : TextWatcher {

        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            var regex = Regex("^[0-9]{3} [0-9]{3} [0-9]{4}$")

            if (!s.toString().matches(regex)) {
                var unformated = s.toString().replace("(", "").replace(")", "").replace("-", "").replace(
                    " ",
                    ""
                )
                if (unformated.length > 10) {
                    if (!s.toString().matches(regex)) {
                        var st = unformated.substring(unformated.length - 10, unformated.length)
                        var final = st.replace("(\\d{3})(\\d{3})(\\d{4})".toRegex(), "$1 $2 $3")
//                        _phoneNumberPrev.value = Event(final)
                    }
                } else {
                }
            }
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
            var regex = Regex("^[0-9]{3} [0-9]{3} [0-9]{4}$")

            if (!s.toString().matches(regex)) {
                var unformated = s.toString().replace("(", "").replace(")", "").replace("-", "").replace(
                    " ",
                    ""
                )
                if (unformated.length > 10) {
                    if (!s.toString().matches(regex)) {
                        var st = unformated.substring(unformated.length - 10, unformated.length)
                        var final = st.replace("(\\d{3})(\\d{3})(\\d{4})".toRegex(), "$1 $2 $3")
                        modifiedText = final
                        _phoneNumberPrev.value = Event(final)
                        _phoneNumberChanged.value = Event(final)
                    }
                } else {
                    modifiedText = ""
                    _phoneNumberChanged.value = Event(s.toString())
                }
            } else {
                if (modifiedText.isEmpty()) {
                    _phoneNumberChanged.value = Event(s.toString())
                }
            }
        }
        override fun afterTextChanged(s: Editable) {
        }
    }
    fun onPrevClicked() {
        _navigateToBack1.value = Event(Unit)
    }

    var registrationId: String = ""
    var channel: String ? = ""
    var email: String? = ""

    private var triggerCaptchaInNextRequest: Boolean = false

    fun onNextClicked() {
        executeWithOptionalCaptcha(triggerCaptchaInNextRequest) { token ->
            Timber.d(phoneNumber)
            compositeDisposable.add(
                phoneCodeRepository.requestOTPV2(
                    phoneNumber,
                    cCode,
                    Constants.CHANNEL_PHONE,
                    token,
                    registrationId,
                    isCheckConsent.value,
                    source = source
                )
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _navigateToCode.value = APIResponse<ResponseOtpReceiveParams>().loading()
                    }
                    .subscribe({
                        if (it.isSuccessful) {
                            it.body()?.let {
                                    otpResponse ->

                                otpResponse.responseData?.firstOrNull()?.let {
                                    user.tempUUID = it.registrationId
                                    user.consentChecked = isCheckConsent.value
//                                    triggerCaptchaInNextRequest = it.triggerCaptchaInNextRequest
                                    val regEdit = preferences.edit()
                                    regEdit.putLong(
                                        Constants.ExpirationTime,
                                        it.otpExpiry.toLong() * 1000
                                    )
                                    regEdit.putBoolean(
                                        Constants.ShowErrorMessage,
                                        it.status == "ERROR"
                                    )
                                    regEdit.putString(Constants.errorMessage, it.message.toString())
                                    regEdit.apply()
                                    savePhone()
                                    _navigateToCode.value = APIResponse<ResponseOtpReceiveParams>().success(
                                        otpResponse
                                    )
                                }
                            }
                        } else {
                            try {
                                val errorResponse: GenericErrorResponse = gson.fromJson(
                                    it.errorBody()?.string(),
                                    GenericErrorResponse::class.java
                                )

                                _showError.value =
                                    APIResponse<ResponseOtpReceiveParams>().error(
                                        errorResponse.getErrorMessage()
                                    )
                            } catch (e: Exception) {
                                e.printStackTrace()
                            }
                        }
                    }, {
                        _showError.value = APIResponse<ResponseOtpReceiveParams>().error(it)
                    })
            )
        }
    }

    private fun executeWithOptionalCaptcha(
        requestCaptcha: Boolean,
        onResult: (token: String?) -> Unit
    ) {
        if (requestCaptcha) {
            viewModelScope.launch {
                val result =
                    recaptchaClient
                        .execute(RecaptchaAction.SIGNUP, timeout = 45000L)

                onResult(result.getOrNull())
            }
        } else {
            onResult(null)
        }
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
