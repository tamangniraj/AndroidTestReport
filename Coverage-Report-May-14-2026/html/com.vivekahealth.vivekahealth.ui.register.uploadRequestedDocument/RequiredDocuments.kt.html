class RequiredDocuments : BaseDialogFragment(), Injectable, NextClickListener, DocClickedListener {

    companion object {
        fun newInstance() = RequiredDocuments()
    }
    val adapter = DocAdapter(
        onClick = {
            viewModel.currentItem = it
            viewModel.currentPerson = it.person!!
            viewModel.currentDoc = it.docName!!.FileName
            photoPicker()
        }
    )
    var successCallbackFrag: SuccessCallbackDocumentRegistration? = null
    var user = RegisterUser()
    fun setSuccessCallback(
        successCallback: SuccessCallbackDocumentRegistration,
        user: RegisterUser
    ) {
        this.successCallbackFrag = successCallback
        this.user = user
    }
    private lateinit var binding: FragmentRegistrationRequiredDocumentsBinding
    private lateinit var cropImageLauncher: ActivityResultLauncher<CropImageContractOptions>
    private lateinit var requestPermissionLauncher: ActivityResultLauncher<String>
    private lateinit var galleryLauncher: ActivityResultLauncher<String>
    private lateinit var cameraLauncher: ActivityResultLauncher<Uri>
    private var uriForFile: Uri? = null

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory

    private lateinit var viewModel: RequiredDocumentsViewModel

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = FragmentRegistrationRequiredDocumentsBinding.inflate(inflater, container, false)
        return binding.root
    }
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setStyle(STYLE_NORMAL, R.style.FullscreenDialogTheme)

        requestPermissionLauncher = registerForActivityResult(
            ActivityResultContracts.RequestPermission()
        ) { isGranted ->
            if (isGranted) {
                launchCamera()
            }
        }

        cropImageLauncher = registerForActivityResult(CropImageContract()) { result ->
            if (result.isSuccessful) {
                result.uriContent?.let { croppedUri ->
                    openUpload(croppedUri)
                }
            } else if (result !is CropImage.CancelledResult) {
                val exception = result.error
                exception?.let {
                    android.widget.Toast.makeText(
                        requireContext(),
                        "Crop failed: ${it.message}",
                        android.widget.Toast.LENGTH_SHORT
                    ).show()
                }
            }
        }

        galleryLauncher = registerForActivityResult(
            ActivityResultContracts.GetContent()
        ) { uri: Uri? ->
            uri?.let {
                cropImageLauncher.launch(
                    CropImageContractOptions(
                        uri = it,
                        cropImageOptions = CropImageOptions(
                            guidelines = CropImageView.Guidelines.ON,
                            outputCompressQuality = 30,
                            cropShape = CropImageView.CropShape.RECTANGLE
                        )
                    )
                )
            }
        }

        cameraLauncher = registerForActivityResult(
            ActivityResultContracts.TakePicture()
        ) { success: Boolean ->
            if (success) {
                uriForFile?.let {
                    cropImageLauncher.launch(
                        CropImageContractOptions(
                            uri = it,
                            cropImageOptions = CropImageOptions(
                                guidelines = CropImageView.Guidelines.ON,
                                outputCompressQuality = 30,
                                cropShape = CropImageView.CropShape.RECTANGLE
                            )
                        )
                    )
                }
            }
        }
    }
    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)

        viewModel = ViewModelProvider(this, viewModelFactory).get(
            RequiredDocumentsViewModel::class.java
        ).apply {
            nextClickListener = this@RequiredDocuments
        }
        if (viewModel.docList.isEmpty()) {
            viewModel.user = EnrollMember(
                Person(
                    firstName = user.firstName,
                    lastName = user.lastName,
                    dateOfBirth = user.dob,
                    gender = user.gender,
                    ssn = user.ssn
                )
            )
            viewModel.getEnrollFromShared()
        }
        binding.viewModel = viewModel

        val animator: RecyclerView.ItemAnimator? = binding.docRecyclerView.itemAnimator
        if (animator is SimpleItemAnimator) {
            (animator as SimpleItemAnimator).supportsChangeAnimations = false
        }

        binding.nextEnable = false

        binding.docRecyclerView.adapter = adapter

        viewModel.dataRetrieved.observe(
            viewLifecycleOwner,
            EventObserver { list ->

                viewModel.createDocList()
            }
        )

        viewModel.navigateToBack.observe(
            viewLifecycleOwner,
            EventObserver {
                successCallbackFrag?.onBackClick()
            }
        )

        viewModel.navigateToNext.observe(
            viewLifecycleOwner,
            EventObserver {
                viewModel.docList.clear()
                successCallbackFrag?.onOkClick(documentId = viewModel.documentID)
            }
        )
        viewModel.navigateToUpload.observe(viewLifecycleOwner) {
            when (it.status) {
                Status.LOADING -> {
                    (activity as RegisterActivity).showLoading(
                        getString(R.string.uploading_document)
                    )
                }
                Status.SUCCESS -> {
                    viewModel.setDocumentName(it.data?.id.toString(), it.data?.fileName.toString())

                    (activity as RegisterActivity).dismissLoading()
                }
                Status.ERROR -> {
                    (activity as RegisterActivity).dismissLoading()
                }
            }
        }

        viewModel.listCreated.observe(
            viewLifecycleOwner,
            EventObserver { list ->

                adapter.submitList(list)
                if (list.indexOf(viewModel.currentItem) != -1) {
                    adapter.notifyItemChanged(list.indexOf(viewModel.currentItem))
                } else {
                    adapter.notifyItemRangeChanged(0, list.size - 1)
                }
                binding.nextEnable = viewModel.checkIfAllComplete()
            }
        )
    }

    private fun photoPicker() {
        // Show picker dialog to choose between camera and gallery
        // with proper permission handling for Android 13+
        val dialog: BottomSheetDialogFragment = PickerSheetFragment.newInstance()
        val bundle = Bundle()
        bundle.putBoolean("home", true)
        dialog.arguments = bundle
        (dialog as PickerSheetFragment).setCallback(object : PickerButtonListener {
            override fun onCamera() {
                requestCameraPermission()
                dialog.dismiss()
            }

            override fun onLibrary() {
                galleryLauncher.launch("image/*")
                dialog.dismiss()
            }

            override fun onFileLibrary() {
            }

            override fun onRemoveDependent() {
            }

            override fun onRemove() {
            }

            override fun onClose() {
                dialog.dismiss()
            }
        })

        dialog.show(childFragmentManager, "tag")
    }

    fun requestCameraPermission() {
        when {
            ContextCompat.checkSelfPermission(
                requireContext(),
                Manifest.permission.CAMERA
            ) == PackageManager.PERMISSION_GRANTED -> {
                launchCamera()
            }
            shouldShowRequestPermissionRationale(Manifest.permission.CAMERA) -> {
                Timber.tag(ContentValues.TAG)
                    .d("requestCameraPermission - Camera Permission NOT Granted")
                requestPermissionLauncher.launch(Manifest.permission.CAMERA)
            }
            else -> {
                requestPermissionLauncher.launch(Manifest.permission.CAMERA)
            }
        }
    }

    private fun launchCamera() {
        try {
            val imageFile = HelperUtils.createImageFile(requireActivity())
            uriForFile = FileProvider.getUriForFile(
                requireActivity(),
                requireContext().packageName + ".provider",
                imageFile
            )
            uriForFile?.let { cameraLauncher.launch(it) }
        } catch (e: IOException) {
            e.printStackTrace()
        }
    }

    // Old onActivityResult removed - now using Activity Result API with registered launchers
    fun handleCropImageResult(uri: String) {
        SCropResultActivity.start(this, null, Uri.parse(uri), null)
    }

    fun openUpload(item: Uri?) {
        val stream = ByteArrayOutputStream()

        val file = File(item?.path.toString())
        val fileExt = MimeTypeMap.getFileExtensionFromUrl(file.toString())

        val originalFileName = File(item?.path.toString()).name

        val imageStream: InputStream? = item?.let { activity?.contentResolver?.openInputStream(it) }

//        var docBitmap = BitmapFactory.decodeFile(item?.path)
        val docBitmap = BitmapFactory.decodeStream(imageStream)

        docBitmap.compress(Bitmap.CompressFormat.JPEG, 50, stream)
        val byteArray = stream.toByteArray()
        val dialog: DialogFragment = UploadDocumentFragment.newInstance()
        val bundle = Bundle()
        viewModel.filePart = HelperUtils.buildImageBodyPart(
            originalFileName,
            docBitmap,
            requireContext()
        )
        bundle.putByteArray("BitMap", byteArray)
        bundle.putString("DocumentName", viewModel.currentDoc)
        bundle.putString(
            "PersonName",
            viewModel.currentPerson.firstName + " " + viewModel.currentPerson.lastName
        )
        dialog.arguments = bundle
        (dialog as UploadDocumentFragment).setCallback(object : UploadDocCallback {
            override fun onActionClick() {
                photoPicker()
                dialog.dismiss()
            }

            override fun onUploadClick() {
                viewModel.uploadPicture(file, viewModel.currentItem)
                dialog.dismiss()
            }
        })

        dialog.show(childFragmentManager, "tag")
    }
}
