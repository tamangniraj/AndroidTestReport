    fun setDocumentName(id: String, fileName: String) {
        job = Coroutines.ioToMainThread(
            {
                when {
                    currentPerson == user.subscriber -> {
                        when {
                            currentItem.id == 1 -> {
                                documentID = id
                            }
                        }
                    }
                }
            },
            {
                var index = docList.indexOf(currentItem)
                if (index != -1) {
                    currentItem.uploaded = true

                    val updatedItems = docList.toMutableList().apply { this[index] = currentItem }

                    docList.clear()
                    docList.addAll(updatedItems)

                    _listCreated.value = Event(docList)
                }
            }
        )
    }
    fun uploadPicture(pic: File, currentDoc: ListDoc) {
        val file: File = pic
//
//        val filePart: MultipartBody.Part = MultipartBody.Part.createFormData(
//            "file",
//
//          pic.name,
//            RequestBody.create("image/*".toMediaTypeOrNull(), pic)
//        )

        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    enrollRepository.uploadDoc(
                        filePart,
                        HelperUtils.fileName(
                            currentDoc.docName!!.FileName,
                            currentPerson.firstName.toString(),
                            currentPerson.lastName.toString()
                        )
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().loading()
                        }
                        .subscribe({
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().success(
                                it
                            )
                        }, {
                            _navigateToUpload.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                                it.localizedMessage!!
                            )
                        })
                )
            }
        )
    }

    fun createDocList() {
        job = Coroutines.ioToMainThread(
            {
                user.subscriber?.let {
                    docList.add(
                        ListDoc(
                            user.subscriber!!.birthCertificateFileId!!.isNotBlank(),
                            DocumentType.BirtCertificate,
                            it,
                            1
                        )
                    )
                }
            },
            {
                _listCreated.value = Event(docList)
            }
        )
    }

    fun checkIfAllComplete(): Boolean {
        return docList.all {
            if (!it.uploaded && it.optional) {
                true
            } else {
                it.uploaded
            }
        }
    }

    fun onPrevClicked() {
        _navigateToBack.value = Event(Unit)
    }
    fun onNextClicked() {
        _navigateToNext.value = Event(Unit)
    }

    fun saveUser() {
        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.enroll, regString)
        regEdit.apply()
    }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
