package com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.specialityDetails

class SpecialityDetailsViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val fspecialityDetailsRepository: SpecialityAllRepository
) : ViewModel(), Observable {

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    var ValueList: ArrayList<DoctorResponseNew> = ArrayList()
    var eligibleValueList: ArrayList<DoctorResponseNew> = ArrayList()
    var listForDoctorAdapter: List<DoctorResponseNew> = listOf()

    private val _DoctorListGenerated = MutableLiveData<Event<List<Value>>>()
    val doctorListGenerated: LiveData<Event<List<Value>>>
        get() = _DoctorListGenerated

    var memberInfoList = fspecialityDetailsRepository.getMemberFromDb().asLiveData()
    var memberInfoDb = fspecialityDetailsRepository.getMemberSid(
        preferences.getString("memberID", "").toString()
    ).asLiveData()

    final var accessToken = preferences.getString(Constants.ACCESS_TOKEN, "").toString()
    private var mutableSignup: MutableLiveData<SpecialityAll>? = null
    fun selectAll(post: String): LiveData<SpecialityAll> {
        mutableSignup = MutableLiveData()
        selectSpecialityAll(post)
        return mutableSignup!!
    }

    private val _logsListGenerated = MutableLiveData<Event<DoctorResponseNew>>()
    val logsListGenerated: LiveData<Event<DoctorResponseNew>>
        get() = _logsListGenerated
    var tempListFlat: ArrayList<Any> = arrayListOf()
    final var memberID = preferences.getString("memberID", "").toString()
    final var latitude = preferences.getString(Constants.LATITUDE, "").toString()
    final var longitude = preferences.getString(Constants.LONGITUDE, "").toString()
    final var accesstoken = preferences.getString(Constants.ACCESS_TOKEN, "").toString()
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }

    var data: Boolean = false
    fun getDoctorList(
        latitude: String,
        longitude: String,
        pageNo: String,
        pageSize: String,
        speciality: String,
        searchText: String,
        searchType: String,
        taxonomyGroup: String,
        locationType: String,
        providerType: String,
        lastDistance: String,
        codeType: String,
        codeValue: String
    ) {
        loading = true
        compositeDisposable.add(
            fspecialityDetailsRepository.getDoctorList(
                latitude, longitude, pageNo, pageSize, speciality,
                searchText, memberID, searchType, taxonomyGroup, locationType, providerType, lastDistance,
                codeType, codeValue
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->

                    // data =  it.doctorList.isNotEmpty()
//                    Value.isSelected = true
                    //   ValueList = it.values

                    eligibleValueList = ValueList
                    listForDoctorAdapter = eligibleValueList.toList()
                    _logsListGenerated.value = Event(it)

                    /*   eligibleValueList.clear()
                       eligibleValueList =  ValueList

                       listForDoctorAdapter = eligibleValueList.toList()



                       loading = false

                       _DoctorListGenerated.value = Event(listForDoctorAdapter)


   */
                }, {
                    loading = false
                })
        )
    }
    private fun selectSpecialityAll(post: String) {
        val call: Call<SpecialityAll> = RetrofitClient.getApi()!!.getAllAllSpeciality(post)
        call.enqueue(object : Callback<SpecialityAll> {
            override fun onResponse(
                call: Call<SpecialityAll>,
                response: Response<SpecialityAll>
            ) {
                try {
                    mutableSignup?.value = response.body()
                } catch (e: Exception) {
                }
            }

            override fun onFailure(call: Call<SpecialityAll>, t: Throwable) {
            }
        })
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
