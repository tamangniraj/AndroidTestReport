package com.vivekahealth.vivekahealth.ui.homePage.shareBenefit.shareBenefitDisclaimer

import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.lifecycle.ViewModelProvider
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.databinding.ShareBenefitDisclaimerFragmentBinding
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseDialogFragment
import com.vivekahealth.vivekahealth.ui.login.enrollmentPending.SuccessCallback
import com.vivekahealth.vivekahealth.utils.EventObserver
import javax.inject.Inject

class ShareBenefitDisclaimer : BaseDialogFragment(), Injectable {

    companion object {
        fun newInstance() = ShareBenefitDisclaimer()
    }

    private lateinit var viewModel: ShareBenefitDisclaimerViewModel
    private lateinit var binding: ShareBenefitDisclaimerFragmentBinding

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory

    var successCallbackFrag: SuccessCallback? = null

    fun setSuccessCallback(successCallback: SuccessCallback) {
        this.successCallbackFrag = successCallback
    }
//    override fun onCreate(savedInstanceState: Bundle?) {
//        super.onCreate(savedInstanceState)
//        setStyle(DialogFragment.STYLE_NORMAL, R.style.BlueButtonTheme)
//
//    }

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = ShareBenefitDisclaimerFragmentBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            ShareBenefitDisclaimerViewModel::class.java
        )
        binding.viewModel = viewModel

        binding.text2.text = getString(
            R.string.viveka_health_encrypts_all_your_personal_information_but_information_you_share_over_email_may_be_over_unencrypted_networks_or_to_insecure_mailboxes,
            getString(R.string.app_name)
        )

        viewModel.navigateToOK.observe(
            viewLifecycleOwner,
            EventObserver {
                successCallbackFrag?.onOkClick()

//            dialog?.dismiss()
            }
        )
    }
}
