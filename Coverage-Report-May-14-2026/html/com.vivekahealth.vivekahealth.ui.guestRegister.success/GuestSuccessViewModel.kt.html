package com.vivekahealth.vivekahealth.ui.guestRegister.success

import android.content.SharedPreferences
import android.os.Bundle
import android.text.Editable
import android.text.TextWatcher
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import com.vivekahealth.vivekahealth.data.remote.guest.GuestRegisterUser
import com.vivekahealth.vivekahealth.data.remote.login.AuthToken
import com.vivekahealth.vivekahealth.data.remote.login.ErrorResponse
import com.vivekahealth.vivekahealth.data.remote.login.GenericErrorResponse
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.guestRegister.repository.GuestRegisterRepository
import com.vivekahealth.vivekahealth.ui.login.repository.LoginRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.HelperUtils
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import java.util.regex.Pattern
import javax.inject.Inject

class GuestSuccessViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val guestRegisterRepository: GuestRegisterRepository,
    var loginRepository: LoginRepository
) :
    ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _passwordChanged = MutableLiveData<Event<String>>()
    val passwordChanged: LiveData<Event<String>>
        get() = _passwordChanged

    private val _confirmPasswordChanged = MutableLiveData<Event<String>>()
    val confirmPasswordChanged: LiveData<Event<String>>
        get() = _confirmPasswordChanged

    private val _navigateToBack = MutableLiveData<Event<Unit>>()
    val navigateToBack: LiveData<Event<Unit>>
        get() = _navigateToBack

    private val _registerSuccess = MutableLiveData<Event<DataResponse>>()
    val registerSuccess: LiveData<Event<DataResponse>>
        get() = _registerSuccess

    private val _loginSuccess = MutableLiveData<Event<AuthToken>>()
    val loginSuccess: LiveData<Event<AuthToken>>
        get() = _loginSuccess

    private val _loading = MutableLiveData<Event<Boolean>>()
    val loading: LiveData<Event<Boolean>>
        get() = _loading

    private val _error = MutableLiveData<Event<String>>()
    val error: LiveData<Event<String>>
        get() = _error

    val passwordError: Boolean = false

    var gson = Gson()

    private var phone: String? = null
    private var cCode: String? = null
    var registrationId: String? = null
    var email: String? = null
    private var password: String? = null
    private var confirmPassword: String? = null
    private var guestRegisterUser: GuestRegisterUser? = null

    @get:Bindable
    val passwordTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
            _passwordChanged.value = Event(s.toString())
            if (s.toString().isNotEmpty()) {
                password = s.toString()
                updateGuestRegisterUser()
            }
        }
    }

    @get:Bindable
    val confirmPasswordTextWatcher = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence, start: Int, count: Int, after: Int) {
            // Do nothing.
        }

        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
        }

        override fun afterTextChanged(s: Editable) {
            if (s.toString().isNotEmpty()) {
                confirmPassword = s.toString()
                _confirmPasswordChanged.value = Event(confirmPassword!!)
                updateGuestRegisterUser()
            }
        }
    }

    fun hasUpperAndLower(pass: String): Boolean {
        val upperLowerPattern = Pattern.compile("(?=.*[a-z])(?=.*[A-Z])")
        return upperLowerPattern.matcher(pass).find()
    }

    fun hasAtLeastOneNumber(pass: String): Boolean {
        val atLeastOneNumber = Pattern.compile("(?=.*\\d)")
        return atLeastOneNumber.matcher(pass).find()
    }

    fun hasOneSpecialCharacter(pass: String): Boolean {
        val atLeastOneSpecialChar = Pattern.compile("(?=.*\\W)")
        return atLeastOneSpecialChar.matcher(pass).find()
    }

    fun isAtLeastEightCharLong(pass: String): Boolean {
        val atLeastEightCharLong = Pattern.compile(".{8,}")
        return atLeastEightCharLong.matcher(pass).find()
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun onPrevClicked() {
        _navigateToBack.value = Event(Unit)
    }

    fun onNextClicked() {
    }

    fun handleArguments(arguments: Bundle?) {
        phone = arguments?.getString("phone")
        cCode = arguments?.getString("cCode")
        registrationId = arguments?.getString(("registrationId"))
        email = arguments?.getString("email")
        updateGuestRegisterUser()
    }

    private fun updateGuestRegisterUser() {
        guestRegisterUser =
            GuestRegisterUser(registrationId, email, phone, cCode, password, confirmPassword)
    }

    fun saveTokens(data: AuthToken) {
        preferences.edit().putString(Constants.ACCESS_TOKEN, data.access_token).apply()
        preferences.edit().putString(Constants.TOKEN_TYPE, data.token_type).apply()
        preferences.edit().putString(Constants.EXPIRES_IN, data.expires_in).apply()
        preferences.edit().putString(Constants.SCOPE, data.scope).apply()
        preferences.edit().putBoolean(Constants.IS_GUEST, data.isGuest).apply()
    }

    fun saveUserNamePassword() {
        guestRegisterUser?.let {
            val encodedEmail = HelperUtils.encrypt(it.email)
            preferences.edit().putString(Constants.encodedEmail, encodedEmail).apply()
            val encodedPassword = HelperUtils.encrypt(it.password)
            preferences.edit().putString(Constants.encodedPassword, encodedPassword).apply()
        }
    }

    fun resendEmail(email: String) {
        compositeDisposable.add(
            guestRegisterRepository.resendActivationCode(email)
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _loading.value = Event(true)
                }.subscribe({
                    if (it.isSuccessful) {
                        it.body()?.let {
                            _loading.value = Event(false)
                            _registerSuccess.value = Event(it)
                        }
                    } else {
                        val errorResponse: GenericErrorResponse = gson.fromJson(
                            it.errorBody()?.string(),
                            GenericErrorResponse::class.java
                        )
                        _error.value = Event(
                            errorResponse.getErrorMessage()
                        )
                        _loading.value = Event(false)
                    }
                }, {
                        onError ->

                    _error.value = Event(onError.message.toString())
                    _loading.value = Event(false)
                })
        )
    }

    fun silentLogin() {
        compositeDisposable.add(
            loginRepository.getLogin(
                guestRegisterUser?.email ?: "",
                guestRegisterUser?.password ?: "",
                "",
                false
            )
                .subscribeOn(Schedulers.io()).observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _loading.value = Event(true)
                }
                .subscribe({
                    if (it.isSuccessful) {
                        it.body()?.let {
                            saveTokens(it)

                            saveUserNamePassword()

                            _loginSuccess.value = Event(it)
                        }
                    } else {
                        val gson = Gson()
                        val type = object : TypeToken<ErrorResponse>() {}.type
                        val errorResponse: ErrorResponse? = gson.fromJson(
                            it.errorBody()!!.charStream(),
                            type
                        )
                        errorResponse?.let { err ->
                            _error.value = Event(err.error.toString())
                        }
                    }
                }, {
                })
        )
    }
}
