package com.vivekahealth.vivekahealth.ui.homePage.termsAndCondition.repository

import com.vivekahealth.vivekahealth.data.remote.guest.GuestClientDetails
import com.vivekahealth.vivekahealth.data.remote.home.ClientDetailResponse
import com.vivekahealth.vivekahealth.ui.homePage.termsAndCondition.service.TermsAndConditionService
import com.vivekahealth.vivekahealth.utils.thread.AppExecutors
import io.reactivex.Single
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class TermsAndConditionRepository @Inject constructor(
    private val appExecutors: AppExecutors,
    private val termsAndConditionService: TermsAndConditionService
) {
    fun getTermsAndCondition(): Single<ClientDetailResponse> = termsAndConditionService.getTermsAndConditionApi().getClientDetail()

    fun getTermsAndConditionGuest(): Single<GuestClientDetails> = termsAndConditionService.getTermsAndConditionApi().getGuestClientDetails()
}
