package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.diagnosis.filter

import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.lifecycle.ViewModelProvider
import com.vivekahealth.vivekahealth.data.local.home.DiagnosisFilterData
import com.vivekahealth.vivekahealth.data.local.home.DiagnosisRange
import com.vivekahealth.vivekahealth.databinding.DiagnosisFilterFragmentBinding
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseBottomSheetDialogFragment
import com.vivekahealth.vivekahealth.utils.EventObserver
import com.vivekahealth.vivekahealth.utils.HelperUtils
import javax.inject.Inject

class DiagnosisFilter : BaseBottomSheetDialogFragment(), Injectable {

    companion object {
        fun newInstance() = DiagnosisFilter()
        const val TAG = "DiagnosisFilterData"
    }

    var filterListener: DiagnosisFilterListener? = null

    var filterOption: DiagnosisFilterData? = null

    var initialRange: DiagnosisRange ? = null
    fun setCallback(
        filterListener: DiagnosisFilterListener,
        filterOption: DiagnosisFilterData?,
        range: DiagnosisRange
    ) {
        this.filterListener = filterListener
        this.filterOption = filterOption
        this.initialRange = range
    }
    private lateinit var binding: DiagnosisFilterFragmentBinding

    private lateinit var viewModel: DiagnosisFilterViewModel
    var sentfilter = false

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = DiagnosisFilterFragmentBinding.inflate(inflater, container, false)

        return binding.root
    }
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
    }

    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            DiagnosisFilterViewModel::class.java
        )

        binding.viewModel = viewModel

        viewModel.minRange = filterOption!!.min
        viewModel.maxRange = filterOption!!.max
        viewModel.rangeMin = HelperUtils.floor(initialRange!!.min.toDouble())
        viewModel.rangeMax = HelperUtils.floor(initialRange!!.max.toDouble())

        val list: MutableList<Float> = emptyArray<Float>().toMutableList()
        list.add(viewModel.minRange)
        list.add(viewModel.maxRange)
        viewModel.rangeValue = list
        binding.rangeSlider.values = list

        binding.rangeSlider.valueFrom = initialRange!!.min
        binding.rangeSlider.valueTo = initialRange!!.max

        binding.rangeSlider.setValues(viewModel.rangeValue[0], viewModel.rangeValue[1])

        viewModel.range = HelperUtils.floor(viewModel.rangeValue[0].toDouble()) + "-" + HelperUtils.floor(
            viewModel.rangeValue[1].toDouble()
        )

        viewModel.navigateToClear.observe(
            viewLifecycleOwner,
            EventObserver {
                viewModel.minRange = initialRange!!.min
                viewModel.maxRange = initialRange!!.max
                binding.rangeSlider.valueFrom = initialRange!!.min
                binding.rangeSlider.valueTo = initialRange!!.max
                val list: MutableList<Float> = emptyArray<Float>().toMutableList()
                list.add(viewModel.minRange)
                list.add(viewModel.maxRange)
                binding.rangeSlider.values = list
                viewModel.range = HelperUtils.floor(viewModel.minRange.toDouble()) + "-" + HelperUtils.floor(
                    viewModel.maxRange.toDouble()
                )

                binding.rangeSlider.setValues(viewModel.minRange, viewModel.maxRange)

//
            }
        )

        viewModel.navigateToValues.observe(
            viewLifecycleOwner,
            EventObserver {
                viewModel.range = HelperUtils.floor(it[0].toDouble()).toString() + "-" + HelperUtils.floor(
                    it[1].toDouble()
                ).toString()

                viewModel.rangeValue = it
                viewModel.minRange = it[0]
                viewModel.maxRange = it[1]
            }
        )
        viewModel.navigateToPres.observe(
            viewLifecycleOwner,
            EventObserver {
                filterListener?.onClose()
            }
        )
        viewModel.navigateToDocFilter.observe(
            viewLifecycleOwner,
            EventObserver {
                filterListener?.takeIf { viewModel.rangeValue.size >= 2 }
                    ?.apply(DiagnosisFilterData(viewModel.rangeValue[0], viewModel.rangeValue[1]))
            }
        )
    }
}
