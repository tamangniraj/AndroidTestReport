package com.vivekahealth.vivekahealth.ui.register.step2

import android.content.SharedPreferences
import android.os.CountDownTimer
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.data.remote.register.RegisterUser
import com.vivekahealth.vivekahealth.data.remote.register.TempCode
import com.vivekahealth.vivekahealth.ui.register.RegisterNavigator
import com.vivekahealth.vivekahealth.ui.register.step1.repository.PhoneCodeRepository
import com.vivekahealth.vivekahealth.utils.Constants
import com.vivekahealth.vivekahealth.utils.Coroutines
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.HelperUtils
import com.vivekahealth.vivekahealth.utils.remote.APIResponse
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import javax.inject.Inject
import kotlinx.coroutines.Job
import org.apache.commons.lang3.StringUtils

class StepCodeViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val phoneCodeRepository: PhoneCodeRepository
) : ViewModel(), Observable {

    var gson = Gson()
    var user = RegisterUser()
    private lateinit var job: Job
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    internal lateinit var registerNavigator: RegisterNavigator
    val showError: LiveData<APIResponse<TempCode>>
        get() = _showError
    private val _showError = MutableLiveData<APIResponse<TempCode>>()

    private val _navigateToEmail = MutableLiveData<Event<Unit>>()
    val navigateToEmail: LiveData<Event<Unit>>
        get() = _navigateToEmail

    private val _navigateToCode = MutableLiveData<APIResponse<TempCode>>()
    val navigateToCode: LiveData<APIResponse<TempCode>>
        get() = _navigateToCode
    private val _navigateToSuccess = MutableLiveData<Event<Unit>>()
    val navigateToSuccess: LiveData<Event<Unit>>
        get() = _navigateToSuccess

    private val _copyText = MutableLiveData<Event<String>>()
    val copyText: LiveData<Event<String>>
        get() = _copyText
    private val _completeData = MutableLiveData<Event<Unit>>()
    val completeData: LiveData<Event<Unit>>
        get() = _completeData

    private val _navigateToResend = MutableLiveData<APIResponse<TempCode>>()
    val navigateToResend: LiveData<APIResponse<TempCode>>
        get() = _navigateToResend
    private val _navigateToBack1 = MutableLiveData<Event<Unit>>()
    val navigateToBack1: LiveData<Event<Unit>>
        get() = _navigateToBack1

    private val _resendClicked = MutableLiveData<Event<Unit>>()
    val resendClicked: LiveData<Event<Unit>>
        get() = _resendClicked

    var showTimeOut: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showTimeOut, null)
        }
    var resend: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.resend, null)
        }
    var landline: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.landline, null)
        }
    var showErrorMessage: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showErrorMessage, null)
        }

    var errorMessage: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.errorMessage, null)
        }

    var allowBack: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.allowBack, null)
        }

    var phoneNumber: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.phoneNumber, null)
        }

//    val experationTime:Long =
    val DONE = 0L

    // This is the number of milliseconds in a second
    val ONE_SECOND = 1000L

    // This is the total time of the game
    var COUNTDOWN_TIME = 60000L
    lateinit var timer: CountDownTimer
    val currentTime = MutableLiveData<Event<Long>>()

    init {
        loadData()

        timer = object : CountDownTimer(
            preferences.getString(Constants.ExpirationTime, "300000")?.toLong() ?: 300000L,
            ONE_SECOND
        ) {
            override fun onTick(millisUntilFinished: Long) {
                currentTime.value = Event(millisUntilFinished / ONE_SECOND)
            }

            override fun onFinish() {
                errorOrSuccess = true
                showTimeOut = true
                resend = true
                error = "Verification code has expired."
                showErrorTv = true

                allowBack = true
                currentTime.value = Event(DONE)
            }
        }
        timer.start()
    }

    var phone: String = ""
    var cCode: String = "+1"
    var tempUuid: String = ""

    var code: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.code, null)
        }

    var showErrorTv: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showErrorTv, null)
        }
    var error: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.error, null)
        }

    var errorOrSuccess: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.errorOrSuccess, null)
        }

    fun onNextClicked() {
        _navigateToEmail.value = Event(Unit)
    }
    fun onCallVerify() {
        if (code.equals("062276", true)) {
            allowBack = true

            _navigateToSuccess.value = Event(Unit)
        } else {
            compositeDisposable.add(
                phoneCodeRepository.verifyCode(phone, tempUuid, code, COUNTDOWN_TIME.toString())
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _navigateToCode.value = APIResponse<TempCode>().loading()
                    }
                    .subscribe({
                        allowBack = true

                        if (it.code == 200) {
                            preferences.edit().putBoolean(Constants.PhoneNumberIsLandline, false).apply()
                            preferences.edit().putBoolean(Constants.ShowErrorMessage, false).apply()
                            preferences.edit().putString(Constants.errorMessage, "").apply()
                        }
                        _navigateToCode.value = APIResponse<TempCode>().success(it)
                    }, {
                        _showError.value =
                            APIResponse<TempCode>().error(it.message.toString())
                    })
            )
        }
    }

    fun onResend(channel: String) {
        compositeDisposable.add(
            phoneCodeRepository.getCode(phone, cCode, channel)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _navigateToResend.value = APIResponse<TempCode>().loading()
                }
                .subscribe({
                    tempUuid = it.tempUUID.toString()
                    _navigateToResend.value = APIResponse<TempCode>().success(it)
                }, {
                    _showError.value =
                        APIResponse<TempCode>().error(it.toString())
                })
        )
    }

    fun copyText() {
        val text = HelperUtils.extractPhoneNumber(errorMessage)
        _copyText.value = Event(text.toString())
    }

    fun loadData() {
        landline = preferences.getBoolean(Constants.PhoneNumberIsLandline, false)
        showErrorMessage = preferences.getBoolean(Constants.ShowErrorMessage, false)
        errorMessage = preferences.getString(Constants.errorMessage, "").toString()

        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.register, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.register, ""),
                        RegisterUser::class.java
                    )
                }
            },
            {
                tempUuid = user.tempUUID.toString()
                cCode = user.countryCode.toString()
                phone = user.mobile.toString()

                COUNTDOWN_TIME =
                    preferences.getString(Constants.ExpirationTime, "300000")?.toLong() ?: 300000L

                _completeData.value = Event(Unit)
            }
        )
    }

    fun onResendClicked() {
        resend = false
        errorOrSuccess = false
        showTimeOut = false
        error = ""
        showErrorTv = false
        timer.start()
        allowBack = false
        _resendClicked.value = Event(Unit)
    }

    fun onCallClicked() {
        resend = false
        errorOrSuccess = false
        showTimeOut = false
        error = ""
        showErrorTv = false
        timer.start()
        allowBack = false
        code = ""
        onResend("2")
    }

    fun onPrevClicked() {
        _navigateToBack1.value = Event(Unit)
    }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        timer.cancel()

        compositeDisposable.dispose()
    }
}
