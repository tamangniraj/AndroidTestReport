    fun getBillPreview(
        memberId: String,
        npi: String,
        ein: String,
        deliveryType: Int,
        networkType: Int,
        feeSchedule: List<FeeSchedule>
    ) {
        val body = AdvancedEobQuickViewRequest(npi, ein, deliveryType, networkType, feeSchedule)
        compositeDisposable.add(
            billPreviewRepository.getBillPreview(memberId, body)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe { loading = true }
                .subscribe({
                    loading = false
                    advancedEobQuickViewResponse = it
                    memberResponsibleAmount = advancedEobQuickViewResponse.memberResponsibleAmount
                    planResponsibleAmount = advancedEobQuickViewResponse.planResponsibleAmount

                    data = getData(advancedEobQuickViewResponse)

                    allowedAmount = advancedEobQuickViewResponse.allowedAmount

                    coPay = advancedEobQuickViewResponse.copayAmount
                    hasCopay = coPay > 0.0
                    coInsurance = advancedEobQuickViewResponse.coinsuranceAmount
                    deductible = advancedEobQuickViewResponse.deductible
                    calculateValues()
                    it.additionalDisclaimers?.let {
                        _additionalisclaimer.postValue(it)
                    }
                }, {
                    loading = false
                })
        )
    }

    /**
     * Create the data for the donut chart.
     */
    private fun getData(response: AdvancedEobQuickViewResponse): PieData {
        val colors = listOf(
            Color.parseColor("#293F7C"),
            Color.parseColor("#6D7CA7")
        )

        val data = mapOf(
            "A" to response.planResponsibleAmount,
            "B" to response.memberResponsibleAmount
        )

        val entries = mutableListOf<PieEntry>()

        data.forEach { (_, value) ->
            entries += PieEntry(value.toFloat())
        }

        val pieDataset = PieDataSet(entries, "Data")
        pieDataset.colors = colors
        pieDataset.setDrawValues(false)
        return PieData(pieDataset)
    }

    /**
     * Calculate values and text shown in the Individual and Family sections.
     */
    fun calculateValues() {
        val deductibleRatios = getDeductibleRatios(tabPosition)
        val oopRatios = getOopRatios(tabPosition)

        deductibleRatio = deductibleRatios.first
        deductibleAfterRatio = deductibleRatios.second
        inverseDeductibleRatio =
            1 - deductibleRatios.first - deductibleRatios.second

        oopRatio = oopRatios.first
        oopAfterRatio = oopRatios.second
        inverseOopRatio = 1 - oopRatios.first - oopRatios.second

        val maxAmounts = getMaxAmounts(tabPosition)

        deductibleMax = maxAmounts.first
        oopMax = maxAmounts.second

        val currentAmounts = getCurrentAmounts(tabPosition)
        currentDeductible = currentAmounts.first
        currentOop = currentAmounts.second

        val projectedAmounts = getProjectedAmounts(tabPosition)
        projectedDeductible = projectedAmounts.first
        projectedOop = projectedAmounts.second

        val remainingAmounts = getRemainingAmounts(tabPosition)
        deductibleRemaining = remainingAmounts.first
        oopRemaining = remainingAmounts.second

        val metValues = getMetValues(tabPosition)
        deductibleMet = metValues.first
        oopMaxMet = metValues.second

        val willBeMetValues = getWillBeMetValues(tabPosition)
        deductibleWillBeMet = willBeMetValues.first
        oopMaxWillBeMet = willBeMetValues.second

        val texts = getInfoTexts(tabPosition)
        deductibleInfo = texts.first
        oopInfo = texts.second

        val hasDeductibleAndOopValues = getHasDeductibleAndOopValues(tabPosition)
        hasDeductible = hasDeductibleAndOopValues.first
        hasOopMax = hasDeductibleAndOopValues.second
    }

    private fun getDeductibleRatios(selectedTab: Int): Pair<Float, Float> {
        val accumulatorDetails = if (selectedTab == 0) {
            advancedEobQuickViewResponse.individualAccumulatorDetails
        } else {
            advancedEobQuickViewResponse.familyAccumulatorDetails
        }
        val deductibleRatio =
            accumulatorDetails.currentDeductible / accumulatorDetails.planDeductibleForYear
        val deductibleAfterRatio =
            (accumulatorDetails.projectedDeductible - accumulatorDetails.currentDeductible) / accumulatorDetails.planDeductibleForYear

        return deductibleRatio.toFloat() to deductibleAfterRatio.toFloat()
    }

    private fun getOopRatios(selectedTab: Int): Pair<Float, Float> {
        val accumulatorDetails = if (selectedTab == 0) {
            advancedEobQuickViewResponse.individualAccumulatorDetails
        } else {
            advancedEobQuickViewResponse.familyAccumulatorDetails
        }
        val oopRatio = accumulatorDetails.currentOop / accumulatorDetails.planOutOfPocketForYear
        val oopAfterRatio =
            (accumulatorDetails.projectedOop - accumulatorDetails.currentOop) / accumulatorDetails.planOutOfPocketForYear

        return oopRatio.toFloat() to oopAfterRatio.toFloat()
    }

    private fun getInfoTexts(selectedTab: Int): Pair<String, String> {
        val accumulatorDetails = if (selectedTab == 0) {
            advancedEobQuickViewResponse.individualAccumulatorDetails
        } else {
            advancedEobQuickViewResponse.familyAccumulatorDetails
        }

        var deductibleInfo = if (accumulatorDetails.planDeductibleForYear == 0.0) {
            resources.getString(R.string.no_deductible)
        } else {
            ""
        }

        var oopInfo = if (accumulatorDetails.planOutOfPocketForYear == 0.0) {
            resources.getString(R.string.no_oop_max)
        } else {
            ""
        }

        if (accumulatorDetails.deductibleMet && accumulatorDetails.planDeductibleForYear > 0L) {
            deductibleInfo = resources.getString(R.string.deductible_met)
        }
        if (accumulatorDetails.oopMaxMet && accumulatorDetails.planOutOfPocketForYear > 0L) {
            oopInfo = resources.getString(R.string.oop_met)
        }

        if (accumulatorDetails.deductibleWillBeMet && !accumulatorDetails.deductibleMet) {
            deductibleInfo = resources.getString(R.string.deductible_will_be_met)
        }
        if (accumulatorDetails.oopMaxWillBeMet && !accumulatorDetails.oopMaxMet) {
            oopInfo = resources.getString(R.string.oop_will_be_met)
        }
        return deductibleInfo to oopInfo
    }

    private fun getHasDeductibleAndOopValues(selectedTab: Int): Pair<Boolean, Boolean> {
        val accumulatorDetails = if (selectedTab == 0) {
            advancedEobQuickViewResponse.individualAccumulatorDetails
        } else {
            advancedEobQuickViewResponse.familyAccumulatorDetails
        }
        val hasDeductible =
            accumulatorDetails.planDeductibleForYear > 0L && !accumulatorDetails.deductibleMet
        val hasOop = accumulatorDetails.planOutOfPocketForYear > 0L && !accumulatorDetails.oopMaxMet
        return hasDeductible to hasOop
    }

    /**
     * Calculate remaining deductible and OOP amounts.
     */
    private fun getRemainingAmounts(selectedTab: Int): Pair<Double, Double> {
        val accumulatorDetails = if (selectedTab == 0) {
            advancedEobQuickViewResponse.individualAccumulatorDetails
        } else {
            advancedEobQuickViewResponse.familyAccumulatorDetails
        }
        val deductibleRemaining =
            accumulatorDetails.planDeductibleForYear - accumulatorDetails.projectedDeductible
        val oopRemaining =
            accumulatorDetails.planOutOfPocketForYear - accumulatorDetails.projectedOop
        return deductibleRemaining to oopRemaining
    }
