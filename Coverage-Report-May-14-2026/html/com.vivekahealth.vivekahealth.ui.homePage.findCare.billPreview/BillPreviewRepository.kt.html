package com.vivekahealth.vivekahealth.ui.homePage.findCare.billPreview

import androidx.lifecycle.LiveData
import androidx.lifecycle.asLiveData
import com.vivekahealth.vivekahealth.data.local.home.MemberInfoDao
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.data.remote.home.findCare.billPreview.AdvancedEobQuickViewRequest
import com.vivekahealth.vivekahealth.data.remote.home.findCare.billPreview.AdvancedEobQuickViewResponse
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityprofile.service.SpecialityProfileService
import io.reactivex.Single
import javax.inject.Inject

class BillPreviewRepository @Inject constructor(
    private val billPreviewService: SpecialityProfileService,
    private val memberInfoDao: MemberInfoDao
) {
    fun getBillPreview(
        memberId: String,
        body: AdvancedEobQuickViewRequest
    ): Single<AdvancedEobQuickViewResponse> =
        billPreviewService.getBillPreviewApi().getAdvancedEobQuickView(memberId, body)

    fun getMemberInfo(memberId: String): LiveData<MemberInfo> {
        return memberInfoDao.getMemberInfoLiveData(memberId).asLiveData()
    }
}
