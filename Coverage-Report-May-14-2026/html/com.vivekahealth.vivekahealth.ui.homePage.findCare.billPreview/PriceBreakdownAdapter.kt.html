package com.vivekahealth.vivekahealth.ui.homePage.findCare.billPreview

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView.ViewHolder
import com.vivekahealth.vivekahealth.databinding.SingleRowProcedureSelectedBinding
import com.vivekahealth.vivekahealth.data.remote.home.findCare.doctorModelsNew.FeeSchedule

class PriceBreakdownAdapter :
    ListAdapter<FeeSchedule, PriceBreakdownAdapter.PriceBreakdownViewHolder>(
        PriceBreakdownDiffCallback()
    ) {
    class PriceBreakdownViewHolder(var binding: SingleRowProcedureSelectedBinding) :
        ViewHolder(binding.root) {
        fun bind(item: FeeSchedule) {
            binding.item = item
        }
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): PriceBreakdownViewHolder {
        val binding = SingleRowProcedureSelectedBinding.inflate(
            LayoutInflater.from(parent.context),
            parent,
            false
        )
        return PriceBreakdownViewHolder(binding)
    }

    override fun onBindViewHolder(holder: PriceBreakdownViewHolder, position: Int) {
        val item = getItem(position)
        holder.bind(item)
    }
}

private class PriceBreakdownDiffCallback : DiffUtil.ItemCallback<FeeSchedule>() {
    override fun areItemsTheSame(oldItem: FeeSchedule, newItem: FeeSchedule): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(oldItem: FeeSchedule, newItem: FeeSchedule): Boolean {
        return oldItem == newItem
    }
}
