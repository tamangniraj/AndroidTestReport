    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.Registration, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.Registration, ""),
                        RegisterUser::class.java
                    )
                }
            },
            {
                cCode = user.countryCode.toString()
                if (cCode.isEmpty()) {
                    cCode = "+1"
                }
                phoneNumber = user.mobile.toString()
                _phoneNumberPrev.value = Event(user.mobile.toString())
            }
        )
    }

    fun savePhone() {
        user.mobile = phoneNumber
        user.countryCode = cCode

        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.Registration, regString)
        regEdit.apply()
    }

            if (!s.toString().matches(regex)) {
                var unformated = s.toString().replace("(", "").replace(")", "").replace("-", "").replace(

                if (unformated.length > 10) {
                    if (!s.toString().matches(regex)) {
                        var st = unformated.substring(unformated.length - 10, unformated.length)
                        var final = st.replace("(\\d{3})(\\d{3})(\\d{4})".toRegex(), "$1 $2 $3")
//                        _phoneNumberPrev.value = Event(final)
                    }
                } else {
                }
            }
        }
        override fun onTextChanged(s: CharSequence, start: Int, before: Int, count: Int) {
            // Filter out non-digits and spaces
            val filteredText = s.toString().filter { it.isDigit() || it == ' ' }

            var regex = Regex("^[0-9]{3} [0-9]{3} [0-9]{4}$")

            if (!filteredText.matches(regex)) {
                var unformated = filteredText.replace("(", "").replace(")", "").replace("-", "").replace(
                    " ",
                    ""
                )
                if (unformated.length > 10) {
                    if (!filteredText.matches(regex)) {
                        var st = unformated.substring(unformated.length - 10, unformated.length)
                        var final = st.replace("(\\d{3})(\\d{3})(\\d{4})".toRegex(), "$1 $2 $3")
                        modifiedText = final
                        _phoneNumberPrev.value = Event(final)
                        _phoneNumberChanged.value = Event(final)
                    }
                } else {
                    modifiedText = ""
                    _phoneNumberChanged.value = Event(filteredText)
                }
            } else {
                if (modifiedText.isEmpty()) {
                    _phoneNumberChanged.value = Event(filteredText)
                }
            }
        }

    fun onNextClicked() {
//        _navigateToCode.value = Event(Unit)

        Timber.d(phoneNumber)
        compositeDisposable.add(
            phoneCodeRepository.getCode(phoneNumber, cCode, "")
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _navigateToCode.value = APIResponse<TempCode>().loading()
                }
                .subscribe({
                    user.tempUUID = it.tempUUID

//                    val gson = Gson()
//                    val regString = gson.toJson(user)
                    val regEdit = preferences.edit()
                    regEdit.putString(Constants.ExpirationTime, it.tokenExpirationTime)
                    regEdit.putBoolean(Constants.PhoneNumberIsLandline, it.phoneNumberIsLandline)
                    regEdit.putBoolean(Constants.ShowErrorMessage, it.showErrorMessage)
                    regEdit.putString(Constants.errorMessage, it.message.toString())
                    regEdit.apply()
                    savePhone()
                    _navigateToCode.value = APIResponse<TempCode>().success(it)
                }, {
                    _showError.value =
                        APIResponse<TempCode>().error(it.message.toString())
                })
        )
    }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
