package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.diagnosis.adapter

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.data.remote.home.DiagnosisHistoryQuickView
import com.vivekahealth.vivekahealth.databinding.DiagnosisDetailsItemBinding

class DiagnosisDetailsListAdapter(private var onClick: (DiagnosisHistoryQuickView) -> Unit) : ListAdapter<DiagnosisHistoryQuickView, RecyclerView.ViewHolder>(
    DiagnosisDetailDiffCallback()
) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        return MemberViewHolder(
            DiagnosisDetailsItemBinding.inflate(
                LayoutInflater.from(parent.context),
                parent,
                false
            ),
            onClick
        )
    }
    override fun getItemId(position: Int): Long {
        return position.toLong()
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val item = getItem(position)
        (holder as MemberViewHolder).binding.viewModel = item
    }
    class MemberViewHolder(
        val binding: DiagnosisDetailsItemBinding,
        private val onClick: (DiagnosisHistoryQuickView) -> Unit
    ) : RecyclerView.ViewHolder(binding.root)
}

private class DiagnosisDetailDiffCallback : DiffUtil.ItemCallback<DiagnosisHistoryQuickView>() {

    override fun areItemsTheSame(
        oldItem: DiagnosisHistoryQuickView,
        newItem: DiagnosisHistoryQuickView
    ): Boolean {
        return oldItem == newItem
    }

    override fun areContentsTheSame(
        oldItem: DiagnosisHistoryQuickView,
        newItem: DiagnosisHistoryQuickView
    ): Boolean {
        return oldItem.icd10Code.code == newItem.icd10Code.code
    }
}
