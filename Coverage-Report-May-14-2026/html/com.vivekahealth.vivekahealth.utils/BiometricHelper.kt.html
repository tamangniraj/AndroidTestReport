package com.vivekahealth.vivekahealth.utils

import android.content.Context
import android.content.pm.PackageManager
import androidx.biometric.BiometricManager
import com.vivekahealth.vivekahealth.R

object BiometricHelper {

    fun getStatus(context: Context): String {
        var isTouchIdAvailable = false
        var isFaceIdAvaialable = false
        var message = ""

        val biometricManager = BiometricManager.from(context)
        when (biometricManager.canAuthenticate(BiometricManager.Authenticators.BIOMETRIC_WEAK)) {
            BiometricManager.BIOMETRIC_SUCCESS -> {
                val packageManager = context.packageManager

                if (packageManager.hasSystemFeature(PackageManager.FEATURE_FINGERPRINT)) {
                    isTouchIdAvailable = true
                }
                if (packageManager.hasSystemFeature(PackageManager.FEATURE_FACE) &&
                    packageManager.hasSystemFeature(PackageManager.FEATURE_CAMERA_FRONT)
                ) {
                    isFaceIdAvaialable = true
                }
            }

            BiometricManager.BIOMETRIC_ERROR_NO_HARDWARE -> {}

            BiometricManager.BIOMETRIC_ERROR_HW_UNAVAILABLE -> {}

            BiometricManager.BIOMETRIC_ERROR_NONE_ENROLLED -> {}
        }

        if (isTouchIdAvailable && isFaceIdAvaialable) {
            message = context.getString(R.string.touch_id_face_id_text)
        } else if (isTouchIdAvailable) {
            message = context.getString(R.string.touch_id_text)
        } else if (isFaceIdAvaialable) {
            message = context.getString(R.string.face_id_text)
        } else {
            message = context.getString(R.string.touch_id_text)
        }

        return message
    }
}
