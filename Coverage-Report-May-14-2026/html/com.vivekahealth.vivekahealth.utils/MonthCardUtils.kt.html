package com.vivekahealth.vivekahealth.utils

object MonthCardUtils {

    /**
     * Returns true if the given [year]/[month] (as Strings) represents a month
     * that is strictly in the future relative to [currentYear]/[currentMonth].
     * Returns false if parsing fails.
     */
    fun isFutureMonthCard(
        year: String,
        month: String,
        currentYear: Int,
        currentMonth: Int
    ): Boolean {
        val itemYear = year.toIntOrNull() ?: return false
        val itemMonth = month.toIntOrNull() ?: return false
        return itemYear > currentYear || (itemYear == currentYear && itemMonth > currentMonth)
    }

    fun shouldLoadEligibility(addedItem: Boolean, isFutureMonth: Boolean): Boolean {
        return !(addedItem && isFutureMonth)
    }
}
