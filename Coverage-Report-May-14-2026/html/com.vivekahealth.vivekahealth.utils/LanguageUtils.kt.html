package com.vivekahealth.vivekahealth.utils

import android.app.Activity
import android.content.Context
import android.content.Intent
import android.content.res.Configuration
import android.os.Build
import java.util.*

object LanguageUtils {

    const val EXTRA_LANGUAGE_CODE = "extra_language_code"

    /**
     * Sets the locale for an activity based on the provided language code
     */
    fun setActivityLocale(activity: Activity, languageCode: String?) {
        if (languageCode.isNullOrEmpty()) return

        val locale = Locale(languageCode)
        Locale.setDefault(locale)

        val configuration = Configuration(activity.resources.configuration)
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N) {
            configuration.setLocale(locale)
        } else {
            @Suppress("DEPRECATION")
            configuration.locale = locale
        }

        activity.resources.updateConfiguration(configuration, activity.resources.displayMetrics)
    }

    /**
     * Creates an Intent with language code for launching activities
     */
    fun createIntentWithLanguage(context: Context, targetActivity: Class<*>, languageCode: String): Intent {
        return Intent(context, targetActivity).apply {
            putExtra(EXTRA_LANGUAGE_CODE, languageCode)
        }
    }

    /**
     * Gets the current device language code
     */
    fun getCurrentLanguageCode(context: Context): String {
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N) {
            context.resources.configuration.locales[0].language
        } else {
            @Suppress("DEPRECATION")
            context.resources.configuration.locale.language
        }
    }

    /**
     * Checks if activity should apply language change
     */
    fun shouldApplyLanguageChange(activity: Activity, languageCode: String?): Boolean {
        if (languageCode.isNullOrEmpty()) return false

        val currentLanguage = getCurrentLanguageCode(activity)
        return !currentLanguage.equals(languageCode, ignoreCase = true)
    }

    /**
     * Updates activity configuration with new language and recreates if needed
     */
    fun updateActivityLanguage(activity: Activity, languageCode: String?) {
        if (shouldApplyLanguageChange(activity, languageCode)) {
            setActivityLocale(activity, languageCode)
        }
    }
}
