package com.vivekahealth.vivekahealth.utils

import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo

object AddressChangedValidator {

    fun isValidZip(zipCode: String?): Boolean {
        val zipRegex = """^\d{5}(-\d{4})?$""".toRegex()
        return zipRegex.matches(zipCode ?: "")
    }

    private fun isRequiredFieldEmpty(
        addressChanged: String?,
        cityChanged: String?,
        state: String?,
        zipChanged: String?
    ): Boolean {
        return addressChanged.isNullOrEmpty() || cityChanged.isNullOrEmpty() || state.isNullOrEmpty() || zipChanged.isNullOrEmpty()
    }

    private fun isSameAsPrevious(
        memberInfo: MemberInfo?,
        apartmentChanged: String?,
        addressChanged: String?,
        cityChanged: String?,
        state: String?,
        zipChanged: String?
    ): Boolean {
        return (memberInfo?.homeAddress?.street1 ?: "") == (addressChanged ?: "") &&
            (memberInfo?.homeAddress?.street2 ?: "") == (apartmentChanged ?: "") &&
            (memberInfo?.homeAddress?.city ?: "") == (cityChanged ?: "") &&
            (memberInfo?.homeAddress?.state ?: "") == (state ?: "") &&
            memberInfo?.homeAddress?.zip == zipChanged
    }

    fun isAddressReady(
        memberInfo: MemberInfo?,
        apartmentChanged: String?,
        addressChanged: String?,
        cityChanged: String?,
        state: String?,
        zipChanged: String?
    ): Boolean {
        val notEmpty = !isRequiredFieldEmpty(addressChanged, cityChanged, state, zipChanged)
        val validZip = isValidZip(zipChanged)
        val hasChanged = !isSameAsPrevious(
            memberInfo,
            apartmentChanged,
            addressChanged,
            cityChanged,
            state,
            zipChanged
        )
        return notEmpty && validZip && hasChanged
    }

    fun isAddressReadyGuest(
        addressChanged: String?,
        cityChanged: String?,
        state: String?,
        zipChanged: String?
    ): Boolean {
        val notEmpty = !isRequiredFieldEmpty(addressChanged, cityChanged, state, zipChanged)
        val validZip = isValidZip(zipChanged)

        return notEmpty && validZip
    }
}
