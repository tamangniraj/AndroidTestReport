package com.vivekahealth.vivekahealth.utils

import android.Manifest
import android.app.Activity
import android.content.ContentValues
import android.os.Build
import android.telephony.PhoneNumberUtils
import android.util.Log
import androidx.core.app.ActivityCompat
import permissions.dispatcher.PermissionUtils.shouldShowRequestPermissionRationale
import java.util.Locale
import timber.log.Timber

object PhoneUtils {
    /**
     * Formats a phone number based on the users locale.
     */
    fun formatPhone(number: String): String {
        return try {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
                PhoneNumberUtils.formatNumber(number, Locale.getDefault().country)
            } else {
                @Suppress("DEPRECATION")
                PhoneNumberUtils.formatNumber(number)
            }
        } catch (e: IllegalStateException) {
            Timber.d("Unable to format phone number: $number")
            number
        }
    }

    fun requestCameraPermission(context: Activity) {
        when {
            context.shouldShowRequestPermissionRationale(Manifest.permission.CAMERA) -> {
                // This case means user previously denied the permission
                // So here we can display an explanation to the user
                // That why exactly we need this permission
                Log.d(ContentValues.TAG, "requestCameraPermission - Camera Permission NOT Granted")
                ActivityCompat.requestPermissions(
                    (context as Activity?)!!,
                    arrayOf(Manifest.permission.CAMERA),
                    Constants.MY_PERMISSIONS_REQUEST_CAMERA
                )
            }
            else -> {
                // Everything is fine you can simply request the permission
                ActivityCompat.requestPermissions(
                    (context as Activity?)!!,
                    arrayOf(Manifest.permission.CAMERA),
                    Constants.MY_PERMISSIONS_REQUEST_CAMERA
                )
            }
        }
    }
}
