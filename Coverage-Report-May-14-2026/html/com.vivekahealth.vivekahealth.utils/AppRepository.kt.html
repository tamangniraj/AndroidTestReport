package com.vivekahealth.vivekahealth.utils

import NetworkStateReceiver
import android.content.Context
import android.os.Build

class AppRepository {

    private val networkStateReceivers = mutableListOf<NetworkStateReceiver>()
    fun registerNetworkStateListener(
        listener: (String, Boolean) -> Unit,
        identifier: String,
        context: Context
    ) {
        val networkStateReceiver = NetworkStateReceiver(listener, identifier)
        networkStateReceivers.add(networkStateReceiver)

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
            networkStateReceiver.register(context)
        }
    }

    fun unregisterNetworkStateListener(
        identifier: String,
        context: Context
    ) {
        val receiverToRemove = networkStateReceivers.find { it.identifier == identifier }
        receiverToRemove?.let {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
                it.unregister(context)
            }
            networkStateReceivers.remove(it)
        }
    }
}
