package com.vivekahealth.vivekahealth.utils

import android.content.Context
import android.graphics.Canvas
import android.graphics.drawable.Drawable
import android.opengl.ETC1.getHeight
import android.opengl.ETC1.getWidth
import android.util.AttributeSet
import com.google.android.material.button.MaterialButton

class DrawableAlignedButton : MaterialButton {
    constructor(context: Context?, attrs: AttributeSet?) : super(context!!, attrs) {}
    constructor(context: Context?) : super(context!!) {}
    constructor(context: Context?, attrs: AttributeSet?, style: Int) : super(
        context!!,
        attrs,
        style
    ) {}

    private var mLeftDrawable: Drawable? = null

    // Overriden to work only with a left drawable.
    override fun setCompoundDrawablesWithIntrinsicBounds(
        left: Drawable?,
        top: Drawable?,
        right: Drawable?,
        bottom: Drawable?
    ) {
        if (left == null) return
        left.setBounds(0, 0, left.intrinsicWidth, left.intrinsicHeight)
        mLeftDrawable = left
    }

    protected override fun onDraw(canvas: Canvas) {
        // transform the canvas so we can draw both image and text at center.
        canvas.save()
        canvas.translate((2 + mLeftDrawable!!.intrinsicWidth / 2).toFloat(), 0F)
        super.onDraw(canvas)
        canvas.restore()
        canvas.save()
        val widthOfText = getPaint().measureText(getText().toString()) as Int
        val left = (getWidth() + widthOfText) / 2 - mLeftDrawable!!.intrinsicWidth - 2
        canvas.translate(
            left.toFloat(),
            ((getHeight() - mLeftDrawable!!.intrinsicHeight) / 2).toFloat()
        )
        mLeftDrawable!!.draw(canvas)
        canvas.restore()
    }

    protected override fun onMeasure(widthMeasureSpec: Int, heightMeasureSpec: Int) {
        super.onMeasure(widthMeasureSpec, heightMeasureSpec)
        var height: Int = getMeasuredHeight()
        height = Math.max(
            height,
            mLeftDrawable!!.intrinsicHeight + getPaddingTop() + getPaddingBottom()
        )
        setMeasuredDimension(getMeasuredWidth(), height)
    }
}
