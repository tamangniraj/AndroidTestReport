package com.vivekahealth.vivekahealth.utils

import com.vivekahealth.vivekahealth.BuildConfig
import com.vivekahealth.vivekahealth.ui.homePage.HomePageViewModel
import java.io.IOException
import java.net.HttpURLConnection
import java.net.URL

class CheckInternetTask(private val viewModel: HomePageViewModel) {

    fun execute() {
        // Perform the background task on a new thread
        kotlin.concurrent.thread {
            val result = try {
                val url = URL(BuildConfig.SERVER_URL)
                val connection = url.openConnection() as HttpURLConnection
                connection.connectTimeout = 10000 // Set connection timeout to 3000 milliseconds
                connection.connect()

                // Check if the response code indicates success
                if (connection.responseCode == HttpURLConnection.HTTP_OK) {
                    false
                } else {
                    true
                }
            } catch (e: IOException) {
                true
            }

            // Update UI on the main thread
            viewModel._isInternetConnected.postValue(result)
        }
    }
}
