package com.vivekahealth.vivekahealth.utils

import android.content.Context
import android.content.SharedPreferences
import android.hardware.fingerprint.FingerprintManager
import android.os.Build
import android.os.CancellationSignal
import androidx.annotation.RequiresApi

@RequiresApi(api = Build.VERSION_CODES.M)
class FingerPrintHandler(
    private val mContext: Context,
    private val mSharedPreferences: SharedPreferences,
    listener: IAuthenticateListener
) :
    FingerprintManager.AuthenticationCallback() {
    private val mCancellationSignal: CancellationSignal?
    private val mListener: IAuthenticateListener

    init {
        mListener = listener
        mCancellationSignal = CancellationSignal()
    }

    fun startAuth(
        fingerprintManager: FingerprintManager,
        cryptoObject: FingerprintManager.CryptoObject?
    ) {
        fingerprintManager.authenticate(cryptoObject, mCancellationSignal, 0, this, null)
    }

    override fun onAuthenticationError(errorCode: Int, errString: CharSequence) {
//        Toast.makeText(mContext, errString, Toast.LENGTH_SHORT).show();
    }

    override fun onAuthenticationHelp(helpCode: Int, helpString: CharSequence) {
//        Toast.makeText(mContext, helpString, Toast.LENGTH_SHORT).show();
    }

    override fun onAuthenticationSucceeded(result: FingerprintManager.AuthenticationResult) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            val cipher = result.cryptoObject.cipher
            val encoded =
                mSharedPreferences.getString(Constants.KEY_PASSWORD, "Re-register")
            val decoded = HelperFingerAuth.decryptString(encoded, cipher)
            mListener.onAuthenticate(decoded)
        }
    }

    override fun onAuthenticationFailed() {
        mListener.onAuthenticateFail()
    }

    fun cancel() {
        try {
            mCancellationSignal?.cancel()
        } catch (e: NullPointerException) {
            e.printStackTrace()
        }
    }
}
