    fun covertTimeToText(dataDate: String): String? {
        var convertTime: String? = null
        val suffix = ""
        try {
            val unixSeconds: Long = dataDate.toLong()
            val date = Date(unixSeconds)

            @SuppressLint("SimpleDateFormat")
            val sdf = SimpleDateFormat("yyyy-MM-dd HH:mm:ss")
            sdf.timeZone = TimeZone.getDefault()

            var dateString = sdf.format(date)

            val dateFormat = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.getDefault())
            val pasTime = dateFormat.parse(dateString)
            val nowTime = Date()
            val dateDiff = nowTime.time - pasTime.time
            val second = TimeUnit.MILLISECONDS.toSeconds(dateDiff)
            val minute = TimeUnit.MILLISECONDS.toMinutes(dateDiff)
            val hour = TimeUnit.MILLISECONDS.toHours(dateDiff)
            val day = TimeUnit.MILLISECONDS.toDays(dateDiff)
            convertTime = if (second < 60) {
                if (second == 1L) {
                    second.toString() + "s"
                } else {
                    second.toString() + "s"
                }
            } else if (minute < 60) {
                if (minute == 1L) {
                    minute.toString() + "min"
                } else {
                    minute.toString() + "min"
                }
            } else if (hour < 24) {
                if (hour == 1L) {
                    hour.toString() + "h"
                } else {
                    hour.toString() + "h"
                }
            } else if (day >= 7) {
                if (day >= 365) {
                    val tempYear = day / 365
                    if (tempYear == 1L) {
                        tempYear.toString() + "y"
                    } else {
                        tempYear.toString() + "y"
                    }
                } else if (day >= 30) {
                    val tempMonth = day / 30
                    if (tempMonth == 1L) {
                        (day / 30).toString() + "mo"
                    } else {
                        (day / 30).toString() + "mo"
                    }
                } else {
                    val tempWeek = day / 7
                    if (tempWeek == 1L) {
                        (day / 7).toString() + "w"
                    } else {
                        (day / 7).toString() + "w"
                    }
                }
            } else {
                if (day == 1L) {
                    day.toString() + "d"
                } else {
                    day.toString() + "d"
                }
            }
        } catch (e: ParseException) {
            e.printStackTrace()
        }
        return convertTime
    }

    fun digitTOMonth(monthNumber: Int?, resources: Resources): String? {
        if (monthNumber in 1..12) {
            // Convert the month number to the corresponding Month enum
            val month = Month.of(monthNumber ?: 1)

            // Get the month name abbreviation (MMM format)
            return month.getDisplayName(TextStyle.SHORT, getCurrentLocale(resources))
        }
        return ""
    }

    fun checkdob(dob: String): Boolean {
        if (StringUtils.isEmpty(dob)) {
            return false
        } else {
            if (StringUtils.isEmpty(dob.replace("-", ""))) {
                return false
            }
        }
        val sdf = SimpleDateFormat("yyyy-MM-dd")
        var strDate: Date? = null
        try {
            strDate = sdf.parse(dob)
        } catch (e: ParseException) {
            e.printStackTrace()
        } catch (e: NullPointerException) {
            e.printStackTrace()
        }
        val today = Date()
        return today.after(strDate)
    }

    @Throws(ParseException::class)
    fun Convert(date: String?): String {
        if (date.isNullOrBlank()) return ""
        return try {
            val inputFormat = SimpleDateFormat("yyyy-MM-dd", Locale.US)
            val outputFormat = SimpleDateFormat("MMddyyyy", Locale.US)

            inputFormat.parse(date)?.let { parsedDate ->
                outputFormat.format(parsedDate)
            } ?: ""
        } catch (e: ParseException) {
            // Log error if needed
            ""
        }
    }

    fun isFutureMonth(toCompare: Date): Boolean {
        val dateFormat = SimpleDateFormat("MM/dd/yyyy", Locale.US)
        val toCompareString = dateFormat.format(toCompare)
        val dateNow = getCurrentDateTime()
        val nowString = dateFormat.format(dateNow)
        return try {
            val parsedToCompare = dateFormat.parse(toCompareString)
            val parsedNow = dateFormat.parse(nowString)
            return parsedToCompare.after(parsedNow)
        } catch (e: ParseException) {
            e.printStackTrace()
            false
        }
    }

    fun isDateWithinValidRange(dateString: String): Boolean {
        val dateFormat = SimpleDateFormat("MM/dd/yyyy", Locale.getDefault())
        val date = dateFormat.parse(dateString) ?: return false

        val calendar = Calendar.getInstance()
        val currentYear = calendar.get(Calendar.YEAR)
        val minYear = currentYear - 100
        val maxYear = currentYear + 100

        calendar.time = date
        val year = calendar.get(Calendar.YEAR)

        return year in minYear..maxYear
    }

    fun toAPIDate(strDate: String?): String? {
        if (strDate.isNullOrEmpty()) {
            return null
        }

        val inputFormat = SimpleDateFormat("MM/dd/yyyy", Locale.US)
        val date = strDate?.let { inputFormat.parse(it) }

        date?.let {
            val outputFormat = SimpleDateFormat("yyyy-MM-dd", Locale.US)
            return outputFormat.format(date)
        }

        return null
    }

    fun String?.convertLicenseDate(): String {
        if (this.isNullOrEmpty()) {
            return ""
        }

        return try {
            // Define the input and output date formats
            val inputFormat = SimpleDateFormat("MMddyyyy", Locale.US)
            val outputFormat = SimpleDateFormat("yyyy-MM-dd", Locale.US)

            // Parse the input date and format it into the desired output format
            val date = inputFormat.parse(this)
            outputFormat.format(date ?: throw IllegalArgumentException("Invalid date"))
        } catch (e: Exception) {
            // Handle invalid input format
            "Invalid date format"
        }
    }

    fun String?.toGenderApi(): String {
        return when (this?.uppercase()) {
            "1", "M" -> "M" // Male
            "2", "F" -> "F" // Female
            "9", "X", "O" -> "O" // Other
            else -> "O" // Default to "O" for undefined or unknown values
        }
    }

    /** Returns true if [dob] (MM/dd/yyyy) represents a person at least 16 years old today. */
    fun isUserMature(dob: String): Boolean {
        if (dob.isBlank()) return false
        return try {
            val dobDate = SimpleDateFormat("MM/dd/yyyy", Locale.US).parse(dob) ?: return false
            val start = DateTime(dobDate)
            val end = DateTime()
            Years.yearsBetween(start, end).years >= 16
        } catch (_: Exception) { false }
    }

    /**
     * Returns the age in full years of a person born on [dob] as of [atDate].
     * Both dates must be in yyyy-MM-dd format. Returns -1 on parse failure.
     */
    fun ageAtDate(dob: String, atDate: String): Int {
        return try {
            val start = DateTime(dob)
            val end = DateTime(atDate)
            Years.yearsBetween(start, end).years
        } catch (_: Exception) { -1 }
    }

    fun String?.toServerDate(
        inputPattern: String = "MM/dd/yyyy",
        outputPattern: String = "yyyy-MM-dd"
    ): String {
        return try {
            if (this.isNullOrBlank()) return ""

            val inputFormat = SimpleDateFormat(inputPattern, Locale.US)
            val outputFormat = SimpleDateFormat(outputPattern, Locale.US)

            val parsedDate = inputFormat.parse(this)
            parsedDate?.let { outputFormat.format(it) } ?: ""
        } catch (e: Exception) {
            ""
        }
    }
