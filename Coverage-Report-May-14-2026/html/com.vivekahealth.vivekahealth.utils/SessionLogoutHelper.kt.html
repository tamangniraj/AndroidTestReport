package com.vivekahealth.vivekahealth.utils

import android.content.Intent
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.Lifecycle
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.ui.login.LoginActivity
import com.vivekahealth.vivekahealth.utils.remote.RetrofitException

object SessionLogoutHelper {
    var isSessionLogOutShowing: Boolean = false
    var isShowing: Boolean = false

    fun handleSessionLogout(context: AppCompatActivity, error: Throwable): Boolean {
        if (error is RetrofitException) {
            if ((
                error.response?.code() == 401 || error.message?.contains(
                        "Too many follow-up requests: 21",
                        ignoreCase = true
                    ) == true
                ) && !isSessionLogOutShowing
            ) {
                showSessionLogoutDialog(context)
                return true
            }
        }

        return false
    }

    fun handleServer(context: AppCompatActivity, error: String): Boolean {
        if (isShowing) {
            return true
        } else {
            isShowing = true
            context.runOnUiThread {
                HelperUtils.dismissAllDialogs(context.supportFragmentManager)

                DialogUtils(
                    context,
                    object : DialogListener {
                        override fun onPositiveClicked() {
                            isShowing = false
                        }

                        override fun onNeutralClicked() {
                        }

                        override fun onNegativeClicked() {
                        }
                    }
                ).sessionDialog(error)
            }

            return true
        }
    }

    fun showSessionLogoutDialog(context: AppCompatActivity) {
        if (isSessionLogOutShowing || context is LoginActivity) return

        isSessionLogOutShowing = true

        context.runOnUiThread {
            HelperUtils.dismissAllDialogs(context.supportFragmentManager)

            val dialog = DialogUtils(
                context,
                object : DialogListener {
                    override fun onPositiveClicked() {
                        val isActivityInForeground =
                            (context as AppCompatActivity).lifecycle.currentState.isAtLeast(
                                Lifecycle.State.RESUMED
                            )

                        if (!isActivityInForeground) {
                            return
                        }
                        HelperUtils.dismissAllDialogs(context.supportFragmentManager)

                        val intent = Intent(context, LoginActivity::class.java)
                        context.startActivity(intent)
                        isSessionLogOutShowing = false
                        context.finish()
                    }

                    override fun onNeutralClicked() {
                    }

                    override fun onNegativeClicked() {
                    }
                }
            )

            dialog.sessionLoggedOutDialog()
        }
    }

    fun showTimeoutError(context: AppCompatActivity): Boolean {
        return if (isShowing) {
            true
        } else {
            isShowing = true
            context.runOnUiThread {
                DialogUtils(
                    context,
                    object : DialogListener {
                        override fun onPositiveClicked() {
                            isShowing = false
                        }

                        override fun onNeutralClicked() {
                        }

                        override fun onNegativeClicked() {
                        }
                    }
                ).dialogWithView(
                    context.getString(R.string.ok_string),
                    "",
                    "",
                    R.layout.layout_network
                )
            }
            true
        }
    }
}
