package com.vivekahealth.vivekahealth.utils

import android.content.Context
import android.content.ContextWrapper
import android.view.View
import android.view.ViewGroup
import android.view.WindowManager
import androidx.fragment.app.DialogFragment

/**
 * Returns a Context whose WindowManager injects the dialog's own window token into every
 * WindowManager.LayoutParams before calling addView().
 *
 * Why this is needed:
 *   SingleDateAndTimePickerDialog (and similar libraries) use
 *   WindowManager.LayoutParams.TYPE_APPLICATION_PANEL, which is a sub-window that anchors
 *   to a parent window via its token.  When the caller is a BottomSheetDialogFragment the
 *   natural context (requireActivity / requireContext) carries the *activity* window token,
 *   so the panel anchors to the activity window and appears BEHIND the dialog.  Overriding
 *   the token with the dialog's decorView.windowToken anchors the panel to the dialog window
 *   instead, making it appear above.
 */
fun DialogFragment.dialogWindowContext(): Context {
    val decorView: View = dialog?.window?.decorView
        ?: return requireActivity() // fallback if dialog isn't attached yet

    val baseWm = requireActivity().getSystemService(Context.WINDOW_SERVICE) as WindowManager

    val patchedWm = object : WindowManager by baseWm {
        override fun addView(view: View, params: ViewGroup.LayoutParams) {
            (params as? WindowManager.LayoutParams)?.token = decorView.windowToken
            baseWm.addView(view, params)
        }
    }

    return object : ContextWrapper(requireActivity()) {
        override fun getSystemService(name: String): Any? =
            if (name == Context.WINDOW_SERVICE) {
                patchedWm
            } else {
                super.getSystemService(name)
            }
    }
}
