package com.vivekahealth.vivekahealth.utils

import android.app.Dialog
import android.content.Context
import android.content.SharedPreferences
import android.graphics.Color
import android.graphics.drawable.ColorDrawable
import android.view.LayoutInflater
import android.widget.TextView
import androidx.annotation.LayoutRes
import androidx.fragment.app.Fragment
import com.afollestad.materialdialogs.MaterialDialog
import com.google.android.material.bottomsheet.BottomSheetDialog
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.databinding.BottomDialogNetworkBinding
import com.vivekahealth.vivekahealth.databinding.EobPriceSuccessBinding
import com.vivekahealth.vivekahealth.databinding.EobSuccessBinding
import com.vivekahealth.vivekahealth.databinding.NotificationRequestBinding
import com.vivekahealth.vivekahealth.databinding.PriceNotEnteredBinding

class DialogUtils(
    val context: Context,

    val listener: DialogListener
) {

    var materialDialog: MaterialDialog? = null

    fun sheetDialog(fragment: Fragment) {
        val dialog = BottomSheetDialog(context)

        val binding = BottomDialogNetworkBinding.inflate(fragment.layoutInflater)
        dialog.window?.setBackgroundDrawable(ColorDrawable(Color.TRANSPARENT))

        binding.includeInSearch.setOnClickListener {
            listener.onNegativeClicked()

            dialog.dismiss()
        }
        binding.dontInclude.setOnClickListener {
            listener.onPositiveClicked()

            dialog.dismiss()
        }

        dialog.setCancelable(false)

        dialog.setContentView(binding.root)

        dialog.show()
    }
    fun sheetDialogEob(fragment: Fragment) {
        val dialog = BottomSheetDialog(fragment.requireContext()) // Use requireContext() to get the context

        // Use ViewBinding to inflate your layout
        val binding = EobSuccessBinding.inflate(LayoutInflater.from(fragment.requireContext()))
        dialog.window?.setBackgroundDrawable(ColorDrawable(Color.TRANSPARENT))

        binding.okBtn.setOnClickListener {
            listener.onPositiveClicked()

            dialog.dismiss()
        }

        dialog.setCancelable(false)

        dialog.setContentView(binding.root)

        dialog.show()
    }

    fun dialogEobPrice(fragment: Fragment, sharedPreferences: SharedPreferences, total: String) {
        val dialog = Dialog(context)
        val binding = EobPriceSuccessBinding.inflate(
            (LayoutInflater.from(fragment.requireContext()))
        )
        binding.total = total

        binding.okBtn.setOnClickListener {
            listener.onPositiveClicked()

            dialog.dismiss()
        }
        binding.check.setOnCheckedChangeListener { buttonView, isChecked ->
            if (isChecked) {
                sharedPreferences.edit().putBoolean(
                    sharedPreferences.getString("memberID", "").toString() + "_" + "doNot_show_price",
                    true
                ).apply()
            } else {
                sharedPreferences.edit().putBoolean(
                    sharedPreferences.getString("memberID", "").toString() + "_" + "doNot_show_price",
                    false
                ).apply()
            }
        }

        dialog.setCancelable(false)

        dialog.setContentView(binding.root)

        dialog.show()
    }

    fun dialogNoPrice(fragment: Fragment) {
        val dialog = Dialog(context)

        val binding = PriceNotEnteredBinding.inflate(
            (LayoutInflater.from(fragment.requireContext()))
        )
        dialog.window?.setBackgroundDrawable(ColorDrawable(Color.TRANSPARENT))

        binding.okBtn.setOnClickListener {
            listener.onPositiveClicked()

            dialog.dismiss()
        }

        dialog.setCancelable(false)

        dialog.setContentView(binding.root)

        dialog.show()
    }
    fun sheetDialogEobNotification(fragment: Fragment) {
        val dialog = BottomSheetDialog(context)

        val binding = NotificationRequestBinding.inflate(fragment.layoutInflater)
        dialog.window?.setBackgroundDrawable(ColorDrawable(Color.TRANSPARENT))

        binding.yes.setOnClickListener {
            listener.onPositiveClicked()

            dialog.dismiss()
        }
        binding.no.setOnClickListener {
            listener.onNegativeClicked()

            dialog.dismiss()
        }

        dialog.setCancelable(false)

        dialog.setContentView(binding.root)

        dialog.show()
    }

    fun dialog(content: String, positiveBtn: String, neutralBtn: String, negativeBtn: String) {
        if (materialDialog?.isShowing == true) {
            materialDialog!!.dismiss()
        }

        if (materialDialog != null) {
            materialDialog = null
        }

        materialDialog = MaterialDialog.Builder(context)
            .content(content).cancelable(false)
            .positiveText(positiveBtn)
            .neutralText(neutralBtn)
            .negativeText(negativeBtn)
            .onPositive { dialog, which ->
                dialog.dismiss()
                listener.onPositiveClicked()
            }.onNeutral { dialog, which ->
                dialog.dismiss()
                listener.onNeutralClicked()
            }.onNegative { dialog, which ->
                dialog.dismiss()

                listener.onNegativeClicked()
            }.show()
    }

    fun dialogTitle(
        content: String,
        title: String,
        positiveBtn: String,
        neutralBtn: String,
        negativeBtn: String
    ) {
        MaterialDialog.Builder(context).title(title)
            .content(content).cancelable(false)
            .positiveText(positiveBtn)
            .neutralText(neutralBtn)
            .negativeText(negativeBtn)
            .onPositive { dialog, which ->

                listener.onPositiveClicked()
                dialog.dismiss()
            }.onNeutral { dialog, which ->
                listener.onNeutralClicked()
                dialog.dismiss()
            }.onNegative { dialog, which ->

                listener.onNegativeClicked()
                dialog.dismiss()
            }
            .show()
    }

    fun dialogWithCustomTextView(
        title: String,
        content: String,
        positiveBtn: String,
        neutralBtn: String,
        negativeBtn: String,
        viewId: TextView
    ) {
        viewId.text = content
        MaterialDialog.Builder(context)
            .title(title)
            .customView(
                viewId,
                false
            )
            .positiveText(positiveBtn)
            .cancelable(true)
            .neutralText(neutralBtn)
            .negativeText(negativeBtn)
            .onPositive { dialog, _ ->
                listener.onPositiveClicked()
                dialog.dismiss()
            }.onNeutral { dialog, _ ->
                listener.onNeutralClicked()
                dialog.dismiss()
            }.onNegative { dialog, _ ->
                listener.onNegativeClicked()
                dialog.dismiss()
            }
            .show()
    }

    fun dialogWithView(
        positiveBtn: String,
        neutralBtn: String,
        negativeBtn: String,
        @LayoutRes viewId: Int
    ) {
        MaterialDialog.Builder(context)
            .customView(
                viewId,
                false
            )
            .positiveText(positiveBtn)
            .cancelable(true)
            .neutralText(neutralBtn)
            .negativeText(negativeBtn)
            .onPositive { dialog, which ->

                listener.onPositiveClicked()
                dialog.dismiss()
            }.onNeutral { dialog, which ->
                listener.onNeutralClicked()
                dialog.dismiss()
            }.onNegative { dialog, which ->

                listener.onNegativeClicked()
                dialog.dismiss()
            }
            .show()
    }

    fun sessionLoggedOutDialog() {
        MaterialDialog.Builder(context)
            .content(R.string.session_expiry_msg)
            .positiveText(R.string.ok_string)
            .onPositive { dialog, _ ->
                listener.onPositiveClicked()
                dialog.dismiss()
            }
            .autoDismiss(false)
            .cancelable(false)
            .show()
    }

    fun sessionDialog(message: String) {
        MaterialDialog.Builder(context)
            .content(message)
            .positiveText(R.string.ok_string)
            .onPositive { dialog, _ ->
                listener.onPositiveClicked()
                dialog.dismiss()
            }
            .autoDismiss(false)
            .cancelable(false)
            .show()
    }
}
