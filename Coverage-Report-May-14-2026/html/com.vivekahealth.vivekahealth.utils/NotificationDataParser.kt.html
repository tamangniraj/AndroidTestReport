package com.vivekahealth.vivekahealth.utils

import android.content.Intent
import android.os.Bundle

/**
 * Helper class to parse and manage notification data across activities.
 * Handles notification routing from MainActivity → LoginActivity → HomePageActivity
 */
data class NotificationDataParser(
    var clickAction: String? = null,
    var isValid: Boolean = false,
    var memberId: String? = null,
    var url: String? = null,
    var extras: Bundle? = null,
    var fromBroadCast: Boolean = false
) {

    companion object {
        const val KEY_CLICK_ACTION = "click_action"
        private const val KEY_MEMBER_ID = "memberId"
        private const val KEY_URL = "url"
        const val FROM_BROADCAST = "from_broadcast"

        // Supported notification actions
        const val ACTION_OPEN_CHAT = "OPEN_CHAT_ACTIVITY"
        const val ACTION_OPEN_DOCUMENTS = "OPEN_DOCUMENTS_LIST_ACTIVITY"
        const val ACTION_OPEN_DOCUMENT_REQUESTS = "OPEN_DOCUMENT_REQUESTS_ACTIVITY"
        const val ACTION_OPEN_NOTIFICATION_CENTER = "OPEN_NOTIFICATION_CENTER"
        const val ACTION_OPEN_SURVEY = "OPEN_SURVEY"
        const val ACTION_OPEN_BENEFIT_SCREEN = "OPEN_BENEFIT_SCREEN_ACTIVITY"

        /**
         * Parse notification data from an Intent
         * @param intent Intent containing notification data
         * @return NotificationDataParser instance with parsed data
         */
        fun fromIntent(intent: Intent?): NotificationDataParser {
            if (intent == null) {
                return NotificationDataParser()
            }

            // Check for clickAction in intent extras first, fallback to intent.action
            val clickAction = intent.getStringExtra(KEY_CLICK_ACTION)
                ?: intent.action
            val url = intent.getStringExtra(KEY_URL)
            val memberId = intent.getStringExtra(KEY_MEMBER_ID)
            val fromBroadCast = intent.getBooleanExtra(FROM_BROADCAST, false)

            return NotificationDataParser(
                clickAction = clickAction,
                isValid = intent.hasExtra(KEY_CLICK_ACTION),
                memberId = memberId,
                url = url,
                extras = intent.extras,
                fromBroadCast = fromBroadCast
            )
        }
    }

    /**
     * Add this notification data to an intent, including all extras from the original intent
     */
    fun addToIntent(intent: Intent): Intent {
        // Copy all extras from the original bundle if available
        extras?.let { originalExtras ->
            intent.putExtras(originalExtras)
        }

        return intent
    }

    override fun toString(): String {
        return "NotificationDataParser(clickAction=$clickAction, isValid=$isValid,  extras=${extras?.keySet()?.joinToString { "," }})"
    }
}
