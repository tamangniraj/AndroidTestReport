package com.vivekahealth.vivekahealth.utils

import android.text.InputFilter
import android.widget.EditText
import androidx.appcompat.widget.SearchView
import com.google.android.material.textfield.TextInputEditText
import com.santalu.maskedittext.MaskEditText

fun EditText.disableEmoji() {
    this.filters = arrayOf(EditTextHelper.getEmojiDisabledString())
}

fun TextInputEditText.disableEmoji() {
    this.filters = arrayOf(EditTextHelper.getEmojiDisabledString())
}

fun MaskEditText.disableEmoji() {
    this.filters = arrayOf(EditTextHelper.getEmojiDisabledString())
}

fun SearchView.disableEmoji() {
    val editText = this.findViewById<EditText>(androidx.appcompat.R.id.search_src_text)
    editText.filters = arrayOf(EditTextHelper.disableNonAscii())
}

private object EditTextHelper {
    fun getEmojiDisabledString(): InputFilter {
        return InputFilter { source, start, end, dest, dstart, dend ->
            val builder = StringBuilder()
            for (index in start until end) {
                val type = Character.getType(source[index])
                if (type != Character.SURROGATE.toInt() && type != Character.OTHER_SYMBOL.toInt()) {
                    builder.append(source[index])
                }
            }
            val newStr = builder.toString()
            newStr
        }
    }

    fun disableNonAscii(): InputFilter {
        val pattern = Regex("[^a-zA-Z0-9.\\- /#` ']")

        return InputFilter { source, start, end, dest, dstart, dend ->
            return@InputFilter when {
                source?.matches(pattern) == true -> ""
                source.length > 1 -> source.trim { it.toString().matches(pattern) } // Put your desired logic here, these sample logic was doing a trim/remove
                else -> null
            }
        }
    }
}
