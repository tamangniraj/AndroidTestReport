fun getLanguageTextAndFlag(langCode: String): Pair<Int, String> {
    return if (langCode.isEmpty()) {
        Pair(getLanguageTextByCode("en"), getFlagByCode("en"))
    } else {
        Pair(getLanguageTextByCode(langCode), getFlagByCode(langCode))
    }
}

fun getLanguageTextByCode(langCode: String): Int {
    return langIdCodePairs.firstOrNull { it.second.equals(langCode, ignoreCase = true) }?.first ?: R.string.english
}

fun getFlagByCode(langCode: String): String {
    return langWithFlag.firstOrNull { it.second.equals(langCode, ignoreCase = true) }?.third ?: ""
}
