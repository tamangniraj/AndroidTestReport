    fun isAppInForeground(context: Context, packageName: String): Boolean {
        val currentMillis = System.currentTimeMillis()
        val usageStatsManager = context.getSystemService(Context.USAGE_STATS_SERVICE) as UsageStatsManager

        // Check if the necessary permission is granted
        val hasUsageStatsPermission = checkUsageStatsPermission(context)
        if (!hasUsageStatsPermission) {
            // Permission is not granted, handle accordingly
            return false
        }

        val usageStatsList = usageStatsManager.queryUsageStats(
            UsageStatsManager.INTERVAL_DAILY,
            currentMillis - 1000 * 10,
            currentMillis
        )

        if (usageStatsList.isNotEmpty()) {
            val recentUsageStats = usageStatsList.first()
            if (recentUsageStats.packageName == packageName) {
                return true
            }
        }

        return false
    }

    fun getMonthIntegers(lookBack: Int, offset: Int, currentMonth: Int, currentYear: Int): List<Int> {
        // Validate input parameters (optional)
        // if (lookBack < 0) {
        //     throw IllegalArgumentException("lookBack must be non-negative.")
        // }

        // Get the current YearMonth
        var yearMonth = YearMonth.now()
        if (currentMonth != 0 && currentYear != 0) {
            yearMonth = YearMonth.of(currentYear, currentMonth)
        }
        val currentYearMonth = yearMonth

        // Calculate the target YearMonth after applying offset
        val targetYearMonth = currentYearMonth.minusMonths(offset.toLong())

        // If offset is zero, use the current month as the target
        val actualTargetYearMonth = if (offset == 0) currentYearMonth else targetYearMonth

        // Calculate the start of the lookBack period
        val startYearMonth = actualTargetYearMonth.minusMonths(lookBack.toLong())

        // Calculate the number of months in the lookBack period
        val monthsInRange = ChronoUnit.MONTHS.between(startYearMonth, actualTargetYearMonth).toInt()

        // Create a list of integers representing the months within the specified range
        return (0 until monthsInRange).map {
            startYearMonth.plusMonths(it.toLong()).monthValue
        }
    }

        lookBack: Int,
        offset: Int,
        currentMonth: Int,
        currentYear: Int,
        resources: Resources
    ): String {
        // Validate input parameters
//        if (lookBack < 0 ) {
//            throw IllegalArgumentException("Both lookBack and offset must be non-negative.")
//        }

        // Get the current YearMonth
        var yearMonth = YearMonth.now()
        if (currentMonth != 0 && currentYear != 0) {
            yearMonth = YearMonth.of(currentYear, currentMonth)
        }
        val currentYearMonth = yearMonth

        // Calculate the target YearMonth after applying offset
        val targetYearMonth = currentYearMonth.minusMonths(offset.toLong())

        // If offset is zero, use the current month as the target
        val actualTargetYearMonth = if (offset == 0) currentYearMonth else targetYearMonth

        // Calculate the start of the lookBack period
        val startYearMonth = actualTargetYearMonth.minusMonths(lookBack.toLong())

        // Calculate the number of months in the lookBack period
        val monthsInRange = ChronoUnit.MONTHS.between(startYearMonth, actualTargetYearMonth).toInt()

        // Create a list of formatted month names within the specified range
        val formattedMonthNames = (0 until monthsInRange).map {
            val month = startYearMonth.plusMonths(it.toLong()).month
            month.getDisplayName(TextStyle.FULL, DateUtils.getCurrentLocale(resources))
        }

        return when {
            monthsInRange <= 1 -> formattedMonthNames.joinToString(" and ")
            else -> formattedMonthNames.dropLast(1).joinToString(", ") + " and ${formattedMonthNames.last()}"
        }
    }
    private fun checkUsageStatsPermission(context: Context): Boolean {
        val packageManager = context.packageManager
        val appOpsManager = context.getSystemService(Context.APP_OPS_SERVICE) as AppOpsManager

        return try {
            val packageInfo = packageManager.getApplicationInfo(context.packageName, 0)
            val mode = appOpsManager.checkOpNoThrow(
                AppOpsManager.OPSTR_GET_USAGE_STATS,
                packageInfo.uid,
                packageInfo.packageName
            )
            mode == AppOpsManager.MODE_ALLOWED
        } catch (e: PackageManager.NameNotFoundException) {
            false
        }
    }

    fun getLocationGeocode(latitude: Double?, longitude: Double?, context: Context): String {
        var stringAddress = ""
        if (isNetworkConnected(context) && !Constants.IS_OFFLINE) {
            if (latitude != null && longitude != null) {
                val geocoder = Geocoder(context, Locale.getDefault())

                try {
                    val address = geocoder.getFromLocation(latitude, longitude, 1)
                    if (address != null && address.isNotEmpty()) {
                        val subThoroughfare = if (!address[0].subThoroughfare.isNullOrEmpty()) {
                            val value = address[0].subThoroughfare
                            "$value "
                        } else {
                            ""
                        }
                        val throughFare = if (!address[0].thoroughfare.isNullOrEmpty()) {
                            val value = address[0].thoroughfare
                            "$value, "
                        } else {
                            ""
                        }
                        val sublocality = if (!address[0].subLocality.isNullOrEmpty()) {
                            val value = address[0].subLocality
                            "$value, "
                        } else {
                            ""
                        }
                        val locality = if (!address[0].locality.isNullOrEmpty()) {
                            val value = address[0].locality
                            "$value, "
                        } else {
                            ""
                        }
                        val state = if (!address[0].adminArea.isNullOrEmpty()) {
                            val value = address[0].adminArea
                            "$value "
                        } else {
                            ""
                        }
                        val zip = if (!address[0].postalCode.isNullOrEmpty()) {
                            val value = address[0].postalCode
                            "$value, "
                        } else {
                            ""
                        }
                        val country = if (!address[0].countryName.isNullOrEmpty()) {
                            val value = address[0].countryName
                            "$value"
                        } else {
                            ""
                        }
                        stringAddress = subThoroughfare + throughFare + sublocality + locality + state + zip + country
                    }
                } catch (e: IOException) {
                    Timber.e(context.packageName, "Error getting location name: ${e.message}")
                }
            }
        }
        return stringAddress
    }

    @Throws(java.lang.Exception::class)
    fun decrypt(value: String?): String {
        if (value == null) {
            return ""
        }

        val key = generateKey()

        val cipher = Cipher.getInstance(Constants.ALGORITHM)

        val decryptedValue64 = Base64.decode(value, Base64.DEFAULT)

        if (decryptedValue64.size < 16) {
            return ""
        }
        val iv = ByteArray(16)
        System.arraycopy(decryptedValue64, 0, iv, 0, 16)
        val ivParams = IvParameterSpec(iv)
        cipher.init(Cipher.DECRYPT_MODE, key, ivParams)
        val decryptedByteValue = cipher.doFinal(decryptedValue64, 16, decryptedValue64.size - 16)
        return String(decryptedByteValue)
    }

    fun encrypt(value: String?): String? {
        requireNotNull(value) { "Input value cannot be null" }
        val key = generateKey()
        val iv = ByteArray(16)
        SecureRandom().nextBytes(iv)
        val ivParams = IvParameterSpec(iv)

        val cipher = Cipher.getInstance(Constants.ALGORITHM) // Use appropriate padding and mode
        cipher.init(Cipher.ENCRYPT_MODE, key, ivParams)
        val encryptedByteValue = cipher.doFinal(value.toByteArray(Charsets.UTF_8))

        // Combine IV and encrypted value
        val combined = ByteArray(iv.size + encryptedByteValue.size)
        System.arraycopy(iv, 0, combined, 0, iv.size)
        System.arraycopy(encryptedByteValue, 0, combined, iv.size, encryptedByteValue.size)

        return Base64.encodeToString(combined, Base64.DEFAULT)
    }

    fun isNetworkAvailable(context: Context): Boolean {
        val connectivityManager =
            context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        val capabilities =
            connectivityManager.getNetworkCapabilities(connectivityManager.activeNetwork)
        if (capabilities != null) {
            if (capabilities.hasTransport(NetworkCapabilities.TRANSPORT_CELLULAR)) {
                Timber.tag("Internet").i("NetworkCapabilities.TRANSPORT_CELLULAR")
                return true
            } else if (capabilities.hasTransport(NetworkCapabilities.TRANSPORT_WIFI)) {
                Timber.tag("Internet").i("NetworkCapabilities.TRANSPORT_WIFI")
                return true
            } else if (capabilities.hasTransport(NetworkCapabilities.TRANSPORT_ETHERNET)) {
                Timber.tag("Internet").i("NetworkCapabilities.TRANSPORT_ETHERNET")
                return true
            }
        }
        return false
    }

    fun isNetworkConnected(context: Context): Boolean {
        val connectivityManager =
            context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        val nw = connectivityManager.activeNetwork ?: return false
        val actNw = connectivityManager.getNetworkCapabilities(nw) ?: return false
        return when {
            actNw.hasTransport(NetworkCapabilities.TRANSPORT_WIFI) -> true
            actNw.hasTransport(NetworkCapabilities.TRANSPORT_CELLULAR) -> true
            // for other device how are able to connect with Ethernet
            actNw.hasTransport(NetworkCapabilities.TRANSPORT_ETHERNET) -> true
            // for check internet over Bluetooth
            actNw.hasTransport(NetworkCapabilities.TRANSPORT_BLUETOOTH) -> true
            else -> false
        }
    }

    fun copyStreamToFile(inputStream: InputStream, outputFile: File) {
        inputStream.use { input ->
            val outputStream = FileOutputStream(outputFile)
            outputStream.use { output ->
                val buffer = ByteArray(4 * 1024) // buffer size
                while (true) {
                    val byteCount = input.read(buffer)
                    if (byteCount < 0) break
                    output.write(buffer, 0, byteCount)
                }
                output.flush()
            }
        }
    }

    fun convertBitmapToFile(fileName: String, bitmap: Bitmap, context: Context): File {

        val bos = ByteArrayOutputStream()
        bitmap.compress(Bitmap.CompressFormat.JPEG, 80 /*ignored for PNG*/, bos)

        try {
            fos = FileOutputStream(file)
        } catch (e: FileNotFoundException) {
            e.printStackTrace()
        }

            fos?.close()
        } catch (e: IOException) {
            e.printStackTrace()
        }
        return file
    }

    @SuppressLint("MissingPermission")
    private fun requestCurrentLocation(fragment: Fragment): Location? {
        var location: Location? = null

        if (fragment.requireContext().hasPermission(Manifest.permission.ACCESS_FINE_LOCATION)) {
            fusedLocationClient =
                LocationServices.getFusedLocationProviderClient(fragment.requireContext())

            // Returns a single current location fix on the device. Unlike getLastLocation() that
            // returns a cached location, this method could cause active location computation on the
            // device. A single fresh location will be returned if the device location can be
            // determined within reasonable time (tens of seconds), otherwise null will be returned.
            //
            // Both arguments are required.
            // PRIORITY type is self-explanatory. (Other options are PRIORITY_BALANCED_POWER_ACCURACY,
            // PRIORITY_LOW_POWER, and PRIORITY_NO_POWER.)
            // The second parameter, [CancellationToken] allows the activity to cancel the request
            // before completion.
            val currentLocationTask: Task<Location> = fusedLocationClient!!.getCurrentLocation(
                LocationRequest.PRIORITY_HIGH_ACCURACY,
                cancellationTokenSource.token
            )

            currentLocationTask.addOnCompleteListener { task: Task<Location> ->
                if (task.isSuccessful && task.result != null) {
                    val result: Location = task.result

                    location = result
//                    "Location (success): ${result.latitude}, ${result.longitude}"
                } else {
                    val exception = task.exception
                    location = null
                    "Location (failure): $exception"
                }

                // this
            }
        }
        return location
    }

    ) {

        }
    }

    fun createMultipartBodyPartFromBitmap(
        bitmap: Bitmap,
        partName: String,
        fileName: String = "image.jpg",
        maxWidth: Int = 1024
    ): MultipartBody.Part {
        val scaled = if (bitmap.width > maxWidth) {
            val scale = maxWidth.toFloat() / bitmap.width
            Bitmap.createScaledBitmap(bitmap, maxWidth, (bitmap.height * scale).toInt(), true)
        } else {
            bitmap
        }

        val byteArrayOutputStream = ByteArrayOutputStream()
        scaled.compress(Bitmap.CompressFormat.JPEG, 80, byteArrayOutputStream)
        if (scaled !== bitmap) scaled.recycle()
        val byteArray = byteArrayOutputStream.toByteArray()

        val requestBody = RequestBody.create(
            "image/jpeg".toMediaTypeOrNull(),
            byteArray
        )
        return MultipartBody.Part.createFormData(partName, fileName, requestBody)
    }

    fun isSameDomain(url1: String?, url2: String?): Boolean {
        fun extractMainDomain(url: String?): String? {
            return try {
                url?.let {
                    val host = URL(it).host
                    // Normalize the host by removing "www."
                    val normalizedHost = host.removePrefix("www.")
                    // Extract the main domain part, assuming the last two segments are the main domain
                    val parts = normalizedHost.split(".")
                    if (parts.size >= 2) {
                        "${parts[parts.size - 2]}.${parts[parts.size - 1]}"
                    } else {
                        normalizedHost
                    }
                }
            } catch (e: MalformedURLException) {
                // Handle malformed URLs by returning null
                return null
            }
        }

        val domain1 = extractMainDomain(url1)
        val domain2 = extractMainDomain(url2)

        return domain1 != null && domain2 != null && domain1 == domain2
    }

    fun debounceAction(

        scope: CoroutineScope,
        action: suspend () -> Unit
    ) {
        debounceJob?.cancel() // Cancel the previous debounce job if it exists

            action() // Execute the action after the debounce interval
        }
    }
