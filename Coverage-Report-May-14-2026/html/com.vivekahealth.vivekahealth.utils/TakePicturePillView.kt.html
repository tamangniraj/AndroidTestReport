package com.vivekahealth.vivekahealth.utils

import android.animation.Animator
import android.animation.AnimatorListenerAdapter
import android.animation.AnimatorSet
import android.animation.ObjectAnimator
import android.content.Context
import android.util.AttributeSet
import android.view.View
import android.view.animation.AccelerateDecelerateInterpolator
import android.widget.ImageView
import android.widget.TextView
import androidx.cardview.widget.CardView
import androidx.constraintlayout.widget.ConstraintLayout
import com.vivekahealth.vivekahealth.R

class TakePicturePillView @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
    defStyleAttr: Int = 0
) : ConstraintLayout(context, attrs, defStyleAttr) {

    private val animatedBackground: CardView
    private val mainPill: CardView
    private var iconImageView: ImageView
    private var msgTextView: TextView
    private var currentAnimator: AnimatorSet? = null

    var isPulseAnimationEnabled: Boolean = true
        get() = field
        set(value) {
            field = value
            if (value) {
                startAnimation()
            } else {
                stopAnimation()
            }
        }

    var isRetake: Boolean = false
        get() = field
        set(value) {
            field = value
            if (value) {
                msgTextView.text = context.getString(R.string.retake)
                iconImageView.setImageResource(R.drawable.document_retake)
                iconImageView.imageTintList = null
            } else {
                msgTextView.text = context.getString(R.string.take_a_photo)
                iconImageView.setImageResource(R.drawable.ic_menu_camera)
                iconImageView.imageTintList = context.getColorStateList(R.color.colorAccent)
            }
        }

    init {
        inflate(context, R.layout.take_picture_pill_view, this)
        animatedBackground = findViewById(R.id.animatedBackground)
        mainPill = findViewById(R.id.mainPill)
        iconImageView = findViewById(R.id.cameraIcon)
        msgTextView = findViewById(R.id.takePictureText)

        // Read custom attributes
        context.theme.obtainStyledAttributes(
            attrs,
            R.styleable.TakePicturePillView,
            0,
            0
        ).apply {
            try {
                isPulseAnimationEnabled = getBoolean(
                    R.styleable.TakePicturePillView_enablePulseAnimation,
                    true // default value
                )
                isRetake = getBoolean(
                    R.styleable.TakePicturePillView_isRetake,
                    false
                )
            } finally {
                recycle()
            }
        }
    }

    override fun onAttachedToWindow() {
        super.onAttachedToWindow()
        if (isPulseAnimationEnabled) {
            startAnimation()
        }
    }

    override fun onDetachedFromWindow() {
        super.onDetachedFromWindow()
        stopAnimation()
    }

    private fun startAnimation() {
        // Cancel any existing animation
        stopAnimation()

        // Reset initial state
        animatedBackground.scaleX = 1f
        animatedBackground.scaleY = 1f
        animatedBackground.alpha = 1f
        animatedBackground.visibility = View.VISIBLE

        val scaleX = ObjectAnimator.ofFloat(animatedBackground, "scaleX", 1f, 1.5f)
        val scaleY = ObjectAnimator.ofFloat(animatedBackground, "scaleY", 1f, 1.5f)
        val alpha = ObjectAnimator.ofFloat(animatedBackground, "alpha", 1f, 0f)

        currentAnimator = AnimatorSet().apply {
            playTogether(scaleX, scaleY, alpha)
            duration = 2000
            interpolator = AccelerateDecelerateInterpolator()
            addListener(object : AnimatorListenerAdapter() {
                override fun onAnimationEnd(animation: Animator) {
                    if (isPulseAnimationEnabled) {
                        // Reset and repeat
                        animatedBackground.scaleX = 1f
                        animatedBackground.scaleY = 1f
                        animatedBackground.alpha = 1f
                        startAnimation()
                    }
                }
            })
            start()
        }
    }

    private fun stopAnimation() {
        currentAnimator?.removeAllListeners()
        currentAnimator?.cancel()
        currentAnimator = null
        animatedBackground.visibility = View.INVISIBLE
        // Reset to initial state
        animatedBackground.scaleX = 1f
        animatedBackground.scaleY = 1f
        animatedBackground.alpha = 1f
    }

    fun enablePulseAnimation(enable: Boolean) {
        isPulseAnimationEnabled = enable
    }

    fun setIsRetake(retake: Boolean) {
        isRetake = retake
    }
}
