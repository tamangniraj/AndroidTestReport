package com.vivekahealth.vivekahealth.utils

import androidx.annotation.StringRes
import com.vivekahealth.vivekahealth.R

enum class DocumentType(@StringRes var TranslatedName: Int, var FileName: String) {
    Ssn(R.string.social_security_card, "Social Security Card"),
    MarriageCertificate(R.string.marriage_certificate, "Marriage Certificate"),
    BirtCertificate(R.string.birth_certificate, "Birth Certificate"),
    FrontDriverLicence(R.string.front_of_drivers_licence, "Front of Driver's License"),
    BackDriverLicence(R.string.back_of_drivers_licence, "Back of Driver's License"),
    FrontInsuranceCard(R.string.front_of_insurance, "Front of Insurance Card"),
    BackInsuranceCard(R.string.back_of_insurance, "Back of Insurance Card"),
    ContinueCoverage(R.string.proof_of_previous_insurance, "Proof of Previous Insurance")
}

fun String.sanitize(): String {
    return this.replace(Regex("[^a-zA-Z0-9._-]"), "_")
}

fun getDocumentType(id: Int): DocumentType {
    return when (id) {
        71 -> DocumentType.Ssn
        50 -> DocumentType.MarriageCertificate
        51 -> DocumentType.BirtCertificate
        103 -> DocumentType.FrontDriverLicence
        104 -> DocumentType.BackDriverLicence
        53 -> DocumentType.ContinueCoverage
        501 -> DocumentType.FrontInsuranceCard
        502 -> DocumentType.BackInsuranceCard
        // Add more mappings if needed
        else -> throw IllegalArgumentException("Unknown document type ID: $id")
    }
}
