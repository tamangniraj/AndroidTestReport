    fun isEmailValid(email: String?): Boolean {
        var isValid = false
        val expression = "^[A-Z0-9a-z._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$"
        val pattern = Pattern.compile(expression, Pattern.CASE_INSENSITIVE)
        val matcher = pattern.matcher(email)
        isValid = matcher.matches()
        return isValid
    }

    fun checkSsnValidity(ssn: String, allowTin: Boolean = true): Boolean {
        val cleaned = ssn.replace("-", "")
        return checkSsnValidityWithOutDash(cleaned, allowTin)

    fun checkSsnValidityWithOutDash(ssn: String, allowTin: Boolean = true): Boolean {
        val regex = if (allowTin) Constants.SSN_REGEX_WITH_TIN else Constants.SSN_REGEX
        return ssn.matches(Regex(regex))

            SimpleDateFormat("yyyyMMdd_HHmmss", Locale.getDefault())

        val imageFileName = "JPEG_" + timeStamp + "_"
        val storageDir = context.getExternalFilesDir(Environment.DIRECTORY_PICTURES)
        return File.createTempFile(

            ".jpg", /* suffix */
            storageDir /* directory */

    fun cleanUpEmail(email: String): String {
        if (email.isEmpty()) {
            return email
        }
        var cleanEmail = email
        cleanEmail = cleanEmail.trim { it <= ' ' }.replace("\\s+".toRegex(), "")
        val atIndex = cleanEmail.indexOf("@")
        if (atIndex != -1) {
            val nameSpaceString = cleanEmail.substring(atIndex)
            val cleanNameSpace = nameSpaceString.replace("\\.{2,}".toRegex(), ".")
            cleanEmail = cleanEmail.replace(nameSpaceString, cleanNameSpace)
        }
        return cleanEmail
    }
