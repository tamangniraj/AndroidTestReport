package com.vivekahealth.vivekahealth.utils

import android.view.View
import androidx.recyclerview.widget.LinearSnapHelper
import androidx.recyclerview.widget.RecyclerView

class SnappingHelper : LinearSnapHelper() {

    override fun calculateDistanceToFinalSnap(
        layoutManager: RecyclerView.LayoutManager,
        targetView: View
    ): IntArray {
        // Disable manual scrolling
        return intArrayOf(0, 0)
    }

    override fun findTargetSnapPosition(
        layoutManager: RecyclerView.LayoutManager,
        velocityX: Int,
        velocityY: Int
    ): Int {
        // Calculate the target position for snapping
        val currentPosition = super.findTargetSnapPosition(layoutManager, velocityX, velocityY)
        return if (velocityX > 0) {
            currentPosition + 1
        } else {
            currentPosition
        }
    }
}
