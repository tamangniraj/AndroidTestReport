package com.vivekahealth.vivekahealth.utils

import android.content.Context
import com.afollestad.materialdialogs.MaterialDialog

class DialogUtilsDismiss(
    val context: Context,

    val listener: DialogListenerDismiss
) {

    var materialDialog: MaterialDialog? = null

    fun dialogTitleDismiss(
        content: String,
        positiveBtn: String,
        neutralBtn: String,
        negativeBtn: String
    ) {
        MaterialDialog.Builder(context)
            .content(content).cancelable(false)
            .positiveText(positiveBtn)
            .neutralText(neutralBtn)
            .negativeText(negativeBtn)
            .onPositive { dialog, which ->
                dialog.dismiss()

                listener.onPositiveClicked(dialog)
            }.onNeutral { dialog, which ->
                listener.onNeutralClicked()
                dialog.dismiss()
            }.onNegative { dialog, which ->

                listener.onNegativeClicked()
                dialog.dismiss()
            }
            .show()
    }
}
