package com.vivekahealth.vivekahealth.utils

import android.content.Context
import androidx.annotation.LayoutRes
import com.afollestad.materialdialogs.MaterialDialog

class DialogUtilsView(
    val context: Context,

    val listener: DialogViewListener
) {

    var materialDialog: MaterialDialog? = null

    fun dialogWithViewText(
        positiveBtn: String,
        neutralBtn: String,
        negativeBtn: String,
        @LayoutRes viewId: Int
    ) {
        MaterialDialog.Builder(context)
            .customView(
                viewId,
                false
            )
            .positiveText(positiveBtn)
            .cancelable(true)
            .neutralText(neutralBtn)
            .negativeText(negativeBtn)
            .onPositive { dialog, which ->

                listener.onPositiveClicked(dialog.customView!!)
                dialog.dismiss()
            }.onNeutral { dialog, which ->
                listener.onNeutralClicked(dialog.customView!!)
                dialog.dismiss()
            }.onNegative { dialog, which ->

                listener.onNegativeClicked(dialog.customView!!)
                dialog.dismiss()
            }
            .show()
    }
}
