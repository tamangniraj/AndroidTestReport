package com.vivekahealth.vivekahealth.ui.homePage.paperless

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import androidx.lifecycle.asLiveData
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import javax.inject.Inject

class GoPaperlessViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository

) : ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    private val _paperlessRadio = MutableLiveData<Event<Boolean>>()
    val paperlessRadio: LiveData<Event<Boolean>>
        get() = _paperlessRadio
    var goPaperless: Boolean = true
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.goPaperless, null)
        }
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    var memberInfoDb = homeRepository.getMemberSid(preferences.getString("memberID", "").toString()).asLiveData()
    var memberInfo: MemberInfo = MemberInfo()
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.memberInfo, null)
        }
    var tree: String = ""
    var water: String = ""
    var waste: String = ""
    var co2: String = ""

    init {
        tree = "102.2"
        water = "275.9"
        co2 = "229.9"
        waste = "15.3"
    }

    private val _navigateToNext = MutableLiveData<Event<Boolean>>()
    val navigateToNext: LiveData<Event<Boolean>>
        get() = _navigateToNext

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    fun no() {
        goPaperless = false
        _paperlessRadio.value = Event(goPaperless)
    }
    fun yes() {
        goPaperless = true
        _paperlessRadio.value = Event(goPaperless)
    }

    fun optIN(goPaperless: Boolean) {
        compositeDisposable.add(
            homeRepository.getGoPeperless(
                preferences.getString("memberID", "").toString(),
                goPaperless
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->

                    memberInfo.optInEdd = goPaperless
                    homeRepository.updateMember(memberInfo)

                    loading = false

                    _navigateToNext.value = Event(goPaperless)
                }, {
                    loading = false
                })
        )
    }

    fun allow() {
        optIN(goPaperless)
    }
    fun deny() {
        _navigateToNext.value = Event(false)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
