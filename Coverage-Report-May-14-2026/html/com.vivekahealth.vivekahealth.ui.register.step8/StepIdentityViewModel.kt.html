    fun ssnRadio() {
        showSsn = true
        showMemberId = false

        _ssnRadio.value = Event(Unit)
    }

    fun memberId() {
        showMemberId = true
        showSsn = false

    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.register, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.register, ""),
                        RegisterUser::class.java
                    )
                    user.firstName?.trim()
                    user.lastName?.trim()
                }
            },
            {
                user.usernameType = "1"
                user.registrationType = ""
                user.electronic_documents = "false"
                if (user.overrideOldRegistration.equals("true", true)) {
                    user.overrideOldRegistration = "true"
                } else {
                    user.overrideOldRegistration = "false"
                }

                user.fcmDTO?.deviceId = deviceId
                user.fcmDTO?.deviceInfo = deviceInfo
                user.fcmDTO?.fcmId = fcmId

                ssn = user.ssn.toString()
                if (ssn.isNotEmpty()) {
                    showSsn = true

                    _ssnChangedPrev.value = Event(user.ssn.toString())
                } else {
//                        showSsn = false
                }

                memberId = user.hipaaNumber.toString()

//                    if (memberId.isNotEmpty() && ssn.isEmpty()) {
//                        showMemberId = true
//                        _idChangedPrev.value = Event(user.hipaaNumber.toString())
//
//                    } else {
//                        showMemberId = false
//
//                    }
            }
        )
    }

    fun saveReg() {
        if (ssnOrMember) {
            user.hipaaNumber = memberId
            user.ssn = ssn.replace("-", "")
        } else {
            user.ssn = ssn.replace("-", "")
        }

        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.register, regString)
        regEdit.apply()
    }

    fun removeReg() {
        val regString = gson.toJson(user)
        val regEdit = preferences.edit()
        regEdit.putString(Constants.register, "")
        regEdit.apply()
    }

    @get:Bindable

    fun verifySsn(ssnLast4: String) {
        val tempUser = user

        tempUser.ssn = ssnLast4
        Timber.d(user.toString())
        compositeDisposable.add(
            registerRepository.verifySsn(tempUser)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _navigateToVerifySSN.value = APIResponse<DataResponse>().loading()
                }
                .subscribe({
                    saveReg()
                    _navigateToVerifySSN.value = APIResponse<DataResponse>().success(it)
                }, {
                    _navigateToVerifySSN.value =
                        APIResponse<DataResponse>().error(it.toString())

//                            _showError.value =
//                                    APIResponse<SuccessResponse>().error(it.toString())
                })
        )
    }

    fun verifyRegistration() {
//            saveReg()

        Timber.d(user.toString())
        compositeDisposable.add(
            registerRepository.verifyRegisterUser(user)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _navigateToVerifyRegistration.value = APIResponse<DataResponse>().loading()
                }
                .subscribe({
                    saveReg()
                    _navigateToVerifyRegistration.value = APIResponse<DataResponse>().success(
                        it

                }, {
                    _navigateToVerifyRegistration.value =
                        APIResponse<DataResponse>().error(it.toString())

//                            _showError.value =
//                                    APIResponse<SuccessResponse>().error(it.toString())
                })
        )
    }
    fun onNextClicked() {
        if (ssnOrMember) {
            user.hipaaNumber = memberId
            user.ssn = ssn.replace("-", "")

            verifyRegistration()
        } else {
            if (user.overrideOldRegistration.equals("true")) {
                registrationAPI(true)
            } else {
                registrationAPI(false)
            }
        }
    }

    fun registrationAPI(overRide: Boolean) {
        saveReg()
        Timber.d(user.toString())
        if (!user.memberId.isNullOrBlank()) {
            user.hipaaNumber = null
        }
        var overRideReg = "false"
        overRideReg = if (user.memberId.isNullOrBlank()) {
            overRide.toString()
        } else {
            "true"
        }
        user.overrideOldRegistration = overRideReg
        compositeDisposable.add(
            registerRepository.registerUser(user)
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    _navigateToSuccess.value = APIResponse<DataResponse>().loading()
                }
                .subscribe({
                    saveReg()
                    _navigateToSuccess.value = APIResponse<DataResponse>().success(it)
                }, {
                    _navigateToSuccess.value =
                        APIResponse<DataResponse>().error(it.toString())

//                            _showError.value =
//                                    APIResponse<SuccessResponse>().error(it.toString())
                })
        )
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
