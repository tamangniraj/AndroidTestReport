package com.vivekahealth.vivekahealth.ui.homePage.findAllActivity

import androidx.lifecycle.LiveData

import java.util.ArrayList

import retrofit2.Callback
import retrofit2.Response

class FindAllActivityViewModel : BaseViewModel<SelectSpecialityVerifyNavigator>() {
    private var mutableSignup: MutableLiveData<DoctorResponseNew>? = null
    private var selectrecent: MutableLiveData<DoctorResponseNew>? = null
    private val TAG = "VerifyLoginViewModel"

    //   var ValueList1: ArrayList<SpecialityAll> = ArrayList()
    var tempListFlat: ArrayList<Any> = arrayListOf()
    var ValueListfind: ArrayList<DoctorResponseNew> = ArrayList()
    var ValueList11: ArrayList<DoctorResponseNew> = ArrayList()
    var tempListFlat1: ArrayList<Any> = arrayListOf()

    var tempListFlat11: ArrayList<Any> = arrayListOf()
    var ValueList111: ArrayList<String> = ArrayList()

    fun getProviderList(
        latitude: String,
        longitude: String,
        pageNo: String,
        pageSize: String,
        speciality: String,
        searchText: String,
        memberId: String,
        searchType: String,
        taxonomyGroup: String,
        locationType: String,
        providerType: String,
        lastDistance: String,
        codeType: String,
        codeValue: String
    ): LiveData<DoctorResponseNew> {
        mutableSignup = MutableLiveData()
        getProviderLists(
            latitude, longitude, pageNo, pageSize, speciality, searchText, memberId, searchType, taxonomyGroup,
            locationType, providerType, lastDistance, codeType, codeValue
        )
        return mutableSignup!!
    }

    private fun getProviderLists(
        latitude: String,
        longitude: String,
        pageNo: String,
        pageSize: String,
        speciality: String,
        searchText: String,
        memberId: String,
        searchType: String,
        taxonomyGroup: String,
        locationType: String,
        providerType: String,
        lastDistance: String,
        codeType: String,
        codeValue: String
    ) {
        val call: Call<DoctorResponseNew> = RetrofitClient.getApi()!!.getAllProviderList(
            latitude, longitude,
            pageNo, pageSize,
            speciality, searchText, memberId, searchType, taxonomyGroup,
            locationType, providerType, "10", codeType, codeValue
        )
        call.enqueue(object : Callback<DoctorResponseNew> {
            override fun onResponse(
                call: Call<DoctorResponseNew>,
                response: Response<DoctorResponseNew>
            ) {
                try {
                    mutableSignup?.value = response.body()
                } catch (e: Exception) {
                }
            }

            override fun onFailure(call: Call<DoctorResponseNew>, t: Throwable) {
                getNavigator()?.onError("Email or Password Invalid")
                //  Log.d(TAG, "onFailure: ${t.localizedMessage}")
            }
        })
    }
}
