    fun updateMaritalStatus(status: Int) {
        var person = Person(
            firstName = firstNameSpouse,
            lastName = lastNameSpouse,
            ssn = ssn,
            birthDate = dateOfBirthSpouse,
            gender = genderSpouse,
            dateOfMarriage = marriageDate,
            healthInsuranceAvailable = spouseInsured,
            provideInsurance = provideInsuranceSpouse,
            marriageCertificateFileId = marriageCertificateFileId,
            ssnCardFileId = ssnCardFileId
        )

        if (StringUtils.equalsIgnoreCase(martialStatus, ("Married"))) {
            maritalStatusId = 2
        } else if (StringUtils.equalsIgnoreCase(martialStatus, ("Single"))) {
            maritalStatusId = 1
        }
        compositeDisposable.add(
            homeRepository.updateMaritalStatus(
                preferences.getString("memberID", "").toString(),
                person,
                maritalStatusId
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({
                    loading = false
                    _navigateToSuccess.value = Event(it)
                }, {
                    loading = false
                })
        )
    }

    fun uploadPicture(pic: File, fileName: String) {
        val file: File = pic

        ThreadUtils.runOnUiThread(
            Runnable {
                compositeDisposable.add(
                    enrollRepository.uploadDoc(
                        filePart,
                        HelperUtils.fileName(fileName, firstNameSpouse, lastNameSpouse)
                    )
                        .subscribeOn(Schedulers.io())
                        .observeOn(AndroidSchedulers.mainThread())
                        .doOnSubscribe {
                            _navigateToUpload.value =
                                APIResponse<DocumentUploadSuccessEnroll>().loading()
                        }
                        .subscribe({
                            _navigateToUpload.value =
                                APIResponse<DocumentUploadSuccessEnroll>().success(it)
                        }, {
                            _navigateToUpload.value =
                                APIResponse<DocumentUploadSuccessEnroll>().error(
                                    it.localizedMessage!!
                                )
                        })
                )
            }
        )
    }

    fun setDocumentName(id: String, fileName: String) {
        when {
            StringUtils.containsIgnoreCase(
                fileName,
                DocumentType.Ssn.FileName.sanitize()
            ) -> {
                ssnCardFileId = id
            }
            StringUtils.containsIgnoreCase(
                fileName,
                DocumentType.MarriageCertificate.FileName.sanitize()
            ) -> {
                marriageCertificateFileId = id
            }
        }
        for (item in docList) {
            if (fileName.contains(
                    docList[docList.indexOf(item)].docName!!.FileName.sanitize(),
                    true
                )
            ) {
                docList[docList.indexOf(item)].uploaded = true
            } else if (fileName.contains(
                    docList[docList.indexOf(item)].docName!!.FileName.sanitize(),
                    true
                )
            ) {
                docList[docList.indexOf(item)].uploaded = true
            }
        }
    }

    fun checkAllDocUploaded(): Boolean {

    fun loadData() {
        job = Coroutines.ioToMainThread(
            {
                if (!StringUtils.equals(preferences.getString(Constants.enroll, ""), "")) {
                    user = gson.fromJson(
                        preferences.getString(Constants.enroll, ""),
                        EnrollMember::class.java
                    )
                }
            },
            {
                setData()
            }
        )
    }

    fun setData() {
        marriageDate = user.subscriber?.dateOfMarriage.toString()

        _datePrev.value = Event(marriageDate)
        firstNameSpouse = user.spouse?.firstName.toString()
        lastNameSpouse = user.spouse?.lastName.toString()

        genderSpouse = user.spouse?.gender.toString()
        if (genderSpouse.equals("m", true)) {
            maleClicked()
        } else if (genderSpouse.equals("f", true)) {
            femaleClicked()
        }

        dateOfBirthSpouse = user.spouse?.dateOfBirth.toString()
        _datePrevDob.value = Event(dateOfBirthSpouse)

        ssn = user.spouse?.ssn.toString()
        if (ssn.isNotEmpty()) {
            _ssnChangedPrev.value = Event(user.spouse?.ssn.toString())
        }

        spouseEmployed = user.spouse?.selfEmployed.toString()

        if (spouseEmployed.equals("true", true)) {
            employedClicked()
        } else if (spouseEmployed.equals("false", true)) {
            unEmployedClicked()
        }
        employerName = user.spouse?.employerName.toString()

        provideInsuranceSpouse = user.spouse?.healthInsuranceAvailable.toString()
        if (provideInsuranceSpouse.equals("true", true)) {
            grantClicked()
        } else if (provideInsuranceSpouse.equals("false", true)) {
            noGrantClicked()
        }

        spouseInsured = user.spouse?.healthInsuranceAvailable.toString()
        if (spouseInsured.equals("true", true)) {
            hasInsuranceClicked()
        } else if (spouseInsured.equals("false", true)) {
            hasNoInsuranceClicked()
        }
    }

    fun getMarriedDivorcedString(): String {
