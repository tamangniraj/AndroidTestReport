package com.vivekahealth.vivekahealth.ui.guestRegister.phone.di

import android.content.SharedPreferences
import com.google.gson.Gson
import com.vivekahealth.vivekahealth.ui.guestRegister.phone.GuestPhoneFragment
import com.vivekahealth.vivekahealth.ui.openEnrollment.registration.FundPhoneFragment
import com.vivekahealth.vivekahealth.ui.register.step1.service.PhoneService
import dagger.Module
import dagger.Provides
import dagger.android.ContributesAndroidInjector
import okhttp3.OkHttpClient

@Module
abstract class GuestPhoneModule {
    @ContributesAndroidInjector
    abstract fun contributesGuessPhoneFragment(): GuestPhoneFragment

    @ContributesAndroidInjector
    abstract fun contributesFundPhoneFragment(): FundPhoneFragment

    companion object {
        @JvmStatic
        @Provides
        fun providePhoneService(
            gson: Gson,
            builder: OkHttpClient.Builder,
            sharedPreferences: SharedPreferences
        ): PhoneService = PhoneService(
            gson = gson,
            builder = builder,
            preferences = sharedPreferences
        )
    }
}
