package com.vivekahealth.vivekahealth.pension.data.model

data class PensionCardData(
    val period: String,
    val creditsEarned: Int,
    val vestingChips: Int,
    val status: WorkStatus,
    val bottomSection: BottomSection? = null
)

data class BottomSection(
    val title: String,
    val description: String
)

enum class WorkStatus(val groupType: String) {
    VESTED("Active"),
    BREAK("Break"),
    FORFEITED("Forfeited");

    companion object {
        fun fromGroupType(type: String): WorkStatus? =
            entries.find { it.groupType == type }
    }
}
