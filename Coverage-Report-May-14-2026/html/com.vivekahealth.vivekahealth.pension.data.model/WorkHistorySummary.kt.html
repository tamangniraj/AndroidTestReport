package com.vivekahealth.vivekahealth.pension.data.model

import com.google.gson.annotations.SerializedName

data class WorkHistorySummary(
    @SerializedName("memberId")
    val memberId: Long,

    @SerializedName("memberName")
    val memberName: String,

    @SerializedName("plans")
    val plans: List<PlanDetail>,

    @SerializedName("totalCreditsAllPlans")
    val totalCreditsAllPlans: Double,

    @SerializedName("totalVestingAllPlans")
    val totalVestingAllPlans: Double,

    @SerializedName("totalForfeitedAllPlans")
    val totalForfeitedAllPlans: Double,

    @SerializedName("vestingStatusMessage")
    val vestingStatusMessage: String
)

data class PlanDetail(
    @SerializedName("planId")
    val planId: String,

    @SerializedName("planName")
    val planName: String,

    @SerializedName("yearGroups")
    val yearGroups: List<YearGroup>,

    @SerializedName("totalCredits")
    val totalCredits: Double,

    @SerializedName("totalVesting")
    val totalVesting: Double,

    @SerializedName("totalForfeited")
    val totalForfeited: Double,

    @SerializedName("isVested")
    val isVested: Boolean,

    @SerializedName("planRules")
    val planRules: PlanRules
)

data class YearGroup(
    @SerializedName("startYear")
    val startYear: Int,

    @SerializedName("endYear")
    val endYear: Int,

    @SerializedName("period")
    val period: String,

    @SerializedName("yearCount")
    val yearCount: Int,

    @SerializedName("groupType")
    val groupType: String,

    @SerializedName("totalCredits")
    val totalCredits: Double,

    @SerializedName("totalVesting")
    val totalVesting: Double,

    @SerializedName("totalForfeited")
    val totalForfeited: Double,

    @SerializedName("isVested")
    val isVested: Boolean,

    @SerializedName("singleYear")
    val singleYear: Boolean,

    @SerializedName("forfeitureExplanation")
    val forfeitureExplanation: String,

    @SerializedName("note")
    var note: Notes ? = null
)

data class Notes(
    @SerializedName("title")
    val title: String ? = "",

    @SerializedName("text")
    val text: String ? = ""
)

data class PlanRules(
    @SerializedName("planId")
    val planId: String,

    @SerializedName("planShortName")
    val planShortName: String,

    @SerializedName("planName")
    val planName: String,

    @SerializedName("breakThresholdHours")
    val breakThresholdHours: Int,

    @SerializedName("vestingYearsPreCutoff")
    val vestingYearsPreCutoff: Int,

    @SerializedName("vestingYearsPostCutoff")
    val vestingYearsPostCutoff: Int,

    @SerializedName("vestingCutoffDate")
    val vestingCutoffDate: String,

    @SerializedName("ruleDescriptions")
    val ruleDescriptions: List<String>
)
