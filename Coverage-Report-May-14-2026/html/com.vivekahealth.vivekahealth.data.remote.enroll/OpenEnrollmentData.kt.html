package com.vivekahealth.vivekahealth.data.remote.enroll

import com.vivekahealth.vivekahealth.data.remote.guest.GuestRegisterUser
import com.vivekahealth.vivekahealth.data.remote.guest.Source
import com.vivekahealth.vivekahealth.ui.enrollVTwo.Enrollment
import com.vivekahealth.vivekahealth.ui.enrollVTwo.UploadedDocument
import com.vivekahealth.vivekahealth.ui.openEnrollment.OpenEnrollSharedViewModel
import com.vivekahealth.vivekahealth.utils.DateUtils

data class OpenEnrollmentData(
    var subscriber: Person? = Person(),
    var spouse: Person? = Person(),
    var children: ArrayList<Person> = arrayListOf(),
    var employment: EmploymentData,
    var type: Int? = null

)

data class EmploymentData(
    val employerName: String = "",
    val selectedPlan: String = "",
    val jobTitle: String = "",
    val dateOfHire: String = ""
)

fun mapToOpenEnrollMember(
    enrollment: Enrollment,
    employment: OpenEnrollSharedViewModel.EmploymentData,
    register: GuestRegisterUser? = null,
    selectedPlan: String,
    dlData: OpenEnrollSharedViewModel.DLData?,
    ssn: String?,
    primaryDocumentList: MutableList<UploadedDocument>?
): OpenEnrollmentData {
    val spouse = enrollment.findSpouse()
    val primary = enrollment.findPrimary()
    val childs = enrollment.findChilds()

    val zips: List<String> = dlData?.zip?.split("-") ?: emptyList()
    var zip: String ? = null
    var zipExtension: String ? = null
    if (zips.isNotEmpty()) {
        if (zips[0].length == 5) {
            zip = zips[0]
        }
        if (zips.size > 1) {
            if (zips[1].length == 4) {
                zipExtension = zips[1]
            }
        }
    }

    return OpenEnrollmentData(
        subscriber = Person(
            signatureFileId = primary?.signatureFileId,
            uploadedDocuments = primaryDocumentList,
            gender = when (dlData?.gender?.uppercase()) { "MALE" -> "M"; "FEMALE" -> "F"; else -> dlData?.gender }, // step 0
            firstName = dlData?.firstName,
            lastName = dlData?.lastName,
            dateOfBirth = DateUtils.toAPIDate(dlData?.dob),
            ssn = ssn,
            street1 = dlData?.streetAddress,
            street2 = dlData?.apt,
            city = dlData?.city,
            state = dlData?.state,
            zip = zip,
            zipExtension = zipExtension,
            country = "US",
            phoneNumber = (register?.phone ?: ""),
            emailAddress = register?.email

        ),
        spouse = spouse?.let {
            Person(
                firstName = it.firstName, // step1
                lastName = it.lastName, // step1
                dateOfMarriage = it.marriageDate, // step2
                gender = it.gender, // step3
                dateOfBirth = it.dob, // step4
                ssn = it.ssn, // step 5
                uploadedDocuments = it.uploadedList
            )
        },
        children = childs?.map { child ->
            Person(
                firstName = child.firstName, // step 1
                lastName = child.lastName, // step 1
                gender = child.gender, // step 2
                dateOfBirth = child.dob, // step 3
                ssn = child.ssn, // step 4
                uploadedDocuments = child.uploadedList
            )
        }?.toCollection(ArrayList()) ?: arrayListOf(),
        employment = EmploymentData(
            employment.employerName,
            selectedPlan = selectedPlan,
            employment.jobTitle,
            employment.dateOfHire
        ),
        type = Source.OPEN_ENROLLMENT.value
    )
}
