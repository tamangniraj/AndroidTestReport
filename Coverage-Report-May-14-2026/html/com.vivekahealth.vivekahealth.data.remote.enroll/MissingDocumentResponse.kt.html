package com.vivekahealth.vivekahealth.data.remote.enroll

import android.os.Parcelable
import com.google.gson.annotations.SerializedName
import kotlinx.parcelize.Parcelize

@Parcelize
data class RootMissingDocumentResponse(
    val documentList: List<MissingDocumentPerson>
) : Parcelable

@Parcelize
data class MissingDocumentPerson(
    @SerializedName("selfId") val selfId: Long? = null,
    @SerializedName("firstName") val firstName: String? = null,
    @SerializedName("lastName") val lastName: String? = null,
    @SerializedName("relationshipType") val relationshipType: Int? = null,
    @SerializedName("documentDetails") val documentDetails: List<MissingDocumentDetail>? = null
) : Parcelable

@Parcelize
data class MissingDocumentDetail(
    @SerializedName("documentType") val documentType: Int? = null,
    @SerializedName("documentName") val documentName: String? = null,
    @SerializedName("translatedDocumentName") val translatedDocumentName: String? = null
) : Parcelable
