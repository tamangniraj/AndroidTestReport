package com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.repository

import com.vivekahealth.vivekahealth.data.local.home.MemberInfoDao
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.service.FindCareSearchService
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityModel.BillingCodePriceResponse
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityModel.BillingData
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityModel.Specialties
import io.reactivex.Single
import javax.inject.Inject
import javax.inject.Singleton
import kotlinx.coroutines.flow.Flow

/**
 * Repository that handles User objects.
 */
@Singleton
class FindCareSearchRepository @Inject constructor(
    private val findCareService: FindCareSearchService,
    private val dao: MemberInfoDao
) {

    fun getPriceFromBillingList(
        billingIds: List<String>,
        latitude: String,
        longitude: String,
        within: String
    ): Single<BillingCodePriceResponse> = findCareService
        .getFindCareSearchApi().getPricingDetailsViaBillingCode(
            BillingData(billingIds),
            latitude,
            longitude,
            within
        )

    fun getProcedureList(
        search: String,
        memberID: String,
        latitude: String,
        longitude: String,
        provider_type: String,
        within: String,
        searchScope: String? = null,
        displayPrice: Boolean ? = true
    ): Single<Specialties> = findCareService.getFindCareSearchApi().getFindCareSearchList(
        search,
        memberID,
        latitude,
        longitude,
        provider_type,
        within,
        searchScope,
        displayPrice = displayPrice
    )

    fun getMemberFromDb(): Flow<List<MemberInfo>> {
        return dao.getAll()
    }

    fun getMemberSid(memberid: String): Flow<MemberInfo> {
        return dao.getMemberInfoLiveData(memberid)
    }
}
