package com.vivekahealth.vivekahealth.ui.homePage.addDependent.dependentList

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.disposables.CompositeDisposable
import javax.inject.Inject

class DependentListSheetViewModel @Inject constructor(private val preferences: SharedPreferences) :
    ViewModel(),
    Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    var selectedItem: MemberInfo = MemberInfo()
    var sortAction: String = ""
    var sortBy: String = ""

    private val _listGeneratedNew = MutableLiveData<Event<ArrayList<MemberInfo>>>()
    val listGeneratedNew: LiveData<Event<ArrayList<MemberInfo>>>
        get() = _listGeneratedNew
    var listMember: ArrayList<MemberInfo> = arrayListOf()
    private val _navigateToList = MutableLiveData<Event<Unit>>()
    val navigateToList: LiveData<Event<Unit>>
        get() = _navigateToList
    private val _navigateToClose = MutableLiveData<Event<Unit>>()
    val navigateToClose: LiveData<Event<Unit>>
        get() = _navigateToClose

    var shouldEnableSelect: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.shouldEnableSelect, null)
        }

    fun apply() {
        _navigateToList.value = Event(Unit)
    }
    fun close() {
        _navigateToClose.value = Event(Unit)
    }

    fun setItemSelected(item: MemberInfo) {
        var index = listMember.indexOf(item)
        for (member in listMember) {
            if (listMember.indexOf(member) == index) {
                listMember[index].apply {
                    isSelected = !isSelected
                }
            } else {
                listMember[listMember.indexOf(member)].apply {
                    isSelected = false
                }
            }
        }
        shouldEnableSelect = listMember.any { it.isSelected }
        _listGeneratedNew.value = Event(listMember)
    }

    fun clearSelection() {
        for (member in listMember) {
            member.isSelected = false
        }
        _listGeneratedNew.value = Event(listMember)
        selectedItem = MemberInfo()
        shouldEnableSelect = false
    }
    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
