package com.vivekahealth.vivekahealth.ui.openEnrollment.enrollment

import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.vivekahealth.vivekahealth.data.local.enroll.ListDoc
import com.vivekahealth.vivekahealth.data.remote.enroll.DocumentUploadSuccessEnroll
import com.vivekahealth.vivekahealth.data.remote.enroll.MissingDocumentPerson
import com.vivekahealth.vivekahealth.data.remote.enroll.Person
import com.vivekahealth.vivekahealth.ui.guestHome.repository.GuestHomeRepository
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.remote.APIResponse
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.launch
import okhttp3.MultipartBody
import javax.inject.Inject

class MissingRequireDocumentViewModel @Inject constructor(
    private val homeRepository: HomeRepository,
    private val guestHomeRepository: GuestHomeRepository
) : ViewModel() {

    private val compositeDisposable = CompositeDisposable()

    private val _documents = MutableLiveData<List<MissingDocumentPerson>>()
    val documents: LiveData<List<MissingDocumentPerson>> get() = _documents

    private val _loading = MutableLiveData<Boolean>()
    val loading: LiveData<Boolean> get() = _loading

    private val _error = MutableLiveData<Event<String>>()
    val error: LiveData<Event<String>> get() = _error

    private val _uploadResult = MutableLiveData<APIResponse<DocumentUploadSuccessEnroll>>()
    val uploadResult: LiveData<APIResponse<DocumentUploadSuccessEnroll>> get() = _uploadResult

    private val _navigateToSuccess = MutableLiveData<Event<Unit>>()
    val navigateToSuccess: LiveData<Event<Unit>> get() = _navigateToSuccess

    var currentItem: ListDoc = ListDoc()
    var currentPerson: Person = Person()
    lateinit var filePart: MultipartBody.Part

    private var debounceJob: Job? = null
    private var isDebouncing = false

    fun fetchRequiredDocuments() {
        compositeDisposable.add(
            guestHomeRepository.getMissingRequiredDocuments()
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe { _loading.value = true }
                .subscribe({ response ->
                    _documents.value = response.documentList
                    _loading.value = false
                }, { throwable ->
                    _loading.value = false
                    _error.value = Event(
                        throwable.localizedMessage ?: "An unexpected error occurred."
                    )
                })
        )
    }

    fun uploadDocument(enrollmentId: Long? = null) {
        singleClick {
            compositeDisposable.add(
                guestHomeRepository.uploadOpenFile(
                    filePart,
                    currentItem.id.toString(),
                    enrollmentId
                )
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
                    .doOnSubscribe {
                        _uploadResult.value = APIResponse<DocumentUploadSuccessEnroll>().loading()
                    }
                    .subscribe({
                        _uploadResult.value = APIResponse<DocumentUploadSuccessEnroll>().success(it)
                    }, {
                        _uploadResult.value = APIResponse<DocumentUploadSuccessEnroll>().error(
                            it.localizedMessage ?: "An unexpected error occurred."
                        )
                    })
            )
        }
    }

    fun onDoneClicked() {
        singleClick { _navigateToSuccess.value = Event(Unit) }
    }

    private fun singleClick(action: () -> Unit) {
        if (isDebouncing) return
        action()
        isDebouncing = true
        debounceJob?.cancel()
        debounceJob = viewModelScope.launch {
            delay(500L)
            isDebouncing = false
        }
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
