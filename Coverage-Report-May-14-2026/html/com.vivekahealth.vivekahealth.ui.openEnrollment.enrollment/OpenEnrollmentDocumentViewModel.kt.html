package com.vivekahealth.vivekahealth.ui.openEnrollment.enrollment

import android.graphics.Bitmap
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.google.android.gms.tasks.OnCompleteListener
import com.google.firebase.messaging.FirebaseMessaging
import com.vivekahealth.vivekahealth.data.local.enroll.RootDocumentResponse
import com.vivekahealth.vivekahealth.ui.enrollVTwo.UploadedDocument
import com.vivekahealth.vivekahealth.ui.guestHome.repository.GuestHomeRepository
import com.vivekahealth.vivekahealth.ui.guestVerify.documents.DocumentType
import com.vivekahealth.vivekahealth.utils.DateUtils
import com.vivekahealth.vivekahealth.utils.Event
import com.vivekahealth.vivekahealth.utils.HelperUtils
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import okhttp3.MultipartBody
import java.util.UUID
import javax.inject.Inject

class OpenEnrollmentDocumentViewModel @Inject constructor(
    private val guestHomeRepository: GuestHomeRepository
) : ViewModel() {

    private val compositeDisposable = CompositeDisposable()

    private val _requiredDocs = MutableLiveData<RootDocumentResponse>()
    val requiredDocs: LiveData<RootDocumentResponse> get() = _requiredDocs

    private val _error = MutableLiveData<Event<Unit>>()
    val error: LiveData<Event<Unit>> get() = _error

    private var isFetchingDocs = false

    fun fetchRequiredDocuments() {
        if (isFetchingDocs || _requiredDocs.value != null) return
        isFetchingDocs = true
        compositeDisposable.add(
            guestHomeRepository.getRequiredDocuments()
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .subscribe({ docs ->
                    isFetchingDocs = false
                    _requiredDocs.value = docs
                }, {
                    isFetchingDocs = false
                    _error.value = Event(Unit)
                })
        )
    }

    val primaryUploadList: MutableList<UploadedDocument>? = mutableListOf()

    fun uploadDocument(bitmap: Bitmap, documentType: DocumentType) {
        val (docTypeId, docTypeName) = when (documentType) {
            DocumentType.DL_FRONT -> 103 to "Front of Driver License"
            DocumentType.DL_BACK -> 104 to "Back of Driver License"
            DocumentType.SSN -> 71 to "SSN Card"
            else -> return
        }

        val filePart: MultipartBody.Part = HelperUtils.createMultipartBodyPartFromBitmap(
            bitmap = bitmap,
            "file",
            "${UUID.randomUUID()}-$docTypeName-${DateUtils.getCurrentLocalTime()}"
        )

        compositeDisposable.add(
            guestHomeRepository.uploadOpenFile(filePart, docTypeId.toString())
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .subscribe({ response ->
                    val uploadedFileId = response.id?.toIntOrNull() ?: return@subscribe
                    primaryUploadList?.add(
                        UploadedDocument(
                            fileTypeId = docTypeId,
                            uploadedFileReferenceId = uploadedFileId
                        )
                    )
                }, {
                    // fire-and-forget: failure is silent; caller proceeds regardless
                })
        )
    }

    fun updateGuestToken() {
        FirebaseMessaging.getInstance().token.addOnCompleteListener(
            OnCompleteListener { task ->
                if (!task.isSuccessful) {
                    return@OnCompleteListener
                }
                val token = task.result
                compositeDisposable.add(
                    guestHomeRepository.updateGuestFcmToken("", token, true)
                        .observeOn(AndroidSchedulers.mainThread())
                        .subscribeOn(Schedulers.io())
                        .subscribe({}, {})
                )
            }
        )
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
