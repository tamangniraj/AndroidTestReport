package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.prescription.prescriptionFilter

import android.app.Dialog
import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.lifecycle.ViewModelProvider
import com.google.android.material.bottomsheet.BottomSheetBehavior
import com.google.android.material.bottomsheet.BottomSheetDialog
import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.AvailPrescriptionFiltersItem
import com.vivekahealth.vivekahealth.databinding.PrescriptionFilterFragmentBinding
import com.vivekahealth.vivekahealth.di.Injectable
import com.vivekahealth.vivekahealth.ui.BaseBottomSheetDialogFragment
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.prescription.adapter.PrescriptionFilterAdapter
import com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.prescription.navagator.PrescriptionFilterListener
import com.vivekahealth.vivekahealth.utils.EventObserver
import javax.inject.Inject

class PrescriptionFilter : BaseBottomSheetDialogFragment(), Injectable {

    companion object {
        fun newInstance() = PrescriptionFilter()
        const val TAG = "PrescriptionFilter"
    }
    private val adapterName = PrescriptionFilterAdapter(onClick = {

        if (it.selected) {
            viewModel.drugNameListID.add(it)
        } else {
            if (viewModel.drugNameListID.contains(it)) {
                viewModel.drugNameListID.remove(it)
            }
        }
        viewModel.count = viewModel.drugNameListID.count() + viewModel.typeListID.count()
    })
    private val adapterType = PrescriptionFilterAdapter(onClick = {
        if (it.selected) {
            viewModel.typeListID.add(it)
        } else {
            if (viewModel.typeListID.contains(it)) {
                viewModel.typeListID.remove(it)
            }
        }
        viewModel.count = viewModel.drugNameListID.count() + viewModel.typeListID.count()
    })
    var filterListener: PrescriptionFilterListener? = null

    var filterOption: List<AvailPrescriptionFiltersItem>? = null

    fun setCallback(
        filterListener: PrescriptionFilterListener,
        filterOption: List<AvailPrescriptionFiltersItem>?
    ) {
        this.filterListener = filterListener
        this.filterOption = filterOption
    }
    private lateinit var binding: PrescriptionFilterFragmentBinding

    private lateinit var viewModel: PrescriptionFilterViewModel
    var sentfilter = false

    @Inject
    lateinit var viewModelFactory: ViewModelProvider.Factory

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        binding = PrescriptionFilterFragmentBinding.inflate(inflater, container, false)
        return binding.root
    }
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
    }

    override fun onCreateDialog(savedInstanceState: Bundle?): Dialog {
        val dialog = super.onCreateDialog(savedInstanceState) as BottomSheetDialog

        dialog.setOnShowListener {
            val bottomSheet = dialog.findViewById<View>(
                com.google.android.material.R.id.design_bottom_sheet
            )

            bottomSheet?.let {
                val behavior = BottomSheetBehavior.from(it)

                behavior.state = BottomSheetBehavior.STATE_EXPANDED
                behavior.isHideable = true
                behavior.skipCollapsed = true
                behavior.isDraggable = false
            }
        }
        return dialog
    }

    override fun onActivityCreated(savedInstanceState: Bundle?) {
        super.onActivityCreated(savedInstanceState)
        viewModel = ViewModelProvider(this, viewModelFactory).get(
            PrescriptionFilterViewModel::class.java
        )

        binding.viewModel = viewModel

        val inStore = arguments?.getBoolean("inStoreSelected")
        val mailOrder = arguments?.getBoolean("mailOrderSelected")

        if (!filterOption.isNullOrEmpty()) {
            val groupedName = filterOption!!.filter { value ->
                value.filterType.equals("drugName", true)
            }

            val groupedType = filterOption!!.filter { value ->
                !value.filterType.equals("drugName", true)
            }

            viewModel.groupedName = groupedName
            viewModel.groupedType = groupedType

            var count = 0
            for (item in groupedName) {
                if (item.selected) {
                    count += 1
                    viewModel.drugNameListID.add(item)
                }
            }

            for (item in groupedType) {
                if (item.selected) {
                    count += 1
                    viewModel.typeListID.add(item)
                }
            }
            viewModel.count = count

            viewModel.showTypeHeader = groupedType.isNotEmpty()
            viewModel.showNameHeader = groupedName.isNotEmpty()

            binding.filterList.adapter = adapterName
            binding.filterList.hasFixedSize()

            binding.typeList.adapter = adapterType
            binding.typeList.hasFixedSize()
            adapterName.submitList(viewModel.groupedName)

            adapterType.submitList(viewModel.groupedType)
        }

//        if (inStore != null) {
//            viewModel.inStore = inStore
//
//        }
//
//        if (mailOrder != null) {
//            viewModel.mailOrder = mailOrder
//
//        }
//

        viewModel.navigateToClear.observe(
            viewLifecycleOwner,
            EventObserver {
                if (!viewModel.groupedName.isNullOrEmpty()) {
                    adapterName.submitList(viewModel.groupedName)
                    adapterType.notifyDataSetChanged()
                }

                if (!viewModel.groupedType.isNullOrEmpty()) {
                    adapterType.submitList(viewModel.groupedType)
                    adapterName.notifyDataSetChanged()
                }
            }
        )
        viewModel.navigateToPres.observe(
            viewLifecycleOwner,
            EventObserver {
                filterListener?.onClose()
            }
        )
        viewModel.navigateToDocFilter.observe(
            viewLifecycleOwner,
            EventObserver {
//            if(!viewModel.drugNameListIDFinal.isNullOrEmpty() or !viewModel.typeListIDFinal.isNullOrEmpty()) {
                filterListener!!.applyFilter(
                    viewModel.drugNameListIDFinal,
                    viewModel.typeListIDFinal
                )
//            }
            }
        )
    }
}
