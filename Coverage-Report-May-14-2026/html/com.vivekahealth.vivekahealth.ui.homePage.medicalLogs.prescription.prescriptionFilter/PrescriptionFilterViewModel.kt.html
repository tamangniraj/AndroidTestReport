package com.vivekahealth.vivekahealth.ui.homePage.medicalLogs.prescription.prescriptionFilter

import android.content.SharedPreferences
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.vivekahealth.vivekahealth.data.local.VivekaDatabase
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import com.vivekahealth.vivekahealth.data.remote.home.medicalLogs.AvailPrescriptionFiltersItem
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.disposables.CompositeDisposable
import java.util.*
import javax.inject.Inject
import kotlin.collections.ArrayList

class PrescriptionFilterViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository,
    private val vivekaDatabase: VivekaDatabase
) : ViewModel(), Observable {

    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }

    private var compositeDisposable: CompositeDisposable = CompositeDisposable()

    private val _navigateToPres = MutableLiveData<Event<Unit>>()
    val navigateToPres: LiveData<Event<Unit>>
        get() = _navigateToPres
    private val _navigateToDocFilter = MutableLiveData<Event<Unit>>()
    val navigateToDocFilter: LiveData<Event<Unit>>
        get() = _navigateToDocFilter

    private val _navigateToClear = MutableLiveData<Event<Unit>>()
    val navigateToClear: LiveData<Event<Unit>>
        get() = _navigateToClear

    private val _dependentListGenerated = MutableLiveData<Event<List<MemberInfo>>>()
    val dependentListGenerated: LiveData<Event<List<MemberInfo>>>
        get() = _dependentListGenerated

    var drugNameList: ArrayList<AvailPrescriptionFiltersItem> = arrayListOf()
    var typeList: ArrayList<AvailPrescriptionFiltersItem> = arrayListOf()

    var listOption: ArrayList<AvailPrescriptionFiltersItem> = arrayListOf()
    var drugNameListID: ArrayList<AvailPrescriptionFiltersItem> = arrayListOf()
    var typeListID: ArrayList<AvailPrescriptionFiltersItem> = arrayListOf()
    var drugNameListIDFinal: ArrayList<AvailPrescriptionFiltersItem> = arrayListOf()
    var typeListIDFinal: ArrayList<AvailPrescriptionFiltersItem> = arrayListOf()
    var count: Int = 0
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.count, null)
        }

    var showTypeHeader: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showTypeHeader, null)
        }
    var showNameHeader: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showNameHeader, null)
        }

    var showType: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showType, null)
        }
    var showName: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.showName, null)
        }
    var mailOrder: Boolean = false
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.mailOrder, null)
        }

    var inStore: Boolean = false
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, BR.inStore, null)
        }

    var groupedName: List<AvailPrescriptionFiltersItem>? = null
    var groupedType: List<AvailPrescriptionFiltersItem>? = null

    fun name() {
        showName = !showName
    }
    fun type() {
        showType = !showType
    }

    fun apply() {
        drugNameListIDFinal = drugNameListID
        typeListIDFinal = typeListID
        _navigateToDocFilter.value = Event(Unit)
    }

    fun close() {
        _navigateToPres.value = Event(Unit)
    }

    fun clear() {
        typeListID.clear()
        drugNameListID.clear()
        count = 0

        if (!groupedType.isNullOrEmpty()) {
            groupedType!!.forEach { it.selected = false }
        }

        if (!groupedName.isNullOrEmpty()) {
            groupedName!!.forEach { it.selected = false }
        }
        _navigateToClear.value = Event(Unit)
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }

    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
