    fun loadData(memberInfo: MemberInfo) {
        val member = memberInfo

        var callingCode = ""
        var numberPhone = ""
        member.phoneNumber?.let { number ->
            numberPhone = number
            member.countryCode?.let {
                callingCode = it
            }
        }

        hasAvatar = memberInfo.hasAvatar.toBooleanStrict()

        name = member.lastName + ", " + member.firstName
        member.email?.let {
            email = it
        }

        address = member.homeAddress?.fullAddress() ?: ""

        phoneNumber = "$callingCode $numberPhone"

        member.maritalStatus?.let {
            martialStat = it
        }
        var dependentList = ""

        createDependentList()
        if (dependentOrdered.isNotEmpty()) {
            for (list in dependentOrdered.filter { !it.hideProfileFromHome && !it.memberAdditionInReview }) {
                dependentList = if (dependentList.isNotBlank()) {
                    if (dependentOrdered.indexOf(list) != dependentOrdered.size - 1) {
                        dependentList + ", " + list.firstName
                    } else {
                        dependentList + " and " + list.firstName
                    }
                } else {
                    list.firstName.toString()
                }
            }
        }

        family = dependentList
    }

    private fun createDependentList() {
        dependentOrdered.clear()

        memberInfo.dependentsArrayList?.let { it1 ->

            memberInfo.dependentsArrayList?.sortByDescending {
                DateUtils.stringServerDateToDate(
                    it.birthDate.toString()
                )
            }

            memberInfo.dependentsArrayList?.reverse()

            for (member in it1) {
                if (StringUtils.equalsIgnoreCase(member.memberRelationship, "Spouse")) {
                    dependentOrdered.add(0, member)
                } else {
                    dependentOrdered.add(member)
                }
            }
        }
    }

    fun getCode(countryName: String) =
        Locale.getISOCountries().find { Locale("", it).displayCountry == countryName }

    fun getCodeHash(countryName: String): String? {
        // Get all country codes in a string array.
        val isoCountryCodes = Locale.getAvailableLocales()

        // Iterate through all country codes:
        for (code in isoCountryCodes) {
            // Create a locale using each country code // Get country name for each code.

            if (code.displayName.equals(countryName)) {
                return code.country
            }
        }

        return "1"
    }

    fun getCountryCode(countryName: String): String {
        val isoCountryCodes: Array<String> = Locale.getISOCountries()
        for (code in isoCountryCodes) {
            val locale = Locale("", code)
            if (countryName.equals(locale.displayCountry, ignoreCase = true)) {
                return code
            }
        }
        return "1"
    }

    fun getCountryDialCode(code: String): String {
        var countryDialCode = ""
        for (i in countryList.indices) {
            val arrDial = countryList[i].split(",".toRegex()).toTypedArray()
            if (arrDial[1].trim { it <= ' ' } == code.trim()) {
                countryDialCode = arrDial[0]
                break
            }
        }
        return countryDialCode
    }

    private var addedAddressAndMaritalTut = false
    private var addedDependentTut = false
    private var addedOtherTutorial = false
    var tutorialShown = false
    fun addAddressAnMaritalTutorial() {
        addedAddressAndMaritalTut = true
        showTutorialIfConditionMatched()
    }

    fun addDependentTutorial() {
        addedDependentTut = true
        showTutorialIfConditionMatched()
    }

    fun addOtherTutorial() {
        addedOtherTutorial = true
        updateTutorialAdditionStatus()
    }

    private fun updateTutorialAdditionStatus() {
        _tutorialsAdded.value = addedAddressAndMaritalTut && addedDependentTut && addedOtherTutorial
    }

    private fun showTutorialIfConditionMatched() {
        if (!preferences.getBoolean(
                Constants.SHOWN_ACC_DET_TUT_FOR_SESSION,
                false
            ) && !tutorialShown && addedAddressAndMaritalTut && addedDependentTut
        ) {
            _showTutorial.value = Event(Unit)
            tutorialShown = true
            preferences.edit().putBoolean(Constants.SHOWN_ACC_DET_TUT_FOR_SESSION, true).apply()
        }
    }

    fun dontShowTutorial() {
        preferences.edit().putBoolean(memberInfo.selfId + "accountDetailTutorialShown", true)
            .apply()
    }

    fun resetTutorial() {
        _tutorialsAdded.value = false
    }

    fun shouldShowTutorial(): Boolean {
        return !preferences.getBoolean(memberInfo.selfId + "accountDetailTutorialShown", false)
    }
