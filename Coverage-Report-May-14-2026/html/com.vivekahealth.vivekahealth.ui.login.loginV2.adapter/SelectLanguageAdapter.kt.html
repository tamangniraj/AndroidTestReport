package com.vivekahealth.vivekahealth.ui.login.loginV2.adapter

import android.annotation.SuppressLint
import android.content.Context
import android.util.TypedValue
import android.view.LayoutInflater
import android.view.ViewGroup
import android.widget.Filter
import android.widget.Filterable
import androidx.recyclerview.widget.RecyclerView
import com.vivekahealth.vivekahealth.R
import com.vivekahealth.vivekahealth.databinding.LanguageRowBinding

class SelectLanguageAdapter(
    context: Context,
    private val languageList: ArrayList<SelectLanguageItem>,
    private var onLangClick: (String) -> Unit
) :
    RecyclerView.Adapter<SelectLanguageViewHolder>(), Filterable {
    private val currentLocale = context.resources.configuration.locale.language
    private val originalList = languageList
    private var filteredList = languageList

    private val searchFilter: Filter = object : Filter() {
        override fun performFiltering(input: CharSequence): FilterResults {
            val query = input.toString().lowercase().trim()

            val filteredList = if (input.isEmpty()) {
                originalList
            } else {
                originalList.filter {
                    it.originalTitle.lowercase().contains(query)
                }
            }
            return FilterResults().apply { values = filteredList }
        }

        @SuppressLint("NotifyDataSetChanged")
        @Suppress("UNCHECKED_CAST")
        override fun publishResults(input: CharSequence, results: FilterResults) {
            (results.values as ArrayList<SelectLanguageItem>).also {
                filteredList = it
                notifyDataSetChanged()
            }
        }
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): SelectLanguageViewHolder {
        val binding = LanguageRowBinding.inflate(LayoutInflater.from(parent.context), parent, false)
        return SelectLanguageViewHolder(binding)
    }

    override fun getItemCount(): Int {
        return filteredList.size
    }

    override fun onBindViewHolder(holder: SelectLanguageViewHolder, position: Int) {
        with(filteredList[position]) {
            holder.bindItem(this, currentLocale)
            holder.binding.root.setOnClickListener {
                onLangClick(this.languageCode)
            }
        }
    }

    override fun getFilter(): Filter {
        return searchFilter
    }
}

class SelectLanguageViewHolder(val binding: LanguageRowBinding) :
    RecyclerView.ViewHolder(binding.root) {
    fun bindItem(item: SelectLanguageItem, currentLocale: String) {
        binding.translatedTextView.text = item.translatedTitle
        binding.originalTextView.text = item.originalTitle
        if (item.languageCode == currentLocale) {
            binding.root.setBackgroundColor(binding.root.context.getColor(R.color.colorAccent))
        } else {
            val outValue = TypedValue()
            binding.root.context.theme
                .resolveAttribute(android.R.attr.selectableItemBackground, outValue, true)
            binding.root.setBackgroundResource(outValue.resourceId)
        }
    }
}

data class SelectLanguageItem(
    val translatedTitle: String,
    val originalTitle: String,
    val languageCode: String
)
