    fun increaseQty() {
        var qty1 = childNumber.toInt()
        if (qty1 < 20) {
            qty1 += 1

            childNumber = qty1.toString()
            user.children.add(Person())
            _childChanged.value = Event(user)
        }
    }

    fun decreaseQty() {
        var qty1 = childNumber.toInt()
        if (qty1 > 0) {
            qty1 -= 1

            childNumber = qty1.toString()

            if (user.children.isNotEmpty()) {
                user.children.removeAt(user.children.size - 1)
            }
            val regString = gson.toJson(user)
            val regEdit = preferences.edit()
            regEdit.putString(Constants.enroll, regString)
            regEdit.apply()
            _childChanged.value = Event(user)
        }
    }

    fun childCheck(): Boolean {
        for (child in user.children) {
            return child.firstName!!.isNotEmpty() && child.lastName!!.isNotEmpty() && child.gender!!.isNotEmpty() && child.dateOfBirth!!.isNotEmpty() && (child.ssn!!.isNotEmpty() || child.ssnOptional) && child.selfEmployed!!.isNotEmpty()
        }
        return false
    }

    fun checkIfAllComplete(): Boolean {
        for (child in user.children) {
            if (((child.ssn.isNullOrBlank() && !child.ssnOptional) || child.selfEmployed.isNullOrBlank() || child.gender.isNullOrBlank() || child.firstName.isNullOrBlank() || child.lastName.isNullOrBlank())) {
