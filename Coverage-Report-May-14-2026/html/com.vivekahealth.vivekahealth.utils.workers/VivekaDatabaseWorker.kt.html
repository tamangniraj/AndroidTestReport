package com.vivekahealth.vivekahealth.utils.workers

import android.content.Context
import android.util.Log
import androidx.work.CoroutineWorker
import androidx.work.WorkerParameters
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import com.google.gson.stream.JsonReader
import com.vivekahealth.vivekahealth.data.remote.home.MemberInfo
import kotlinx.coroutines.coroutineScope

class VivekaDatabaseWorker(
    context: Context,
    workerParams: WorkerParameters
) : CoroutineWorker(context, workerParams) {
    override suspend fun doWork(): Result = coroutineScope {
        try {
            applicationContext.assets.open("member.json").use { inputStream ->
                JsonReader(inputStream.reader()).use { jsonReader ->
                    val Type = object : TypeToken<List<MemberInfo>>() {}.type
                    val plantList: List<MemberInfo> = Gson().fromJson(jsonReader, Type)

//                    val database = VivekaDatabase.getInstance(applicationContext)
//                    database.

                    Result.success()
                }
            }
        } catch (ex: Exception) {
            Log.e(TAG, "Error seeding database", ex)
            Result.failure()
        }
    }

    companion object {
        private const val TAG = "SeedDatabaseWorker"
    }
}
