package com.vivekahealth.vivekahealth.ui.homePage.findCare.map

import android.content.SharedPreferences
import android.graphics.Bitmap
import android.graphics.drawable.BitmapDrawable
import android.graphics.drawable.Drawable
import android.location.Location
import androidx.databinding.Bindable
import androidx.databinding.Observable
import androidx.databinding.PropertyChangeRegistry
import androidx.databinding.library.baseAdapters.BR
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.lifecycle.ViewModel
import com.bumptech.glide.load.DataSource
import com.bumptech.glide.load.engine.GlideException
import com.bumptech.glide.request.RequestListener
import com.bumptech.glide.request.target.Target
import com.bumptech.glide.request.transition.DrawableCrossFadeTransition
import com.vivekahealth.vivekahealth.data.local.home.findCare.ProviderCallData
import com.vivekahealth.vivekahealth.data.remote.home.findCare.FindCareFilterModel
import com.vivekahealth.vivekahealth.data.remote.home.findCare.doctorModelsNew.DoctorResponseNew
import com.vivekahealth.vivekahealth.data.remote.home.findCare.doctorModelsNew.Value
import com.vivekahealth.vivekahealth.ui.homePage.findcareMyDoctorViewAll.repository.FindAllMapRepository
import com.vivekahealth.vivekahealth.ui.homePage.findcareMyDoctorViewAll.repository.FindCareViewAllRepository
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.repository.FindCareSearchRepository
import com.vivekahealth.vivekahealth.ui.homePage.repository.HomeRepository
import com.vivekahealth.vivekahealth.utils.Event
import io.reactivex.android.schedulers.AndroidSchedulers
import io.reactivex.disposables.CompositeDisposable
import io.reactivex.schedulers.Schedulers
import javax.inject.Inject

class MapViewViewModel @Inject constructor(
    private val preferences: SharedPreferences,
    private val homeRepository: HomeRepository,
    private val findCareSearchRepository: FindCareSearchRepository,
    private val findAllMapRepository: FindAllMapRepository,
    private val findCareViewAllRepository: FindCareViewAllRepository

) : ViewModel(), Observable {
    private val callbacks: PropertyChangeRegistry by lazy { PropertyChangeRegistry() }
    private var compositeDisposable: CompositeDisposable = CompositeDisposable()
    var data: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.data, null)
        }
    var reason: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.reason, null)
        }
    var listItem: ArrayList<Value> = arrayListOf()
    var filterData: FindCareFilterModel = FindCareFilterModel()
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.filterData, null)
        }
    var lastSelectedPosition: Int = 0
    var currentLocation: Location? = null
    var mapReady: Boolean = false
    var search: Boolean = false
    var title: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.title, null)
        }
    var imageProfile: Bitmap? = null
    var subTitle: String = ""
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.subTitle, null)
        }
    var loading: Boolean = false
        @Bindable get
        set(value) {
            field = value
            callbacks.notifyCallbacks(this, BR.loading, null)
        }
    private val _scrolled = MutableLiveData<Event<Unit>>()
    val scrolled: LiveData<Event<Unit>>
        get() = _scrolled
    private val _imageLoaded = MutableLiveData<Event<Unit>>()
    val imageLoaded: LiveData<Event<Unit>>
        get() = _imageLoaded
    private val _mapReadyState = MutableLiveData<Event<Unit>>()
    val mapReadyState: LiveData<Event<Unit>>
        get() = _mapReadyState
    private val _dataUpdated = MutableLiveData<Event<DoctorResponseNew>>()
    val dataUpdated: LiveData<Event<DoctorResponseNew>>
        get() = _dataUpdated

    private val _showFilter = MutableLiveData<Event<Unit>>()
    val showFilter: LiveData<Event<Unit>>
        get() = _showFilter
    var searchText: String = ""
    var providerType: String = ""
    var within: String = ""
    var page: Int = 1
    var pageSize: String = "10"
    var searchType: String = ""
    var taxonomyGroup: String = ""

    var locationType: String = ""
    var lastDistance: String = ""
    var codeType: String = ""
    var codeValue: String = ""
    var isLastPage = false
    var isScrolling = false
    var imageBitmap: Bitmap? = null
    var key: String = ""

    @get:Bindable
    val listener = object : RequestListener<Drawable> {
        override fun onLoadFailed(
            e: GlideException?,
            model: Any?,
            target: Target<Drawable>?,
            isFirstResource: Boolean
        ): Boolean {
            loading = false

            return false
        }

        override fun onResourceReady(
            resource: Drawable?,
            model: Any?,
            target: Target<Drawable>?,
            dataSource: DataSource?,
            isFirstResource: Boolean
        ): Boolean {
            imageBitmap = (resource as BitmapDrawable).bitmap
            target?.onResourceReady(
                resource,
                DrawableCrossFadeTransition(10, isFirstResource)
            )
            _imageLoaded.value = Event(Unit)
            loading = false

            return true
        }
    }
    var selfId: String = preferences.getString("memberID", "").toString()

    var grouped: ArrayList<Value> = arrayListOf()
    var providerCallData: ProviderCallData? = ProviderCallData()
    private val _listView = MutableLiveData<Event<Unit>>()
    val listView: LiveData<Event<Unit>>
        get() = _listView
    private val _providerListGenerated = MutableLiveData<Event<ArrayList<Value>>>()
    val providerListGenerated: LiveData<Event<ArrayList<Value>>>
        get() = _providerListGenerated
    private val _mapPinSelect = MutableLiveData<Event<Unit>>()
    val mapPinSelect: LiveData<Event<Unit>>
        get() = _mapPinSelect

    var showFilterView: Boolean = true
        @Bindable get
        set(value) {
            field = value

            callbacks.notifyCallbacks(this, com.vivekahealth.vivekahealth.BR.showFilterView, null)
        }
    fun getProfile(): String {
        val memberId = preferences.getString("memberID", "").toString()
        return preferences.getString("uri$memberId", "").toString()
    }

    init {
    }

    fun updateMapeState() {
        selfId = preferences.getString("memberID", "").toString()
        _mapReadyState.value = Event(Unit)
    }
    fun selectIndex(index: Int) {
        try {
            val item = listItem[index]
            listItem.forEach {
                it.selected = it == item
            }
            _mapPinSelect.value = Event(Unit)
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    fun listView() {
        _listView.value = Event(Unit)
    }

    fun getDataService() {
        if (search) {
            getSearchData()
        } else {
            getProviderList()
        }
    }

    fun getProviderList() {
        if (key.equals("doctor", true)) {
            showFilterView = false
            filterData.speciality = "My Doctor"
            filterData = filterData
            getDoctorList()
        } else {
            showFilterView = true

            getProvider()
        }
    }
    fun openFilter() {
        _showFilter.value = Event(Unit)
    }

    fun getDoctorList() {
        compositeDisposable.add(
            findCareViewAllRepository.getDoctorList(
                preferences.getString("memberID", "").toString(),
                filterData.current?.latitude.toString(),
                filterData.current?.longitude.toString(),
                "",
                ""
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->

                    loading = false

                    page++
                    listItem.addAll(it.values)

                    listItem[lastSelectedPosition].selected = true
                    isLastPage = it.paging?.totalPages == page
                    data = listItem.isEmpty()
                    _providerListGenerated.value = Event(listItem)
                }, {
                    loading = false
                    data = listItem.isEmpty()
                })
        )
    }

    fun getProvider() {
        val list = filterData.selectedSubSpeciality.map { it.speciality }
        val listTaxonomy = filterData.taxonomyCodes

        var tempSpeciality = list.joinToString(separator = "~") { item -> item }
        val tempTaxonomyCode = listTaxonomy.joinToString(separator = ",") { st -> st }
        compositeDisposable.add(
            findAllMapRepository.getProviderList(
                filterData.current?.latitude.toString(), filterData.current?.longitude.toString(), page, pageSize, tempSpeciality,
                filterData.searchText.toString(), preferences.getString("memberID", "").toString(), filterData.searchType.toString(), filterData.taxonomyGroup.toString(), filterData.locationType.toString(), filterData.providerType.toString(), filterData.radius.toString(), filterData.codeType.toString(), filterData.codeValue.toString(),
                tempTaxonomyCode
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->

                    loading = false

                    page++
                    listItem.addAll(it.values)
                    if (listItem.isNotEmpty()) {
                        listItem[lastSelectedPosition].selected = true
                    }

                    isLastPage = it.paging?.totalPages == page
                    data = listItem.isEmpty()

                    _providerListGenerated.value = Event(listItem)
                }, {
                    data = listItem.isEmpty()

                    loading = false
                })
        )
    }
    fun getSearchData() {
        compositeDisposable.add(
            findCareSearchRepository.getProcedureList(
                filterData.searchText.toString(),
                preferences.getString("memberID", "").toString(),
                filterData.current?.latitude.toString(),
                filterData.current?.longitude.toString(),
                filterData.providerType.toString(),
                filterData.radius.toString()
            )
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
                .doOnSubscribe {
                    loading = true
                }
                .subscribe({ it ->
                    try {
                        loading = false
                        listItem.addAll(it.doctors)
                        listItem[lastSelectedPosition].selected = true
                        _providerListGenerated.value = Event(listItem)
                    } catch (exp: IndexOutOfBoundsException) {
                        exp.printStackTrace()
                    }
                }, {
                    loading = false
                })
        )
    }

    override fun addOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.add(callback)
    }

    override fun removeOnPropertyChangedCallback(callback: Observable.OnPropertyChangedCallback?) {
        callbacks.remove(callback)
    }
    override fun onCleared() {
        super.onCleared()
        compositeDisposable.dispose()
    }
}
