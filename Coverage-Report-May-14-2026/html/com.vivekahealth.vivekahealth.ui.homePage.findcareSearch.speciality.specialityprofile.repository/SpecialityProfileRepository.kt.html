package com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityprofile.repository

import com.vivekahealth.vivekahealth.data.remote.home.findCare.BillPreviewModel
import com.vivekahealth.vivekahealth.data.remote.home.findCare.providerRecommended.RecommendedProcedure
import com.vivekahealth.vivekahealth.data.remote.home.findCare.providerRecommended.RecommendedRequest
import com.vivekahealth.vivekahealth.data.remote.register.DataResponse
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityprofile.service.SpecialityProfileService
import com.vivekahealth.vivekahealth.ui.homePage.findcareSearch.speciality.specialityprofile.responce.Procedures
import io.reactivex.Single
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Repository that handles User objects.
 */
@Singleton
class SpecialityProfileRepository @Inject constructor(
    private val findCareService: SpecialityProfileService
) {

    fun getProcedureList(
        npa: String,
        ein: String,
        pageNo: Int,
        pageSize: Int,
        sortBy: String,
        sortOrder: String,
        searchText: String,
        codeType: String?,
        codeValue: String?
    ): Single<Procedures> = findCareService.getSpecialityApi().getProvidersList(
        npa,
        ein,
        pageNo,
        pageSize,
        sortBy,
        sortOrder,
        searchText,
        codeType,
        codeValue
    )

    fun getRecommendedProcedureList(
        recommendedRequest: RecommendedRequest
    ): Single<RecommendedProcedure> = findCareService.getSpecialityApi().getRecommendedProvidersList(
        recommendedRequest
    )
    fun getBillPreview(selfId: String, billPreviewModel: BillPreviewModel): Single<DataResponse> = findCareService.getSpecialityApi().getGenerateEOB(
        selfId,
        billPreviewModel
    )
}
